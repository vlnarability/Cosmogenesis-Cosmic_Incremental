    31→/**
    32→ * PrestigeTab — the divine awakening roadmap.
    33→ *
    34→ * Shows all 10 prestige layers as a vertical timeline. Only Layer 1 (Evolution)
    35→ * is currently unlockable (after first Galactic win); Layers 2-10 are rendered
    36→ * as "???" locked cards so the player can see the full progression arc without
    37→ * the design being spoiled.
    38→ *
    39→ * Unlock logic:
    40→ *  - Layer 1 is revealed once `galacticWins > 0` (set in triggerPrestige on
    41→ *    first Galactic win, alongside `unlockedChallenges`).
    42→ *  - Layers 2-10 are revealed only when the previous layer is "completed".
    43→ *    For now, NO layer is ever marked completed (the Evolution Challenges
    44→ *    system grants mastery but does not gate the next layer). So Layers 2-10
    45→ *    remain locked until future implementation.
    46→ */
    47→
    48→// Layer 1's mechanic summary, shown as a stable label rather than the long
    49→// prose `mechanic` field (which is intended for the codex/tooltip).
    50→const LAYER_1_MECHANIC_LABEL = "Challenges / Masteries";
    51→
    52→// Layer 1's completion threshold — 10 challenge families, one mastered trial
    53→// per family completes the layer (per the design doc).
    54→const LAYER_1_TOTAL_CHALLENGES = CHALLENGES.length;
    55→
    56→type LayerStatus = "locked" | "in_progress" | "mastered";
    57→
    58→/**
    59→ * Determine the status of a layer given the current game state.
    60→ *
    61→ *  - Layer 1 (Evolution) unlocks after first Galactic win; becomes "mastered"
    62→ *    once all 10 challenges are complete.
    63→ *  - Layer 2 (Enlightenment) unlocks once the player has mastered all 10
    64→ *    Evolution challenges (i.e. `enlightenmentUnlocked === true`); becomes
    65→ *    "mastered" once all *implemented* foresight nodes have been purchased
    66→ *    AND a route has been chosen.
    67→ *  - Layers 3-10 are still locked — their unlock chain is not yet implemented.
    68→ */
    69→function getLayerStatus(
    70→  layer: PrestigeLayer,
    71→  galacticWins: number,
    72→  completedChallengesCount: number,
    73→  enlightenmentUnlocked: boolean,
    74→  purchasedForesightCount: number,
    75→  totalForesightCount: number,
    76→  selectedRoute: string | null,
    77→): LayerStatus {
    78→  if (layer.order === 1) {
    79→    if (galacticWins <= 0) return "locked";
    80→    if (completedChallengesCount >= LAYER_1_TOTAL_CHALLENGES) return "mastered";
    81→    return "in_progress";
    82→  }
    83→  if (layer.order === 2) {
    84→    if (!enlightenmentUnlocked) return "locked";
    85→    const allForesightPurchased = purchasedForesightCount >= totalForesightCount;
    86→    if (allForesightPurchased && selectedRoute) return "mastered";
    87→    return "in_progress";
    88→  }
    89→  // Layers 3-10 — not implemented yet.
    90→  return "locked";
    91→}

   325→            ? { boxShadow: "inset 0 1px 0 0 oklch(0.68 0.14 55 / 0.2), 0 0 0 1px oklch(0.68 0.14 55 / 0.3)" }

   329→        {/* === LOCKED LAYER === */}

   331→          <LockedLayerContent layer={layer} isLast={isLast} />

   333→          /* === ENLIGHTENMENT (Layer 2) === */

   335→            layer={layer}
   336→            status={status}
   337→            isCurrent={isCurrent}
   338→            divinity={divinity}
   339→            selectedRoute={selectedRoute}
   340→            purchasedForesight={purchasedForesight}
   341→            totalImplementedForesight={totalImplementedForesight}
   342→            onViewEnlightenment={onViewEnlightenment}
   343→          />
   344→        ) : (
   345→          /* === UNLOCKED LAYER (Layer 1: Evolution) === */
   346→          <UnlockedLayerContent
   347→            layer={layer}
   348→            status={status}
   349→            isCurrent={isCurrent}
   350→            completedChallenges={completedChallenges}
   351→            totalChallenges={totalChallenges}
   352→            mechanicLabel={mechanicLabel}
   353→            onViewChallenges={onViewChallenges}
