    28→    28→// ===== Part 6 — Layer-progress helpers =====
    29→    29→// Per-layer (L_n → L_n+1) unlock progress. Returns the current count and the max (target) count.
    30→    30→// nextLayerKey uses the runtime unlockedLayers keys (e.g. "enlightenment" for L2).
    31→    31→interface NextLayerProgressDef {
    32→    32→  currentLayerName: string; // L_n name (the layer the player needs to have unlocked to progress)
    33→    33→  nextLayerName: string;    // L_n+1 name (the layer the progress bar points toward)
    34→    34→  nextLayerKey: string;     // key in unlockedLayers (or "evolution" for L1)
    35→    35→  current: number;
    36→    36→  max: number;
    37→    37→  metricLabel: string;      // description of the metric (e.g. "Trials completed")
    38→    38→}
    39→    39→
    40→    40→// ===== Part 6 — Layer-next-progress component =====
    41→    41→// Renders a progress bar showing how close the player is to unlocking the NEXT layer (L_(n+1)).
    42→    42→// - If next layer is already unlocked: "✓ {nextLayerName} Unlocked"
    43→    43→// - If current layer (prereq for next) is NOT unlocked: "Requires {currentLayerName}"
    44→    44→// - Otherwise: progress bar with the layer-unlock metric (e.g. "5/10 Foresight nodes").
    45→    45→function LayerNextProgress({
    46→    46→  progress,
    47→    47→  currentLayerUnlocked,
    48→    48→  nextLayerUnlocked,
    49→    49→}: {
    50→    50→  progress: NextLayerProgressDef;
    51→    51→  currentLayerUnlocked: boolean;
    52→    52→  nextLayerUnlocked: boolean;
    53→    53→}) {
    54→    54→  if (nextLayerUnlocked) {
    55→    55→    return (
    56→    56→      <div className="mt-2 flex items-center gap-1.5 text-[0.65rem] text-emerald-300">
    57→    57→        <CheckCircle2 className="w-3 h-3" />
    58→    58→        <span className="font-semibold">{progress.nextLayerName} Unlocked</span>
    59→    59→      </div>
    60→    60→    );
    61→    61→  }
    62→    62→  if (!currentLayerUnlocked) {
    63→    63→    return (
    64→    64→      <div className="mt-2 flex items-center gap-1.5 text-[0.65rem] text-muted-foreground">
    65→    65→        <Lock className="w-3 h-3" />
    66→    66→        <span>Requires <span className="font-semibold text-foreground/80">{progress.currentLayerName}</span> to be unlocked first</span>
    67→    67→      </div>
    68→    68→    );
    69→    69→  }
    70→    70→  const pct = progress.max > 0 ? Math.min(100, (progress.current / progress.max) * 100) : 0;
    71→    71→  return (
    72→    72→    <div className="mt-2">
    73→    73→      <div className="flex items-center justify-between text-[0.6rem] text-muted-foreground">
    74→    74→        <span className="flex items-center gap-1">
    75→    75→          <ArrowRight className="w-2.5 h-2.5" />
    76→    76→          Toward {progress.nextLayerName}
    77→    77→          <span className="text-muted-foreground/70">· {progress.metricLabel}</span>
    78→    78→        </span>
    79→    79→        <span className="tabular-nums">{progress.current}/{progress.max}</span>
    80→    80→      </div>
    81→    81→      <div className="h-1.5 bg-muted/40 rounded-full overflow-hidden mt-0.5">
    82→    82→        <div className="h-full bg-emerald-400/70 transition-all" style={{ width: `${pct}%` }} />
    83→    83→      </div>
    84→    84→    </div>
    85→    85→  );
    86→    86→}
