    47→/**
    48→ * SingularityModal — Layer 6 of the prestige arc.
    49→ *
    50→ * The god enters the system. Will becomes logic; logic becomes law. The god
    51→ * no longer intervenes — it configures. Three pillars:
    52→ *
    53→ *   1. **Relic Loadouts** (8) — pre-run choices, each with one large upside,
    54→ *      one meaningful downside, and one special interaction. Equipped BEFORE
    55→ *      a run starts (up to `relicSlots`, default 3).
    56→ *   2. **Compression** — mark mastered stages as auto-skip on future runs.
    57→ *      Permanent QoL feature once unlocked. Only stages the player has
    58→ *      cleared at least once can be compressed (Galactic cannot be compressed).
    59→ *   3. **Logic Cores** (6) — advanced automation priorities. Only one may be
    60→ *      active at a time. Setting / switching costs Divinity.
    61→ *
    62→ * The modal is gated on `singularityUnlocked === true` — the dialog won't
    63→ * open until the player has enacted at least 2 divine laws (lifetime) AND
    64→ * survived at least one heresy crisis.
    65→ */

    75→  const singularityStats = useGameStore((s) => s.singularityStats);
    76→  const stageClearCounts = useGameStore((s) => s.stageClearCounts);
    77→
    78→  if (!singularityUnlocked) return null;
    79→
    80→  const slotsUsed = equippedRelics.length;
    81→  const slotsAvailable = relicSlots - slotsUsed;
    82→  const compressedCount = compressedStages.length;
    83→
    84→  return (
    85→    <Dialog
    86→      open={showSingularity && singularityUnlocked}
    87→      onOpenChange={setShowSingularity}
    88→    >
    89→      <DialogContent className="max-w-5xl max-h-[90vh] overflow-y-auto glass-strong">
    90→        <DialogHeader>
    91→          <DialogTitle className="flex items-center gap-2 text-lg">
    92→            <Cpu className="w-5 h-5 text-cyan-400" />
    93→            Singularity
    94→            <span className="text-xs text-muted-foreground font-normal hidden sm:inline">
    95→              the god enters the system
    96→            </span>
    97→          </DialogTitle>
    98→          <DialogDescription className="flex items-center justify-between flex-wrap gap-2">
    99→            <span className="max-w-xl">
   100→              The boundary between god and system dissolves. Will becomes logic;
   101→              logic becomes world. The god no longer intervenes — it configures.
   102→              Relics define how a run feels; logic cores decide where it goes.
   103→            </span>
   104→            <div className="flex items-center gap-2 flex-wrap">
   105→              <Badge variant="outline" className="text-violet-300 border-violet-400/40">
   106→                <Sparkles className="w-3 h-3 mr-1" />
   107→                {formatNumber(divinity, 0)} Divinity
   108→              </Badge>
   109→              <Badge
   110→                variant="outline"
   111→                className={cn(
   112→                  "border-cyan-400/40",
   113→                  slotsAvailable === 0
   114→                    ? "text-rose-300 border-rose-500/60"

   117→              >
   118→                <Layers className="w-3 h-3 mr-1" />

   357→              "w-full h-7 text-xs",

   359→                ? "border-rose-500/40 text-rose-300"

   370→          </Button>
   371→        )}
   372→      </div>

   381→function CompressionSection({
   382→  compressedStages,
   383→  stageClearCounts,
   384→  onToggle,
   385→}: {
   386→  compressedStages: string[];
   387→  stageClearCounts: Record<string, number>;
   388→  onToggle: (stageId: string) => void;
   389→}) {
   390→  return (
   391→    <section className="rounded-lg border border-border/60 bg-muted/10 p-3">
   392→      <div className="flex items-center gap-2 mb-1 flex-wrap">
   393→        <Gauge className="w-4 h-4 text-emerald-300" />
   394→        <h3 className="text-sm font-bold uppercase tracking-wide text-foreground">
   395→          Compression
   396→        </h3>
   397→        <span className="text-[0.65rem] text-muted-foreground">
   398→          skip mastered stages with a "what would have been" build · permanent quality-of-life feature
   399→        </span>
   400→        <Badge variant="outline" className="text-[0.6rem] ml-auto text-emerald-300 border-emerald-400/40">
   401→          {compressedStages.length} / {COMPRESSION_PROFILES.length - 1} compressed
   402→        </Badge>
   403→      </div>
   404→      <p className="text-[0.7rem] text-muted-foreground italic mb-2 leading-snug">
   405→        When a stage is compressed, future runs auto-skip through it: the run
   406→        marks the stage as cleared, grants the resources + population a
   407→        mid-build would have earned, and jumps to the stage after. Galactic
   408→        cannot be compressed (it triggers prestige).
   409→      </p>
   410→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-2">
   411→        {COMPRESSION_PROFILES.map((profile) => {
   412→          const stage = STAGES.find((s) => s.id === profile.stageId);
   413→          if (!stage) return null;
   414→          const clearCount = (stageClearCounts as Record<string, number>)?.[stage.id] || 0;
   415→          const canCompress = canCompressStage(stage.id, clearCount);
   416→          const compressed = compressedStages.includes(stage.id);
   417→          return (
   418→            <CompressionCard
   419→              key={stage.id}
   420→              stageName={stage.name}
   421→              stageIcon={stage.icon}
   422→              stageId={stage.id}
   423→              clearCount={clearCount}
   424→              canCompress={canCompress}
   425→              compressed={compressed}
   426→              profile={profile}
   427→              onToggle={() => onToggle(stage.id)}
   428→            />
   429→          );
   430→        })}
   431→      </div>
   432→    </section>
   433→  );
   434→}
   435→
   436→function CompressionCard({
   437→  stageName,
   438→  stageIcon,
   439→  stageId,
   440→  clearCount,
   441→  canCompress,
   442→  compressed,
   443→  profile,
   444→  onToggle,
   445→}: {
   446→  stageName: string;
   447→  stageIcon: string;
   448→  stageId: string;
   449→  clearCount: number;
   450→  canCompress: boolean;
   451→  compressed: boolean;
   452→  profile: ReturnType<typeof getCompressionProfile>;
   453→  onToggle: () => void;
   454→}) {
   455→  const resourceBits: string[] = [];
   456→  if (profile) {
   457→    for (const [r, v] of Object.entries(profile.resourceGrant)) {
   458→      resourceBits.push(`+${v} ${resourceName(r)}`);
   459→    }
   460→  }
   461→  return (
   462→    <div
   463→      className={cn(
   464→        "rounded-md border p-2 transition-all",
   465→        compressed
   466→          ? "border-emerald-500/50 bg-emerald-500/10 ring-1 ring-emerald-500/30"
   467→          : canCompress
   468→            ? "border-border/60 bg-muted/20 hover:bg-muted/30"
   469→            : "border-border/40 bg-muted/10 opacity-60",
   470→      )}
   471→    >
   472→      <div className="flex items-center justify-between gap-1 mb-1">
   473→        <div className="flex items-center gap-1.5 min-w-0">
   474→          <span className="text-base shrink-0">{stageIcon}</span>
   475→          <div className="min-w-0">
   476→            <div className="text-xs font-bold leading-tight truncate">
   477→              {stageName}
   478→            </div>
   479→            <div className="text-[0.55rem] uppercase tracking-wide text-muted-foreground leading-tight">
   480→              cleared {clearCount}× {clearCount === 1 ? "run" : "runs"}
   481→            </div>
   482→          </div>
   483→        </div>
   484→        {compressed ? (
   485→          <Badge className="bg-emerald-500/20 text-emerald-300 border border-emerald-500/40 shrink-0 text-[0.55rem]">
   486→            <CheckCircle2 className="w-2.5 h-2.5 mr-0.5" />
   487→            ON
   488→          </Badge>
   489→        ) : canCompress ? (
   490→          <Badge variant="outline" className="text-[0.55rem] text-muted-foreground shrink-0">
   491→            OFF
   492→          </Badge>
   493→        ) : (
   494→          <Lock className="w-3 h-3 text-muted-foreground shrink-0" />
   495→        )}
   496→      </div>
   497→      {profile && (
   498→        <div className="text-[0.65rem] text-foreground/70 leading-tight mb-1">
   499→          Skip grants: {resourceBits.join(", ")} · +{profile.populationGrant} pop
   500→        </div>
   501→      )}
   502→      {!canCompress && stageId !== "galactic" && (
   503→        <div className="text-[0.6rem] text-amber-300/80 italic">
   504→          Clear this stage at least once to enable compression.
   505→        </div>
   506→      )}
   507→      {stageId === "galactic" && (
   508→        <div className="text-[0.6rem] text-muted-foreground/80 italic">
   509→          Cannot compress — triggers prestige.
   510→        </div>
   511→      )}
   512→      {canCompress && (
   513→        <Button
   514→          size="sm"
   515→          variant="outline"
   516→          onClick={onToggle}
   517→          className={cn(
   518→            "w-full h-6 text-[0.65rem] mt-1",
   519→            compressed
   520→              ? "border-rose-500/40 text-rose-300 hover:bg-rose-500/10"
   521→              : "border-emerald-500/40 text-emerald-300 hover:bg-emerald-500/10",
   522→          )}
   523→        >
   524→          {compressed ? "Disable Compression" : "Enable Compression"}
   525→        </Button>
   526→      )}
   527→    </div>
   528→  );
   529→}

   566→        </h3>
   567→        <span className="text-[0.65rem] text-muted-foreground">
   568→          advanced automation priorities · only one may be active at a time
   569→        </span>
   570→        <Badge variant="outline" className="text-[0.6rem] ml-auto text-cyan-300 border-cyan-400/40">
   571→          {activeCore ? (
   572→            <>
   573→              <Zap className="w-3 h-3 mr-1 text-cyan-300 animate-pulse" />
   574→              {activeCore.name}
   575→            </>
   576→          ) : (
   577→            <>
   578→              <Lock className="w-3 h-3 mr-1" />
   579→              None Active
   580→            </>
   581→          )}
   582→        </Badge>
   583→      </div>
   584→      {activeCore && (
   585→        <div className="rounded-md border border-cyan-500/40 bg-cyan-500/10 p-2 mb-2">
   586→          <div className="text-[0.6rem] uppercase tracking-wide text-cyan-300/80 font-semibold mb-0.5">
   587→            Active Core
   588→          </div>
   589→          <div className="text-xs text-foreground/90 leading-snug">
   590→            {activeCore.icon} <span className="font-bold">{activeCore.name}</span>:{" "}
   591→            {activeCore.effectLabel}
   592→          </div>
   593→          <Button
   594→            size="sm"
   595→            variant="outline"
   596→            onClick={onDeactivate}
   597→            className="mt-2 h-6 text-[0.65rem] border-rose-500/40 text-rose-300 hover:bg-rose-500/10"
   598→          >
   599→            <X className="w-3 h-3 mr-1" />
   600→            Deactivate Core
   601→          </Button>
   602→        </div>
   603→      )}
   604→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-2">
   605→        {cores.map((core) => (
   606→          <LogicCoreCard
   607→            key={core.id}
   608→            core={core}
   609→            divinity={divinity}
   610→            active={activeLogicCore === core.id}
   611→            onActivate={() => onActivate(core.id)}
   612→          />
   613→        ))}
   614→      </div>
   615→    </section>
   616→  );
   617→}

   619→function LogicCoreCard({
   620→  core,
   621→  divinity,
   622→  active,
   623→  onActivate,
   624→}: {
   625→  core: LogicCoreDef;
   626→  divinity: number;
   627→  active: boolean;
   628→  onActivate: () => void;
   629→}) {
   630→  const canAfford = divinity >= core.divinityCost;
   631→  const Icon = PRIORITY_ICONS[core.priority] || Cpu;
   632→  const disabled = active || !canAfford;
   633→
   634→  // Build effect summary
   635→  const effectBits: string[] = [];
   636→  const e = core.effects;
   637→  if (e.popGrowthMult) effectBits.push(`+${Math.round(e.popGrowthMult * 100)}% pop growth`);
   638→  if (e.happinessMult) effectBits.push(`+${Math.round(e.happinessMult * 100)}% happiness`);
   639→  if (e.militaryMult) effectBits.push(`+${Math.round(e.militaryMult * 100)}% military`);
   640→  if (e.scienceMult) effectBits.push(`+${Math.round(e.scienceMult * 100)}% science`);
   641→  if (e.goldMult) effectBits.push(`+${Math.round(e.goldMult * 100)}% gold & production`);
   642→  if (e.productionMult) effectBits.push(`+${Math.round(e.productionMult * 100)}% all production`);
   643→
   644→  return (
   645→    <div
   646→      className={cn(
   647→        "stat-card relative flex flex-col",
   648→        active && "border-cyan-500/50 bg-cyan-500/5 ring-1 ring-cyan-500/30",
   649→      )}
   650→      style={active ? { boxShadow: `inset 0 0 0 1px ${core.color}40` } : undefined}
   651→    >
   652→      <div className="flex items-start gap-2 mb-1.5">
   653→        <div
   654→          className="w-7 h-7 rounded shrink-0 flex items-center justify-center"
   655→          style={{ background: `${core.color}20`, border: `1px solid ${core.color}40` }}
   656→        >
   657→          <Icon className="w-3.5 h-3.5" style={{ color: core.color }} />
   658→        </div>
   659→        <div className="min-w-0 flex-1">
   660→          <div className="text-sm font-bold leading-tight">
   661→            {core.name}
   662→          </div>
   663→          <div className="text-[0.6rem] uppercase tracking-wide text-muted-foreground">
   664→            {core.priority} priority · {core.divinityCost} Divinity
   665→          </div>
   666→        </div>
   667→        {active && (
   668→          <Badge className="bg-cyan-500/20 text-cyan-300 border border-cyan-500/40 shrink-0 text-[0.6rem]">
   669→            <CheckCircle2 className="w-3 h-3 mr-0.5" />
   670→            Active
   671→          </Badge>
   672→        )}
   673→      </div>
   674→
   675→      <p className="text-xs text-muted-foreground italic leading-snug mb-2 line-clamp-3">
   676→        &ldquo;{core.desc}&rdquo;
   677→      </p>
   678→
   679→      {/* Effects */}
   680→      <div className="rounded-md bg-muted/20 border border-border/40 px-2 py-1 mb-1.5">
   681→        <div className="text-[0.55rem] uppercase tracking-wide text-cyan-300/70 font-semibold mb-0.5">
   682→          Effects
   683→        </div>
   684→        <div className="text-[0.7rem] text-foreground/90 leading-tight space-y-0.5">
   685→          {effectBits.length > 0 ? (
   686→            effectBits.map((bit, i) => (
   687→              <div key={i} className="flex items-center gap-1">
   688→                <ChevronRight className="w-2.5 h-2.5 text-cyan-400/70" />
   689→                {bit}
   690→              </div>
   691→            ))
   692→          ) : (
   693→            <div className="text-muted-foreground italic">No mechanical effect</div>
   694→          )}
   695→        </div>
   696→      </div>
   697→
   698→      {/* Action button */}
   699→      <div className="mt-auto pt-1">
   700→        {active ? (
   701→          <div className="h-7 w-full flex items-center justify-center rounded-md bg-cyan-500/10 border border-cyan-500/30 text-cyan-300 text-xs font-semibold">
   702→            <Cpu className="w-3 h-3 mr-1" />
   703→            Active
   704→          </div>
   705→        ) : (
   706→          <Button
   707→            size="sm"
   708→            variant="outline"
   709→            disabled={disabled}
   710→            onClick={onActivate}
   711→            className={cn(
   712→              "w-full h-7 text-xs",
   713→              !canAfford
   714→                ? "border-rose-500/40 text-rose-300"
   715→                : "border-cyan-500/40 text-cyan-300 hover:bg-cyan-500/10",
   716→            )}
   717→          >
   718→            {!canAfford
   719→              ? `Need ${core.divinityCost} Div`
   720→              : `Activate (${core.divinityCost} Div)`}
   721→          </Button>
   722→        )}
   723→      </div>
   724→    </div>
   725→  );
   726→}
