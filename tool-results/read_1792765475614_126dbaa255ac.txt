    43→    43→function mulberry32(seed: number): () => number {
    44→    44→  let t = seed >>> 0;
    45→    45→  return () => {
    46→    46→    t = (t + 0x6d2b79f5) >>> 0;
    47→    47→    let r = Math.imul(t ^ (t >>> 15), 1 | t);
    48→    48→    r = (r + Math.imul(r ^ (r >>> 7), 61 | r)) ^ r;
    49→    49→    return ((r ^ (r >>> 14)) >>> 0) / 4294967296;
    50→    50→  };
    51→    51→}

    62→    62→// Generate stable 3D positions for worlds. Each world gets a deterministic
    63→    63→// random position in a sphere. We use a sqrt-radius distribution so worlds
    64→    64→// are spread more on the outer shell (less clumping at the center) and a
    65→    65→// wide radius range (60..380) so dots are visibly separate in 3D space.
    66→    66→function computeWorldDots(worldTiers: Record<string, number>): WorldDot[] {
    67→    67→  const rng = mulberry32(0xfeedface);
    68→    68→  const dots: WorldDot[] = [];
    69→    69→  for (const tier of FUSION_TIERS) {
    70→    70→    const count = Math.max(0, Math.floor(worldTiers[tier.id] || 0));
    71→    71→    if (count === 0) continue;
    72→    72→    // Cap the visualized count per tier so the 3D scene stays performant.
    73→    73→    const maxDots = Math.min(count, tier.id === "1" ? 220 : 40);
    74→    74→    for (let i = 0; i < maxDots; i++) {
    75→    75→      // Uniform direction (theta + phi via acos trick).
    76→    76→      const theta = rng() * Math.PI * 2;
    77→    77→      const phi = Math.acos(2 * rng() - 1);
    78→    78→      // sqrt distribution → more dots near the outer shell (galaxy look).
    79→    79→      // Radius range 60..380 gives ~6× spread between nearest + farthest,
    80→    80→      // so dots are clearly separated at any rotation angle.
    81→    81→      const r = 60 + Math.sqrt(rng()) * 320;
    82→    82→      dots.push({
    83→    83→        x: r * Math.sin(phi) * Math.cos(theta),
    84→    84→        y: r * Math.sin(phi) * Math.sin(theta),
    85→    85→        z: r * Math.cos(phi),
    86→    86→        tier: tier.id,
    87→    87→        color: TIER_COLORS[tier.id] || "#fff",
    88→    88→        size: TIER_SIZES[tier.id] || 5,
    89→    89→      });
    90→    90→    }
    91→    91→  }
    92→    92→  return dots;
    93→    93→}

    95→    95→// Background starfield — distant points positioned in 3D, rendered with
    96→    96→// reduced opacity for parallax depth. Pushed far back so they don't
    97→    97→// overlap with the world sphere.
    98→    98→function computeStarfield(): WorldDot[] {
    99→    99→  const rng = mulberry32(0x12345678);
   100→   100→  const stars: WorldDot[] = [];
   101→   101→  for (let i = 0; i < 100; i++) {
   102→   102→    const theta = rng() * Math.PI * 2;
   103→   103→    const phi = Math.acos(2 * rng() - 1);
   104→   104→    const r = 500 + rng() * 200; // far outside the world sphere
   105→   105→    stars.push({
   106→   106→      x: r * Math.sin(phi) * Math.cos(theta),
   107→   107→      y: r * Math.sin(phi) * Math.sin(theta),
   108→   108→      z: r * Math.cos(phi),
   109→   109→      tier: "1",
   110→   110→      color: "#ffffff",
   111→   111→      size: 1.5 + rng() * 1.5,
   112→   112→    });
   113→   113→  }
   114→   114→  return stars;
   115→   115→}

   117→   117→// ===========================================================================
   118→   118→// WorldVisualization3D — pure CSS 3D transforms. No external deps.
   119→   119→// Features:
   120→   120→//   • Drag to rotate 360° (rotateX clamped ±89°, rotateY unbounded).
   121→   121→//   • Auto-rotate 6°/s on Y axis when not dragging.
   122→   122→//   • Zoom in/out via buttons (+/-), mouse wheel, or pinch (touch).
   123→   123→//   • Sharper dot rendering: solid dot + thin border + small glow (no
   124→   124→//     soft blur). `will-change: transform` to hint the compositor.
   125→   125→//   • Reset button restores default angle + zoom.
   126→   126→// ===========================================================================
   127→   127→
   128→   128→const ZOOM_MIN = 0.4;
   129→   129→const ZOOM_MAX = 3.0;
   130→   130→const ZOOM_STEP = 0.15;
   131→   131→
   132→   132→export function WorldVisualization3D({ dots }: { dots: WorldDot[] }) {
   133→   133→  // Rotation state — degrees. Drag to update; auto-rotate when idle.
   134→   134→  const [rotX, setRotX] = useState(-18);
   135→   135→  const [rotY, setRotY] = useState(25);
   136→   136→  const [zoom, setZoom] = useState(1.0);
   137→   137→  const [isDragging, setIsDragging] = useState(false);
   138→   138→  const dragStart = useRef<{ x: number; y: number; rotX: number; rotY: number } | null>(null);
   139→   139→  // Pinch-tracking refs (touch). Stores the initial distance + zoom.
   140→   140→  const pinchStart = useRef<{ dist: number; zoom: number } | null>(null);
   141→   141→  const containerRef = useRef<HTMLDivElement>(null);
   142→   142→  const stars = useMemo(() => computeStarfield(), []);
   143→   143→
   144→   144→  // Auto-rotate when not dragging. Subtle continuous Y rotation.
   145→   145→  useEffect(() => {
   146→   146→    if (isDragging) return;
   147→   147→    let raf: number;
   148→   148→    let last = performance.now();
   149→   149→    const tick = (now: number) => {
   150→   150→      const dt = (now - last) / 1000;
   151→   151→      last = now;
   152→   152→      // 6°/s slow rotation around Y axis.
   153→   153→      setRotY((prev) => (prev + 6 * dt) % 360);
   154→   154→      raf = requestAnimationFrame(tick);
   155→   155→    };
   156→   156→    raf = requestAnimationFrame(tick);
   157→   157→    return () => cancelAnimationFrame(raf);
   158→   158→  }, [isDragging]);
   159→   159→
   160→   160→  const onPointerDown = useCallback((e: React.PointerEvent) => {
   161→   161→    e.preventDefault();
   162→   162→    try { (e.target as Element).setPointerCapture(e.pointerId); } catch {}
   163→   163→    dragStart.current = { x: e.clientX, y: e.clientY, rotX, rotY };
   164→   164→    setIsDragging(true);
   165→   165→  }, [rotX, rotY]);
   166→   166→
   167→   167→  const onPointerMove = useCallback((e: React.PointerEvent) => {
   168→   168→    if (!dragStart.current) return;
   169→   169→    const dx = e.clientX - dragStart.current.x;
   170→   170→    const dy = e.clientY - dragStart.current.y;
   171→   171→    // 0.4° per pixel of drag — feels responsive without being twitchy.
   172→   172→    const newRotY = (dragStart.current.rotY + dx * 0.4) % 360;
   173→   173→    let newRotX = dragStart.current.rotX - dy * 0.4;
   174→   174→    // Clamp X rotation to [-89°, 89°] so the scene never fully flips.
   175→   175→    newRotX = Math.max(-89, Math.min(89, newRotX));
   176→   176→    setRotY(newRotY);
   177→   177→    setRotX(newRotX);
   178→   178→  }, []);
   179→   179→
   180→   180→  const onPointerUp = useCallback((e: React.PointerEvent) => {
   181→   181→    if (dragStart.current) {
   182→   182→      try { (e.target as Element).releasePointerCapture(e.pointerId); } catch {}
   183→   183→    }
   184→   184→    dragStart.current = null;
   185→   185→    pinchStart.current = null;
   186→   186→    setIsDragging(false);
   187→   187→  }, []);
   188→   188→
   189→   189→  // Zoom helpers — clamp to [ZOOM_MIN, ZOOM_MAX].
   190→   190→  const clampZoom = useCallback((v: number) => Math.max(ZOOM_MIN, Math.min(ZOOM_MAX, v)), []);
   191→   191→  const zoomIn = useCallback(() => setZoom((z) => clampZoom(z + ZOOM_STEP * 2)), [clampZoom]);
   192→   192→  const zoomOut = useCallback(() => setZoom((z) => clampZoom(z - ZOOM_STEP * 2)), [clampZoom]);
   193→   193→
   194→   194→  // Mouse-wheel zoom — intercepts wheel so the page doesn't scroll.
   195→   195→  const onWheel = useCallback((e: React.WheelEvent) => {
   196→   196→    e.preventDefault();
   197→   197→    // deltaY > 0 = scroll down = zoom out. Step scaled by scroll amount.
   198→   198→    const step = ZOOM_STEP * Math.min(3, Math.abs(e.deltaY) / 50);
   199→   199→    const delta = e.deltaY > 0 ? -step : step;
   200→   200→    setZoom((z) => clampZoom(z + delta));
   201→   201→  }, [clampZoom]);
   202→   202→
   203→   203→  // Touch pinch — when 2 fingers are down, the distance between them
   204→   204→  // drives zoom. We track the initial distance + zoom and apply the ratio.
   205→   205→  const onTouchMove = useCallback((e: React.TouchEvent) => {
   206→   206→    if (e.touches.length !== 2) return;
   207→   207→    e.preventDefault();
   208→   208→    const t1 = e.touches[0];
   209→   209→    const t2 = e.touches[1];
   210→   210→    const dist = Math.hypot(t2.clientX - t1.clientX, t2.clientY - t1.clientY);
   211→   211→    if (!pinchStart.current) {
   212→   212→      pinchStart.current = { dist, zoom };
   213→   213→      return;
   214→   214→    }
   215→   215→    const ratio = dist / pinchStart.current.dist;
   216→   216→    setZoom(clampZoom(pinchStart.current.zoom * ratio));
   217→   217→  }, [zoom, clampZoom]);
   218→   218→
   219→   219→  const resetView = useCallback(() => {
   220→   220→    setRotX(-18);
   221→   221→    setRotY(25);
   222→   222→    setZoom(1.0);
   223→   223→  }, []);
   224→   224→
   225→   225→  return (
   226→   226→    <div className="relative w-full" style={{ height: "clamp(360px, 60vh, 600px)" }}>
   227→   227→      <div
   228→   228→        ref={containerRef}
   229→   229→        className="relative w-full h-full overflow-hidden rounded-lg border border-violet-500/20 bg-gradient-to-b from-[#020817] via-[#0a0a23] to-[#050518] touch-none select-none"
   230→   230→        // Perspective scales inversely with zoom: higher zoom = smaller
   231→   231→        // perspective = closer view. Base 1100px gives comfortable depth.
   232→   232→        style={{ perspective: `${1100 / zoom}px`, perspectiveOrigin: "50% 50%" }}
   233→   233→        onPointerDown={onPointerDown}
   234→   234→        onPointerMove={onPointerMove}
   235→   235→        onPointerUp={onPointerUp}
   236→   236→        onPointerCancel={onPointerUp}
   237→   237→        onWheel={onWheel}
   238→   238→        onTouchMove={onTouchMove}
   239→   239→        role="application"
   240→   240→        aria-label="Drag to rotate · scroll to zoom · 3D world visualization"
   241→   241→      >
   242→   242→        {/* Inner 3D scene — preserve-3d so children are positioned in 3D space.
   243→   243→            Scale by zoom so the whole sphere grows/shrinks together. */}
   244→   244→        <div
   245→   245→          className="absolute left-1/2 top-1/2"
   246→   246→          style={{
   247→   247→            transformStyle: "preserve-3d",
   248→   248→            transform: `translate(-50%, -50%) scale3d(${zoom}, ${zoom}, ${zoom}) rotateX(${rotX}deg) rotateY(${rotY}deg)`,
   249→   249→            willChange: "transform",
   250→   250→          }}
   251→   251→        >
   252→   252→          {/* Starfield — far back, low opacity. */}
   253→   253→          {stars.map((s, i) => (
   254→   254→            <div
   255→   255→              key={`star-${i}`}
   256→   256→              className="absolute rounded-full"
   257→   257→              style={{
   258→   258→                width: `${s.size}px`,
   259→   259→                height: `${s.size}px`,
   260→   260→                background: s.color,
   261→   261→                opacity: 0.4,
   262→   262→                transform: `translate3d(${s.x}px, ${s.y}px, ${s.z}px)`,
   263→   263→                transformStyle: "preserve-3d",
   264→   264→                willChange: "transform",
   265→   265→              }}
   266→   266→            />
   267→   267→          ))}
   268→   268→
   269→   269→          {/* World dots — solid dot + thin border + small glow. */}
   270→   270→          {dots.map((dot, i) => (
   271→   271→            <div
   272→   272→              key={`world-${i}`}
   273→   273→              className="absolute rounded-full"
   274→   274→              style={{
   275→   275→                width: `${dot.size}px`,
   276→   276→                height: `${dot.size}px`,
   277→   277→                marginLeft: `-${dot.size / 2}px`,
   278→   278→                marginTop: `-${dot.size / 2}px`,
   279→   279→                background: dot.color,
   280→   280→                // Sharp 1px border for definition + small glow for visibility.
   281→   281→                // Old boxShadow was 2-layer with 1.6× radius → too soft.
   282→   282→                border: `1px solid ${dot.color}`,
   283→   283→                boxShadow: `0 0 ${Math.max(4, dot.size * 0.6)}px ${dot.size * 0.25}px ${dot.color}66`,
   284→   284→                opacity: 0.95,
   285→   285→                transform: `translate3d(${dot.x}px, ${dot.y}px, ${dot.z}px)`,
   286→   286→                transformStyle: "preserve-3d",
   287→   287→                willChange: "transform",
   288→   288→              }}
   289→   289→              title={`${FUSION_TIERS.find(t => t.id === dot.tier)?.name || dot.tier} world`}
   290→   290→            >
   291→   291→              {dot.tier === "void" && (
   292→   292→                <div
   293→   293→                  className="absolute inset-0 rounded-full border-2"
   294→   294→                  style={{
   295→   295→                    borderColor: "#a855f7",
   296→   296→                    transform: "scale(1.8)",
   297→   297→                    opacity: 0.6,
   298→   298→                  }}
   299→   299→                />
   300→   300→              )}
   301→   301→            </div>
   302→   302→          ))}
   303→   303→        </div>
   304→   304→
   305→   305→        {/* Top-left: hint badge + zoom controls + Reset */}
   306→   306→        <div className="absolute top-2 left-2 flex items-center gap-1 pointer-events-none flex-wrap">
   307→   307→          <Badge variant="outline" className="text-[0.55rem] text-violet-200 border-violet-400/40 bg-slate-950/70 backdrop-blur pointer-events-auto">
   308→   308→            <Orbit className="w-2.5 h-2.5 mr-0.5" /> Drag · scroll to zoom
   309→   309→          </Badge>
   310→   310→          <div className="flex items-center gap-0.5 pointer-events-auto bg-slate-950/70 backdrop-blur rounded-md border border-violet-400/40">
   311→   311→            <Button
   312→   312→              size="sm"

   401→   401→

   403→   403→  const evolutionSliders = useGameStore((s) => s.evolutionSliders || { speed: 0, aggression: 0, curiosity: 0, faith: 0 });
   404→   404→  const setEvolutionSlider = useGameStore((s) => s.setEvolutionSlider);
   405→   405→
   406→   406→  const rawCount = rawWorldCount(worldTiers);
   407→   407→  const effectiveCount = effectiveWorldCount(worldTiers);
   408→   408→  const flameMult = eternalFlameMultiplier(eternalFlameInvested);

   412→   412→  const sliderSpeedMult = 1 + (evolutionSliders.speed || 0) * 0.10;
   413→   413→  const sliderSum =
   414→   414→    (evolutionSliders.speed || 0) +
   415→   415→    (evolutionSliders.aggression || 0) +
   416→   416→    (evolutionSliders.curiosity || 0) +
   417→   417→    (evolutionSliders.faith || 0);
   418→   418→
   419→   419→  const divPerSec = effectiveCount > 0 && peakProductionRate > 0
   420→   420→    ? effectiveCount * peakProductionRate * 0.01 * flameMult * (activeWorldArchetype ? 1.30 : 1) * sliderSpeedMult
   421→   421→    : 0;
   422→   422→
   423→   423→  return (

   426→   426→      <div className="stat-card flex items-center justify-between gap-3 flex-wrap">
   427→   427→        <div className="flex items-center gap-2">
   428→   428→          <Globe className="w-6 h-6 text-violet-300" />
   429→   429→          <div>
   430→   430→            <h2 className="text-lg font-bold">Divine Engine</h2>
   431→   431→            <p className="text-xs text-muted-foreground">The machinery of reality — worlds producing Divinity</p>
   432→   432→          </div>
   433→   433→        </div>
   434→   434→        <div className="flex gap-2 flex-wrap">
   435→   435→          <Badge variant="outline" className="text-violet-300 border-violet-400/40">
   436→   436→            <Sparkles className="w-3 h-3 mr-1" />
   437→   437→            {formatCompact(divinity)} Divinity
   438→   438→          </Badge>
   439→   439→          <Badge variant="outline" className="text-amber-300 border-amber-400/40">
   440→   440→            <TrendingUp className="w-3 h-3 mr-1" />
   441→   441→            {divPerSec > 0 ? `${formatCompact(divPerSec)} Div/s` : "0 Div/s"}
   442→   442→          </Badge>
   443→   443→          {voidShards > 0 && (
   444→   444→            <Badge variant="outline" className="text-purple-300 border-purple-400/40">
   445→   445→              🔮 {formatCompact(voidShards)} Void Shards
   446→   446→            </Badge>
   447→   447→          )}
   448→   448→        </div>
   449→   449→      </div>
   450→   450→
   451→   451→      {/* ===== STATS GRID ===== */}
   452→   452→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   453→   453→        <div className="stat-card text-center">
   454→   454→          <div className="text-[0.6rem] text-muted-foreground uppercase">Raw Worlds</div>
   455→   455→          <div className="text-xl font-bold text-violet-300">{formatNumber(rawCount, 0)}</div>
   456→   456→          <div className="text-[0.55rem] text-muted-foreground">toward cap</div>
   457→   457→        </div>
   458→   458→        <div className="stat-card text-center">
   459→   459→          <div className="text-[0.6rem] text-muted-foreground uppercase">Effective</div>
   460→   460→          <div className="text-xl font-bold text-cyan-300">{formatCompact(effectiveCount)}</div>
   461→   461→          <div className="text-[0.55rem] text-muted-foreground">production-weighted</div>
   462→   462→        </div>
   463→   463→        <div className="stat-card text-center">
   464→   464→          <div className="text-[0.6rem] text-muted-foreground uppercase">Peak Prod</div>
   465→   465→          <div className="text-xl font-bold text-amber-300">{formatCompact(peakProductionRate)}</div>
   466→   466→          <div className="text-[0.55rem] text-muted-foreground">resources/s</div>
   467→   467→        </div>
   468→   468→        <div className="stat-card text-center">
   469→   469→          <div className="text-[0.6rem] text-muted-foreground uppercase">Flame</div>
   470→   470→          <div className="text-xl font-bold text-orange-300">×{flameMult.toFixed(2)}</div>
   471→   471→          <div className="text-[0.55rem] text-muted-foreground">{formatCompact(eternalFlameInvested)} invested</div>
   472→   472→        </div>
   473→   473→      </div>
   474→   474→
   475→   475→      {/* ===== WORLD TIER VISUALIZATION (3D) ===== */}
   476→   476→      {rawCount > 0 && (
   477→   477→        <div className="stat-card">
   478→   478→          <div className="flex items-center gap-2 mb-2 flex-wrap">
   479→   479→            <Layers className="w-4 h-4 text-violet-300" />
   480→   480→            <span className="text-sm font-semibold">3D World Visualization</span>
   481→   481→            <span className="text-[0.65rem] text-muted-foreground">
   482→   482→              — each dot is a world, colored by fusion tier · drag to rotate (360°)
   483→   483→            </span>
   484→   484→          </div>
   485→   485→          <WorldVisualization3D dots={dots} />
   486→   486→        </div>
   487→   487→      )}
   488→   488→
   489→   489→      {/* ===== IDEA-2: WORLD ARCHETYPE INJECTION ===== */}
   490→   490→      <div className="stat-card">
   491→   491→        <div className="flex items-center gap-2 mb-2">
   492→   492→          <Users className="w-4 h-4 text-cyan-300" />
   493→   493→          <span className="text-sm font-semibold">World Archetype</span>
   494→   494→          <span className="text-[0.65rem] text-muted-foreground">— choose what evolves inside your worlds (+30% Divinity/s when active)</span>
   495→   495→        </div>
   496→   496→        <Select
   497→   497→          value={activeWorldArchetype ?? "__none__"}
   498→   498→          onValueChange={(v) => setWorldArchetype(v === "__none__" ? null : v)}
   499→   499→        >
   500→   500→          <SelectTrigger className="h-9 text-sm">
   501→   501→            <SelectValue placeholder="No archetype (no bonus)" />
   502→   502→          </SelectTrigger>
   503→   503→          <SelectContent>
   504→   504→            <SelectItem value="__none__">No archetype (no bonus)</SelectItem>
   505→   505→            {ARCHETYPES.map((a) => (
   506→   506→              <SelectItem key={a.id} value={a.id}>
   507→   507→                {a.glyph} {a.name}
   508→   508→              </SelectItem>
   509→   509→            ))}
   510→   510→          </SelectContent>
   511→   511→        </Select>
   512→   512→        {activeWorldArchetype && (
   513→   513→          <p className="text-[0.6rem] text-cyan-300/80 mt-1">
   514→   514→            Active: {ARCHETYPES.find(a => a.id === activeWorldArchetype)?.name} — +30% Divinity/s from all worlds.

   517→   517→      </div>

   530→   530→        </div>
   531→   531→        <div className="space-y-3">

   535→   535→            icon="⚡"
   536→   536→            value={evolutionSliders.speed}
   537→   537→            max={10}
   538→   538→            color="text-amber-300"
   539→   539→            display={`×${(1 + (evolutionSliders.speed || 0) * 0.10).toFixed(2)} Div/s`}
   540→   540→            onValueChange={(v) => setEvolutionSlider("speed", v)}
   541→   541→            help="+10% Divinity/s per point (×1.0 → ×2.0)."

   548→   548→            max={10}
   549→   549→            color="text-rose-300"

   553→   553→          />
   554→   554→          {/* Curiosity */}

   556→   556→            label="Curiosity"

   559→   559→            max={10}

   566→   566→          <SliderRow

   570→   570→            max={10}
   571→   571→            color="text-violet-300"

   575→   575→          />
   576→   576→        </div>
   577→   577→        <div className="text-[0.6rem] text-muted-foreground italic mt-2 flex items-center gap-1">
   578→   578→          <Flame className="w-2.5 h-2.5" />
   579→   579→          Distribute 10 points across the 4 sliders. Set one higher and the pool shrinks for the others. Reallocate freely at any time.
   580→   580→        </div>
   581→   581→      </div>

   583→   583→      {/* ===== QUICK ACCESS BUTTONS ===== */}

   586→   586→        <div className="flex flex-wrap gap-2">

   588→   588→            <Eye className="w-3 h-3" /> Constellations
