     1→// Genesis — Layer 4 prestige content
     2→//
     3→// When a god reaches Genesis it becomes a true creator. Before life begins, it
     4→// defines cradle worlds, sacred geography, and dormant seeds — the conditions
     5→// under which the next universe will wake. The next run inherits these
     6→// authored choices as starting state.
     7→//
     8→// Five authored facets compose a Cradle World:
     9→//
    10→//   1. Cradle World (8)         — origin world type that biases starting
    11→//                                  resources / archetype / caps / EP reward.
    12→//   2. Prime Condition (6)      — starting boon (resources / pop / cap /
    13→//                                  production / starting tech).
    14→//   3. Sacred Geography (4)     — map template with run-long production /
    15→//                                  cap / pop-growth trade-offs.
    16→//   4. Dormant Seed (4)         — hidden mechanic that awakens mid-run when
    17→//                                  the player reaches the awakening stage.
    18→//   5. Difficulty Tier (4)     — EP reward multiplier (Easy ×0.7 → Extreme ×2.5).
    19→//
    20→// State fields (planned for GameState — to be wired in a follow-up task):
    21→//   - worldConfig: WorldConfig | null                 // authored choices; null before Layer 4
    22→//   - dormantSeedAwakened: boolean                    // whether the dormant seed has awoken this run
    23→//
    24→// Advancement to Layer 5 (Apotheosis) requires authoring a complete Cradle
    25→// World (all 5 facets set) — see prestigeLayers.ts
    26→// `genesis.completeState: "Author a complete Cradle World (all 5 facets set)"`.
    27→
    28→import type { ResourceId, StageId } from "../state/types";
    29→import { RESOURCES } from "./resources";
    30→
    31→// =====================================================================
    32→// Cap bonus model — shared by Cradle Worlds, Prime Conditions, Sacred
    33→// Geography, and (when awakened) the Vault Seed dormant seed.
    34→// =====================================================================
    35→
    36→/**
    37→ * A capacity bonus applied to either a specific list of resources or to
    38→ * every resource ("all"). `value` is additive on top of the baseline cap,
    39→ * e.g. `0.50` = +50% cap, `-0.10` = -10% cap.
    40→ *
    41→ * `target: "all"` expands to every entry in `RESOURCES` (see `worldCapMult`).
    42→ * For per-resource boosts, list the specific ResourceIds (e.g. Crystal
    43→ * Cradle's +50% to stone & alloys).
    44→ */
    45→export interface WorldCapBonus {
    46→  target: ResourceId[] | "all";
    47→  value: number;
    48→}
    49→

   287→// 4. Dormant Seeds (4) — hidden mechanics that awaken mid-run
   288→// =====================================================================
   289→
   290→/**
   291→ * A dormant seed sits silent at run start, then awakens when the player
   292→ * reaches `awakeningStage`. Exactly one of the effect fields below should
   293→ * be set (validator-enforced).
   294→ *
   295→ * Effect field → consumer wiring:
   296→ *   - `productionMult`: add to the global production multiplier for the
   297→ *     remainder of the run (Growth Seed).
   298→ *   - `capBonuses`: apply as an additional cap bonus to all resources
   299→ *     for the remainder of the run (Vault Seed).
   300→ *   - `divinityFlat`: one-time flat grant to the `divinity` resource when
   301→ *     the seed awakens (Wisdom Seed).
   302→ *   - `freeRandomTech`: the store grants one free random tech from the
   303→ *     currently-available tech pool (Ascension Seed). This is a flag-style
   304→ *     effect; the consumer performs the grant and clears the flag, like
   305→ *     Transcendence's `break_through` ritual.
   306→ */
   307→export interface DormantSeedDef {
   308→  /** Stable identifier (matches `worldConfig.dormantSeed` in GameState). */

   310→  /** Human-readable dormant seed name. */
   311→  name: string;
   312→  /** Narrative / mechanical flavour of the seed and its awakening. */
   313→  desc: string;
   314→  /** Emoji glyph shown on the seed card. */
   315→  icon: string;
   316→  /** Stage at which the seed awakens. */
   317→  awakeningStage: StageId;
   318→  /** Additive all-production multiplier granted for the rest of the run. */
   319→  productionMult?: number;
   320→  /** Global capacity bonus granted for the rest of the run. */
   321→  capBonuses?: WorldCapBonus[];
   322→  /** One-time flat grant to `divinity` upon awakening. */
   323→  divinityFlat?: number;

   428→ * A fully-authored Cradle World. All five facets must be set for the
   429→ * Genesis layer to count as "complete" (see prestigeLayers.ts
   430→ * `genesis.completeState`).
   431→ *
   432→ * Stored on `GameState.worldConfig` after Layer 4 unlocks. Consumers
   433→ * (initialRunState, tick loop, EP calc) read the helpers below to
   434→ * extract the mechanical implications of the chosen facets.
   435→ */
   436→export interface WorldConfig {
   437→  cradleWorld: string;
   438→  primeCondition: string;
   439→  sacredGeography: string;
   440→  dormantSeed: string;

   442→}
   443→
   444→// =====================================================================
   445→// Module-load sanity checks
   446→// =====================================================================
   447→
   448→/**
   449→ * Validate the genesis data at module load:
   450→ *   - All ids within each family are unique.
   451→ *   - All cradle world `archetypeBias` ids exist in the archetype set
   452→ *     (checked against the 13 known archetypes).
   453→ *   - Each dormant seed sets exactly one of `productionMult` /
   454→ *     `capBonuses` / `divinityFlat` / `freeRandomTech`.
   455→ *   - Dormant seed `divinityFlat` is positive when set.
   456→ *   - Difficulty `epMultiplier` is positive.
   457→ *
   458→ * Throws on any violation (programmer error, not a runtime / save data
   459→ * error). The check is O(N) and runs once, so it's a no-op on the hot
   460→ * path. Matches the pattern established by `transcendence.ts` and
   461→ * `foresight.ts`.
   462→ */
   463→(function validateGenesisData(): void {
   464→  const KNOWN_ARCHETYPES = new Set<string>([
   465→    "humanoid", "mammalian", "reptilian", "avian", "arthropoid",
   466→    "molluscoid", "fungoid", "plantoid", "aquatic",
   467→    "lithoid", "necroid", "toxoid", "extremophile",
   468→  ]);
   469→
   470→  const seenCradle = new Set<string>();
   471→  for (const c of CRADLE_WORLDS) {
   472→    if (seenCradle.has(c.id)) {
   473→      throw new Error(`genesis.ts: duplicate cradle world id "${c.id}"`);
   474→    }
   475→    seenCradle.add(c.id);
   476→    if (c.archetypeBias && !KNOWN_ARCHETYPES.has(c.archetypeBias)) {
   477→      throw new Error(
   478→        `genesis.ts: cradle world "${c.id}" has unknown archetypeBias "${c.archetypeBias}"`
   479→      );
   480→    }
   481→    if (c.productionMult !== undefined && c.productionMult < 0) {
   482→      throw new Error(`genesis.ts: cradle world "${c.id}" has negative productionMult`);
   483→    }
   484→    if (c.epRewardMult !== undefined && c.epRewardMult < 0) {
   485→      throw new Error(`genesis.ts: cradle world "${c.id}" has negative epRewardMult`);
   486→    }
   487→  }
   488→
   489→  const seenPrime = new Set<string>();
   490→  for (const p of PRIME_CONDITIONS) {
   491→    if (seenPrime.has(p.id)) {
   492→      throw new Error(`genesis.ts: duplicate prime condition id "${p.id}"`);
   493→    }
   494→    seenPrime.add(p.id);
   495→    if (p.productionMult !== undefined && p.productionMult < 0) {
   496→      throw new Error(`genesis.ts: prime condition "${p.id}" has negative productionMult`);
   497→    }
   498→    if (p.startPopulation !== undefined && p.startPopulation < 0) {
   499→      throw new Error(`genesis.ts: prime condition "${p.id}" has negative startPopulation`);
   500→    }
   501→  }
   502→
   503→  const seenGeo = new Set<string>();
   504→  for (const g of SACRED_GEOGRAPHIES) {
   505→    if (seenGeo.has(g.id)) {
   506→      throw new Error(`genesis.ts: duplicate sacred geography id "${g.id}"`);
   507→    }
   508→    seenGeo.add(g.id);
   509→  }
   510→
   511→  const seenSeed = new Set<string>();
   512→  for (const s of DORMANT_SEEDS) {
   513→    if (seenSeed.has(s.id)) {
   514→      throw new Error(`genesis.ts: duplicate dormant seed id "${s.id}"`);
   515→    }
   516→    seenSeed.add(s.id);
   517→    const effects = [
   518→      s.productionMult !== undefined,
   519→      s.capBonuses !== undefined,
   520→      s.divinityFlat !== undefined,
   521→      s.freeRandomTech !== undefined,
   522→    ].filter(Boolean).length;
   523→    if (effects !== 1) {
   524→      throw new Error(
   525→        `genesis.ts: dormant seed "${s.id}" must set exactly one effect (set ${effects})`
   526→      );
   527→    }
   528→    if (s.divinityFlat !== undefined && s.divinityFlat <= 0) {
   529→      throw new Error(`genesis.ts: dormant seed "${s.id}" has non-positive divinityFlat`);
   530→    }
   531→    if (s.productionMult !== undefined && s.productionMult < 0) {
   532→      throw new Error(`genesis.ts: dormant seed "${s.id}" has negative productionMult`);
   533→    }
   534→  }
   535→
   536→  const seenDiff = new Set<string>();
   537→  for (const d of DIFFICULTY_TIERS) {
   538→    if (seenDiff.has(d.id)) {
   539→      throw new Error(`genesis.ts: duplicate difficulty id "${d.id}"`);
   540→    }
   541→    seenDiff.add(d.id);
   542→    if (d.epMultiplier <= 0) {
   543→      throw new Error(`genesis.ts: difficulty "${d.id}" has non-positive epMultiplier`);
   544→    }
   545→  }
   546→})();

   552→/**
   553→ * Apply a `WorldCapBonus` to an accumulator map. "all" expands to every
   554→ * entry in `RESOURCES`; otherwise the listed ResourceIds are incremented.
   555→ *
   556→ * Internal helper used by `worldCapMult`.
   557→ */
   558→function applyCapBonus(
   559→  acc: Partial<Record<ResourceId, number>>,
   560→  bonus: WorldCapBonus
   561→): void {
   562→  if (bonus.target === "all") {
   563→    for (const r of RESOURCES) {
   564→      acc[r.id] = (acc[r.id] ?? 0) + bonus.value;
   565→    }
   566→  } else {
   567→    for (const rid of bonus.target) {
   568→      acc[rid] = (acc[rid] ?? 0) + bonus.value;
   569→    }
   570→  }
   571→}

   573→/**
   574→ * Sum the additive production multiplier from all authored facets.
   575→ *
   576→ * Sources (all additive, e.g. 0.05 + 0.10 = 0.15 = +15%):
   577→ *   - Cradle World:    Ashen Womb (+5%)
   578→ *   - Prime Condition: Stable Geology (+5%), Lucky Stars (+10%)
   579→ *   - Sacred Geography: Archipelago (-5%), Great Continent (+10%),
   580→ *                       Ringed Ocean (-10%), Skylands (+15%)
   581→ *   - Dormant Seed:    Growth Seed (+10%) — only counted when `awakened`
   582→ *
   583→ * @param config     The authored WorldConfig.
   584→ * @param awakened   Whether the dormant seed has awakened this run.
   585→ * @returns Additive production multiplier (0.10 = +10% all production).
   586→ *
   587→ * @example
   588→ *   worldProductionMult({cradleWorld:"ashen_womb", primeCondition:"lucky_stars",
   589→ *     sacredGeography:"great_continent", dormantSeed:"growth_seed",
   590→ *     difficulty:"normal"}, false)
   591→ *     // → 0.25  (0.05 + 0.10 + 0.10; growth seed not yet awakened)
   592→ *   worldProductionMult({...same...}, true)
   593→ *     // → 0.35  (+0.10 from awakened Growth Seed)
   594→ */
   595→export function worldProductionMult(
   596→  config: WorldConfig,
   597→  awakened: boolean
   598→): number {
   599→  let total = 0;
   600→
   601→  const cradle = CRADLE_WORLD_MAP[config.cradleWorld];
   602→  if (cradle?.productionMult) total += cradle.productionMult;
   603→
   604→  const prime = PRIME_CONDITION_MAP[config.primeCondition];
   605→  if (prime?.productionMult) total += prime.productionMult;
   606→
   607→  const geo = SACRED_GEOGRAPHY_MAP[config.sacredGeography];
   608→  if (geo?.productionMult) total += geo.productionMult;
   609→
   610→  if (awakened) {
   611→    const seed = DORMANT_SEED_MAP[config.dormantSeed];
   612→    if (seed?.productionMult) total += seed.productionMult;
   613→  }
   614→
   615→  return total;
   616→}

   618→/**
   619→ * Sum the additive capacity multiplier per resource from all authored
   620→ * facets.
   621→ *
   622→ * Sources:
   623→ *   - Cradle World:    Crystal Cradle (+50% stone & alloys),
   624→ *                       Frozen Origin (+20% all)
   625→ *   - Prime Condition: Fertile Soil (+10% food)
   626→ *   - Sacred Geography: Archipelago (+15% all), Great Continent (+5% all),
   627→ *                       Skylands (-10% all)
   628→ *   - Dormant Seed:    Vault Seed (+15% all) — only counted when `awakened`
   629→ *
   630→ * Global ("all") bonuses are expanded across every resource in
   631→ * `RESOURCES`; per-resource bonuses only affect the listed resources.
   632→ *
   633→ * @param config     The authored WorldConfig.
   634→ * @param awakened   Whether the dormant seed has awakened this run.
   635→ * @returns Per-resource additive cap multiplier map (0.50 = +50% cap).
   636→ *
   637→ * @example
   638→ *   worldCapMult({cradleWorld:"crystal_cradle", primeCondition:"fertile_soil",
   639→ *     sacredGeography:"archipelago", dormantSeed:"vault_seed",
   640→ *     difficulty:"normal"}, false)
   641→ *     // → { stone: 0.65, alloys: 0.65, food: 0.25, ...allOthers: 0.15 }
   642→ *        (crystal +50% to stone/alloys, fertile +10% to food,
   643→ *         archipelago +15% to all, vault seed not yet awakened)
   644→ *   worldCapMult({...same...}, true)
   645→ *     // → vault seed adds another +0.15 to every resource
   646→ */
   647→export function worldCapMult(
   648→  config: WorldConfig,
   649→  awakened: boolean
   650→): Partial<Record<ResourceId, number>> {
   651→  const acc: Partial<Record<ResourceId, number>> = {};
   652→
   653→  const cradle = CRADLE_WORLD_MAP[config.cradleWorld];
   654→  if (cradle?.capBonuses) {
   655→    for (const b of cradle.capBonuses) applyCapBonus(acc, b);
   656→  }
   657→
   658→  const prime = PRIME_CONDITION_MAP[config.primeCondition];
   659→  if (prime?.capBonuses) {
   660→    for (const b of prime.capBonuses) applyCapBonus(acc, b);
   661→  }
   662→
   663→  const geo = SACRED_GEOGRAPHY_MAP[config.sacredGeography];
   664→  if (geo?.capBonuses) {
   665→    for (const b of geo.capBonuses) applyCapBonus(acc, b);
   666→  }
   667→
   668→  if (awakened) {
   669→    const seed = DORMANT_SEED_MAP[config.dormantSeed];
   670→    if (seed?.capBonuses) {
   671→      for (const b of seed.capBonuses) applyCapBonus(acc, b);
   672→    }
   673→  }
   674→
   675→  return acc;
   676→}

   703→/**
   704→ * Aggregate the starting resource bonuses from the authored Cradle
   705→ * World and Prime Condition.
   706→ *
   707→ * Sources:
   708→ *   - Cradle World:    Primordial Soup (+10 glucose, +5 proteins),
   709→ *                       Ashen Womb (+15 ATP), Verdant Eden (+10 food, +5 water,
   710→ *                       +5 happiness), Salt Abyss (+5 proteins, +5 lipids),
   711→ *                       Rotbloom Basin (+10 organic matter, +3 culture)
   712→ *   - Prime Condition: Fertile Soil (+5 happiness), Warm Currents (+5 water,
   713→ *                       +5 food), Ancient Relics (+5 knowledge, +3 culture)
   714→ *
   715→ * `happiness` is included in the returned map; `worldStartHappinessBonus`
   716→ * reads it back out as a convenience.
   717→ *
   718→ * Dormant seeds do NOT grant starting resources (they awaken mid-run),
   719→ * so they are not consulted here.
   720→ *
   721→ * @param config  The authored WorldConfig.
   722→ * @returns Partial resource → amount map (additive on top of baseline).
   723→ *
   724→ * @example
   725→ *   worldStartResourceBonus({cradleWorld:"primordial_soup",
   726→ *     primeCondition:"warm_currents", ...})
   727→ *     // → { glucose: 10, proteins: 5, water: 5, food: 5 }
   728→ */
   729→export function worldStartResourceBonus(
   730→  config: WorldConfig
   731→): Partial<Record<ResourceId, number>> {
   732→  const acc: Partial<Record<ResourceId, number>> = {};
   733→
   734→  const cradle = CRADLE_WORLD_MAP[config.cradleWorld];
   735→  if (cradle?.startResources) {
   736→    for (const [rid, amt] of Object.entries(cradle.startResources)) {
   737→      acc[rid as ResourceId] = (acc[rid as ResourceId] ?? 0) + (amt as number);
   738→    }
   739→  }
   740→
   741→  const prime = PRIME_CONDITION_MAP[config.primeCondition];
   742→  if (prime?.startResources) {
   743→    for (const [rid, amt] of Object.entries(prime.startResources)) {
   744→      acc[rid as ResourceId] = (acc[rid as ResourceId] ?? 0) + (amt as number);
   745→    }
   746→  }
   747→
   748→  return acc;
   749→}

   751→/**
   752→ * Total EP reward multiplier — the product of the difficulty tier's
   753→ * base multiplier and the cradle world's EP reward bonus.
   754→ *
   755→ * Sources:
   756→ *   - Difficulty:    Easy ×0.7, Normal ×1.0, Hard ×1.6, Extreme ×2.5
   757→ *   - Cradle World:  Void Basin (+25% EP reward)
   758→ *
   759→ * The cradle world's `epRewardMult` is additive on top of the difficulty
   760→ * base, so the total is `difficulty.epMultiplier * (1 + cradle.epRewardMult)`.
   761→ * For Void Basin on Normal: `1.0 * (1 + 0.25) = 1.25`.
   762→ *
   763→ * @param config  The authored WorldConfig.
   764→ * @returns Final EP multiplier to apply to prestige-earned EP.
   765→ *
   766→ * @example
   767→ *   worldEpMultiplier({cradleWorld:"void_basin", difficulty:"normal", ...})
   768→ *     // → 1.25  (1.0 × 1.25)
   769→ *   worldEpMultiplier({cradleWorld:"primordial_soup", difficulty:"extreme", ...})
   770→ *     // → 2.5   (2.5 × 1.0)
   771→ *   worldEpMultiplier({cradleWorld:"void_basin", difficulty:"hard", ...})
   772→ *     // → 2.0   (1.6 × 1.25)
   773→ */
   774→export function worldEpMultiplier(config: WorldConfig): number {
   775→  const diff = DIFFICULTY_MAP[config.difficulty];
   776→  const base = diff?.epMultiplier ?? 1.0;
   777→
   778→  const cradle = CRADLE_WORLD_MAP[config.cradleWorld];
   779→  const bonus = cradle?.epRewardMult ?? 0;
   780→
   781→  return base * (1 + bonus);
   782→}
