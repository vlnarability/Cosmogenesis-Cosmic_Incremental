    20→// ===== Part 6 — Layer-progress helpers =====
    21→// Per-layer (L_n → L_n+1) unlock progress. Returns the current count and the max (target) count.
    22→// nextLayerKey uses the runtime unlockedLayers keys (e.g. "enlightenment" for L2).
    23→interface NextLayerProgressDef {
    24→  currentLayerName: string; // L_n name (the layer the player needs to have unlocked to progress)
    25→  nextLayerName: string;    // L_n+1 name (the layer the progress bar points toward)
    26→  nextLayerKey: string;     // key in unlockedLayers (or "evolution" for L1)
    27→  current: number;
    28→  max: number;
    29→  metricLabel: string;      // description of the metric (e.g. "Trials completed")
    30→}
    31→
    32→// ===== Part 6 — Layer-next-progress component =====
    33→// Renders a progress bar showing how close the player is to unlocking the NEXT layer (L_(n+1)).
    34→// - If next layer is already unlocked: "✓ {nextLayerName} Unlocked"
    35→// - If current layer (prereq for next) is NOT unlocked: "Requires {currentLayerName}"
    36→// - Otherwise: progress bar with the layer-unlock metric (e.g. "5/10 Foresight nodes").
    37→function LayerNextProgress({
    38→  progress,
    39→  currentLayerUnlocked,
    40→  nextLayerUnlocked,
    41→}: {
    42→  progress: NextLayerProgressDef;
    43→  currentLayerUnlocked: boolean;
    44→  nextLayerUnlocked: boolean;
    45→}) {
    46→  if (nextLayerUnlocked) {
    47→    return (
    48→      <div className="mt-2 flex items-center gap-1.5 text-[0.65rem] text-emerald-300">
    49→        <CheckCircle2 className="w-3 h-3" />
    50→        <span className="font-semibold">{progress.nextLayerName} Unlocked</span>
    51→      </div>
    52→    );
    53→  }
    54→  if (!currentLayerUnlocked) {
    55→    return (
    56→      <div className="mt-2 flex items-center gap-1.5 text-[0.65rem] text-muted-foreground">
    57→        <Lock className="w-3 h-3" />
    58→        <span>Requires <span className="font-semibold text-foreground/80">{progress.currentLayerName}</span> to be unlocked first</span>
    59→      </div>
    60→    );
    61→  }
    62→  const pct = progress.max > 0 ? Math.min(100, (progress.current / progress.max) * 100) : 0;
    63→  return (
    64→    <div className="mt-2">
    65→      <div className="flex items-center justify-between text-[0.6rem] text-muted-foreground">
    66→        <span className="flex items-center gap-1">
    67→          <ArrowRight className="w-2.5 h-2.5" />
    68→          Toward {progress.nextLayerName}
    69→          <span className="text-muted-foreground/70">· {progress.metricLabel}</span>
    70→        </span>
    71→        <span className="tabular-nums">{progress.current}/{progress.max}</span>
    72→      </div>
    73→      <div className="h-1.5 bg-muted/40 rounded-full overflow-hidden mt-0.5">
    74→        <div className="h-full bg-emerald-400/70 transition-all" style={{ width: `${pct}%` }} />
    75→      </div>
    76→    </div>
    77→  );
    78→}

   721→function ProgressRow({ label, value, max }: { label: string; value: number; max: number }) {
   722→  const pct = max > 0 ? Math.min(100, (value / max) * 100) : 0;
   723→  return (
   724→    <div className="mt-1">
   725→      <div className="flex items-center justify-between text-[0.6rem] text-muted-foreground">
   726→        <span>{label}</span>
   727→        <span>{value}/{max}</span>
   728→      </div>
   729→      <div className="h-1 bg-muted/40 rounded-full overflow-hidden mt-0.5">
   730→        <div className="h-full bg-amber-400/70 transition-all" style={{ width: `${pct}%` }} />
   731→      </div>
   732→    </div>
   733→  );
   734→}

   736→function LayerRow({
   737→  layer,
   738→  unlocked,
   739→  isLast,
   740→}: {
   741→  layer: PrestigeLayer;
   742→  unlocked: boolean;

   776→function formatUnlock(layer: PrestigeLayer): string {
   777→  switch (layer.unlockCondition.type) {
   778→    case "default":
   779→      return "Default";
   780→    case "galacticWins":
   781→      return `${layer.unlockCondition.value} wins`;
   782→    case "challengesCompleted":
   783→      return `${layer.unlockCondition.value} trials`;
   784→    case "totalRuns":
   785→      return `${layer.unlockCondition.value} runs`;
   786→    default:
   787→      return "Locked";
   788→  }
   789→}
