    10→import { TECH_MAP, TECHS } from "../data/techs";
    11→import { UPGRADE_MAP, UPGRADES, upgradeCost } from "../data/upgrades";
    12→import { STORY_MAP, STORY_TRIGGERS } from "../data/story";
    13→import { ARCHETYPE_MAP, dominantArchetype } from "../data/archetypes";
    14→import { ACHIEVEMENTS, achievementBonus, type AchievementCheckCtx } from "../data/achievements";

    16→
    17→const STORAGE_KEY = "evolution_idle_v2";
    18→
    19→function initialRunState(): Partial<GameState> {
    20→  const resources = emptyResources();
    21→  const capacities = emptyCapacities();
    22→  resources.atp = 5;
    23→  resources.glucose = 8;
    24→  resources.happiness = 75;
    25→  return {
    26→    stageIndex: 0,
    27→    time: 0,
    28→    population: 8,
    29→    populationProgress: 0,
    30→    maxPopulation: 8,
    31→    resources,
    32→    capacities,
    33→    ownedSystems: {},
    34→    systemEnabled: {},
    35→    technologies: {},
    36→    lockedTechs: {},
    37→    archetypeAffinity: {},
    38→    lockedArchetype: null,
    39→    log: ["New run started. The first spark stirs."],
    40→  };
    41→}
    42→
    43→function initialMetaState(): Partial<GameState> {
    44→  return {
    45→    evolutionPoints: 0,
    46→    totalRuns: 0,
    47→    galacticWins: 0,
    48→    stageClearCounts: Object.fromEntries(STAGE_IDS.map((s) => [s, 0])) as Record<StageId, number>,
    49→    fastestCellClear: 0,
    50→    totalEventsResolved: 0,
    51→    totalPlayTime: 0,
    52→    totalActions: 0,
    53→    totalSystemsBuilt: 0,
    54→    totalTechResearched: 0,
    55→    upgrades: Object.fromEntries(UPGRADES.map((u) => [u.id, 0])),
    56→    storyUnlocked: { first_spark: true },
    57→    storyAcknowledged: {},
    58→    archive: [],
    59→    archivedArchetypes: [],
    60→    unlockedLayers: { evolution: true },
    61→    achievements: {},
    62→    newAchievements: [],
    63→    currentTab: "actions",
    64→    speed: 1,
    65→    paused: false,
    66→    showShop: false,
    67→    showEvolve: false,
    68→    showSettings: false,
    69→    activeStoryPopup: "first_spark",
    70→    activeEvent: null,
    71→    eventCooldown: 60, // first event can fire after 60s
    72→    eventHistory: [],
    73→    hasSeenIntro: false,
    74→    lastSaved: Date.now(),
    75→    tutorialStep: 0,
    76→    tutorialActive: true,
    77→    tutorialDismissed: false,
    78→    eventFrequency: "normal",
    79→  };
    80→}
    81→
    82→export const useGameStore = create<GameStore>()(
    83→  persist(
    84→    (set, get) => ({
    85→      ...initialRunState(),
    86→      ...initialMetaState(),
    87→
    88→      // ============ TICK ============
    89→      tick: (dt) => {
    90→        const s = get();
    91→        if (s.paused) return;
    92→        const realDt = dt * s.speed;
    93→
    94→        const autoLvl = s.upgrades["inherited_efficiency"] || 0;
    95→        const autoMult = 1 + autoLvl * 0.15;
    96→        const techMult = techMultiplier(s, "production");
    97→        const capBoostLvl = s.upgrades["expansive_vaults"] || 0;
    98→        const capMult = 1 + capBoostLvl * 0.20;
    99→        const hasAutoBalancer = (s.upgrades["auto_balancer"] || 0) > 0;
   100→
   101→        // Archetype bonus
   102→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   103→        const archAllMult = archetype?.bonus.allProductionMult || 0;
   104→        const archPopMult = archetype?.bonus.populationGrowthMult || 0;
   105→        const archCapMult = archetype?.bonus.capMult || 0;
   106→
   107→        const resources = { ...s.resources };
   108→        let capacities = recomputeCapacitiesFresh(s, capMult + archCapMult);
   109→
   110→        // Apply system production — with cap-aware pausing
   111→        for (const [sysId, count] of Object.entries(s.ownedSystems)) {
   112→          if (!count) continue;
   113→          const sys = SYSTEM_MAP[sysId];
   114→          if (!sys) continue;
   115→          if (sys.requiredTech && !s.technologies[sys.requiredTech]) continue;
   116→
   117→          // Check if system is manually disabled
   118→          const manuallyEnabled = s.systemEnabled[sysId] !== false; // default true
   119→          if (!manuallyEnabled && !hasAutoBalancer) continue;
   120→
   121→          // Cap-aware logic: if all output resources are at cap, idle the system
   122→          // (don't produce, don't consume upkeep — no waste)
   123→          const outputsCapped = Object.keys(sys.produces).length > 0 &&
   124→            Object.keys(sys.produces).every((r) => {
   125→              const cap = capacities[r] || 0;
   126→              return (resources[r] || 0) >= cap - 0.01;
   127→            });
   128→
   129→          if (outputsCapped) {
   130→            // System idles — no production, no upkeep
   131→            continue;
   132→          }
   133→
   134→          const stageMult = sys.stage === STAGES[s.stageIndex].id ? 1 : 0.5;
   135→          const mult = autoMult * techMult * stageMult * (1 + archAllMult);
   136→
   137→          // Per-resource archetype bonus
   138→          const produces = sys.produces as Record<string, number>;
   139→          for (const [r, v] of Object.entries(produces)) {
   140→            const archResBonus = archetype?.bonus.productionMult?.[r] || 0;
   141→            const rate = v * count * mult * (1 + archResBonus);
   142→            resources[r] = (resources[r] || 0) + rate * realDt;
   143→          }
   144→          // Upkeep
   145→          if (sys.upkeep) {
   146→            const upkeep = sys.upkeep as Record<string, number>;
   147→            for (const [r, v] of Object.entries(upkeep)) {
   148→              resources[r] = (resources[r] || 0) - v * count * realDt;
   149→            }
   150→          }
   151→        }
   152→
   153→        // Clamp resources to capacities (no overflow)
   154→        for (const r of Object.keys(resources)) {
   155→          const cap = capacities[r];
   156→          if (cap != null && resources[r] > cap) resources[r] = cap;
   157→          if (resources[r] < 0) resources[r] = 0;
   158→        }
   159→
   160→        // Slow passive ATP for cell (keeps early game from stalling)
   161→        if (STAGES[s.stageIndex].id === "cell") {
   162→          resources.atp = Math.min(capacities.atp || 30, resources.atp + 0.1 * realDt);
   163→        }
   164→
   165→        // Happiness drift toward 50 baseline if no source
   166→        // (prevents permanent high happiness from start)
   167→        // — disabled: let systems/tech maintain it
   168→
   169→        // Population growth with progress bar
   170→        // Growth rate scales with happiness and food/water availability
   171→        let popGrowthRate = 0.015; // base per second
   172→        const happiness = resources.happiness || 50;
   173→        popGrowthRate *= (1 + (happiness - 40) / 200); // happiness 100 → +30%, happiness 20 → -10%
   174→        popGrowthRate *= (1 + archPopMult);
   175→        // Penalty if food/water are critically low (creature+)
   176→        const curStage = STAGES[s.stageIndex];
   177→        if (["creature", "tribal", "civilization", "empire"].includes(curStage.id)) {
   178→          const foodCap = capacities.food || 1;
   179→          const waterCap = capacities.water || 1;
   180→          const foodPct = (resources.food || 0) / foodCap;
   181→          const waterPct = (resources.water || 0) / waterCap;
   182→          if (foodPct < 0.1) popGrowthRate *= 0.3;
   183→          if (waterPct < 0.1) popGrowthRate *= 0.3;
   184→        }
   185→
   186→        let populationProgress = (s.populationProgress || 0) + popGrowthRate * realDt;
   187→        let population = s.population;
   188→        while (populationProgress >= 1) {
   189→          populationProgress -= 1;
   190→          population += 1;
   191→        }
   192→
   193→        const maxPopulation = Math.max(s.maxPopulation || 0, population);
   194→
   195→        const newTime = s.time + realDt;
   196→
   197→        // Check story triggers
   198→        const newStory = { ...s.storyUnlocked };
   199→        let activePopup = s.activeStoryPopup;
   200→        const ctx = { ...s, resources, ownedSystems: s.ownedSystems };
   201→        for (const trig of STORY_TRIGGERS) {
   202→          if (!newStory[trig.id] && trig.condition(ctx)) {
   203→            newStory[trig.id] = true;
   204→            if (!activePopup) activePopup = trig.id;
   205→          }
   206→        }
   207→
   208→        // Check achievements periodically (every ~2 seconds of game time)
   209→        let achievements = s.achievements;
   210→        let newAchievements = s.newAchievements;
   211→        const checkInterval = 2.0;
   212→        const lastCheck = (s as any)._lastAchCheck || 0;
   213→        if (newTime - lastCheck >= checkInterval) {
   214→          const curState = { ...s, resources, time: newTime } as GameState;
   215→          const result = checkAchievements(curState);
   216→          if (result.newOnes.length > 0) {
   217→            achievements = result.earned;
   218→            newAchievements = [...(newAchievements || []), ...result.newOnes];
   219→          } else if (Object.keys(result.earned).length !== Object.keys(achievements || {}).length) {
   220→            achievements = result.earned;
   221→          }
   222→          (s as any)._lastAchCheck = newTime; // not persisted, just a runtime marker
   223→        }
   224→
   225→        // Random event firing — only if no popup/modal is active
   226→        let activeEvent = s.activeEvent;
   227→        let eventCooldown = (s.eventCooldown || 0) - realDt;
   228→        if (eventCooldown < 0) eventCooldown = 0;
   229→        const eventFreq = s.eventFrequency || "normal";
   230→        if (eventFreq !== "off" && !activeEvent && !activePopup && eventCooldown === 0) {
   231→          const stageId = STAGES[s.stageIndex].id;
   232→          const possible = eventsForStage(stageId, newTime);
   233→          if (possible.length > 0) {
   234→            // Weighted random selection based on rarity
   235→            const totalWeight = possible.reduce((sum, e) => sum + eventWeight(e), 0);
   236→            const roll = Math.random() * totalWeight;
   237→            let cum = 0;
   238→            for (const ev of possible) {
   239→              cum += eventWeight(ev);
   240→              if (roll < cum) {
   241→                activeEvent = ev.id;
   242→                // Cooldown depends on frequency setting
   243→                const base = eventFreq === "frequent" ? 30 : 90;
   244→                const range = eventFreq === "frequent" ? 30 : 90;
   245→                eventCooldown = base + Math.random() * range;
   246→                break;
   247→              }
   248→            }
   249→          }
   250→        }
   251→
   252→        set({
   253→          resources,
   254→          capacities,
   255→          time: newTime,
   256→          population,
   257→          maxPopulation,
   258→          populationProgress,
   259→          storyUnlocked: newStory,
   260→          activeStoryPopup: activePopup,
   261→          achievements,
   262→          newAchievements,
   263→          activeEvent,
   264→          eventCooldown,
   265→          totalPlayTime: (s.totalPlayTime || 0) + realDt,
   266→        });
   267→      },
   268→
   269→      // ============ ACTIONS ============
   270→      performAction: (actionId) => {
   271→        const s = get();
   272→        const action = ACTION_MAP[actionId];
   273→        if (!action) return;
   274→        if (action.stage !== STAGES[s.stageIndex].id) return;
   275→
   276→        const res = { ...s.resources };
   277→
   278→        // Check if all output resources are at cap — if so, block (no waste)
   279→        const outputs = Object.entries(action.produces);
   280→        if (outputs.length > 0) {
   281→          const allCapped = outputs.every(([r]) => {
   282→            const cap = s.capacities[r] || 0;
   283→            return (res[r] || 0) >= cap - 0.001;
   284→          });
   285→          if (allCapped) return; // action wasted — block it
   286→        }
   287→
   288→        // Check costs
   289→        if (action.cost) {
   290→          for (const [r, v] of Object.entries(action.cost)) {
   291→            if ((res[r] || 0) < (v as number)) return;
   292→          }
   293→          for (const [r, v] of Object.entries(action.cost)) {
   294→            res[r] -= v as number;
   295→          }
   296→        }
   297→
   298→        // Manual action multiplier
   299→        const manualLvl = s.upgrades["quickened_hands"] || 0;
   300→        const mult = 1 + manualLvl * 0.25;
   301→        const techMult = techMultiplier(s, "manual");
   302→        const finalMult = mult * techMult;
   303→
   304→        // Archetype per-resource bonus
   305→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   306→
   307→        for (const [r, v] of Object.entries(action.produces)) {
   308→          const cap = s.capacities[r] || 0;
   309→          const archResBonus = archetype?.bonus.productionMult?.[r] || 0;
   310→          const totalArchMult = 1 + archResBonus + (archetype?.bonus.allProductionMult || 0);
   311→          const amount = (v as number) * finalMult * totalArchMult;
   312→          res[r] = Math.min(cap, (res[r] || 0) + amount);
   313→        }
   314→
   315→        set({
   316→          resources: res,
   317→          totalActions: (s.totalActions || 0) + 1,
   318→        });
   319→      },
   320→
   321→      buySystem: (systemId, qty) => {
   322→        const s = get();
   323→        const sys = SYSTEM_MAP[systemId];
   324→        if (!sys) return;
   325→        if (sys.stage !== STAGES[s.stageIndex].id) return;
   326→        // Check tech prerequisite
   327→        if (sys.requiredTech && !s.technologies[sys.requiredTech]) return;
   328→
   329→        const costReduction = (s.upgrades["frugality"] || 0) * 0.04;
   330→        let owned = s.ownedSystems[systemId] || 0;
   331→        if (sys.maxOwned && owned >= sys.maxOwned) return;
   332→
   333→        const res = { ...s.resources };
   334→        const ownedSystems = { ...s.ownedSystems };
   335→        const systemEnabled = { ...s.systemEnabled };
   336→        const archetypeAffinity = { ...s.archetypeAffinity };
   337→
   338→        let bought = 0;
   339→        for (let i = 0; i < qty; i++) {
   340→          if (sys.maxOwned && owned + bought >= sys.maxOwned) break;
   341→          const cost = systemCost(sys, owned + bought, costReduction);
   342→          let canAfford = true;
   343→          for (const [r, v] of Object.entries(cost)) {
   344→            if ((res[r] || 0) < (v as number)) {
   345→              canAfford = false;
   346→              break;
   347→            }
   348→          }
   349→          if (!canAfford) break;
   350→          for (const [r, v] of Object.entries(cost)) {
   351→            res[r] -= v as number;
   352→          }
   353→          bought++;
   354→        }
   355→
   356→        if (bought === 0) return;
   357→
   358→        ownedSystems[systemId] = owned + bought;
   359→        // New systems are enabled by default
   360→        if (systemEnabled[systemId] === undefined) systemEnabled[systemId] = true;
   361→
   362→        // Apply archetype affinity grant
   363→        if (sys.grantsAffinity) {
   364→          const arch = sys.grantsAffinity.archetype;
   365→          archetypeAffinity[arch] = (archetypeAffinity[arch] || 0) + sys.grantsAffinity.amount * bought;
   366→        }
   367→
   368→        // Trigger story
   369→        const newStory = { ...s.storyUnlocked };
   370→        let activePopup = s.activeStoryPopup;
   371→        for (const trig of STORY_TRIGGERS) {
   372→          if (!newStory[trig.id] && trig.condition({ ...s, resources: res, ownedSystems })) {
   373→            newStory[trig.id] = true;
   374→            if (!activePopup) activePopup = trig.id;
   375→          }
   376→        }
   377→
   378→        set({
   379→          resources: res,
   380→          ownedSystems,
   381→          systemEnabled,
   382→          archetypeAffinity,
   383→          storyUnlocked: newStory,
   384→          activeStoryPopup: activePopup,
   385→          totalSystemsBuilt: (s.totalSystemsBuilt || 0) + bought,
   386→        });
   387→        get().addToLog(`Built ${bought}× ${sys.name}.`);
   388→      },
   389→
   390→      toggleSystem: (systemId) => {
   391→        const s = get();
   392→        if (!s.ownedSystems[systemId]) return;
   393→        const systemEnabled = { ...s.systemEnabled };
   394→        systemEnabled[systemId] = systemEnabled[systemId] === false ? true : false;
   395→        set({ systemEnabled });
   396→      },
   397→
   398→      enableAllSystems: (enabled) => {
   399→        const s = get();
   400→        const systemEnabled: Record<string, boolean> = {};
   401→        for (const id of Object.keys(s.ownedSystems)) {
   402→          systemEnabled[id] = enabled;
   403→        }
   404→        set({ systemEnabled });
   405→      },
   406→
   407→      buyTech: (techId) => {
   408→        const s = get();
   409→        const tech = TECH_MAP[techId];
   410→        if (!tech) return;
   411→        if (s.technologies[techId]) return;
   412→        // Locked techs cannot be researched (a mutually-exclusive sibling was already chosen)
   413→        if (s.lockedTechs && s.lockedTechs[techId]) return;
   414→        if (tech.stage !== STAGES[s.stageIndex].id) return;
   415→        if (tech.requires) {
   416→          for (const req of tech.requires) {
   417→            if (!s.technologies[req]) return;
   418→          }
   419→        }
   420→        const res = { ...s.resources };
   421→        for (const [r, v] of Object.entries(tech.cost)) {
   422→          if ((res[r] || 0) < (v as number)) return;
   423→        }
   424→        for (const [r, v] of Object.entries(tech.cost)) {
   425→          res[r] -= v as number;
   426→        }
   427→        const technologies = { ...s.technologies, [techId]: true };
   428→        const archetypeAffinity = { ...s.archetypeAffinity };
   429→        if (tech.grantsAffinity) {
   430→          archetypeAffinity[tech.grantsAffinity.archetype] =
   431→            (archetypeAffinity[tech.grantsAffinity.archetype] || 0) + tech.grantsAffinity.amount;
   432→        }
   433→        // Enforce mutually-exclusive tech path choices: lock all siblings
   434→        // listed in `exclusiveWith` so they can never be researched this run.
   435→        const lockedTechs = { ...(s.lockedTechs || {}) };
   436→        if (tech.exclusiveWith && tech.exclusiveWith.length > 0) {
   437→          for (const siblingId of tech.exclusiveWith) {
   438→            // Don't lock a sibling that is somehow already researched (defensive)
   439→            if (!technologies[siblingId]) {
   440→              lockedTechs[siblingId] = true;
   441→            }
   442→          }
   443→        }
   444→        set({
   445→          resources: res,
   446→          technologies,
   447→          lockedTechs,
   448→          archetypeAffinity,
   449→          totalTechResearched: (s.totalTechResearched || 0) + 1,
   450→        });
   451→        get().addToLog(`Researched ${tech.name}.`);
   452→      },
   453→
   454→      buyUpgrade: (upgradeId) => {
   455→        const s = get();
   456→        const up = UPGRADE_MAP[upgradeId];
   457→        if (!up) return;
   458→        if (up.requiresWins && s.galacticWins < up.requiresWins) return;
   459→        const owned = s.upgrades[upgradeId] || 0;
   460→        if (owned >= up.maxLevel) return;
   461→        const cost = upgradeCost(up, owned);
   462→        if (s.evolutionPoints < cost) return;
   463→        const upgrades = { ...s.upgrades, [upgradeId]: owned + 1 };
   464→        set({
   465→          evolutionPoints: s.evolutionPoints - cost,
   466→          upgrades,
   467→        });
   468→        get().addToLog(`Purchased ${up.name} (Lv ${owned + 1}).`);
   469→      },
   470→
   471→      evolveStage: () => {
   472→        const s = get();
   473→        const stage = STAGES[s.stageIndex];
   474→        const next = nextStage(stage.id);
   475→        if (!next) {
   476→          get().triggerPrestige();
   477→          return;
   478→        }
   479→        const reqs = stage.evolveRequires;
   480→        const evolveBoost = 1 - Math.min(0.30, (s.upgrades["evolutionary_momentum"] || 0) * 0.05);
   481→        const totalSystems = Object.values(s.ownedSystems).reduce((a, b) => a + b, 0);
   482→        const totalTech = Object.keys(s.technologies).length;
   483→
   484→        if (s.population < reqs.minPopulation * evolveBoost) return;
   485→        if (totalSystems < reqs.minSystems * evolveBoost) return;
   486→        if (totalTech < reqs.minTech * evolveBoost) return;
   487→
   488→        const nextStageIdx = s.stageIndex + 1;
   489→        const nextStageDef = STAGES[nextStageIdx];
   490→
   491→        const resources = { ...s.resources };
   492→        // Give starting resources for the new stage so the player can build the first systems
   493→        // This prevents the "can't build anything" progression blocker
   494→        const stageStartBonus: Record<string, number> = {
   495→          creature: { food: 15, water: 10, materials: 5 },
   496→          tribal: { wood: 15, stone: 5, clay: 3 },
   497→          civilization: { production: 15, gold: 5 },
   498→          empire: { influence: 5, gold: 20 },
   499→          solar: { energy: 20, alloys: 5 },
   500→          galactic: { data: 10, energy: 30 },
   501→        };
   502→        const startBonus = stageStartBonus[nextStageDef.id];
   503→        if (startBonus) {
   504→          for (const [r, v] of Object.entries(startBonus)) {
   505→            resources[r] = (resources[r] || 0) + v;
   506→          }
   507→        }
   508→        const capBoostLvl = s.upgrades["expansive_vaults"] || 0;
   509→        const capMult = 1 + capBoostLvl * 0.20;
   510→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   511→        const archCapMult = archetype?.bonus.capMult || 0;
   512→        const capacities = recomputeCapacitiesFresh(s, capMult + archCapMult);
   513→
   514→        const newStory = { ...s.storyUnlocked };
   515→        const storyId = STORY_MAP[
   516→          nextStageDef.id === "creature" ? "first_body_stirs"
   517→          : nextStageDef.id === "tribal" ? "fire_remembered"
   518→          : nextStageDef.id === "civilization" ? "cities_learn_to_dream"
   519→          : nextStageDef.id === "empire" ? "distance_begins_to_obey"
   520→          : nextStageDef.id === "solar" ? "sky_stops_being_a_wall"
   521→          : "galaxy_takes_a_shape"
   522→        ];
   523→        if (storyId) newStory[storyId.id] = true;
   524→
   525→        // Lock archetype when evolving from Creature → Tribal
   526→        let lockedArchetype = s.lockedArchetype;
   527→        if (stage.id === "creature" && !lockedArchetype) {
   528→          const dom = dominantArchetype(s.archetypeAffinity);

   871→function techMultiplier(s: GameState, target: "production" | "manual" | "capacity"): number {
   872→  let mult = 1;
   873→  for (const [techId, owned] of Object.entries(s.technologies)) {
   874→    if (!owned) continue;
   875→    const tech = TECH_MAP[techId];
   876→    if (!tech || !tech.multiplier) continue;
   877→    if (tech.multiplier.target === target || tech.multiplier.target === "production") {
   878→      mult += tech.multiplier.value;
   879→    }
   880→  }
   881→  // Achievement production bonus
   882→  const ctx = buildAchievementCtx(s);
   883→  mult += achievementBonus(s.achievements || {}, ctx, "production");
   884→  return mult;
   885→}

   887→function buildAchievementCtx(s: GameState): AchievementCheckCtx {
   888→  const systemsDiscoveredCount = Object.keys(s.ownedSystems).filter((id) => (s.ownedSystems[id] || 0) > 0).length;
   889→  const techResearchedCount = Object.keys(s.technologies).filter((id) => s.technologies[id]).length;
   890→  const upgradesOwnedCount = Object.values(s.upgrades).filter((c) => c > 0).length;
   891→  const storyUnlockedCount = Object.keys(s.storyUnlocked).filter((id) => s.storyUnlocked[id]).length;
   892→  return {
   893→    galacticWins: s.galacticWins,
   894→    totalRuns: s.totalRuns,
   895→    stageClearCounts: s.stageClearCounts,
   896→    archivedArchetypes: s.archivedArchetypes || [],
   897→    storyUnlockedCount,
   898→    systemsDiscoveredCount,
   899→    techResearchedCount,
   900→    upgradesOwnedCount,
   901→    maxPopulation: s.maxPopulation || 0,
   902→    fastestCellClear: s.fastestCellClear || 0,
   903→    totalEventsResolved: s.totalEventsResolved || 0,
   904→  };
   905→}

   907→// Check all achievements; return new ones earned
   908→function checkAchievements(s: GameState): { earned: Record<string, boolean>; newOnes: string[] } {
   909→  const ctx = buildAchievementCtx(s);
   910→  const earned = { ...(s.achievements || {}) };
   911→  const newOnes: string[] = [];
   912→  for (const ach of ACHIEVEMENTS) {
   913→    if (!earned[ach.id] && ach.check(ctx)) {
   914→      earned[ach.id] = true;
   915→      newOnes.push(ach.id);
   916→    }
   917→  }
   918→  return { earned, newOnes };
   919→}

   921→function computeScore(s: GameState): number {
   922→  const totalSystems = Object.values(s.ownedSystems).reduce((a, b) => a + b, 0);
   923→  const totalTech = Object.keys(s.technologies).length;
   924→  const resourceSum = Object.values(s.resources).reduce((a, b) => a + b, 0);
   925→  return Math.floor(
   926→    s.population * 2 +
   927→    totalSystems * 5 +
   928→    totalTech * 10 +
   929→    resourceSum * 0.5 +
   930→    s.time * 0.1
   931→  );
   932→}

   934→// Recompute all capacities from scratch based on owned systems
   935→function recomputeCapacitiesFresh(s: GameState, capMult: number): Record<string, number> {
   936→  const base = emptyCapacities();
   937→  const caps: Record<string, number> = {};
   938→  for (const r of Object.keys(base)) {
   939→    let cap = (base[r] || 50) * capMult;
   940→    for (const sys of SYSTEMS) {
   941→      const count = s.ownedSystems[sys.id] || 0;
   942→      if (!count || !sys.capacityBoost) continue;
   943→      const boost = sys.capacityBoost[r];
   944→      if (boost) cap += (boost as number) * count * capMult;
   945→    }
   946→    caps[r] = cap;
   947→  }
   948→  return caps;
   949→}

   951→// Export helper for components to compute whether an action is capped
   952→export function isActionCapped(
   953→  produces: Record<string, number>,
   954→  resources: Record<string, number>,
   955→  capacities: Record<string, number>
   956→): boolean {
   957→  const outputs = Object.entries(produces);
   958→  if (outputs.length === 0) return false;
   959→  return outputs.every(([r]) => {
   960→    const cap = capacities[r] || 0;
   961→    return (resources[r] || 0) >= cap - 0.001;
   962→  });
   963→}

   965→// Export helper for whether a system is currently capped (idling)
   966→export function isSystemCapped(
   967→  sysId: string,
   968→  resources: Record<string, number>,
   969→  capacities: Record<string, number>
   970→): boolean {
   971→  const sys = SYSTEM_MAP[sysId];
   972→  if (!sys) return false;
   973→  const produces = Object.keys(sys.produces);
   974→  if (produces.length === 0) return false;
   975→  return produces.every((r) => {
   976→    const cap = capacities[r] || 0;
   977→    return (resources[r] || 0) >= cap - 0.01;
   978→  });
   979→}
