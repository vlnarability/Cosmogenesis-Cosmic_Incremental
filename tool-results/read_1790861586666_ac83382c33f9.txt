    82→    82→export const useGameStore = create<GameStore>()(
    83→    83→  persist(
    84→    84→    (set, get) => ({
    85→    85→      ...initialRunState(),
    86→    86→      ...initialMetaState(),
    87→    87→
    88→    88→      // ============ TICK ============
    89→    89→      tick: (dt) => {
    90→    90→        const s = get();
    91→    91→        if (s.paused) return;
    92→    92→        const realDt = dt * s.speed;
    93→    93→
    94→    94→        const autoLvl = s.upgrades["inherited_efficiency"] || 0;
    95→    95→        const autoMult = 1 + autoLvl * 0.15;
    96→    96→        const techMult = techMultiplier(s, "production");
    97→    97→        const capBoostLvl = s.upgrades["expansive_vaults"] || 0;
    98→    98→        const capMult = 1 + capBoostLvl * 0.20;
    99→    99→        const hasAutoBalancer = (s.upgrades["auto_balancer"] || 0) > 0;
   100→   100→
   101→   101→        // Archetype bonus
   102→   102→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   103→   103→        const archAllMult = archetype?.bonus.allProductionMult || 0;
   104→   104→        const archPopMult = archetype?.bonus.populationGrowthMult || 0;
   105→   105→        const archCapMult = archetype?.bonus.capMult || 0;
   106→   106→
   107→   107→        const resources = { ...s.resources };
   108→   108→        let capacities = recomputeCapacitiesFresh(s, capMult + archCapMult);
   109→   109→
   110→   110→        // Apply system production — with cap-aware pausing
   111→   111→        for (const [sysId, count] of Object.entries(s.ownedSystems)) {
   112→   112→          if (!count) continue;
   113→   113→          const sys = SYSTEM_MAP[sysId];
   114→   114→          if (!sys) continue;
   115→   115→          if (sys.requiredTech && !s.technologies[sys.requiredTech]) continue;
   116→   116→
   117→   117→          // Check if system is manually disabled
   118→   118→          const manuallyEnabled = s.systemEnabled[sysId] !== false; // default true
   119→   119→          if (!manuallyEnabled && !hasAutoBalancer) continue;
   120→   120→
   121→   121→          // Cap-aware logic: if all output resources are at cap, idle the system
   122→   122→          // (don't produce, don't consume upkeep — no waste)
   123→   123→          const outputsCapped = Object.keys(sys.produces).length > 0 &&
   124→   124→            Object.keys(sys.produces).every((r) => {
   125→   125→              const cap = capacities[r] || 0;
   126→   126→              return (resources[r] || 0) >= cap - 0.01;
   127→   127→            });
   128→   128→
   129→   129→          if (outputsCapped) {
   130→   130→            // System idles — no production, no upkeep
   131→   131→            continue;
   132→   132→          }
   133→   133→
   134→   134→          const stageMult = sys.stage === STAGES[s.stageIndex].id ? 1 : 0.5;
   135→   135→          const mult = autoMult * techMult * stageMult * (1 + archAllMult);
   136→   136→
   137→   137→          // Per-resource archetype bonus
   138→   138→          const produces = sys.produces as Record<string, number>;
   139→   139→          for (const [r, v] of Object.entries(produces)) {
   140→   140→            const archResBonus = archetype?.bonus.productionMult?.[r] || 0;
   141→   141→            const rate = v * count * mult * (1 + archResBonus);
   142→   142→            resources[r] = (resources[r] || 0) + rate * realDt;
   143→   143→          }
   144→   144→          // Upkeep
   145→   145→          if (sys.upkeep) {
   146→   146→            const upkeep = sys.upkeep as Record<string, number>;
   147→   147→            for (const [r, v] of Object.entries(upkeep)) {
   148→   148→              resources[r] = (resources[r] || 0) - v * count * realDt;
   149→   149→            }
   150→   150→          }
   151→   151→        }
   152→   152→
   153→   153→        // Clamp resources to capacities (no overflow)
   154→   154→        for (const r of Object.keys(resources)) {
   155→   155→          const cap = capacities[r];
   156→   156→          if (cap != null && resources[r] > cap) resources[r] = cap;
   157→   157→          if (resources[r] < 0) resources[r] = 0;
   158→   158→        }
   159→   159→
   160→   160→        // Slow passive ATP for cell (keeps early game from stalling)
   161→   161→        if (STAGES[s.stageIndex].id === "cell") {
   162→   162→          resources.atp = Math.min(capacities.atp || 30, resources.atp + 0.1 * realDt);
   163→   163→        }
   164→   164→
   165→   165→        // Happiness drift toward 50 baseline if no source
   166→   166→        // (prevents permanent high happiness from start)
   167→   167→        // — disabled: let systems/tech maintain it
   168→   168→
   169→   169→        // Population growth with progress bar
   170→   170→        // Growth rate scales with happiness and food/water availability
   171→   171→        let popGrowthRate = 0.015; // base per second
   172→   172→        const happiness = resources.happiness || 50;
   173→   173→        popGrowthRate *= (1 + (happiness - 40) / 200); // happiness 100 → +30%, happiness 20 → -10%
   174→   174→        popGrowthRate *= (1 + archPopMult);
   175→   175→        // Penalty if food/water are critically low (creature+)
   176→   176→        const curStage = STAGES[s.stageIndex];
   177→   177→        if (["creature", "tribal", "civilization", "empire"].includes(curStage.id)) {
   178→   178→          const foodCap = capacities.food || 1;
   179→   179→          const waterCap = capacities.water || 1;
   180→   180→          const foodPct = (resources.food || 0) / foodCap;
   181→   181→          const waterPct = (resources.water || 0) / waterCap;
   182→   182→          if (foodPct < 0.1) popGrowthRate *= 0.3;
   183→   183→          if (waterPct < 0.1) popGrowthRate *= 0.3;
   184→   184→        }
   185→   185→
   186→   186→        let populationProgress = (s.populationProgress || 0) + popGrowthRate * realDt;
   187→   187→        let population = s.population;
   188→   188→        while (populationProgress >= 1) {
   189→   189→          populationProgress -= 1;
   190→   190→          population += 1;
   191→   191→        }
   192→   192→
   193→   193→        const maxPopulation = Math.max(s.maxPopulation || 0, population);
   194→   194→
   195→   195→        const newTime = s.time + realDt;
   196→   196→
   197→   197→        // Check story triggers
   198→   198→        const newStory = { ...s.storyUnlocked };
   199→   199→        let activePopup = s.activeStoryPopup;
   200→   200→        const ctx = { ...s, resources, ownedSystems: s.ownedSystems };
   201→   201→        for (const trig of STORY_TRIGGERS) {
   202→   202→          if (!newStory[trig.id] && trig.condition(ctx)) {
   203→   203→            newStory[trig.id] = true;
   204→   204→            if (!activePopup) activePopup = trig.id;
   205→   205→          }
   206→   206→        }
   207→   207→
   208→   208→        // Check achievements periodically (every ~2 seconds of game time)
   209→   209→        let achievements = s.achievements;
   210→   210→        let newAchievements = s.newAchievements;
   211→   211→        const checkInterval = 2.0;
   212→   212→        const lastCheck = (s as any)._lastAchCheck || 0;
   213→   213→        if (newTime - lastCheck >= checkInterval) {
   214→   214→          const curState = { ...s, resources, time: newTime } as GameState;
   215→   215→          const result = checkAchievements(curState);
   216→   216→          if (result.newOnes.length > 0) {
   217→   217→            achievements = result.earned;
   218→   218→            newAchievements = [...(newAchievements || []), ...result.newOnes];
   219→   219→          } else if (Object.keys(result.earned).length !== Object.keys(achievements || {}).length) {
   220→   220→            achievements = result.earned;
   221→   221→          }
   222→   222→          (s as any)._lastAchCheck = newTime; // not persisted, just a runtime marker
   223→   223→        }
   224→   224→
   225→   225→        // Random event firing — only if no popup/modal is active
   226→   226→        let activeEvent = s.activeEvent;
   227→   227→        let eventCooldown = (s.eventCooldown || 0) - realDt;
   228→   228→        if (eventCooldown < 0) eventCooldown = 0;
   229→   229→        const eventFreq = s.eventFrequency || "normal";
   230→   230→        if (eventFreq !== "off" && !activeEvent && !activePopup && eventCooldown === 0) {
   231→   231→          const stageId = STAGES[s.stageIndex].id;
   232→   232→          const possible = eventsForStage(stageId, newTime);
   233→   233→          if (possible.length > 0) {
   234→   234→            // Weighted random selection based on rarity
   235→   235→            const totalWeight = possible.reduce((sum, e) => sum + eventWeight(e), 0);
   236→   236→            const roll = Math.random() * totalWeight;
   237→   237→            let cum = 0;
   238→   238→            for (const ev of possible) {
   239→   239→              cum += eventWeight(ev);
   240→   240→              if (roll < cum) {
   241→   241→                activeEvent = ev.id;
   242→   242→                // Cooldown depends on frequency setting
   243→   243→                const base = eventFreq === "frequent" ? 30 : 90;
   244→   244→                const range = eventFreq === "frequent" ? 30 : 90;
   245→   245→                eventCooldown = base + Math.random() * range;
   246→   246→                break;
   247→   247→              }
   248→   248→            }
   249→   249→          }
   250→   250→        }
   251→   251→
   252→   252→        set({
   253→   253→          resources,
   254→   254→          capacities,
   255→   255→          time: newTime,
   256→   256→          population,
   257→   257→          maxPopulation,
   258→   258→          populationProgress,
   259→   259→          storyUnlocked: newStory,
   260→   260→          activeStoryPopup: activePopup,
   261→   261→          achievements,
   262→   262→          newAchievements,
   263→   263→          activeEvent,
   264→   264→          eventCooldown,
   265→   265→          totalPlayTime: (s.totalPlayTime || 0) + realDt,
   266→   266→        });
   267→   267→      },
   268→   268→
   269→   269→      // ============ ACTIONS ============
   270→   270→      performAction: (actionId) => {
   271→   271→        const s = get();
   272→   272→        const action = ACTION_MAP[actionId];
   273→   273→        if (!action) return;
   274→   274→        if (action.stage !== STAGES[s.stageIndex].id) return;
   275→   275→
   276→   276→        const res = { ...s.resources };
   277→   277→
   278→   278→        // Check if all output resources are at cap — if so, block (no waste)
   279→   279→        const outputs = Object.entries(action.produces);
   280→   280→        if (outputs.length > 0) {
   281→   281→          const allCapped = outputs.every(([r]) => {
   282→   282→            const cap = s.capacities[r] || 0;
   283→   283→            return (res[r] || 0) >= cap - 0.001;
   284→   284→          });
   285→   285→          if (allCapped) return; // action wasted — block it
   286→   286→        }
   287→   287→
   288→   288→        // Check costs
   289→   289→        if (action.cost) {
   290→   290→          for (const [r, v] of Object.entries(action.cost)) {
   291→   291→            if ((res[r] || 0) < (v as number)) return;
   292→   292→          }
   293→   293→          for (const [r, v] of Object.entries(action.cost)) {
   294→   294→            res[r] -= v as number;
   295→   295→          }
   296→   296→        }
   297→   297→
   298→   298→        // Manual action multiplier
   299→   299→        const manualLvl = s.upgrades["quickened_hands"] || 0;
   300→   300→        const mult = 1 + manualLvl * 0.25;
   301→   301→        const techMult = techMultiplier(s, "manual");
   302→   302→        const finalMult = mult * techMult;
   303→   303→
   304→   304→        // Archetype per-resource bonus
   305→   305→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   306→   306→
   307→   307→        for (const [r, v] of Object.entries(action.produces)) {
   308→   308→          const cap = s.capacities[r] || 0;
   309→   309→          const archResBonus = archetype?.bonus.productionMult?.[r] || 0;
   310→   310→          const totalArchMult = 1 + archResBonus + (archetype?.bonus.allProductionMult || 0);
   311→   311→          const amount = (v as number) * finalMult * totalArchMult;
   312→   312→          res[r] = Math.min(cap, (res[r] || 0) + amount);
   313→   313→        }
   314→   314→
   315→   315→        set({
   316→   316→          resources: res,
   317→   317→          totalActions: (s.totalActions || 0) + 1,
   318→   318→        });
   319→   319→      },
   320→   320→
   321→   321→      buySystem: (systemId, qty) => {
   322→   322→        const s = get();
   323→   323→        const sys = SYSTEM_MAP[systemId];
   324→   324→        if (!sys) return;
   325→   325→        if (sys.stage !== STAGES[s.stageIndex].id) return;
   326→   326→        // Check tech prerequisite
   327→   327→        if (sys.requiredTech && !s.technologies[sys.requiredTech]) return;
   328→   328→
   329→   329→        const costReduction = (s.upgrades["frugality"] || 0) * 0.04;
   330→   330→        let owned = s.ownedSystems[systemId] || 0;
   331→   331→        if (sys.maxOwned && owned >= sys.maxOwned) return;
   332→   332→
   333→   333→        const res = { ...s.resources };
   334→   334→        const ownedSystems = { ...s.ownedSystems };
   335→   335→        const systemEnabled = { ...s.systemEnabled };
   336→   336→        const archetypeAffinity = { ...s.archetypeAffinity };
   337→   337→
   338→   338→        let bought = 0;
   339→   339→        for (let i = 0; i < qty; i++) {
   340→   340→          if (sys.maxOwned && owned + bought >= sys.maxOwned) break;
   341→   341→          const cost = systemCost(sys, owned + bought, costReduction);
   342→   342→          let canAfford = true;
   343→   343→          for (const [r, v] of Object.entries(cost)) {
   344→   344→            if ((res[r] || 0) < (v as number)) {
   345→   345→              canAfford = false;
   346→   346→              break;
   347→   347→            }
   348→   348→          }
   349→   349→          if (!canAfford) break;
   350→   350→          for (const [r, v] of Object.entries(cost)) {
   351→   351→            res[r] -= v as number;
   352→   352→          }
   353→   353→          bought++;
   354→   354→        }
   355→   355→
   356→   356→        if (bought === 0) return;
   357→   357→
   358→   358→        ownedSystems[systemId] = owned + bought;
   359→   359→        // New systems are enabled by default
   360→   360→        if (systemEnabled[systemId] === undefined) systemEnabled[systemId] = true;
   361→   361→
   362→   362→        // Apply archetype affinity grant
   363→   363→        if (sys.grantsAffinity) {
   364→   364→          const arch = sys.grantsAffinity.archetype;
   365→   365→          archetypeAffinity[arch] = (archetypeAffinity[arch] || 0) + sys.grantsAffinity.amount * bought;
   366→   366→        }
   367→   367→
   368→   368→        // Trigger story
   369→   369→        const newStory = { ...s.storyUnlocked };
   370→   370→        let activePopup = s.activeStoryPopup;
   371→   371→        for (const trig of STORY_TRIGGERS) {
   372→   372→          if (!newStory[trig.id] && trig.condition({ ...s, resources: res, ownedSystems })) {
   373→   373→            newStory[trig.id] = true;
   374→   374→            if (!activePopup) activePopup = trig.id;
   375→   375→          }
   376→   376→        }
   377→   377→
   378→   378→        set({
   379→   379→          resources: res,
   380→   380→          ownedSystems,
   381→   381→          systemEnabled,
   382→   382→          archetypeAffinity,
   383→   383→          storyUnlocked: newStory,
   384→   384→          activeStoryPopup: activePopup,
   385→   385→          totalSystemsBuilt: (s.totalSystemsBuilt || 0) + bought,
   386→   386→        });
   387→   387→        get().addToLog(`Built ${bought}× ${sys.name}.`);
   388→   388→      },
   389→   389→
   390→   390→      toggleSystem: (systemId) => {
   391→   391→        const s = get();
   392→   392→        if (!s.ownedSystems[systemId]) return;
   393→   393→        const systemEnabled = { ...s.systemEnabled };
   394→   394→        systemEnabled[systemId] = systemEnabled[systemId] === false ? true : false;
   395→   395→        set({ systemEnabled });
   396→   396→      },
   397→   397→
   398→   398→      enableAllSystems: (enabled) => {
   399→   399→        const s = get();
   400→   400→        const systemEnabled: Record<string, boolean> = {};
   401→   401→        for (const id of Object.keys(s.ownedSystems)) {
   402→   402→          systemEnabled[id] = enabled;
   403→   403→        }
   404→   404→        set({ systemEnabled });
   405→   405→      },
   406→   406→
   407→   407→      buyTech: (techId) => {
   408→   408→        const s = get();
   409→   409→        const tech = TECH_MAP[techId];
   410→   410→        if (!tech) return;
   411→   411→        if (s.technologies[techId]) return;
   412→   412→        // Locked techs cannot be researched (a mutually-exclusive sibling was already chosen)
   413→   413→        if (s.lockedTechs && s.lockedTechs[techId]) return;
   414→   414→        if (tech.stage !== STAGES[s.stageIndex].id) return;
   415→   415→        if (tech.requires) {
   416→   416→          for (const req of tech.requires) {
   417→   417→            if (!s.technologies[req]) return;
   418→   418→          }
   419→   419→        }
   420→   420→        const res = { ...s.resources };
   421→   421→        for (const [r, v] of Object.entries(tech.cost)) {
   422→   422→          if ((res[r] || 0) < (v as number)) return;
   423→   423→        }
   424→   424→        for (const [r, v] of Object.entries(tech.cost)) {
   425→   425→          res[r] -= v as number;
   426→   426→        }
   427→   427→        const technologies = { ...s.technologies, [techId]: true };
   428→   428→        const archetypeAffinity = { ...s.archetypeAffinity };
   429→   429→        if (tech.grantsAffinity) {
   430→   430→          archetypeAffinity[tech.grantsAffinity.archetype] =
   431→   431→            (archetypeAffinity[tech.grantsAffinity.archetype] || 0) + tech.grantsAffinity.amount;
   432→   432→        }
   433→   433→        // Enforce mutually-exclusive tech path choices: lock all siblings
   434→   434→        // listed in `exclusiveWith` so they can never be researched this run.
   435→   435→        const lockedTechs = { ...(s.lockedTechs || {}) };
   436→   436→        if (tech.exclusiveWith && tech.exclusiveWith.length > 0) {
   437→   437→          for (const siblingId of tech.exclusiveWith) {
   438→   438→            // Don't lock a sibling that is somehow already researched (defensive)
   439→   439→            if (!technologies[siblingId]) {
   440→   440→              lockedTechs[siblingId] = true;
   441→   441→            }
   442→   442→          }
   443→   443→        }
   444→   444→        set({
   445→   445→          resources: res,
   446→   446→          technologies,
   447→   447→          lockedTechs,
   448→   448→          archetypeAffinity,
   449→   449→          totalTechResearched: (s.totalTechResearched || 0) + 1,
   450→   450→        });

   871→   871→function techMultiplier(s: GameState, target: "production" | "manual" | "capacity"): number {
   872→   872→  let mult = 1;
   873→   873→  for (const [techId, owned] of Object.entries(s.technologies)) {
   874→   874→    if (!owned) continue;
   875→   875→    const tech = TECH_MAP[techId];
   876→   876→    if (!tech || !tech.multiplier) continue;
   877→   877→    if (tech.multiplier.target === target || tech.multiplier.target === "production") {
   878→   878→      mult += tech.multiplier.value;
   879→   879→    }
   880→   880→  }
   881→   881→  // Achievement production bonus
   882→   882→  const ctx = buildAchievementCtx(s);
   883→   883→  mult += achievementBonus(s.achievements || {}, ctx, "production");
   884→   884→  return mult;
   885→   885→}

   909→   909→  const ctx = buildAchievementCtx(s);

   921→   921→function computeScore(s: GameState): number {
   922→   922→  const totalSystems = Object.values(s.ownedSystems).reduce((a, b) => a + b, 0);
   923→   923→  const totalTech = Object.keys(s.technologies).length;
   924→   924→  const resourceSum = Object.values(s.resources).reduce((a, b) => a + b, 0);
   925→   925→  return Math.floor(
   926→   926→    s.population * 2 +
   927→   927→    totalSystems * 5 +
   928→   928→    totalTech * 10 +
   929→   929→    resourceSum * 0.5 +
   930→   930→    s.time * 0.1
   931→   931→  );
   932→   932→}

   934→   934→// Recompute all capacities from scratch based on owned systems
   935→   935→function recomputeCapacitiesFresh(s: GameState, capMult: number): Record<string, number> {
   936→   936→  const base = emptyCapacities();
   937→   937→  const caps: Record<string, number> = {};
   938→   938→  for (const r of Object.keys(base)) {
   939→   939→    let cap = (base[r] || 50) * capMult;
   940→   940→    for (const sys of SYSTEMS) {
   941→   941→      const count = s.ownedSystems[sys.id] || 0;
   942→   942→      if (!count || !sys.capacityBoost) continue;
   943→   943→      const boost = sys.capacityBoost[r];
   944→   944→      if (boost) cap += (boost as number) * count * capMult;
   945→   945→    }
   946→   946→    caps[r] = cap;
   947→   947→  }
   948→   948→  return caps;
   949→   949→}

   951→   951→// Export helper for components to compute whether an action is capped
   952→   952→export function isActionCapped(
   953→   953→  produces: Record<string, number>,
   954→   954→  resources: Record<string, number>,
   955→   955→  capacities: Record<string, number>
   956→   956→): boolean {
   957→   957→  const outputs = Object.entries(produces);
   958→   958→  if (outputs.length === 0) return false;
   959→   959→  return outputs.every(([r]) => {
   960→   960→    const cap = capacities[r] || 0;
   961→   961→    return (resources[r] || 0) >= cap - 0.001;
   962→   962→  });
   963→   963→}
