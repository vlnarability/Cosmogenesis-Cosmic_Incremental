    44→// FIX-PRESTIGE-DIVINE-CONSTELLATIONS — Constellation UI redesign.
    45→// ===========================================================================
    46→//
    47→// The L2 constellation map is now a wide, horizontally-scrollable sky:
    48→//   - 3000px wide × 500px tall panorama of all 88 IAU constellations
    49→//   - Stars are positioned by mapping the existing normalized (0-100) coords
    50→//     to (0-SKY_W, 0-SKY_H) pixels
    51→//   - Click and drag, or use ◀ ▶ arrows, to pan the sky
    52→//   - The sky wraps infinitely (render 3 copies: left, center, right) — when
    53→//     you scroll past the end, it loops back to the beginning seamlessly
    54→//   - Click a star → opens a popup showing star name, constellation name,
    55→//     reward for completing the constellation, cost in Divinity, and an
    56→//     "Illuminate" button. Replaces the old direct-click illumination.
    57→//
    58→// The store action `illuminateStar(constellationId, starId)` is preserved.
    59→// ===========================================================================
    60→
    61→// Sky dimensions — the wide panorama.
    62→const SKY_W = 3000;
    63→const SKY_H = 500;
    64→
    65→// Per-instance offsets for the infinite-wrap 3-copy trick.
    66→// We render the sky at three x-offsets: -SKY_W (left), 0 (center), +SKY_W
    67→// (right). The wrapper translates them all by `scrollX`. As the user scrolls,
    68→// when |scrollX| exceeds SKY_W, we wrap by adding/subtracting SKY_W so the
    69→// visible area always shows at least one fully-populated instance. The wrap
    70→// is invisible because all 3 instances render identical content.
    71→const SKY_OFFSETS: number[] = [-SKY_W, 0, SKY_W];
    72→
    73→// FIX-UI-TABS / react-hooks/set-state-in-effect — module-level helper
    74→// for the infinite-wrap. Moved out of useEffect (which linted as a
    75→// cascading-render trigger) and applied at each setScrollX() call site.
    76→function wrapScrollX(x: number): number {
    77→  if (x < -SKY_W) return x + SKY_W;
    78→  if (x > SKY_W) return x - SKY_W;
    79→  return x;
    80→}

   140→  }
   141→  return out;

   144→// Per-constellation label anchor (top-right of the constellation's bounding
   145→// box). Precomputed so we don't run Math.min/max on every render.
   146→interface LabelAnchor { cId: string; left: number; top: number; }
   147→const LABEL_ANCHORS: LabelAnchor[] = CONSTELLATIONS.map((c) => {
   148→  const xs = c.stars.map((s) => s.x);
   149→  const ys = c.stars.map((s) => s.y);
   150→  const maxX = Math.max(...xs);
   151→  const minY = Math.min(...ys);
   152→  return { cId: c.id, left: maxX, top: minY };
   153→});

   155→// Greek-letter → readable glyph, for star designations in the popup.
   156→const GREEK: Record<string, string> = {
   157→  alpha: "α", beta: "β", gamma: "γ", delta: "δ", epsilon: "ε", eps: "ε",
   158→  zeta: "ζ", eta: "η", theta: "θ", iota: "ι", kappa: "κ", lambda: "λ",
   159→  mu: "μ", nu: "ν", xi: "ξ", omicron: "ο", pi: "π", rho: "ρ", sigma: "σ",
   160→  tau: "τ", upsilon: "υ", phi: "φ", chi: "χ", psi: "ψ", omega: "ω",
   161→};
   162→
   163→// Convert a star id like "aries_alpha" or "taurus_eps" or "orion_pi3" into a
   164→// readable designation like "α Arietis". Falls back to the raw id if parsing
   165→// fails (no constellation ever uses an unknown suffix pattern, but be safe).
   166→function starDesignation(starId: string, constellationName: string): string {
   167→  const parts = starId.split("_");
   168→  if (parts.length < 2) return starId;
   169→  const suffix = parts.slice(1).join("_");
   170→  const m = suffix.match(/^([a-z]+)(\d*)$/);
   171→  if (m) {
   172→    const greek = GREEK[m[1]];
   173→    if (greek) return `${greek}${m[2]} ${constellationName}`;
   174→  }
   175→  return `${suffix} ${constellationName}`;
   176→}

   178→// Format the constellation's completion bonus as a list of pill badges.
   179→function bonusSummary(bonus: ConstellationDef["bonus"]): { label: string; value: string; cls: string }[] {
   180→  const out: { label: string; value: string; cls: string }[] = [];
   181→  if (bonus.productionMult) out.push({ label: "Production", value: `+${(bonus.productionMult * 100).toFixed(1)}%`, cls: "text-cyan-300 border-cyan-400/40" });
   182→  if (bonus.capMult) out.push({ label: "Capacity", value: `+${(bonus.capMult * 100).toFixed(1)}%`, cls: "text-amber-300 border-amber-400/40" });
   183→  if (bonus.epMult) out.push({ label: "EP", value: `+${(bonus.epMult * 100).toFixed(1)}%`, cls: "text-emerald-300 border-emerald-400/40" });
   184→  if (bonus.popGrowthMult) out.push({ label: "Pop growth", value: `+${(bonus.popGrowthMult * 100).toFixed(1)}%`, cls: "text-violet-300 border-violet-400/40" });
   185→  if (bonus.costMult) out.push({ label: "Cost", value: `${(bonus.costMult * 100).toFixed(1)}%`, cls: "text-pink-300 border-pink-400/40" });
   186→  if (bonus.divinityMult) out.push({ label: "Divinity", value: `+${(bonus.divinityMult * 100).toFixed(0)}%`, cls: "text-yellow-300 border-yellow-400/40" });
   187→  return out;
   188→}
