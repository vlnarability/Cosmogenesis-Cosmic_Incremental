     1→"use client";
     2→
     3→import { useState, useMemo, useCallback } from "react";
     4→import { useGameStore } from "@/game/state/store";
     5→import {
     6→  CONSTELLATIONS,
     7→  CONSTELLATION_MAP,
     8→  constellationBonus,
     9→  starCost,
    10→  countCompletedConstellations,
    11→  countLitStars,
    12→  canAdvanceToTranscendence,
    13→  CONSTELLATIONS_REQUIRED_FOR_LAYER_3,
    14→  CONSTELLATION_BASE_COST,
    15→  WITHIN_CONSTELLATION_GROWTH,
    16→  COST_GROWTH_PER_STARTED,
    17→  NEW_CONSTELLATION_PENALTY,
    18→  CATEGORY_LABEL,
    19→  CATEGORY_ACCENT,
    20→  type ConstellationState,
    21→  type ConstellationCategory,
    22→} from "@/game/data/constellations";
    23→import {
    24→  FORESIGHT_ROUTES,
    25→  routeBonus,
    26→} from "@/game/data/foresight";
    27→import { DialogHeader, DialogTitle, DialogDescription } from "@/components/ui/dialog";
    28→import { Button } from "@/components/ui/button";
    29→import { Badge } from "@/components/ui/badge";
    30→import { Separator } from "@/components/ui/separator";
    31→import { ScrollArea } from "@/components/ui/scroll-area";
    32→import { Lock, Eye, Sparkles, Crosshair, Zap, Orbit } from "lucide-react";
    33→import { formatNumber } from "../shared/format";
    34→import { BackgroundLayerBadge } from "../shared/BackgroundLayerBadge";
    35→
    36→// SVG viewBox dimensions for the star map.
    37→const MAP_W = 1000;
    38→const MAP_H = 600;
    39→
    40→/**
    41→ * Layer 2 — Enlightenment (Constellation Map).
    42→ *
    43→ * Renders a star-map SVG (1000×600 viewBox) of all curated constellations.
    44→ * Each star is a dim dot by default; lit stars glow. Constellation connection
    45→ * lines appear dim until started, then bright. Click a star to illuminate it
    46→ * (calls `illuminateStar` — the only purchase action on this layer). Sidebar
    47→ * shows progress stats, a focused-constellation selector that pans/zooms the
    48→ * SVG, a cost-formula explainer, and the constellation list.
    49→ *
    50→ * L2 active abilities (Stargaze / Supernova / Black Hole / Cosmic Bridge)
    51→ * were removed — stars are lit only by purchasing with Divinity.
    52→ */
    53→export function EnlightenmentContent() {
    54→  const closeLayerModal = useGameStore((s) => s.closeLayerModal);
    55→  const unlockedLayers = useGameStore((s) => s.unlockedLayers);
    56→  const divinity = useGameStore((s) => s.divinity);
    57→  // L2 state
    58→  const constellationStars = useGameStore((s) => s.constellationStars || {});
    59→  const constellationProgress = useGameStore((s) => s.constellationProgress || {});
    60→  const constellationsStarted = useGameStore((s) => s.constellationsStarted || {});
    61→  const constellationsCompleted = useGameStore((s) => s.constellationsCompleted || {});
    62→  const constellationRevealed = useGameStore((s) => s.constellationRevealed || {});
    63→  const activeRoute = useGameStore((s) => s.activeForesightRoute);
    64→  // Layer shop upgrades
    65→  const upgrades = useGameStore((s) => s.upgrades || {});
    66→
    67→  // Actions
    68→  const illuminateStar = useGameStore((s) => s.illuminateStar);
    69→  const setForesightRoute = useGameStore((s) => s.setForesightRoute);
    70→
    71→  // Local UI state
    72→  const [focusedConstellationId, setFocusedConstellationId] = useState<string | null>(null);
    73→  const [hoveredStar, setHoveredStar] = useState<string | null>(null);
    74→
    75→  const isUnlocked = !!unlockedLayers.enlightenment;
    76→
    77→  const constellationState: ConstellationState = useMemo(
    78→    () => ({
    79→      constellationStars,
    80→      constellationProgress,
    81→      constellationsStarted,
    82→      constellationsCompleted,
    83→    }),
    84→    [constellationStars, constellationProgress, constellationsStarted, constellationsCompleted]
    85→  );
    86→
    87→  const rb = routeBonus(activeRoute);
    88→  const cb = constellationBonus(constellationState);
    89→  const completedCount = countCompletedConstellations(constellationState);
    90→  const litStarsCount = countLitStars(constellationState);
    91→  const startedCount = Object.values(constellationsStarted).filter(Boolean).length;
    92→  const totalStars = useMemo(
    93→    () => CONSTELLATIONS.reduce((acc, c) => acc + c.stars.length, 0),
    94→    []
    95→  );
    96→
    97→  // Foresight discount level (shop upgrade)
    98→  const fdLvl = (upgrades?.foresight_discount as number) || 0;
    99→  const foresightDiscountMult = Math.pow(0.80, fdLvl);
   100→
   101→  // Compute the cost for the focused constellation's next star (or first star).
   102→  const focusedNextCost = useMemo(() => {
   103→    const cid = focusedConstellationId;
   104→    if (!cid) return null;
   105→    const cdef = CONSTELLATION_MAP[cid];
   106→    if (!cdef) return null;
   107→    // First unlit star in the focused constellation
   108→    const nextStar = cdef.stars.find((st) => !constellationStars[st.id]);
   109→    if (!nextStar) return { starId: null, cost: 0, isNew: false, constellationId: cid, name: cdef.name };
   110→    const cost = starCost(cid, constellationState, nextStar.id, { foresightDiscountMult });
   111→    const isNew = (constellationProgress[cid] || 0) === 0;
   112→    return { starId: nextStar.id, cost, isNew, constellationId: cid, name: cdef.name };
   113→  }, [focusedConstellationId, constellationStars, constellationState, foresightDiscountMult, constellationProgress]);
   114→
   115→  // Compute a "first star of any unstarted constellation" cost preview (for the
   116→  // "new constellation penalty" tooltip).
   117→  const newConstellationPreview = useMemo(() => {
   118→    const cid = CONSTELLATIONS.find((c) => !constellationsStarted[c.id])?.id;
   119→    if (!cid) return null;
   120→    const cdef = CONSTELLATION_MAP[cid];
   121→    if (!cdef) return null;
   122→    const firstStar = cdef.stars[0];
   123→    const cost = starCost(cid, constellationState, firstStar.id, { foresightDiscountMult });
   124→    return { constellationId: cid, name: cdef.name, cost };
   125→  }, [constellationsStarted, constellationState, foresightDiscountMult]);
   126→
   127→  // SVG pan/zoom target based on focused constellation.
   128→  const viewBox = useMemo(() => {
   129→    if (!focusedConstellationId) return `0 0 ${MAP_W} ${MAP_H}`;
   130→    const cdef = CONSTELLATION_MAP[focusedConstellationId];
   131→    if (!cdef) return `0 0 ${MAP_W} ${MAP_H}`;
   132→    const xs = cdef.stars.map((s) => s.x);
   133→    const ys = cdef.stars.map((s) => s.y);
   134→    const minX = Math.min(...xs);
   135→    const maxX = Math.max(...xs);
   136→    const minY = Math.min(...ys);
   137→    const maxY = Math.max(...ys);
   138→    const pad = 6;
   139→    const w = (maxX - minX + pad * 2) * (MAP_W / 100);
   140→    const h = (maxY - minY + pad * 2) * (MAP_H / 100);
   141→    const cx = (minX + maxX) / 2 * (MAP_W / 100);
   142→    const cy = (minY + maxY) / 2 * (MAP_H / 100);
   143→    // Clamp to map bounds.
   144→    const halfW = w / 2;
   145→    const halfH = h / 2;
   146→    const x0 = Math.max(0, Math.min(MAP_W - w, cx - halfW));
   147→    const y0 = Math.max(0, Math.min(MAP_H - h, cy - halfH));
   148→    return `${x0} ${y0} ${w} ${h}`;
   149→  }, [focusedConstellationId]);
   150→
   151→  // Click a star: illuminate it (the only L2 purchase action — Divinity cost
   152→  // is validated inside the store action).
   153→  const onStarClick = useCallback(
   154→    (constellationId: string, starId: string) => {
   155→      if (!isUnlocked) return;
   156→      illuminateStar(constellationId, starId);
   157→    },
   158→    [illuminateStar, isUnlocked]
   159→  );
   160→
   161→  return (
   162→    <>
   163→      <DialogHeader>
   164→        <DialogTitle className="flex items-center gap-2 text-lg flex-wrap">
   165→          <Eye className="w-5 h-5" />
   166→          Layer 2 — Enlightenment (Constellation Map)
   167→          <BackgroundLayerBadge layerId="enlightenment" />
   168→        </DialogTitle>
   169→        <DialogDescription className="flex items-center justify-between flex-wrap gap-2">
   170→          <span>
   171→            Illuminate stars within {CONSTELLATIONS.length} IAU constellations. Completing a constellation grants a permanent bonus.
   172→          </span>
   173→          <Badge variant="outline" className="text-violet-300 border-violet-400/40">
   174→            {formatNumber(divinity, 0)} Divinity
   175→          </Badge>
   176→        </DialogDescription>
   177→      </DialogHeader>
   178→
   179→        {!isUnlocked ? (
   180→          <div className="text-center text-muted-foreground py-10">
   181→            <Lock className="w-8 h-8 mx-auto mb-2 opacity-50" />
   182→            <p className="text-sm">Enlightenment is sealed.</p>
   183→            <p className="text-xs mt-1">Master 3 distinct Layer 1 trials to unlock the constellation map.</p>
   184→          </div>
   185→        ) : (
   186→          <div className="flex-1 overflow-hidden grid grid-cols-1 lg:grid-cols-[1fr_320px] gap-3 min-h-0">
   187→            {/* ===== STAR MAP ===== */}
   188→            <div className="relative rounded-lg overflow-hidden border border-violet-500/20 bg-gradient-to-b from-[#020817] via-[#0a0a23] to-[#050518]">
   189→              {/* Subtle starfield background */}
   190→              <div
   191→                className="absolute inset-0 opacity-60 pointer-events-none"
   192→                style={{
   193→                  backgroundImage: `
   194→                    radial-gradient(1px 1px at 12% 18%, #fff, transparent),
   195→                    radial-gradient(1px 1px at 28% 42%, #fff, transparent),
   196→                    radial-gradient(1.5px 1.5px at 47% 12%, #fff, transparent),
   197→                    radial-gradient(1px 1px at 63% 56%, #fff, transparent),
   198→                    radial-gradient(1px 1px at 78% 28%, #fff, transparent),
   199→                    radial-gradient(1.5px 1.5px at 88% 70%, #fff, transparent),
   200→                    radial-gradient(1px 1px at 8% 72%, #fff, transparent),
   201→                    radial-gradient(1px 1px at 33% 88%, #fff, transparent),
   202→                    radial-gradient(1px 1px at 70% 85%, #fff, transparent),
   203→                    radial-gradient(1px 1px at 95% 8%, #fff, transparent),
   204→                    radial-gradient(1px 1px at 18% 50%, #fff, transparent),
   205→                    radial-gradient(1px 1px at 53% 75%, #fff, transparent)
   206→                  `,
   207→                  backgroundRepeat: "no-repeat",
   208→                  backgroundSize: "100% 100%",
   209→                }}
   210→              />
   211→              <svg
   212→                viewBox={viewBox}
   213→                className="relative w-full h-full"
   214→                style={{ aspectRatio: `${MAP_W} / ${MAP_H}`, transition: "viewBox 0.6s ease" as any }}
   215→                preserveAspectRatio="xMidYMid meet"
   216→              >
   217→                {/* Connection lines */}
   218→                {CONSTELLATIONS.flatMap((c) =>
   219→                  c.connections.map(([aId, bId], i) => {
   220→                    const aStar = c.stars.find((s) => s.id === aId);
   221→                    const bStar = c.stars.find((s) => s.id === bId);
   222→                    if (!aStar || !bStar) return null;
   223→                    const x1 = (aStar.x / 100) * MAP_W;
   224→                    const y1 = (aStar.y / 100) * MAP_H;
   225→                    const x2 = (bStar.x / 100) * MAP_W;
   226→                    const y2 = (bStar.y / 100) * MAP_H;
   227→                    const isCompleted = !!constellationsCompleted[c.id];
   228→                    const isStarted = !!constellationsStarted[c.id];
   229→                    const isRevealed = !!constellationRevealed[c.id];
   230→                    const dim = !isStarted && !isRevealed;
   231→                    const stroke = isCompleted
   232→                      ? CATEGORY_ACCENT[c.category]
   233→                      : isStarted
   234→                        ? `${CATEGORY_ACCENT[c.category]}80`
   235→                        : "#1e293b";
   236→                    return (
   237→                      <line
   238→                        key={`${c.id}-${i}`}
   239→                        x1={x1}
   240→                        y1={y1}
   241→                        x2={x2}
   242→                        y2={y2}
   243→                        stroke={stroke}
   244→                        strokeWidth={isCompleted ? 2.2 : 1}
   245→                        strokeLinecap="round"
   246→                        opacity={dim ? 0.35 : 0.9}
   247→                      />
   248→                    );
   249→                  })
   250→                )}
   251→                {/* Stars */}
   252→                {CONSTELLATIONS.map((c) =>
   253→                  c.stars.map((star) => {
   254→                    const cx = (star.x / 100) * MAP_W;
   255→                    const cy = (star.y / 100) * MAP_H;
   256→                    const lit = !!constellationStars[star.id];
   257→                    const isStarted = !!constellationsStarted[c.id];
   258→                    const accent = CATEGORY_ACCENT[c.category];
   259→                    const isFocused = focusedConstellationId === c.id;
   260→                    const isHovered = hoveredStar === star.id;
   261→                    const isNewConstellation = (constellationProgress[c.id] || 0) === 0;
   262→                    const cost = starCost(c.id, constellationState, star.id, { foresightDiscountMult });
   263→                    const canAfford = divinity >= cost;
   264→                    const clickable = !lit && canAfford;
   265→                    const r = lit ? 4.5 : 2.2;
   266→                    return (
   267→                      <g
   268→                        key={star.id}
   269→                        transform={`translate(${cx},${cy})`}
   270→                        style={{ cursor: clickable ? "pointer" : "default" }}
   271→                        onClick={() => clickable && onStarClick(c.id, star.id)}
   272→                        onMouseEnter={() => setHoveredStar(star.id)}
   273→                        onMouseLeave={() => setHoveredStar(null)}
   274→                      >
   275→                        {/* Glow halo for lit/hovered */}
   276→                        {lit && (
   277→                          <circle r={r * 3} fill={accent} opacity={0.18} />
   278→                        )}
   279→                        {isHovered && !lit && (
   280→                          <circle r={r * 2.5} fill={accent} opacity={0.12} />
   281→                        )}
   282→                        {/* The star itself */}
   283→                        <circle
   284→                          r={r}
   285→                          fill={lit ? accent : isStarted ? "#475569" : "#1e293b"}
   286→                          stroke={lit ? "#fff" : isFocused ? accent : "transparent"}
   287→                          strokeWidth={lit || isFocused ? 0.8 : 0}
   288→                          opacity={lit ? 1 : isStarted ? 0.8 : 0.55}
   289→                        />
   290→                        {/* Cost label on hover for unlit affordable stars */}
   291→                        {isHovered && !lit && (
   292→                          <text
   293→                            y={-8}
   294→                            textAnchor="middle"
   295→                            fill={canAfford ? "#a3e635" : "#f87171"}
   296→                            fontSize="9"
   297→                            fontFamily="monospace"
   298→                            pointerEvents="none"
   299→                          >
   300→                            {isNewConstellation && startedCount > 0 ? `${cost}✦` : cost}
   301→                          </text>
   302→                        )}
   303→                      </g>
   304→                    );
   305→                  })
   306→                )}
   307→                {/* Constellation labels */}
   308→                {CONSTELLATIONS.map((c) => {
   309→                  const xs = c.stars.map((s) => s.x);
   310→                  const ys = c.stars.map((s) => s.y);
   311→                  const maxX = Math.max(...xs);
   312→                  const minY = Math.min(...ys);
   313→                  const lx = (maxX / 100) * MAP_W + 8;
   314→                  const ly = (minY / 100) * MAP_H + 4;
   315→                  const isCompleted = !!constellationsCompleted[c.id];
   316→                  const isStarted = !!constellationsStarted[c.id];
   317→                  const isRevealed = !!constellationRevealed[c.id];
   318→                  const dim = !isStarted && !isRevealed;
   319→                  return (
   320→                    <text
   321→                      key={`label-${c.id}`}
   322→                      x={lx}
   323→                      y={ly}
   324→                      fill={isCompleted ? CATEGORY_ACCENT[c.category] : isStarted ? "#cbd5e1" : "#475569"}
   325→                      fontSize="10"
   326→                      fontFamily="serif"
   327→                      fontStyle="italic"
   328→                      opacity={dim ? 0.35 : 0.95}
   329→                      pointerEvents="none"
   330→                    >
   331→                      {c.i18nName}
   332→                    </text>
   333→                  );
   334→                })}
   335→              </svg>
   336→              {/* Map legend */}
   337→              <div className="absolute bottom-2 left-2 flex flex-wrap items-center gap-x-3 gap-y-1 rounded-md border border-slate-700/50 bg-slate-950/70 px-2.5 py-1 text-[0.6rem] text-slate-300 backdrop-blur">
   338→                {(["zodiac", "northern", "equatorial", "southern"] as ConstellationCategory[]).map((cat) => (
   339→                  <span key={cat} className="flex items-center gap-1">
   340→                    <span className="inline-block w-2 h-2 rounded-full" style={{ background: CATEGORY_ACCENT[cat] }} />
   341→                    {CATEGORY_LABEL[cat]}
   342→                  </span>
   343→                ))}
   344→              </div>
   345→            </div>
   346→
   347→            {/* ===== SIDEBAR ===== */}
   348→            <div className="flex flex-col gap-2 overflow-hidden min-h-0">
   349→              <ScrollArea className="flex-1 pr-2">
   350→                <div className="space-y-3">
   351→                  {/* Progress stats */}
   352→                  <div className="stat-card">
   353→                    <div className="text-[0.7rem] uppercase tracking-wide text-muted-foreground mb-1.5">
   354→                      Constellation progress
   355→                    </div>
   356→                    <div className="grid grid-cols-2 gap-2 text-xs">
   357→                      <div className="flex flex-col">
   358→                        <span className="text-[0.6rem] text-muted-foreground">Completed</span>
   359→                        <span className="text-amber-300 font-semibold">
   360→                          {completedCount} / {CONSTELLATIONS_REQUIRED_FOR_LAYER_3}
   361→                          <span className="text-muted-foreground text-[0.6rem] ml-1">→ L3</span>
   362→                        </span>
   363→                      </div>
   364→                      <div className="flex flex-col">
   365→                        <span className="text-[0.6rem] text-muted-foreground">Stars lit</span>
   366→                        <span className="text-cyan-300 font-semibold">
   367→                          {litStarsCount} / {totalStars}
   368→                        </span>
   369→                      </div>
   370→                      <div className="flex flex-col">
   371→                        <span className="text-[0.6rem] text-muted-foreground">Started</span>
   372→                        <span className="text-violet-300 font-semibold">
   373→                          {startedCount} / {CONSTELLATIONS.length}
   374→                        </span>
   375→                      </div>
   376→                      <div className="flex flex-col">
   377→                        <span className="text-[0.6rem] text-muted-foreground">Divinity</span>
   378→                        <span className="text-violet-300 font-semibold">
   379→                          {formatNumber(divinity, 0)}
   380→                        </span>
   381→                      </div>
   382→                    </div>
   383→                    <Separator className="my-2" />
   384→                    <div className="flex flex-wrap gap-1">
   385→                      <Badge variant="outline" className="text-cyan-300 border-cyan-400/40">+{(cb.productionMult * 100).toFixed(1)}% prod</Badge>
   386→                      <Badge variant="outline" className="text-amber-300 border-amber-400/40">+{(cb.capMult * 100).toFixed(1)}% cap</Badge>
   387→                      <Badge variant="outline" className="text-emerald-300 border-emerald-400/40">+{(cb.epMult * 100).toFixed(1)}% EP</Badge>
   388→                      <Badge variant="outline" className="text-violet-300 border-violet-400/40">+{(cb.popGrowthMult * 100).toFixed(1)}% pop</Badge>
   389→                      {cb.costMult < 0 && (
   390→                        <Badge variant="outline" className="text-pink-300 border-pink-400/40">{(cb.costMult * 100).toFixed(1)}% cost</Badge>
   391→                      )}
   392→                      {cb.divinityMult > 0 && (
   393→                        <Badge variant="outline" className="text-yellow-300 border-yellow-400/40">+{(cb.divinityMult * 100).toFixed(0)}% Divinity</Badge>
   394→                      )}
   395→                    </div>
   396→                    <div className="mt-2 text-[0.6rem] text-muted-foreground">
   397→                      Per-star bonus: <span className="text-cyan-300">+0.5% prod</span> × {litStarsCount} lit star{litStarsCount === 1 ? "" : "s"} = <span className="text-cyan-300">+{(litStarsCount * 0.5).toFixed(1)}%</span>
   398→                    </div>
   399→                    <Badge
   400→                      variant="outline"
   401→                      className={`mt-2 ${canAdvanceToTranscendence(constellationState) ? "text-emerald-300 border-emerald-400/40" : "text-muted-foreground"}`}
   402→                    >
   403→                      {canAdvanceToTranscendence(constellationState)
   404→                        ? "Transcendence unlocked!"
   405→                        : `${CONSTELLATIONS_REQUIRED_FOR_LAYER_3 - completedCount} more constellation(s) to advance`}
   406→                    </Badge>
   407→                  </div>
   408→
   409→                  {/* Cost preview for focused constellation */}
   410→                  {focusedNextCost && (
   411→                    <div className="stat-card">
   412→                      <div className="text-[0.7rem] uppercase tracking-wide text-muted-foreground mb-1 flex items-center gap-1">
   413→                        <Crosshair className="w-3 h-3" /> Focused: {focusedNextCost.name}
   414→                      </div>
   415→                      {focusedNextCost.starId ? (
   416→                        <div className="text-xs space-y-1">
   417→                          <div className="flex items-center justify-between">
   418→                            <span className="text-muted-foreground">Next star:</span>
   419→                            <Badge variant="outline" className={divinity >= focusedNextCost.cost ? "text-cyan-300 border-cyan-400/40" : "text-rose-300 border-rose-400/40"}>
   420→                              {focusedNextCost.cost} Div
   421→                            </Badge>
   422→                          </div>
   423→                          {focusedNextCost.isNew && startedCount > 0 && (
   424→                            <div className="text-[0.6rem] text-amber-300/80 flex items-center gap-1">
   425→                              <Zap className="w-3 h-3" /> new-constellation penalty (×{Math.pow(NEW_CONSTELLATION_PENALTY, startedCount).toFixed(1)} = {NEW_CONSTELLATION_PENALTY}^{startedCount})
   426→                            </div>
   427→                          )}
   428→                          <Button
   429→                            size="sm"
   430→                            className="w-full mt-1"
   431→                            disabled={divinity < focusedNextCost.cost}
   432→                            onClick={() => illuminateStar(focusedNextCost.constellationId, focusedNextCost.starId!)}
   433→                          >
   434→                            Illuminate star ({focusedNextCost.cost} Div)
   435→                          </Button>
   436→                        </div>
   437→                      ) : (
   438→                        <div className="text-xs text-emerald-300 flex items-center gap-1">
   439→                          <Sparkles className="w-3 h-3" /> Constellation complete.
   440→                        </div>
   441→                      )}
   442→                    </div>
   443→                  )}
   444→
   445→                  {/* Cost preview: starting a NEW constellation */}
   446→                  {newConstellationPreview && (
   447→                    <div className="stat-card">
   448→                      <div className="text-[0.7rem] uppercase tracking-wide text-muted-foreground mb-1 flex items-center gap-1">
   449→                        <Orbit className="w-3 h-3" /> Start a new constellation
   450→                      </div>
   451→                      <div className="text-xs flex items-center justify-between">
   452→                        <span className="text-muted-foreground">{newConstellationPreview.name}</span>
