   353→function highestUnlockedLayerThemeId(unlockedLayers: Record<string, boolean>): string | null {
   354→  for (const layerId of LAYER_ORDER_FOR_THEME) {
   355→    if (unlockedLayers[layerId]) {
   356→      return LAYER_ID_TO_THEME[layerId] || null;
   357→    }
   358→  }
   359→  return null;
   360→}

   956→        popGrowthRate *= (1 + archPopMult);
   957→        // Layers 2-6: aggregate pop-growth bonus

   959→

   963→        // These are independent of resource production cap logic — a brood
   964→        // chamber keeps hatching even when its secondary outputs are capped.
   965→        let systemPopGrowth = 0;
   966→        for (const [sysId, count] of Object.entries(s.ownedSystems)) {
   967→          if (!count) continue;

   969→          if (!sys || !sys.popGrowth) continue;
   970→          // Respect manual disable (unless Auto-Balancer is active)
   971→          const manuallyEnabled = s.systemEnabled[sysId] !== false;
