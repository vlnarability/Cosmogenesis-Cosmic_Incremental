   151→export const useGameStore = create<GameStore>()(
   152→  persist(
   153→    (set, get) => ({
   154→      ...initialRunState(),
   155→      ...initialMetaState(),
   156→
   157→      // ============ TICK ============
   158→      tick: (dt) => {
   159→        const s = get();
   160→        if (s.paused) return;
   161→        const realDt = dt * s.speed;
   162→
   163→        const autoLvl = s.upgrades["inherited_efficiency"] || 0;
   164→        const autoMult = 1 + autoLvl * 0.15;
   165→        const techMult = techMultiplier(s, "production");
   166→        const capBoostLvl = s.upgrades["expansive_vaults"] || 0;
   167→        const capMult = 1 + capBoostLvl * 0.20;
   168→        const hasAutoBalancer = (s.upgrades["auto_balancer"] || 0) > 0;
   169→
   170→        // Archetype bonus
   171→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   172→        const archAllMult = archetype?.bonus.allProductionMult || 0;
   173→        const archPopMult = archetype?.bonus.populationGrowthMult || 0;
   174→        const archCapMult = archetype?.bonus.capMult || 0;
   175→
   176→        // ===== Enlightenment route bonus (Layer 2)
   177→        // The chosen route provides persistent bonuses that stack with archetype
   178→        // and challenge-mastery bonuses. Foresight passives apply on top.
   179→        const route = s.selectedRoute ? ROUTE_MAP[s.selectedRoute] : null;
   180→        const routeAllMult = route?.bonus.allProductionMult || 0;
   181→        const routePopMult = route?.bonus.populationGrowthMult || 0;
   182→        const foresightAllMult = foresightBonus(s.foresightNodes || {}, "production");
   183→        const foresightPopMult = foresightBonus(s.foresightNodes || {}, "popGrowth");
   184→
   185→        // ===== Challenge debuff (applied for the duration of an active challenge run)
   186→        const activeChallenge = s.activeChallenge ? CHALLENGE_MAP[s.activeChallenge] : null;
   187→        const challengeEffects = activeChallenge?.debuff.effects || {};
   188→        const chFoodMult = challengeEffects.foodMult ?? 1;
   189→        const chScienceMult = challengeEffects.scienceMult ?? 1;
   190→        const chProductionMult = challengeEffects.productionMult ?? 1;
   191→        const chMilitaryMult = challengeEffects.militaryMult ?? 1;
   192→        const chWaterUpkeepMult = challengeEffects.waterUpkeepMult ?? 1;
   193→        const chHappinessDrain = challengeEffects.happinessDrain ?? 0;
   194→        const chPopGrowthMult = challengeEffects.popGrowthMult ?? 1;
   195→
   196→        // ===== Transcendence (Layer 3) — transient ritual effects + script automation.
   197→        // - Surge: multiply all production while active (set by the production_surge ritual).
   198→        // - Divine Shield: suppress event firing while active.
   199→        // - Ritual cooldowns + script cooldowns tick down each tick.
   200→        const surgeMult = s.activeSurge && s.activeSurge.remaining > 0 ? s.activeSurge.mult : 1;
   201→        // Decrement transient cooldowns/timers by realDt (clamped at 0).
   202→        const ritualCooldowns = tickDownCooldowns(s.ritualCooldowns || {}, realDt);
   203→        const scriptCooldowns = tickDownCooldowns(s.scriptCooldowns || {}, realDt);
   204→        let divineShieldRemaining = Math.max(0, (s.divineShieldRemaining || 0) - realDt);
   205→        let activeSurge = s.activeSurge;
   206→        if (activeSurge) {
   207→          const remaining = activeSurge.remaining - realDt;
   208→          activeSurge = remaining > 0 ? { mult: activeSurge.mult, remaining } : null;
   209→        }
   210→
   211→        // ===== Genesis (Layer 4) — compiled world effects.
   212→        // The cradle, prime condition, geography, and any awakened dormant
   213→        // seed contribute their bonuses here. `worldEffects.resourceMult[r]`
   214→        // is additive with archetype + route + foresight per-resource bonuses.
   215→        // `worldEffects.capacityMult` is additive with archCapMult. `upkeepMult`
   216→        // and `eventFreqMult` are multiplicative.
   217→        const worldEffects = compileWorldEffects({
   218→          selectedCradle: s.selectedCradle,
   219→          selectedPrimeCondition: s.selectedPrimeCondition,
   220→          selectedGeography: s.selectedGeography,
   221→          selectedSeed: s.selectedSeed,
   222→          awakenedSeeds: s.awakenedSeeds || {},
   223→        });
   224→        const worldCapMult = worldEffects.capacityMult || 0;
   225→        const worldPopMult = worldEffects.popGrowthMult || 0;
   226→
   227→        const resources = { ...s.resources };
   228→        let capacities = recomputeCapacitiesFresh(s, capMult + archCapMult + worldCapMult);
   229→
   230→        // Apply system production — with cap-aware pausing
   231→        for (const [sysId, count] of Object.entries(s.ownedSystems)) {
   232→          if (!count) continue;
   233→          const sys = SYSTEM_MAP[sysId];
   234→          if (!sys) continue;
   235→          if (sys.requiredTech && !s.technologies[sys.requiredTech]) continue;
   236→
   237→          // Check if system is manually disabled
   238→          const manuallyEnabled = s.systemEnabled[sysId] !== false; // default true
   239→          if (!manuallyEnabled && !hasAutoBalancer) continue;
   240→
   241→          // Cap-aware logic: if all output resources are at cap, idle the system
   242→          // (don't produce, don't consume upkeep — no waste)
   243→          const outputsCapped = Object.keys(sys.produces).length > 0 &&
   244→            Object.keys(sys.produces).every((r) => {
   245→              const cap = capacities[r] || 0;
   246→              return (resources[r] || 0) >= cap - 0.01;
   247→            });
   248→
   249→          if (outputsCapped) {
   250→            // System idles — no production, no upkeep
   251→            continue;
   252→          }
   253→
   254→          const stageMult = sys.stage === STAGES[s.stageIndex].id ? 1 : 0.5;
   255→          const mult = autoMult * techMult * stageMult *
   256→            (1 + archAllMult + routeAllMult + foresightAllMult) * chProductionMult *
   257→            surgeMult;
   258→
   259→          // Per-resource archetype + route + foresight bonus + challenge resource-specific debuff
   260→          // Genesis (Layer 4) world effects: per-resource production multiplier
   261→          // (additive with archetype + route + foresight bonuses). Stacks with all.
   262→          const produces = sys.produces as Record<string, number>;
   263→          for (const [r, v] of Object.entries(produces)) {
   264→            const archResBonus = archetype?.bonus.productionMult?.[r] || 0;
   265→            const routeResBonus = route?.bonus.productionMult?.[r] || 0;
   266→            const foresightResBonus = foresightBonus(s.foresightNodes || {}, "resource", r);
   267→            const worldResBonus = worldEffects.resourceMult?.[r] || 0;
   268→            // Apply per-resource challenge multipliers
   269→            let chResMult = 1;
   270→            if (r === "food") chResMult *= chFoodMult;
   271→            if (r === "science") chResMult *= chScienceMult;
   272→            if (r === "military_power") chResMult *= chMilitaryMult;
   273→            const rate = v * count * mult *
   274→              (1 + archResBonus + routeResBonus + foresightResBonus + worldResBonus) * chResMult;
   275→            resources[r] = (resources[r] || 0) + rate * realDt;
   276→          }
   277→          // Upkeep — Genesis (Layer 4) world upkeep multiplier is multiplicative
   278→          // (so e.g. arid's 1.25× makes water cost 25% more on every system).
   279→          const worldUpkeepMult = worldEffects.upkeepMult ?? 1;
   280→          if (sys.upkeep) {
   281→            const upkeep = sys.upkeep as Record<string, number>;
   282→            for (const [r, v] of Object.entries(upkeep)) {
   283→              const upkMult = (r === "water" ? chWaterUpkeepMult : 1) * worldUpkeepMult;
   284→              resources[r] = (resources[r] || 0) - v * count * realDt * upkMult;
   285→            }
   286→          }
   287→        }
   288→
   289→        // Apply challenge happiness drain (flat per second)
   290→        if (chHappinessDrain > 0) {
   291→          const happCap = capacities.happiness || 100;
   292→          resources.happiness = Math.max(0, (resources.happiness || 0) - chHappinessDrain * realDt);
   293→          // also clamp to cap (already done below, but keep happ bounded)
   294→          if (resources.happiness > happCap) resources.happiness = happCap;
   295→        }
   296→
   297→        // Clamp resources to capacities (no overflow)
   298→        for (const r of Object.keys(resources)) {
   299→          const cap = capacities[r];
   300→          if (cap != null && resources[r] > cap) resources[r] = cap;
   301→          if (resources[r] < 0) resources[r] = 0;
   302→        }
   303→
   304→        // Slow passive ATP for cell (keeps early game from stalling)
   305→        if (STAGES[s.stageIndex].id === "cell") {
   306→          resources.atp = Math.min(capacities.atp || 30, resources.atp + 0.1 * realDt);
   307→        }
   308→
   309→        // Happiness drift toward 50 baseline if no source
   310→        // (prevents permanent high happiness from start)
   311→        // — disabled: let systems/tech maintain it
   312→
   313→        // Population growth with progress bar
   314→        // Growth rate scales with happiness and food/water availability
   315→        let popGrowthRate = 0.015; // base per second
   316→        const happiness = resources.happiness || 50;
   317→        popGrowthRate *= (1 + (happiness - 40) / 200); // happiness 100 → +30%, happiness 20 → -10%
   318→        popGrowthRate *= (1 + archPopMult + routePopMult + foresightPopMult + worldPopMult);
   319→        // Apply challenge population-growth debuff
   320→        popGrowthRate *= chPopGrowthMult;
   321→        // Penalty if food/water are critically low (creature+)
   322→        const curStage = STAGES[s.stageIndex];
   323→        if (["creature", "tribal", "civilization", "empire"].includes(curStage.id)) {
   324→          const foodCap = capacities.food || 1;
   325→          const waterCap = capacities.water || 1;
   326→          const foodPct = (resources.food || 0) / foodCap;
   327→          const waterPct = (resources.water || 0) / waterCap;
   328→          if (foodPct < 0.1) popGrowthRate *= 0.3;
   329→          if (waterPct < 0.1) popGrowthRate *= 0.3;
   330→        }
   331→
   332→        let populationProgress = (s.populationProgress || 0) + popGrowthRate * realDt;
   333→        let population = s.population;
   334→        while (populationProgress >= 1) {
   335→          populationProgress -= 1;
   336→          population += 1;
   337→        }
   338→
   339→        const maxPopulation = Math.max(s.maxPopulation || 0, population);
   340→
   341→        const newTime = s.time + realDt;
   342→
   343→        // Check story triggers
   344→        const newStory = { ...s.storyUnlocked };
   345→        let activePopup = s.activeStoryPopup;
   346→        const ctx = { ...s, resources, ownedSystems: s.ownedSystems };
   347→        for (const trig of STORY_TRIGGERS) {
   348→          if (!newStory[trig.id] && trig.condition(ctx)) {
   349→            newStory[trig.id] = true;
   350→            if (!activePopup) activePopup = trig.id;
   351→          }
   352→        }
   353→
   354→        // Check achievements periodically (every ~2 seconds of game time)
   355→        let achievements = s.achievements;
   356→        let newAchievements = s.newAchievements;
   357→        const checkInterval = 2.0;
   358→        const lastCheck = (s as any)._lastAchCheck || 0;
   359→        if (newTime - lastCheck >= checkInterval) {
   360→          const curState = { ...s, resources, time: newTime } as GameState;
   361→          const result = checkAchievements(curState);
   362→          if (result.newOnes.length > 0) {
   363→            achievements = result.earned;
   364→            newAchievements = [...(newAchievements || []), ...result.newOnes];
   365→          } else if (Object.keys(result.earned).length !== Object.keys(achievements || {}).length) {
   366→            achievements = result.earned;
   367→          }
   368→          (s as any)._lastAchCheck = newTime; // not persisted, just a runtime marker
   369→        }
   370→
   371→        // Random event firing — only if no popup/modal is active.
   372→        // Divine Shield (Transcendence ritual) suppresses event firing while active.
   373→        // Genesis (Layer 4) world effects: `eventFreqMult` multiplies the
   374→        // cooldown — values > 1 mean events come MORE often (cooldown
   375→        // shrinks), values < 1 mean events come LESS often (cooldown grows).
   376→        // (The cradle's "eventFreqMult" is a multiplier on event frequency,
   377→        // not on the cooldown duration, so we divide the cooldown by it.)
   378→        let activeEvent = s.activeEvent;
   379→        const worldEventFreqMult = worldEffects.eventFreqMult ?? 1;
   380→        let eventCooldown = Math.max(0, (s.eventCooldown || 0) - realDt);
   381→        const eventFreq = s.eventFrequency || "normal";
   382→        const shieldActive = divineShieldRemaining > 0;
   383→        if (eventFreq !== "off" && !activeEvent && !activePopup && eventCooldown === 0 && !shieldActive) {
   384→          const stageId = STAGES[s.stageIndex].id;
   385→          const possible = eventsForStage(stageId, newTime);
   386→          if (possible.length > 0) {
   387→            // Weighted random selection based on rarity
   388→            const totalWeight = possible.reduce((sum, e) => sum + eventWeight(e), 0);
   389→            const roll = Math.random() * totalWeight;
   390→            let cum = 0;
   391→            for (const ev of possible) {
   392→              cum += eventWeight(ev);
   393→              if (roll < cum) {
   394→                activeEvent = ev.id;
   395→                // Cooldown depends on frequency setting — divide by the world's
   396→                // event frequency multiplier so a 1.5× world fires events 50%
   397→                // more often (cooldowns are 1/1.5 = 0.67× as long).
   398→                const base = eventFreq === "frequent" ? 30 : 90;
   399→                const range = eventFreq === "frequent" ? 30 : 90;
   400→                eventCooldown = (base + Math.random() * range) / worldEventFreqMult;
   401→                break;
   402→              }
   403→            }
   404→          }
   405→        }
   406→
   407→        // ===== Transcendence (Layer 3) — Script automation.
   408→        // For each purchased script, evaluate its condition. If met (and the
   409→        // script's cooldown has elapsed for one-shot scripts), apply its action.
   410→        // Continuous scripts (sustainer) apply every tick while their condition holds.
   411→        // Crisis Manager auto-resolves the active event when its cooldown is up.
   412→        const scriptCtx = {
   413→          resources,
   414→          population,
   415→          activeEvent,
   416→        };
   417→        let scriptLogs: string[] = [];
   418→        for (const scriptDef of SCRIPTS) {
   419→          // Only purchased scripts are active.
   420→          if (!s.activeScripts?.[scriptDef.id]) continue;
   421→          const onCooldown = (scriptCooldowns[scriptDef.id] || 0) > 0;
   422→          if (onCooldown) continue;
   423→          // Evaluate the condition
   424→          if (!evalScriptCondition(scriptDef, scriptCtx)) continue;
   425→          // Apply the action
   426→          const result = applyScriptAction(scriptDef, {
   427→            resources,
   428→            population,
   429→            capacities,
   430→            activeEvent,
   431→          });
   432→          if (!result) continue;
   433→          // Apply mutations
   434→          for (const [k, v] of Object.entries(result.resourceChanges || {})) {
   435→            const cap = capacities[k] || Infinity;
   436→            resources[k] = Math.max(0, Math.min(cap, (resources[k] || 0) + (v as number)));
   437→          }
   438→          if (result.popChange) {
   439→            population = Math.max(0, population + result.popChange);
   440→          }
   441→          if (result.eventResolved) {
   442→            activeEvent = null;
   443→            eventCooldown = Math.max(eventCooldown, 60);
   444→          }
   445→          // Set cooldown for one-shot scripts
   446→          if (scriptDef.oneShot && scriptDef.cooldown) {
   447→            scriptCooldowns[scriptDef.id] = scriptDef.cooldown;
   448→          }
   449→          if (result.logMsg) scriptLogs.push(result.logMsg);
   450→        }
   451→
   452→        set({
   453→          resources,
   454→          capacities,
   455→          time: newTime,
   456→          population,
   457→          maxPopulation,
   458→          populationProgress,
   459→          storyUnlocked: newStory,
   460→          activeStoryPopup: activePopup,
   461→          achievements,
   462→          newAchievements,
   463→          activeEvent,
   464→          eventCooldown,
   465→          totalPlayTime: (s.totalPlayTime || 0) + realDt,
   466→          // Transcendence (Layer 3) — persist transient timers
   467→          ritualCooldowns,
   468→          scriptCooldowns,
   469→          divineShieldRemaining,
   470→          activeSurge,
   471→        });
   472→
   473→        // Log any script actions (after the set so the log appears in order)
   474→        for (const msg of scriptLogs) {
   475→          get().addToLog(msg);
   476→        }
   477→
   478→        // Check if the active challenge's completion condition has been met.
   479→        // If so, mark it complete and clear the active challenge slot.
   480→        // (Done after the set so completion sees updated state.)
   481→        if (s.activeChallenge) {
   482→          get().checkChallengeCompletion();
   483→        }
   484→      },
   485→
   486→      // ============ ACTIONS ============
   487→      performAction: (actionId) => {
   488→        const s = get();
   489→        const action = ACTION_MAP[actionId];
   490→        if (!action) return;
   491→        if (action.stage !== STAGES[s.stageIndex].id) return;
   492→
   493→        const res = { ...s.resources };
   494→
   495→        // Check if all output resources are at cap — if so, block (no waste)
   496→        const outputs = Object.entries(action.produces);
   497→        if (outputs.length > 0) {
   498→          const allCapped = outputs.every(([r]) => {
   499→            const cap = s.capacities[r] || 0;
   500→            return (res[r] || 0) >= cap - 0.001;
   501→          });
   502→          if (allCapped) return; // action wasted — block it
   503→        }
   504→
   505→        // Check costs
   506→        if (action.cost) {
   507→          for (const [r, v] of Object.entries(action.cost)) {
   508→            if ((res[r] || 0) < (v as number)) return;
   509→          }
   510→          for (const [r, v] of Object.entries(action.cost)) {
   511→            res[r] -= v as number;
   512→          }
   513→        }
   514→
   515→        // Manual action multiplier
   516→        const manualLvl = s.upgrades["quickened_hands"] || 0;
   517→        const mult = 1 + manualLvl * 0.25;
   518→        const techMult = techMultiplier(s, "manual");
   519→        // Transcendence surge — multiplies manual production too while active
   520→        const surgeMult = s.activeSurge && s.activeSurge.remaining > 0 ? s.activeSurge.mult : 1;
   521→        const finalMult = mult * techMult * surgeMult;
   522→
   523→        // Archetype + Enlightenment route per-resource bonus
   524→        // Genesis (Layer 4) world effects — compiled here too so manual actions
   525→        // benefit from the same per-resource bonuses as auto production.
   526→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   527→        const route = s.selectedRoute ? ROUTE_MAP[s.selectedRoute] : null;
   528→        const routeAllMult = route?.bonus.allProductionMult || 0;
   529→        const foresightAllMult = foresightBonus(s.foresightNodes || {}, "production");
   530→        const worldEffects = compileWorldEffects({
   531→          selectedCradle: s.selectedCradle,
   532→          selectedPrimeCondition: s.selectedPrimeCondition,
   533→          selectedGeography: s.selectedGeography,
   534→          selectedSeed: s.selectedSeed,
   535→          awakenedSeeds: s.awakenedSeeds || {},
   536→        });
   537→
   538→        for (const [r, v] of Object.entries(action.produces)) {
   539→          const cap = s.capacities[r] || 0;
   540→          const archResBonus = archetype?.bonus.productionMult?.[r] || 0;
   541→          const routeResBonus = route?.bonus.productionMult?.[r] || 0;
   542→          const foresightResBonus = foresightBonus(s.foresightNodes || {}, "resource", r);
   543→          const worldResBonus = worldEffects.resourceMult?.[r] || 0;
   544→          const totalMult = 1 + archResBonus + routeResBonus + foresightResBonus + worldResBonus +
   545→            (archetype?.bonus.allProductionMult || 0) + routeAllMult + foresightAllMult;
   546→          const amount = (v as number) * finalMult * totalMult;
   547→          res[r] = Math.min(cap, (res[r] || 0) + amount);
   548→        }
   549→
   550→        // Mark first-hunt flag so STORY_TRIGGERS can fire the first_hunt story
   551→        // beat on the next tick. The hunt action lives in Tribal stage.
   552→        const updates: Partial<GameState> = {
   553→          resources: res,
   554→          totalActions: (s.totalActions || 0) + 1,
   555→        };
   556→        if (actionId === "hunt" && !s.flags?.firstHunt) {
   557→          updates.flags = { ...(s.flags || {}), firstHunt: true };
   558→        }
   559→        set(updates);
   560→      },
   561→
   562→      buySystem: (systemId, qty) => {
   563→        const s = get();
   564→        const sys = SYSTEM_MAP[systemId];
   565→        if (!sys) return;
   566→        if (sys.stage !== STAGES[s.stageIndex].id) return;
   567→        // Check tech prerequisite
   568→        if (sys.requiredTech && !s.technologies[sys.requiredTech]) return;
   569→        // Locked systems (via exclusiveWith variant choices) cannot be purchased.
   570→        if (s.lockedSystems && s.lockedSystems[systemId]) return;
   571→
   572→        const costReduction = (s.upgrades["frugality"] || 0) * 0.04;
   573→        let owned = s.ownedSystems[systemId] || 0;
   574→        if (sys.maxOwned && owned >= sys.maxOwned) return;
   575→
   576→        const res = { ...s.resources };
   577→        const ownedSystems = { ...s.ownedSystems };
   578→        const systemEnabled = { ...s.systemEnabled };
   579→        const archetypeAffinity = { ...s.archetypeAffinity };
   580→
   581→        let bought = 0;
   582→        for (let i = 0; i < qty; i++) {
   583→          if (sys.maxOwned && owned + bought >= sys.maxOwned) break;
   584→          const cost = systemCost(sys, owned + bought, costReduction);
   585→          let canAfford = true;
   586→          for (const [r, v] of Object.entries(cost)) {
   587→            if ((res[r] || 0) < (v as number)) {
   588→              canAfford = false;
   589→              break;
   590→            }
   591→          }
   592→          if (!canAfford) break;
   593→          for (const [r, v] of Object.entries(cost)) {
   594→            res[r] -= v as number;
   595→          }
   596→          bought++;
   597→        }
   598→
   599→        if (bought === 0) return;
   600→
   601→        ownedSystems[systemId] = owned + bought;
   602→        // New systems are enabled by default
   603→        if (systemEnabled[systemId] === undefined) systemEnabled[systemId] = true;
   604→
   605→        // Apply archetype affinity grant
   606→        if (sys.grantsAffinity) {
   607→          const arch = sys.grantsAffinity.archetype;
   608→          archetypeAffinity[arch] = (archetypeAffinity[arch] || 0) + sys.grantsAffinity.amount * bought;
   609→        }
   610→
   611→        // Trigger story
   612→        const newStory = { ...s.storyUnlocked };
   613→        let activePopup = s.activeStoryPopup;
   614→        for (const trig of STORY_TRIGGERS) {
   615→          if (!newStory[trig.id] && trig.condition({ ...s, resources: res, ownedSystems })) {
   616→            newStory[trig.id] = true;
   617→            if (!activePopup) activePopup = trig.id;
   618→          }
   619→        }
   620→
   621→        // Enforce mutually-exclusive variant choices: when buying the first
   622→        // instance of a system that is part of an exclusive group, permanently
   623→        // lock all sibling variants so they can never be purchased this run.
   624→        // (Only the FIRST purchase locks siblings — once chosen, the choice is
   625→        // final for this run, but the chosen system can still be bought up to
   626→        // its maxOwned.)
   627→        const lockedSystems = { ...(s.lockedSystems || {}) };
   628→        if (sys.exclusiveWith && sys.exclusiveWith.length > 0 && owned === 0 && bought > 0) {
   629→          for (const siblingId of sys.exclusiveWith) {
   630→            // Don't lock a sibling that is somehow already owned (defensive)
   631→            if (!(ownedSystems[siblingId] > 0)) {
   632→              lockedSystems[siblingId] = true;
   633→            }
   634→          }
   635→        }
   636→
   637→        set({
   638→          resources: res,
   639→          ownedSystems,
   640→          systemEnabled,
   641→          archetypeAffinity,
   642→          storyUnlocked: newStory,
   643→          activeStoryPopup: activePopup,
   644→          lockedSystems,
   645→          totalSystemsBuilt: (s.totalSystemsBuilt || 0) + bought,
   646→        });
   647→        get().addToLog(`Built ${bought}× ${sys.name}.`);
   648→      },
   649→
   650→      toggleSystem: (systemId) => {
   651→        const s = get();
   652→        if (!s.ownedSystems[systemId]) return;
   653→        const systemEnabled = { ...s.systemEnabled };
   654→        systemEnabled[systemId] = systemEnabled[systemId] === false ? true : false;
   655→        set({ systemEnabled });
   656→      },
   657→
   658→      enableAllSystems: (enabled) => {
   659→        const s = get();
   660→        const systemEnabled: Record<string, boolean> = {};
   661→        for (const id of Object.keys(s.ownedSystems)) {
   662→          systemEnabled[id] = enabled;
   663→        }
   664→        set({ systemEnabled });
   665→      },
   666→
   667→      buyTech: (techId) => {
   668→        const s = get();
   669→        const tech = TECH_MAP[techId];
   670→        if (!tech) return;
   671→        if (s.technologies[techId]) return;
   672→        // Locked techs cannot be researched (a mutually-exclusive sibling was already chosen)
   673→        if (s.lockedTechs && s.lockedTechs[techId]) return;
   674→        if (tech.stage !== STAGES[s.stageIndex].id) return;
   675→        if (tech.requires) {
   676→          for (const req of tech.requires) {
   677→            if (!s.technologies[req]) return;
   678→          }
   679→        }
   680→        const res = { ...s.resources };
   681→        for (const [r, v] of Object.entries(tech.cost)) {
   682→          if ((res[r] || 0) < (v as number)) return;
   683→        }
   684→        for (const [r, v] of Object.entries(tech.cost)) {
   685→          res[r] -= v as number;
   686→        }
   687→        const technologies = { ...s.technologies, [techId]: true };
   688→        const archetypeAffinity = { ...s.archetypeAffinity };
   689→        if (tech.grantsAffinity) {
   690→          archetypeAffinity[tech.grantsAffinity.archetype] =
   691→            (archetypeAffinity[tech.grantsAffinity.archetype] || 0) + tech.grantsAffinity.amount;
   692→        }
   693→        // Enforce mutually-exclusive tech path choices: lock all siblings
   694→        // listed in `exclusiveWith` so they can never be researched this run.
   695→        const lockedTechs = { ...(s.lockedTechs || {}) };
   696→        if (tech.exclusiveWith && tech.exclusiveWith.length > 0) {
   697→          for (const siblingId of tech.exclusiveWith) {
   698→            // Don't lock a sibling that is somehow already researched (defensive)
   699→            if (!technologies[siblingId]) {
   700→              lockedTechs[siblingId] = true;
   701→            }
   702→          }
   703→        }
   704→        set({
   705→          resources: res,
   706→          technologies,
   707→          lockedTechs,
   708→          archetypeAffinity,
   709→          totalTechResearched: (s.totalTechResearched || 0) + 1,
   710→        });
   711→        get().addToLog(`Researched ${tech.name}.`);
   712→      },
   713→
   714→      buyUpgrade: (upgradeId) => {
   715→        const s = get();
   716→        const up = UPGRADE_MAP[upgradeId];
   717→        if (!up) return;
   718→        if (up.requiresWins && s.galacticWins < up.requiresWins) return;
   719→        const owned = s.upgrades[upgradeId] || 0;
   720→        if (owned >= up.maxLevel) return;
   721→        const cost = upgradeCost(up, owned);
   722→        if (s.evolutionPoints < cost) return;
   723→        const upgrades = { ...s.upgrades, [upgradeId]: owned + 1 };
   724→        set({
   725→          evolutionPoints: s.evolutionPoints - cost,
   726→          upgrades,
   727→        });
   728→        get().addToLog(`Purchased ${up.name} (Lv ${owned + 1}).`);
   729→      },
   730→
   731→      evolveStage: () => {
   732→        const s = get();
   733→        const stage = STAGES[s.stageIndex];
   734→        const next = nextStage(stage.id);
   735→        if (!next) {
   736→          get().triggerPrestige();
   737→          return;
   738→        }
   739→        const reqs = stage.evolveRequires;
   740→        const evolveBoost = 1 - Math.min(0.30, (s.upgrades["evolutionary_momentum"] || 0) * 0.05);
   741→        const totalSystems = Object.values(s.ownedSystems).reduce((a, b) => a + b, 0);
   742→        const totalTech = Object.keys(s.technologies).length;
   743→
   744→        if (s.population < reqs.minPopulation * evolveBoost) return;
   745→        if (totalSystems < reqs.minSystems * evolveBoost) return;
   746→        if (totalTech < reqs.minTech * evolveBoost) return;
   747→
   748→        const nextStageIdx = s.stageIndex + 1;
   749→        const nextStageDef = STAGES[nextStageIdx];
   750→
   751→        // ===== Genesis (Layer 4) — dormant seed awakening.
   752→        // If the player has planted a dormant seed whose awakening stage is the
   753→        // new stage, the seed awakens here. The reward is applied immediately:
   754→        //   - resourceMult becomes part of the world effects (applied via
   755→        //     compileWorldEffects on every tick / action from this moment on)
   756→        //   - divinityGain is added to the player's Divinity balance
   757→        // The awakening fires a log entry; no story popup (the seed's
   758→        // awakening IS the surprise — a quiet moment that bends the run).
   759→        let divinityGain = 0;
   760→        let awakenedSeeds = s.awakenedSeeds;
   761→        const plantedSeedId = s.selectedSeed;
   762→        const plantedSeed = plantedSeedId ? DORMANT_SEED_MAP[plantedSeedId] : null;
   763→        if (plantedSeed && plantedSeed.awakeningStage === nextStageDef.id &&
   764→            !(s.awakenedSeeds?.[plantedSeed.id])) {
   765→          // Awaken the seed.
   766→          awakenedSeeds = { ...(s.awakenedSeeds || {}), [plantedSeed.id]: true };
   767→          divinityGain = plantedSeed.reward.divinityGain || 0;
   768→          // Log it (deferred so it lands after the evolve log).
   769→          const seedName = plantedSeed.name;
   770→          const rewardBits: string[] = [];
   771→          if (plantedSeed.reward.divinityGain) {
   772→            rewardBits.push(`+${plantedSeed.reward.divinityGain} Divinity`);
   773→          }
   774→          if (plantedSeed.reward.resourceMult) {
   775→            for (const [r, v] of Object.entries(plantedSeed.reward.resourceMult)) {
   776→              const pct = Math.round((v as number) * 100);
   777→              rewardBits.push(`+${pct}% ${r}`);
   778→            }
   779→          }
   780→          setTimeout(() => {
   781→            try {
   782→              useGameStore.getState().addToLog(
   783→                `Dormant seed awakened — ${seedName}. The world shifts: ${rewardBits.join(", ")}.`
   784→              );
   785→            } catch { /* ignore */ }
   786→          }, 0);
   787→        }
   788→
   789→        const resources = { ...s.resources };
   790→        // Give starting resources for the new stage so the player can build the first systems
   791→        // This prevents the "can't build anything" progression blocker
   792→        const stageStartBonus: Record<string, number> = {
   793→          creature: { food: 15, water: 10, materials: 5 },
   794→          tribal: { wood: 15, stone: 5, clay: 3 },
   795→          civilization: { production: 15, gold: 5 },
   796→          empire: { influence: 5, gold: 20 },
   797→          solar: { energy: 20, alloys: 5 },
   798→          galactic: { data: 10, energy: 30 },
   799→        };
   800→        const startBonus = stageStartBonus[nextStageDef.id];
   801→        if (startBonus) {
   802→          for (const [r, v] of Object.entries(startBonus)) {
   803→            resources[r] = (resources[r] || 0) + v;
   804→          }
   805→        }
   806→        const capBoostLvl = s.upgrades["expansive_vaults"] || 0;
   807→        const capMult = 1 + capBoostLvl * 0.20;
   808→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   809→        const archCapMult = archetype?.bonus.capMult || 0;
   810→        // Genesis (Layer 4) — world capacity mult applies to the fresh stage's caps too.
   811→        const worldCapMultForEvolve = compileWorldEffects({
   812→          selectedCradle: s.selectedCradle,
   813→          selectedPrimeCondition: s.selectedPrimeCondition,
   814→          selectedGeography: s.selectedGeography,
   815→          selectedSeed: s.selectedSeed,
   816→          awakenedSeeds,
   817→        }).capacityMult || 0;
   818→        const capacities = recomputeCapacitiesFresh(s, capMult + archCapMult + worldCapMultForEvolve);
   819→
   820→        const newStory = { ...s.storyUnlocked };
   821→
   822→        // Map: NEW stage → that stage's primary intro story ID. This story is
   823→        // shown by the StageTransition modal (which displays the new stage's
   824→        // `storyIntro` text) AND marked as unlocked for the journal so the
   825→        // player can re-read it later.
   826→        const introStoryId =
   827→          nextStageDef.id === "creature" ? "first_body_stirs"
   828→          : nextStageDef.id === "tribal" ? "fire_remembered"
   829→          : nextStageDef.id === "civilization" ? "cities_learn_to_dream"
   830→          : nextStageDef.id === "empire" ? "distance_begins_to_obey"
   831→          : nextStageDef.id === "solar" ? "sky_stops_being_a_wall"
   832→          : "galaxy_takes_a_shape"; // → galactic
   833→
   834→        // Map: OLD stage → transition outro story ID. This is the "<stage>_endures"
   835→        // beat that looks back at the stage the player just left. Fired as a
   836→        // StoryPopup AFTER the StageTransition modal closes, so the player gets
   837→        // two narrative beats per evolve: the new stage's intro (in the modal),
   838→        // and the old stage's outro (in the popup).
   839→        const outroStoryId =
   840→          stage.id === "cell" ? "cell_endures"
   841→          : stage.id === "creature" ? "creature_endures"
   842→          : stage.id === "tribal" ? "tribal_endures"
   843→          : stage.id === "civilization" ? "civilization_endures"
   844→          : stage.id === "empire" ? "empire_endures"
   845→          : stage.id === "solar" ? "solar_endures"
   846→          : null;
   847→
   848→        if (introStoryId) newStory[introStoryId] = true;
   849→        if (outroStoryId) newStory[outroStoryId] = true;
   850→
   851→        // Lock archetype when evolving from Creature → Tribal
   852→        let lockedArchetype = s.lockedArchetype;
   853→        if (stage.id === "creature" && !lockedArchetype) {
   854→          const dom = dominantArchetype(s.archetypeAffinity);
   855→          if (dom) {
   856→            lockedArchetype = dom;
   857→            get().addToLog(`Lineage locked: ${ARCHETYPE_MAP[dom].name}.`);
   858→          } else {
   859→            lockedArchetype = "humanoid"; // default fallback
   860→            get().addToLog(`Lineage locked: Humanoid (no drift).`);
   861→          }
   862→        }
   863→
   864→        const stageClearCounts = { ...s.stageClearCounts, [stage.id]: (s.stageClearCounts[stage.id] || 0) + 1 };
   865→
   866→        // Track fastest Cell clear
   867→        let fastestCellClear = s.fastestCellClear || 0;
   868→        if (stage.id === "cell") {
   869→          const clearTime = s.time;
   870→          if (fastestCellClear === 0 || clearTime < fastestCellClear) {
   871→            fastestCellClear = clearTime;
   872→          }
   873→        }
   874→
   875→        // Check achievements after stage clear
   876→        const postEvolveState = { ...s, stageClearCounts, lockedArchetype, fastestCellClear } as GameState;
   877→        const achResult = checkAchievements(postEvolveState);
   878→        const newAchievements = achResult.newOnes.length > 0
   879→          ? [...(s.newAchievements || []), ...achResult.newOnes]
   880→          : s.newAchievements;
   881→
   882→        set({
   883→          stageIndex: nextStageIdx,
   884→          resources,
   885→          capacities,
   886→          storyUnlocked: newStory,
   887→          // Fire the OLD stage's transition outro as the StoryPopup. The new
   888→          // stage's intro is shown by the StageTransition modal (z-60, sits
   889→          // above the StoryPopup). When the player dismisses the modal, the
   890→          // outro popup becomes visible — a "looking back" beat before the
   891→          // new era begins in earnest.
   892→          activeStoryPopup: outroStoryId || introStoryId || s.activeStoryPopup,
   893→          stageClearCounts,
   894→          fastestCellClear,
   895→          lockedArchetype,
   896→          achievements: achResult.earned,
   897→          newAchievements,
   898→          // Genesis (Layer 4) — record awakened seed + apply divinity gain.
   899→          awakenedSeeds,
   900→          divinity: (s.divinity || 0) + divinityGain,
   901→        });
   902→        get().addToLog(`Evolved to ${nextStageDef.name}. ${nextStageDef.tagline}`);
   903→      },
   904→
   905→      triggerPrestige: () => {
   906→        const s = get();
   907→        const stage = STAGES[s.stageIndex];
   908→        const score = computeScore(s);
   909→        const stageMult = [1, 2, 4, 8, 16, 32, 64][s.stageIndex] || 1;
   910→        // ===== Genesis (Layer 4) — world EP modifier.
   911→        // The selected cradle's `epModifier` multiplies EP earned. Easy worlds
   912→        // (0.7×) yield fewer EP; extreme worlds (2.5×) yield dramatically more.
   913→        // Stacks with the stage multiplier (already baked into the formula below).
   914→        const cradle = s.selectedCradle ? CRADLE_WORLD_MAP[s.selectedCradle] : null;
   915→        const epModifier = cradle?.epModifier ?? 1.0;
   916→        const epEarned = Math.max(1, Math.floor(Math.sqrt(score / 100) * stageMult * epModifier));
   917→
   918→        const archive = [...s.archive];
   919→        const archName = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype]?.name : "Humanoid";
   920→        archive.unshift({
   921→          runId: `run_${s.totalRuns + 1}`,
   922→          timestamp: Date.now(),
   923→          stage: stage.id,
   924→          archetype: s.lockedArchetype || "humanoid",
   925→          score,
   926→          epEarned,
   927→          duration: s.time,
   928→          ending: s.stageIndex >= 6 ? "Galactic Ascension" : `${stage.name} Evolution (${archName})`,
   929→          // Genesis (Layer 4) — record which cradle was authored for this run
   930→          // (so the archive entry can show the difficulty + EP modifier that
   931→          // shaped the run). `notes` is a free-form field on ArchiveEntry.
   932→          notes: cradle ? `${cradle.name} · ${epModifier}× EP` : undefined,
   933→        });
   934→
   935→        const archivedArchetypes = Array.from(new Set([
   936→          ...(s.archivedArchetypes || []),
   937→          ...(s.lockedArchetype ? [s.lockedArchetype] : []),
   938→        ]));
   939→
   940→        // ===== Challenges: unlock on first Galactic win + abandon any active challenge
   941→        const wasFirstGalacticWin = s.stageIndex >= 6 && !s.unlockedChallenges;
   942→        const unlockedChallenges = s.unlockedChallenges || (s.stageIndex >= 6);
   943→        const completedChallenges = s.completedChallenges || {};
   944→        // If a challenge was active when prestiging without completing it, that's
   945→        // an abandonment — log it. The fresh run will start with no active challenge.
   946→        if (s.activeChallenge) {
   947→          const chName = CHALLENGE_MAP[s.activeChallenge]?.name || s.activeChallenge;
   948→          // Defer logging until after state has been set so it appears after the
   949→          // prestige log entry.
   950→          setTimeout(() => get().addToLog(`Challenge abandoned: ${chName}. The trial is incomplete.`), 0);
   951→        }
   952→
   953→        // ===== Enlightenment (Layer 2): unlock when all 10 Evolution challenges are mastered.
   954→        // Also re-checked defensively here in case checkChallengeCompletion missed
   955→        // it (e.g., legacy saves where the 10th challenge was completed before
   956→        // this code shipped).
   957→        const completedChallengesCount = Object.values(completedChallenges).filter(Boolean).length;
   958→        const wasEnlightenmentUnlocked = !!s.enlightenmentUnlocked;
   959→        const enlightenmentUnlocked = wasEnlightenmentUnlocked ||
   960→          completedChallengesCount >= CHALLENGES.length;
   961→
   962→        // Divinity is earned only once Enlightenment is unlocked. Strong runs
   963→        // (higher stage + higher score) yield proportionally more.
   964→        const divinityEarned = enlightenmentUnlocked
   965→          ? Math.max(1, Math.floor(score / 500 * stageMult))
   966→          : 0;
   967→
   968→        const fresh = initialRunState();
   969→        const metaState: Partial<GameState> = {
   970→          evolutionPoints: s.evolutionPoints + epEarned,
   971→          totalRuns: s.totalRuns + 1,
   972→          galacticWins: s.galacticWins + (s.stageIndex >= 6 ? 1 : 0),
   973→          archive,
   974→          archivedArchetypes,
   975→          upgrades: s.upgrades,
   976→          storyUnlocked: { ...s.storyUnlocked, god_opens_its_eyes: true },
   977→          storyAcknowledged: s.storyAcknowledged,
   978→          unlockedLayers: s.unlockedLayers,
   979→          stageClearCounts: s.stageClearCounts,
   980→          fastestCellClear: s.fastestCellClear,

  1414→      performRitual: (ritualId) => {
  1415→        const s = get();
  1416→        if (!s.transcendenceUnlocked) return false;
  1417→        const ritual = RITUAL_MAP[ritualId];
  1418→        if (!ritual) return false;
  1419→        // One-time-per-run: if ritual cooldown is set (i.e., already performed
  1420→        // this run), block. We use the ritual's presence in ritualCooldowns as
  1421→        // the "already performed" marker — once performed, the cooldown is set
  1422→        // to a large value (Number.MAX_SAFE_INTEGER) so the button stays

  1425→        // Check Divinity cost
  1426→        if ((s.divinity || 0) < ritual.divinityCost) return false;
  1427→
  1428→        // ===== Apply the ritual effect =====
  1429→        const updates: Partial<GameState> = {
  1430→          divinity: s.divinity - ritual.divinityCost,
  1431→        };
  1432→        const effect = ritual.effect;
  1433→        let logMsg = `Ritual performed: ${ritual.name}. `;
  1434→        let surgeApplied = false;
  1435→        switch (effect.type) {
  1436→          case "surge": {
  1437→            // Stack with existing surge if any (most recent surge takes priority
  1438→            // by resetting remaining to its duration).
  1439→            const mult = effect.surgeMult ?? 2;
  1440→            const duration = effect.surgeDuration ?? 60;
  1441→            updates.activeSurge = { mult, remaining: duration };
  1442→            logMsg += `Production ×${mult} for ${duration}s.`;
  1443→            surgeApplied = true;
  1444→            break;
  1445→          }
  1446→          case "stabilize": {
  1447→            const floor = effect.stabilizeHappiness ?? 75;
  1448→            const cur = s.resources.happiness || 0;
  1449→            const cap = s.capacities.happiness || 100;
  1450→            if (cur < floor) {
