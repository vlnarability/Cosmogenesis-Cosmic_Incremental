     1→"use client";
     2→
     3→import { create } from "zustand";
     4→import { persist, createJSONStorage } from "zustand/middleware";
     5→import type { GameState, GameStore, StageId, TabId } from "./types";
     6→import { emptyResources, emptyCapacities } from "../data/resources";
     7→import { STAGES, STAGE_IDS, nextStage } from "../data/stages";
     8→import { ACTION_MAP } from "../data/actions";
     9→import { SYSTEMS, SYSTEM_MAP, systemCost } from "../data/systems";
    10→import { TECH_MAP, TECHS } from "../data/techs";
    11→import { UPGRADE_MAP, UPGRADES, upgradeCost } from "../data/upgrades";
    12→import { STORY_MAP, STORY_TRIGGERS } from "../data/story";
    13→import { ARCHETYPE_MAP, dominantArchetype } from "../data/archetypes";
    14→import { ACHIEVEMENTS, achievementBonus, type AchievementCheckCtx } from "../data/achievements";
    15→import { EVENTS, eventsForStage, eventWeight, EVENT_MAP, RARITY_INFO } from "../data/events";
    16→
    17→const STORAGE_KEY = "evolution_idle_v2";
    18→
    19→function initialRunState(): Partial<GameState> {
    20→  const resources = emptyResources();
    21→  const capacities = emptyCapacities();
    22→  resources.atp = 5;
    23→  resources.glucose = 8;
    24→  resources.happiness = 75;
    25→  return {
    26→    stageIndex: 0,
    27→    time: 0,
    28→    population: 8,
    29→    populationProgress: 0,
    30→    maxPopulation: 8,
    31→    resources,
    32→    capacities,
    33→    ownedSystems: {},
    34→    systemEnabled: {},
    35→    technologies: {},
    36→    archetypeAffinity: {},
    37→    lockedArchetype: null,
    38→    log: ["New run started. The first spark stirs."],
    39→  };
    40→}
    41→
    42→function initialMetaState(): Partial<GameState> {
    43→  return {
    44→    evolutionPoints: 0,
    45→    totalRuns: 0,
    46→    galacticWins: 0,
    47→    stageClearCounts: Object.fromEntries(STAGE_IDS.map((s) => [s, 0])) as Record<StageId, number>,
    48→    fastestCellClear: 0,
    49→    totalEventsResolved: 0,
    50→    totalPlayTime: 0,
    51→    totalActions: 0,
    52→    totalSystemsBuilt: 0,
    53→    totalTechResearched: 0,
    54→    upgrades: Object.fromEntries(UPGRADES.map((u) => [u.id, 0])),
    55→    storyUnlocked: { first_spark: true },
    56→    storyAcknowledged: {},
    57→    archive: [],
    58→    archivedArchetypes: [],
    59→    unlockedLayers: { evolution: true },
    60→    achievements: {},
    61→    newAchievements: [],
    62→    currentTab: "actions",
    63→    speed: 1,
    64→    paused: false,
    65→    showShop: false,
    66→    showEvolve: false,
    67→    showSettings: false,
    68→    activeStoryPopup: "first_spark",
    69→    activeEvent: null,
    70→    eventCooldown: 60, // first event can fire after 60s
    71→    eventHistory: [],
    72→    hasSeenIntro: false,
    73→    lastSaved: Date.now(),
    74→    tutorialStep: 0,
    75→    tutorialActive: true,
    76→    tutorialDismissed: false,
    77→    eventFrequency: "normal",
    78→  };
    79→}
    80→
    81→export const useGameStore = create<GameStore>()(
    82→  persist(
    83→    (set, get) => ({
    84→      ...initialRunState(),
    85→      ...initialMetaState(),
    86→
    87→      // ============ TICK ============
    88→      tick: (dt) => {
    89→        const s = get();
    90→        if (s.paused) return;
    91→        const realDt = dt * s.speed;
    92→
    93→        const autoLvl = s.upgrades["inherited_efficiency"] || 0;
    94→        const autoMult = 1 + autoLvl * 0.15;
    95→        const techMult = techMultiplier(s, "production");
    96→        const capBoostLvl = s.upgrades["expansive_vaults"] || 0;
    97→        const capMult = 1 + capBoostLvl * 0.20;
    98→        const hasAutoBalancer = (s.upgrades["auto_balancer"] || 0) > 0;
    99→
   100→        // Archetype bonus
   101→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   102→        const archAllMult = archetype?.bonus.allProductionMult || 0;
   103→        const archPopMult = archetype?.bonus.populationGrowthMult || 0;
   104→        const archCapMult = archetype?.bonus.capMult || 0;
   105→
   106→        const resources = { ...s.resources };
   107→        let capacities = recomputeCapacitiesFresh(s, capMult + archCapMult);
   108→
   109→        // Apply system production — with cap-aware pausing
   110→        for (const [sysId, count] of Object.entries(s.ownedSystems)) {
   111→          if (!count) continue;
   112→          const sys = SYSTEM_MAP[sysId];
   113→          if (!sys) continue;
   114→          if (sys.requiredTech && !s.technologies[sys.requiredTech]) continue;
   115→
   116→          // Check if system is manually disabled
   117→          const manuallyEnabled = s.systemEnabled[sysId] !== false; // default true
   118→          if (!manuallyEnabled && !hasAutoBalancer) continue;
   119→
   120→          // Cap-aware logic: if all output resources are at cap, idle the system
   121→          // (don't produce, don't consume upkeep — no waste)
   122→          const outputsCapped = Object.keys(sys.produces).length > 0 &&
   123→            Object.keys(sys.produces).every((r) => {
   124→              const cap = capacities[r] || 0;
   125→              return (resources[r] || 0) >= cap - 0.01;
   126→            });
   127→
   128→          if (outputsCapped) {
   129→            // System idles — no production, no upkeep
   130→            continue;
   131→          }
   132→
   133→          const stageMult = sys.stage === STAGES[s.stageIndex].id ? 1 : 0.5;
   134→          const mult = autoMult * techMult * stageMult * (1 + archAllMult);
   135→
   136→          // Per-resource archetype bonus
   137→          const produces = sys.produces as Record<string, number>;
   138→          for (const [r, v] of Object.entries(produces)) {
   139→            const archResBonus = archetype?.bonus.productionMult?.[r] || 0;
   140→            const rate = v * count * mult * (1 + archResBonus);
   141→            resources[r] = (resources[r] || 0) + rate * realDt;
   142→          }
   143→          // Upkeep
   144→          if (sys.upkeep) {
   145→            const upkeep = sys.upkeep as Record<string, number>;
   146→            for (const [r, v] of Object.entries(upkeep)) {
   147→              resources[r] = (resources[r] || 0) - v * count * realDt;
   148→            }
   149→          }
   150→        }
   151→
   152→        // Clamp resources to capacities (no overflow)
   153→        for (const r of Object.keys(resources)) {
   154→          const cap = capacities[r];
   155→          if (cap != null && resources[r] > cap) resources[r] = cap;
   156→          if (resources[r] < 0) resources[r] = 0;
   157→        }
   158→
   159→        // Slow passive ATP for cell (keeps early game from stalling)
   160→        if (STAGES[s.stageIndex].id === "cell") {
   161→          resources.atp = Math.min(capacities.atp || 30, resources.atp + 0.1 * realDt);
   162→        }
   163→
   164→        // Happiness drift toward 50 baseline if no source
   165→        // (prevents permanent high happiness from start)
   166→        // — disabled: let systems/tech maintain it
   167→
   168→        // Population growth with progress bar
   169→        // Growth rate scales with happiness and food/water availability
   170→        let popGrowthRate = 0.015; // base per second
   171→        const happiness = resources.happiness || 50;
   172→        popGrowthRate *= (1 + (happiness - 40) / 200); // happiness 100 → +30%, happiness 20 → -10%
   173→        popGrowthRate *= (1 + archPopMult);
   174→        // Penalty if food/water are critically low (creature+)
   175→        const curStage = STAGES[s.stageIndex];
   176→        if (["creature", "tribal", "civilization", "empire"].includes(curStage.id)) {
   177→          const foodCap = capacities.food || 1;
   178→          const waterCap = capacities.water || 1;
   179→          const foodPct = (resources.food || 0) / foodCap;
   180→          const waterPct = (resources.water || 0) / waterCap;
   181→          if (foodPct < 0.1) popGrowthRate *= 0.3;
   182→          if (waterPct < 0.1) popGrowthRate *= 0.3;
   183→        }
   184→
   185→        let populationProgress = (s.populationProgress || 0) + popGrowthRate * realDt;
   186→        let population = s.population;
   187→        while (populationProgress >= 1) {
   188→          populationProgress -= 1;
   189→          population += 1;
   190→        }
   191→
   192→        const maxPopulation = Math.max(s.maxPopulation || 0, population);
   193→
   194→        const newTime = s.time + realDt;
   195→
   196→        // Check story triggers
   197→        const newStory = { ...s.storyUnlocked };
   198→        let activePopup = s.activeStoryPopup;
   199→        const ctx = { ...s, resources, ownedSystems: s.ownedSystems };
   200→        for (const trig of STORY_TRIGGERS) {
   201→          if (!newStory[trig.id] && trig.condition(ctx)) {
   202→            newStory[trig.id] = true;
   203→            if (!activePopup) activePopup = trig.id;
   204→          }
   205→        }
   206→
   207→        // Check achievements periodically (every ~2 seconds of game time)
   208→        let achievements = s.achievements;
   209→        let newAchievements = s.newAchievements;
   210→        const checkInterval = 2.0;
   211→        const lastCheck = (s as any)._lastAchCheck || 0;
   212→        if (newTime - lastCheck >= checkInterval) {
   213→          const curState = { ...s, resources, time: newTime } as GameState;
   214→          const result = checkAchievements(curState);
   215→          if (result.newOnes.length > 0) {
   216→            achievements = result.earned;
   217→            newAchievements = [...(newAchievements || []), ...result.newOnes];
   218→          } else if (Object.keys(result.earned).length !== Object.keys(achievements || {}).length) {
   219→            achievements = result.earned;
   220→          }
   221→          (s as any)._lastAchCheck = newTime; // not persisted, just a runtime marker
   222→        }
   223→
   224→        // Random event firing — only if no popup/modal is active
   225→        let activeEvent = s.activeEvent;
   226→        let eventCooldown = (s.eventCooldown || 0) - realDt;
   227→        if (eventCooldown < 0) eventCooldown = 0;
   228→        const eventFreq = s.eventFrequency || "normal";
   229→        if (eventFreq !== "off" && !activeEvent && !activePopup && eventCooldown === 0) {
   230→          const stageId = STAGES[s.stageIndex].id;
   231→          const possible = eventsForStage(stageId, newTime);
   232→          if (possible.length > 0) {
   233→            // Weighted random selection based on rarity
   234→            const totalWeight = possible.reduce((sum, e) => sum + eventWeight(e), 0);
   235→            const roll = Math.random() * totalWeight;
   236→            let cum = 0;
   237→            for (const ev of possible) {
   238→              cum += eventWeight(ev);
   239→              if (roll < cum) {
   240→                activeEvent = ev.id;
   241→                // Cooldown depends on frequency setting
   242→                const base = eventFreq === "frequent" ? 30 : 90;
   243→                const range = eventFreq === "frequent" ? 30 : 90;
   244→                eventCooldown = base + Math.random() * range;
   245→                break;
   246→              }
   247→            }
   248→          }
   249→        }
   250→
   251→        set({
   252→          resources,
   253→          capacities,
   254→          time: newTime,
   255→          population,
   256→          maxPopulation,
   257→          populationProgress,
   258→          storyUnlocked: newStory,
   259→          activeStoryPopup: activePopup,
   260→          achievements,
   261→          newAchievements,
   262→          activeEvent,
   263→          eventCooldown,
   264→          totalPlayTime: (s.totalPlayTime || 0) + realDt,
   265→        });
   266→      },
   267→
   268→      // ============ ACTIONS ============
   269→      performAction: (actionId) => {
   270→        const s = get();
   271→        const action = ACTION_MAP[actionId];
   272→        if (!action) return;
   273→        if (action.stage !== STAGES[s.stageIndex].id) return;
   274→
   275→        const res = { ...s.resources };
   276→
   277→        // Check if all output resources are at cap — if so, block (no waste)
   278→        const outputs = Object.entries(action.produces);
   279→        if (outputs.length > 0) {
   280→          const allCapped = outputs.every(([r]) => {
   281→            const cap = s.capacities[r] || 0;
   282→            return (res[r] || 0) >= cap - 0.001;
   283→          });
   284→          if (allCapped) return; // action wasted — block it
   285→        }
   286→
   287→        // Check costs
   288→        if (action.cost) {
   289→          for (const [r, v] of Object.entries(action.cost)) {
   290→            if ((res[r] || 0) < (v as number)) return;
   291→          }
   292→          for (const [r, v] of Object.entries(action.cost)) {
   293→            res[r] -= v as number;
   294→          }
   295→        }
   296→
   297→        // Manual action multiplier
   298→        const manualLvl = s.upgrades["quickened_hands"] || 0;
   299→        const mult = 1 + manualLvl * 0.25;
   300→        const techMult = techMultiplier(s, "manual");
   301→        const finalMult = mult * techMult;
   302→
   303→        // Archetype per-resource bonus
   304→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   305→
   306→        for (const [r, v] of Object.entries(action.produces)) {
   307→          const cap = s.capacities[r] || 0;
   308→          const archResBonus = archetype?.bonus.productionMult?.[r] || 0;
   309→          const totalArchMult = 1 + archResBonus + (archetype?.bonus.allProductionMult || 0);
   310→          const amount = (v as number) * finalMult * totalArchMult;
   311→          res[r] = Math.min(cap, (res[r] || 0) + amount);
   312→        }
   313→
   314→        set({
   315→          resources: res,
   316→          totalActions: (s.totalActions || 0) + 1,
   317→        });
   318→      },
   319→
   320→      buySystem: (systemId, qty) => {
   321→        const s = get();
   322→        const sys = SYSTEM_MAP[systemId];
   323→        if (!sys) return;
   324→        if (sys.stage !== STAGES[s.stageIndex].id) return;
   325→        // Check tech prerequisite
   326→        if (sys.requiredTech && !s.technologies[sys.requiredTech]) return;
   327→
   328→        const costReduction = (s.upgrades["frugality"] || 0) * 0.04;
   329→        let owned = s.ownedSystems[systemId] || 0;
   330→        if (sys.maxOwned && owned >= sys.maxOwned) return;
   331→
   332→        const res = { ...s.resources };
   333→        const ownedSystems = { ...s.ownedSystems };
   334→        const systemEnabled = { ...s.systemEnabled };
   335→        const archetypeAffinity = { ...s.archetypeAffinity };
   336→
   337→        let bought = 0;
   338→        for (let i = 0; i < qty; i++) {
   339→          if (sys.maxOwned && owned + bought >= sys.maxOwned) break;
   340→          const cost = systemCost(sys, owned + bought, costReduction);
   341→          let canAfford = true;
   342→          for (const [r, v] of Object.entries(cost)) {
   343→            if ((res[r] || 0) < (v as number)) {
   344→              canAfford = false;
   345→              break;
   346→            }
   347→          }
   348→          if (!canAfford) break;
   349→          for (const [r, v] of Object.entries(cost)) {
   350→            res[r] -= v as number;
   351→          }
   352→          bought++;
   353→        }
   354→
   355→        if (bought === 0) return;
   356→
   357→        ownedSystems[systemId] = owned + bought;
   358→        // New systems are enabled by default
   359→        if (systemEnabled[systemId] === undefined) systemEnabled[systemId] = true;
   360→
   361→        // Apply archetype affinity grant
   362→        if (sys.grantsAffinity) {
   363→          const arch = sys.grantsAffinity.archetype;
   364→          archetypeAffinity[arch] = (archetypeAffinity[arch] || 0) + sys.grantsAffinity.amount * bought;
   365→        }
   366→
   367→        // Trigger story
   368→        const newStory = { ...s.storyUnlocked };
   369→        let activePopup = s.activeStoryPopup;
   370→        for (const trig of STORY_TRIGGERS) {
   371→          if (!newStory[trig.id] && trig.condition({ ...s, resources: res, ownedSystems })) {
   372→            newStory[trig.id] = true;
   373→            if (!activePopup) activePopup = trig.id;
   374→          }
   375→        }
   376→
   377→        set({
   378→          resources: res,
   379→          ownedSystems,
   380→          systemEnabled,
   381→          archetypeAffinity,
   382→          storyUnlocked: newStory,
   383→          activeStoryPopup: activePopup,
   384→          totalSystemsBuilt: (s.totalSystemsBuilt || 0) + bought,
   385→        });
   386→        get().addToLog(`Built ${bought}× ${sys.name}.`);
   387→      },
   388→
   389→      toggleSystem: (systemId) => {
   390→        const s = get();
   391→        if (!s.ownedSystems[systemId]) return;
   392→        const systemEnabled = { ...s.systemEnabled };
   393→        systemEnabled[systemId] = systemEnabled[systemId] === false ? true : false;
   394→        set({ systemEnabled });
   395→      },
   396→
   397→      enableAllSystems: (enabled) => {
   398→        const s = get();
   399→        const systemEnabled: Record<string, boolean> = {};
   400→        for (const id of Object.keys(s.ownedSystems)) {
   401→          systemEnabled[id] = enabled;
   402→        }
   403→        set({ systemEnabled });
   404→      },
   405→
   406→      buyTech: (techId) => {
   407→        const s = get();
   408→        const tech = TECH_MAP[techId];
   409→        if (!tech) return;
   410→        if (s.technologies[techId]) return;
   411→        if (tech.stage !== STAGES[s.stageIndex].id) return;
   412→        if (tech.requires) {
   413→          for (const req of tech.requires) {
   414→            if (!s.technologies[req]) return;
   415→          }
   416→        }
   417→        const res = { ...s.resources };
   418→        for (const [r, v] of Object.entries(tech.cost)) {
   419→          if ((res[r] || 0) < (v as number)) return;
   420→        }
   421→        for (const [r, v] of Object.entries(tech.cost)) {
   422→          res[r] -= v as number;
   423→        }
   424→        const technologies = { ...s.technologies, [techId]: true };
   425→        const archetypeAffinity = { ...s.archetypeAffinity };
   426→        if (tech.grantsAffinity) {
   427→          archetypeAffinity[tech.grantsAffinity.archetype] =
   428→            (archetypeAffinity[tech.grantsAffinity.archetype] || 0) + tech.grantsAffinity.amount;
   429→        }
   430→        set({
   431→          resources: res,
   432→          technologies,
   433→          archetypeAffinity,
   434→          totalTechResearched: (s.totalTechResearched || 0) + 1,
   435→        });
   436→        get().addToLog(`Researched ${tech.name}.`);
   437→      },
   438→
   439→      buyUpgrade: (upgradeId) => {
   440→        const s = get();
   441→        const up = UPGRADE_MAP[upgradeId];
   442→        if (!up) return;
   443→        if (up.requiresWins && s.galacticWins < up.requiresWins) return;
   444→        const owned = s.upgrades[upgradeId] || 0;
   445→        if (owned >= up.maxLevel) return;
   446→        const cost = upgradeCost(up, owned);
   447→        if (s.evolutionPoints < cost) return;
   448→        const upgrades = { ...s.upgrades, [upgradeId]: owned + 1 };
   449→        set({
   450→          evolutionPoints: s.evolutionPoints - cost,
   451→          upgrades,
   452→        });
   453→        get().addToLog(`Purchased ${up.name} (Lv ${owned + 1}).`);
   454→      },
   455→
   456→      evolveStage: () => {
   457→        const s = get();
   458→        const stage = STAGES[s.stageIndex];
   459→        const next = nextStage(stage.id);
   460→        if (!next) {
   461→          get().triggerPrestige();
   462→          return;
   463→        }
   464→        const reqs = stage.evolveRequires;
   465→        const evolveBoost = 1 - Math.min(0.30, (s.upgrades["evolutionary_momentum"] || 0) * 0.05);
   466→        const totalSystems = Object.values(s.ownedSystems).reduce((a, b) => a + b, 0);
   467→        const totalTech = Object.keys(s.technologies).length;
   468→
   469→        if (s.population < reqs.minPopulation * evolveBoost) return;
   470→        if (totalSystems < reqs.minSystems * evolveBoost) return;
   471→        if (totalTech < reqs.minTech * evolveBoost) return;
   472→
   473→        const nextStageIdx = s.stageIndex + 1;
   474→        const nextStageDef = STAGES[nextStageIdx];
   475→
   476→        const resources = { ...s.resources };
   477→        // Give starting resources for the new stage so the player can build the first systems
   478→        // This prevents the "can't build anything" progression blocker
   479→        const stageStartBonus: Record<string, number> = {
   480→          creature: { food: 15, water: 10, materials: 5 },
   481→          tribal: { wood: 15, stone: 5, clay: 3 },
   482→          civilization: { production: 15, gold: 5 },
   483→          empire: { influence: 5, gold: 20 },
   484→          solar: { energy: 20, alloys: 5 },
   485→          galactic: { data: 10, energy: 30 },
   486→        };
   487→        const startBonus = stageStartBonus[nextStageDef.id];
   488→        if (startBonus) {
   489→          for (const [r, v] of Object.entries(startBonus)) {
   490→            resources[r] = (resources[r] || 0) + v;
   491→          }
   492→        }
   493→        const capBoostLvl = s.upgrades["expansive_vaults"] || 0;
   494→        const capMult = 1 + capBoostLvl * 0.20;
   495→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   496→        const archCapMult = archetype?.bonus.capMult || 0;
   497→        const capacities = recomputeCapacitiesFresh(s, capMult + archCapMult);
   498→
   499→        const newStory = { ...s.storyUnlocked };
   500→        const storyId = STORY_MAP[
   501→          nextStageDef.id === "creature" ? "first_body_stirs"
   502→          : nextStageDef.id === "tribal" ? "fire_remembered"
   503→          : nextStageDef.id === "civilization" ? "cities_learn_to_dream"
   504→          : nextStageDef.id === "empire" ? "distance_begins_to_obey"
   505→          : nextStageDef.id === "solar" ? "sky_stops_being_a_wall"
   506→          : "galaxy_takes_a_shape"
   507→        ];
   508→        if (storyId) newStory[storyId.id] = true;
   509→
   510→        // Lock archetype when evolving from Creature → Tribal
   511→        let lockedArchetype = s.lockedArchetype;
   512→        if (stage.id === "creature" && !lockedArchetype) {
   513→          const dom = dominantArchetype(s.archetypeAffinity);
   514→          if (dom) {
   515→            lockedArchetype = dom;
   516→            get().addToLog(`Lineage locked: ${ARCHETYPE_MAP[dom].name}.`);
   517→          } else {
   518→            lockedArchetype = "humanoid"; // default fallback
   519→            get().addToLog(`Lineage locked: Humanoid (no drift).`);
   520→          }
   521→        }
   522→
   523→        const stageClearCounts = { ...s.stageClearCounts, [stage.id]: (s.stageClearCounts[stage.id] || 0) + 1 };
   524→
   525→        // Track fastest Cell clear
   526→        let fastestCellClear = s.fastestCellClear || 0;
   527→        if (stage.id === "cell") {
   528→          const clearTime = s.time;
   529→          if (fastestCellClear === 0 || clearTime < fastestCellClear) {

   854→function techMultiplier(s: GameState, target: "production" | "manual" | "capacity"): number {
   855→  let mult = 1;
   856→  for (const [techId, owned] of Object.entries(s.technologies)) {
   857→    if (!owned) continue;
   858→    const tech = TECH_MAP[techId];
   859→    if (!tech || !tech.multiplier) continue;
   860→    if (tech.multiplier.target === target || tech.multiplier.target === "production") {
   861→      mult += tech.multiplier.value;
   862→    }
   863→  }
   864→  // Achievement production bonus
   865→  const ctx = buildAchievementCtx(s);
   866→  mult += achievementBonus(s.achievements || {}, ctx, "production");
   867→  return mult;
   868→}

   870→function buildAchievementCtx(s: GameState): AchievementCheckCtx {
   871→  const systemsDiscoveredCount = Object.keys(s.ownedSystems).filter((id) => (s.ownedSystems[id] || 0) > 0).length;
   872→  const techResearchedCount = Object.keys(s.technologies).filter((id) => s.technologies[id]).length;
   873→  const upgradesOwnedCount = Object.values(s.upgrades).filter((c) => c > 0).length;
   874→  const storyUnlockedCount = Object.keys(s.storyUnlocked).filter((id) => s.storyUnlocked[id]).length;
   875→  return {
   876→    galacticWins: s.galacticWins,
   877→    totalRuns: s.totalRuns,
   878→    stageClearCounts: s.stageClearCounts,
   879→    archivedArchetypes: s.archivedArchetypes || [],
   880→    storyUnlockedCount,
   881→    systemsDiscoveredCount,
   882→    techResearchedCount,
   883→    upgradesOwnedCount,
   884→    maxPopulation: s.maxPopulation || 0,
   885→    fastestCellClear: s.fastestCellClear || 0,
   886→    totalEventsResolved: s.totalEventsResolved || 0,
   887→  };
   888→}

   890→// Check all achievements; return new ones earned
   891→function checkAchievements(s: GameState): { earned: Record<string, boolean>; newOnes: string[] } {
   892→  const ctx = buildAchievementCtx(s);
   893→  const earned = { ...(s.achievements || {}) };
   894→  const newOnes: string[] = [];
   895→  for (const ach of ACHIEVEMENTS) {
   896→    if (!earned[ach.id] && ach.check(ctx)) {
   897→      earned[ach.id] = true;
   898→      newOnes.push(ach.id);
   899→    }
   900→  }
   901→  return { earned, newOnes };
   902→}

   904→function computeScore(s: GameState): number {
   905→  const totalSystems = Object.values(s.ownedSystems).reduce((a, b) => a + b, 0);
   906→  const totalTech = Object.keys(s.technologies).length;
   907→  const resourceSum = Object.values(s.resources).reduce((a, b) => a + b, 0);
   908→  return Math.floor(
   909→    s.population * 2 +
   910→    totalSystems * 5 +
   911→    totalTech * 10 +
   912→    resourceSum * 0.5 +
   913→    s.time * 0.1
   914→  );
   915→}

   917→// Recompute all capacities from scratch based on owned systems
   918→function recomputeCapacitiesFresh(s: GameState, capMult: number): Record<string, number> {
   919→  const base = emptyCapacities();
   920→  const caps: Record<string, number> = {};
   921→  for (const r of Object.keys(base)) {
   922→    let cap = (base[r] || 50) * capMult;
   923→    for (const sys of SYSTEMS) {
   924→      const count = s.ownedSystems[sys.id] || 0;
   925→      if (!count || !sys.capacityBoost) continue;
   926→      const boost = sys.capacityBoost[r];
   927→      if (boost) cap += (boost as number) * count * capMult;
   928→    }
   929→    caps[r] = cap;
   930→  }
   931→  return caps;
   932→}

   934→// Export helper for components to compute whether an action is capped
   935→export function isActionCapped(
   936→  produces: Record<string, number>,
   937→  resources: Record<string, number>,
   938→  capacities: Record<string, number>
   939→): boolean {
   940→  const outputs = Object.entries(produces);
   941→  if (outputs.length === 0) return false;
   942→  return outputs.every(([r]) => {
   943→    const cap = capacities[r] || 0;
   944→    return (resources[r] || 0) >= cap - 0.001;
   945→  });
   946→}

   948→// Export helper for whether a system is currently capped (idling)
   949→export function isSystemCapped(
   950→  sysId: string,
   951→  resources: Record<string, number>,
   952→  capacities: Record<string, number>
   953→): boolean {
   954→  const sys = SYSTEM_MAP[sysId];
   955→  if (!sys) return false;
   956→  const produces = Object.keys(sys.produces);
   957→  if (produces.length === 0) return false;
   958→  return produces.every((r) => {
   959→    const cap = capacities[r] || 0;
   960→    return (resources[r] || 0) >= cap - 0.01;
   961→  });
   962→}
