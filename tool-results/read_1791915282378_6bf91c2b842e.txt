   321→// ===== Active realm view — shows the actual mini-game UI =====
   322→function ActiveRealmView({
   323→  realmId,
   324→  state,
   325→  onClose,
   326→  onBreakthrough,
   327→  onDiscontentAction,
   328→  onFaminePump,
   329→  onIgnoranceBuyNode,
   330→  onStagnationUpgrade,
   331→  onWeaknessPlaceTower,
   332→  onPurityBuildSystem,
   333→  onAsceticBuildSystem,
   334→  stageIndex,
   335→  liveElapsed = 0,
   336→}: {
   337→  realmId: string;
   338→  state: Record<string, any>;
   339→  onClose: () => void;
   340→  onBreakthrough: () => void;
   341→  onDiscontentAction: (action: "celebrate" | "tax" | "ignore") => void;
   342→  onFaminePump: (resourceId: "a" | "b" | "c") => void;
   343→  onIgnoranceBuyNode: (nodeId: number) => void;
   344→  onStagnationUpgrade: (stage: number) => void;
   345→  onWeaknessPlaceTower: () => void;
   346→  onPurityBuildSystem: () => void;
   347→  onAsceticBuildSystem: (systemId: number) => void;
   348→  stageIndex: number;
   349→  liveElapsed?: number; // FEATURE 1 — live timer
   350→}) {
   351→  const realm = TRIAL_REALMS.find((r) => r.id === realmId);
   352→  if (!realm) return null;
   353→
   354→  return (
   355→    <div className="space-y-3">
   356→      <div className="flex items-center justify-between">
   357→        <div className="flex items-center gap-2">
   358→          <span className="text-3xl">{realm.icon}</span>
   359→          <div>
   360→            <div className="text-base font-semibold">{realm.name}</div>
   361→            <div className="text-[0.65rem] text-muted-foreground">{realm.tagline}</div>
   362→          </div>
   363→          {/* FEATURE 1 — live timer in the active realm header */}
   364→          <Badge variant="outline" className="ml-2 text-[0.6rem] text-cyan-300 border-cyan-400/40 animate-pulse">
   365→            <Hourglass className="w-2.5 h-2.5 mr-0.5" /> Time: {Math.floor(liveElapsed)}s
   366→          </Badge>
   367→        </div>
   368→        <Button size="sm" variant="outline" onClick={onClose}>Exit Realm</Button>
   369→      </div>
   370→
   371→      {realmId === "realm_growth" && (
   372→        <RealmGrowthGame state={state} onBreakthrough={onBreakthrough} />
   373→      )}
   374→      {realmId === "realm_famine" && (
   375→        <RealmFamineGame state={state} onPump={onFaminePump} />
   376→      )}
   377→      {realmId === "realm_discontent" && (
   378→        <RealmDiscontentGame state={state} onAction={onDiscontentAction} />
   379→      )}
   380→      {realmId === "realm_ignorance" && (
   381→        <RealmIgnoranceGame state={state} onBuyNode={onIgnoranceBuyNode} />
   382→      )}
   383→      {realmId === "realm_stagnation" && (
   384→        <RealmStagnationGame state={state} onUpgrade={onStagnationUpgrade} />
   385→      )}
   386→      {realmId === "realm_weakness" && (
   387→        <RealmWeaknessGame state={state} onPlaceTower={onWeaknessPlaceTower} />
   388→      )}
   389→      {realmId === "realm_late_bloom" && (
   390→        <RealmLateBloomGame state={state} />
   391→      )}
   392→      {realmId === "realm_purity" && (
   393→        <RealmPurityGame state={state} onBuildSystem={onPurityBuildSystem} />
   394→      )}
   395→      {realmId === "realm_ascetic" && (
   396→        <RealmAsceticGame state={state} onBuildSystem={onAsceticBuildSystem} />
   397→      )}
   398→      {realmId === "realm_swiftness" && (
   399→        <RealmSwiftnessGame state={state} stageIndex={stageIndex} />
   400→      )}
   401→    </div>
   402→  );
   403→}

   405→// ===== Realm of Growth — energy bar that fills passively, click to Break Through =====
   406→function RealmGrowthGame({ state, onBreakthrough }: { state: Record<string, any>; onBreakthrough: () => void }) {
   407→  const energy = state.energy || 0;
   408→  const breaks = state.breaks || 0;
   409→  const completed = !!state.completed;
   410→  const canBreak = energy >= 100 && !completed;
   411→
   412→  return (
   413→    <div className="stat-card space-y-3">
   414→      <div className="text-[0.7rem] text-muted-foreground">
   415→        Energy fills at 0.5/s. When the bar is full, click Break Through to convert energy into +50% population.
   416→      </div>
   417→
   418→      <div className="space-y-1">
   419→        <div className="flex justify-between text-xs">
   420→          <span className="text-emerald-300/80">Energy</span>
   421→          <span className="font-mono">{energy.toFixed(1)} / 100</span>
   422→        </div>
   423→        <div className="h-6 rounded-md bg-emerald-950/40 border border-emerald-500/30 overflow-hidden">

  1125→function formatEffects(eff: Record<string, unknown>): string {
  1126→  const parts: string[] = [];
  1127→  if (eff.productionMult !== undefined) {
  1128→    const v = eff.productionMult as number;
  1129→    parts.push(v >= 0 ? `+${(v * 100).toFixed(0)}% prod` : `${(v * 100).toFixed(0)}% prod`);
  1130→  }
  1131→  if (eff.costMult !== undefined) {
  1132→    const v = eff.costMult as number;
  1133→    if (v >= 1) parts.push(`costs ×${v.toFixed(2)}`);
  1134→    else parts.push(`costs ${(v * 100).toFixed(0)}%`);
  1135→  }
  1136→  if (eff.popGrowthMult !== undefined) {
  1137→    const v = eff.popGrowthMult as number;
  1138→    parts.push(v >= 1 ? `pop ×${v.toFixed(2)}` : `pop ${(v * 100).toFixed(0)}%`);
  1139→  }
  1140→  if (eff.capMult !== undefined) {
  1141→    const v = eff.capMult as number;
  1142→    parts.push(v >= 1 ? `cap ×${v.toFixed(2)}` : `cap ${(v * 100).toFixed(0)}%`);
  1143→  }
  1144→  if (eff.epMult !== undefined) {
  1145→    const v = eff.epMult as number;
  1146→    parts.push(`+${(v * 100).toFixed(0)}% EP`);
  1147→  }
  1148→  if (eff.disableTech) parts.push("no tech");
  1149→  if (eff.disableMilitary) parts.push("no military");
  1150→  if (eff.disableEvents) parts.push("no events");
  1151→  if (eff.disablePause) parts.push("no pause");
  1152→  if (eff.startPop !== undefined) parts.push(`start pop = ${eff.startPop}`);
  1153→  if (eff.maxSystems !== undefined) parts.push(`max ${eff.maxSystems} systems`);
  1154→  return parts.join(" · ");
  1155→}
