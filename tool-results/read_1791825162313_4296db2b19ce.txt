     1→// ===========================================================================
     2→// LAYER 2 — ENLIGHTENMENT (CONSTELLATION MAP)
     3→// ===========================================================================
     4→// After 3 trials are mastered, the god opens its eyes. The Enlightenment layer
     5→// is a star map: the player illuminates individual stars within real IAU
     6→// constellations. Completing a constellation grants a permanent bonus.
     7→// Currency = Divinity (earned at prestige while this layer is unlocked).
     8→// Complete CONSTELLATIONS_REQUIRED_FOR_LAYER_3 constellations to advance to
     9→// Layer 3 (Transcendence).
    10→//
    11→// ===========================================================================
    12→// ARCHITECTURE NOTE — The full IAU list has 88 constellations. This file ships
    13→// a curated subset of 28 prominent constellations (all 12 zodiac + 16 of the
    14→// brightest northern / southern / equatorial patterns). The data shape
    15→// (`ConstellationDef`) supports all 88 unchanged — to extend, just append more
    16→// entries to `CONSTELLATIONS`. Cost model, UI, and bonus aggregation are
    17→// constellation-count-agnostic.
    18→// ===========================================================================
    19→
    20→export type ConstellationCategory =
    21→  | "zodiac"
    22→  | "northern"
    23→  | "southern"
    24→  | "equatorial";
    25→
    26→export interface ConstellationStar {
    27→  /** Globally-unique star id, e.g. "orion_alpha". */
    28→  id: string;
    29→  /** Normalized X coordinate on the star map, 0-100 (left → right). */
    30→  x: number;
    31→  /** Normalized Y coordinate on the star map, 0-100 (top → bottom). */
    32→  y: number;
    33→}
    34→
    35→export interface ConstellationBonus {
    36→  /** Additive production multiplier (0.05 = +5%). */
    37→  productionMult?: number;
    38→  /** Additive capacity multiplier. */
    39→  capMult?: number;
    40→  /** Additive EP multiplier (applied at prestige). */
    41→  epMult?: number;
    42→  /** Additive population growth multiplier. */
    43→  popGrowthMult?: number;
    44→  /** Cost reduction (negative = cheaper systems/techs). */
    45→  costMult?: number;
    46→  /** Bonus Divinity earned on prestige. */
    47→  divinityMult?: number;
    48→}
    49→
    50→export interface ConstellationDef {
    51→  /** Slug id, e.g. "orion". */
    52→  id: string;
    53→  /** Display name, e.g. "Orion". */
    54→  name: string;
    55→  /** IAU designation (Latin), e.g. "Orion". */
    56→  i18nName: string;
    57→  /** Stars with normalized 0-100 coordinates on the star map. */
    58→  stars: ConstellationStar[];
    59→  /** Pairs of star ids that form the constellation's connecting lines. */
    60→  connections: [string, string][];
    61→  /** Permanent reward for completing this constellation. */
    62→  bonus: ConstellationBonus;
    63→  /** Flavor grouping. */
    64→  category: ConstellationCategory;
    65→}

    67→// ---------------------------------------------------------------------------
    68→// COST MODEL CONSTANTS
    69→// ---------------------------------------------------------------------------
    70→// **Critical**: matches the user-supplied spec exactly.
    71→//   baseCost = 10 Divinity.
    72→//   withinConstellationGrowth = 1.5 — each subsequent star in the SAME
    73→//     constellation multiplies the cost by 1.5 (cumulative).
    74→//   costGrowth = 2.0 — multiplier per OTHER started constellation (i.e. any
    75→//     constellation the player has already begun illuminating that isn't the
    76→//     one being illuminated right now).
    77→//
    78→// Concretely (baseCost = 10):
    79→//   A·star1:  10 × 1.5^0 × 2.0^0 = 10       (A is the first started → N=0)
    80→//   A·star2:  10 × 1.5^1 × 2.0^0 = 15        (A is the only started, N=0 still)
    81→//   A·star3:  10 × 1.5^2 × 2.0^0 = 22.5
    82→//   B·star1:  10 × 1.5^0 × 2.0^1 = 20        (A is started, B is new → N=1)
    83→//   B·star2:  10 × 1.5^1 × 2.0^1 = 30        (A+B started, B excluded → N=1)
    84→//   Cass·star1: 10 × 1.5^0 × 2.0^2 = 40      (A+B started, Cass is new → N=2)
    85→//
    86→// Where N = (# of started constellations) - (1 if THIS constellation is
    87→// already started, else 0). I.e. N counts OTHER started constellations.
    88→//
    89→// The "new constellation penalty" the user describes is the costGrowth^N
    90→// term: it only kicks in when this is a brand-new constellation (because at
    91→// that point every other started constellation counts toward N). For
    92→// already-started constellations, N stays fixed at the count when you began
    93→// that constellation, which means subsequent stars in a "penalized"
    94→// constellation inherit the same N multiplier — encouraging the player to
    95→// finish a constellation before starting a new one.
    96→// ---------------------------------------------------------------------------
    97→export const CONSTELLATION_BASE_COST = 10;
    98→export const WITHIN_CONSTELLATION_GROWTH = 1.5;
    99→export const COST_GROWTH_PER_STARTED = 2.0;

   781→// ---------------------------------------------------------------------------
   782→// COST MODEL
   783→// ---------------------------------------------------------------------------
   784→
   785→/**
   786→ * Shape of the constellation-related state used by the cost model and bonus
   787→ * aggregation. Matches the GameState subset, but parameterized for testing.
   788→ */
   789→export interface ConstellationState {
   790→  constellationStars: Record<string, boolean>;
   791→  constellationProgress: Record<string, number>;
   792→  constellationsStarted: Record<string, boolean>;
   793→  constellationsCompleted: Record<string, boolean>;
   794→}
   795→
   796→/**
   797→ * Compute the Divinity cost of illuminating the next star in a constellation.
   798→ *
   799→ * Formula:
   800→ *   baseCost × (withinConstellationGrowth ^ litInThisConstellation)
   801→ *             × (costGrowth ^ N)
   802→ *
   803→ * Where:
   804→ *   `litInThisConstellation` = count of already-lit stars in `constellationId`.
   805→ *   `N` = number of OTHER started constellations (i.e. startedCount minus 1 if
   806→ *        this constellation is already started). This is the "new constellation
   807→ *        penalty" — it kicks in when starting a brand-new constellation
   808→ *        (because every other started constellation counts toward N) and STICKS
   809→ *        for every subsequent star within that constellation, encouraging the
   810→ *        player to finish a constellation before starting a new one.
   811→ *
   812→ * Concrete example (baseCost = 10):
   813→ *   A·star1 (no started):        10 × 1.5^0 × 2.0^0 = 10
   814→ *   A·star2 (A started):          10 × 1.5^1 × 2.0^0 = 15
   815→ *   A·star3 (A started):          10 × 1.5^2 × 2.0^0 = 22.5
   816→ *   B·star1 (A started, B new):   10 × 1.5^0 × 2.0^1 = 20  (new constellation)
   817→ *   B·star2 (A+B started):         10 × 1.5^1 × 2.0^1 = 30  (penalty sticks!)
   818→ *   Cass·star1 (A+B started):      10 × 1.5^0 × 2.0^2 = 40  (two others)
   819→ */
   820→export function starCost(
   821→  constellationId: string,
   822→  state: ConstellationState,
   823→  starId?: string,
   824→  opts?: { baseCost?: number; foresightDiscountMult?: number }
   825→): number {
   826→  const baseCost = opts?.baseCost ?? CONSTELLATION_BASE_COST;
   827→  const litInThis = state.constellationProgress?.[constellationId] || 0;
   828→  // N = number of OTHER started constellations (exclude the current one if
   829→  // already started). This is the "new constellation penalty" — it sticks
   830→  // for every star within a started constellation, so switching to B before
   831→  // finishing A makes EVERY B star cost 2× more, not just the first.
   832→  const startedCount = Object.keys(state.constellationsStarted || {}).filter(
   833→    (k) => state.constellationsStarted[k]
   834→  ).length;
   835→  const thisStarted = !!state.constellationsStarted?.[constellationId];
   836→  const N = Math.max(0, startedCount - (thisStarted ? 1 : 0));
   837→  let cost =
   838→    baseCost *
   839→    Math.pow(WITHIN_CONSTELLATION_GROWTH, litInThis) *
   840→    Math.pow(COST_GROWTH_PER_STARTED, N);
   841→  // LAYER UPGRADE — foresight_discount: multiplicative -20% per level (max 5).
   842→  if (opts?.foresightDiscountMult && opts.foresightDiscountMult < 1) {
   843→    cost *= opts.foresightDiscountMult;
   844→  }
   845→  return Math.ceil(cost);
   846→}

   848→// ---------------------------------------------------------------------------
   849→// BONUS AGGREGATION
   850→// ---------------------------------------------------------------------------
   851→
   852→export interface ConstellationAggregateBonus {
   853→  productionMult: number;
   854→  capMult: number;
   855→  epMult: number;
   856→  popGrowthMult: number;
   857→  costMult: number;
   858→  divinityMult: number;
   859→}
   860→
   861→/**
   862→ * Sum of:
   863→ *   - per-star production bonus (+0.5% per lit star).
   864→ *   - per-completed-constellation bonus (defined in `bonus` field).
   865→ *
   866→ * The first argument is the constellation subset of GameState. Pass the L2
   867→ * store state directly to compute the live bonus.
   868→ */
   869→export function constellationBonus(state: ConstellationState): ConstellationAggregateBonus {
   870→  const out: ConstellationAggregateBonus = {
   871→    productionMult: 0,
   872→    capMult: 0,
   873→    epMult: 0,
   874→    popGrowthMult: 0,
   875→    costMult: 0,
   876→    divinityMult: 0,
   877→  };
   878→  // Per-star production bonus (cumulative across all lit stars).
   879→  const litStars = Object.values(state.constellationStars || {}).filter(Boolean).length;
   880→  out.productionMult += litStars * PER_STAR_PRODUCTION_BONUS;
   881→  // Per-completed-constellation bonus.
   882→  for (const c of CONSTELLATIONS) {
   883→    if (!state.constellationsCompleted?.[c.id]) continue;
   884→    out.productionMult += c.bonus.productionMult || 0;
   885→    out.capMult += c.bonus.capMult || 0;
   886→    out.epMult += c.bonus.epMult || 0;
   887→    out.popGrowthMult += c.bonus.popGrowthMult || 0;
   888→    out.costMult += c.bonus.costMult || 0;
   889→    out.divinityMult += c.bonus.divinityMult || 0;
   890→  }
   891→  return out;
   892→}

   894→/** Count how many constellations are fully completed. */
   895→export function countCompletedConstellations(state: ConstellationState): number {
   896→  return CONSTELLATIONS.filter((c) => state.constellationsCompleted?.[c.id]).length;
   897→}
   898→
   899→/** Count total lit stars across all constellations. */
   900→export function countLitStars(state: ConstellationState): number {
   901→  return Object.values(state.constellationStars || {}).filter(Boolean).length;
   902→}

   904→/**
   905→ * Returns true if the player has completed enough constellations to advance
   906→ * to Layer 3 (Transcendence).
   907→ */
   908→export function canAdvanceToTranscendence(state: ConstellationState): boolean {
   909→  return countCompletedConstellations(state) >= CONSTELLATIONS_REQUIRED_FOR_LAYER_3;
