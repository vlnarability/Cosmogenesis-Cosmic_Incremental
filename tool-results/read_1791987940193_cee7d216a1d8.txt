     1→"use client";
     2→

     4→import { useGameStore } from "@/game/state/store";
     5→import {
     6→  RELIC_LOADOUTS,
     7→  relicBonus,
     8→} from "@/game/data/singularity";
     9→import {
    10→  UNIVERSAL_GALAXIES,
    11→  UNIVERSAL_GALAXY_MAP,

    22→  galaxiesInAnchorRadius,

    25→  totalNewSystemsClaimed,
    26→  totalRivalSystemsClaimed,
    27→  totalUniversalSystemsClaimed,
    28→  type UniversalGalaxy,
    29→} from "@/game/data/universalMap";
    30→import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription } from "@/components/ui/dialog";
    31→import { Button } from "@/components/ui/button";
    32→import { Badge } from "@/components/ui/badge";
    33→import { Separator } from "@/components/ui/separator";
    34→import {
    35→  Lock, CheckCircle2, Anchor, Shield, Swords, Orbit, Sparkles, Crosshair, Globe2,
    36→} from "lucide-react";
    37→import { formatNumber, formatCompact } from "../shared/format";
    38→import { BackgroundLayerBadge } from "../shared/BackgroundLayerBadge";
    39→
    40→// SVG viewport for the universal map (galaxy pos is in 0-100 units).
    41→const MAP_SIZE = 100;
    42→
    43→export function SingularityModal() {
    44→  const show = useGameStore((s) => s.showSingularity);
    45→  const setShow = useGameStore((s) => s.setShowSingularity);
    46→  const unlockedLayers = useGameStore((s) => s.unlockedLayers);
    47→  const singularityCores = useGameStore((s) => s.singularityCores);
    48→  const faith = useGameStore((s) => s.faith);
    49→  const equippedRelics = useGameStore((s) => s.equippedRelics || {});
    50→
    51→  // L6 Universal Map runtime state
    52→  const universalGalaxies = useGameStore((s) => s.universalGalaxies || []);
    53→  const rifts = useGameStore((s) => s.rifts || []);
    54→  const realityAnchors = useGameStore((s) => s.realityAnchors || []);
    55→  const riftsClosed = useGameStore((s) => s.riftsClosed || 0);
    56→  const anchorsPlaced = useGameStore((s) => s.anchorsPlaced || 0);
    57→  const riftSpawnTimer = useGameStore((s) => s.riftSpawnTimer || 0);
    58→
    59→  // Actions
    60→  const toggleRelicLoadout = useGameStore((s) => s.toggleRelicLoadout);
    61→  const closeRift = useGameStore((s) => s.closeRift);
    62→  const placeRealityAnchor = useGameStore((s) => s.placeRealityAnchor);
    63→  const attackUniversalGalaxy = useGameStore((s) => s.attackUniversalGalaxy);
    64→
    65→  // Selected galaxy (for the detail panel). null = none.
    66→  const [selectedGalaxyId, setSelectedGalaxyId] = useState<string | null>(null);
    67→
    68→  const isUnlocked = !!unlockedLayers.singularity;
    69→  const rb = relicBonus(equippedRelics);
    70→  const equippedCount = RELIC_LOADOUTS.filter((r) => equippedRelics[r.id]).length;
    71→
    72→  // Build lookup maps for fast access in the SVG layer.
    73→  const stateById = useMemo(() => {
    74→    const m: Record<string, (typeof universalGalaxies)[number]> = {};
    75→    for (const g of universalGalaxies) m[g.id] = g;
    76→    return m;
    77→  }, [universalGalaxies]);
    78→
    79→  const riftsByGalaxyId = useMemo(() => {
    80→    const m: Record<string, (typeof rifts)[number][]> = {};
    81→    for (const r of rifts) {
    82→      if (!m[r.galaxyId]) m[r.galaxyId] = [];
    83→      m[r.galaxyId].push(r);
    84→    }
    85→    return m;
    86→  }, [rifts]);
    87→
    88→  const anchorGalaxyIds = useMemo(() => realityAnchors.map((a) => a.galaxyId), [realityAnchors]);
    89→
    90→  // Aggregate stats for the sidebar.
    91→  const totalSystemsClaimed = totalUniversalSystemsClaimed(universalGalaxies);
    92→  const newSystemsClaimed = totalNewSystemsClaimed(universalGalaxies);
    93→  const rivalSystemsClaimed = totalRivalSystemsClaimed(universalGalaxies);
    94→  const worldBonusFromL6 = newSystemsClaimed * WORLDS_PER_NEW_SYSTEM_L6;
    95→
    96→  const nextAnchorCost = anchorPlaceCost(anchorsPlaced);
    97→  const l6Complete = isUniversalMapComplete(riftsClosed, anchorsPlaced);
    98→
    99→  const selectedGalaxy = selectedGalaxyId ? UNIVERSAL_GALAXY_MAP[selectedGalaxyId] : null;
   100→  const selectedState = selectedGalaxyId ? stateById[selectedGalaxyId] : null;
   101→  const selectedRifts = selectedGalaxyId ? (riftsByGalaxyId[selectedGalaxyId] || []) : [];
   102→  const selectedAnchored = selectedGalaxyId ? isGalaxyAnchored(selectedGalaxyId, anchorGalaxyIds) : false;
   103→  const selectedIsNewL6 = !!selectedGalaxy?.isNewL6;
   104→  const selectedAttackCost = selectedIsNewL6
   105→    ? newGalaxyAttackCost(newSystemsClaimed)
   106→    : 0;
   107→
   108→  // SVG-friendly coordinate helper: galaxy.pos (0-100) → svg viewBox (0-100).
   109→  const galaxyColor = (g: UniversalGalaxy): string => {
   110→    const st = stateById[g.id];
   111→    const owner = st?.ownerId || g.ownerId;
   112→    if (owner === "player") return "#fbbf24"; // amber-400 — owned by player
   113→    if (owner.startsWith("rival_")) return "#f472b6"; // pink-400 — rival-owned
   114→    return "#475569"; // slate-600 — neutral
   115→  };
   116→
   117→  return (
   118→    <Dialog open={show} onOpenChange={setShow}>
   119→      <DialogContent className="max-w-6xl max-h-[90vh] overflow-y-auto glass-strong">
   120→        <DialogHeader>
   121→          <DialogTitle className="flex items-center gap-2 text-lg flex-wrap">
   122→            <Orbit className="w-5 h-5 text-cyan-300" />
   123→            Layer 6 — Singularity (Universal Map)
   124→            <BackgroundLayerBadge layerId="singularity" />
   125→          </DialogTitle>
   126→          <DialogDescription asChild>
   127→            <div className="flex items-center justify-between flex-wrap gap-2">
   128→              <span>
   129→                Defend the 50-galaxy universe from Old-God rifts. Close rifts (Singularity Cores),
   130→                place Reality Anchors (Singularity Cores) to immunize galaxies, and conquer
   131→                neutral galaxies (Faith) for <span className="text-cyan-300">+198 worlds</span> each.
   132→              </span>
   133→              <div className="flex gap-1 flex-wrap">
   134→                <Badge variant="outline" className="text-cyan-300 border-cyan-400/40">
   135→                  <Sparkles className="w-3 h-3 mr-0.5" /> {formatNumber(singularityCores, 0)} Cores
   136→                </Badge>
   137→                <Badge variant="outline" className={l6Complete ? "text-emerald-300 border-emerald-400/40" : "text-amber-300 border-amber-400/40"}>
   138→                  <Shield className="w-3 h-3 mr-0.5" /> {riftsClosed}/{RIFTS_REQUIRED_FOR_L7} rifts
   139→                </Badge>
   140→                <Badge variant="outline" className={l6Complete ? "text-emerald-300 border-emerald-400/40" : "text-amber-300 border-amber-400/40"}>
   141→                  <Anchor className="w-3 h-3 mr-0.5" /> {anchorsPlaced}/{ANCHORS_REQUIRED_FOR_L7} anchors
   142→                </Badge>
   143→              </div>
   144→            </div>
   145→          </DialogDescription>
   146→        </DialogHeader>
   147→
   148→        {!isUnlocked ? (
   149→          <div className="text-center text-muted-foreground py-10">
   150→            <Lock className="w-8 h-8 mx-auto mb-2 opacity-50" />
   151→            <p className="text-sm">Singularity is sealed.</p>
   152→            <p className="text-xs mt-1">Enact 3+ Divine Laws (Layer 5) to unlock Singularity.</p>
   153→          </div>
   154→        ) : (
   155→          <div className="grid grid-cols-1 lg:grid-cols-[1fr_280px] gap-3">
   156→            {/* ===== UNIVERSAL MAP (SVG) ===== */}
   157→            <section className="stat-card p-2">
   158→              <div className="flex items-center justify-between mb-2">
   159→                <div className="text-[0.7rem] uppercase tracking-wide text-cyan-300/80 flex items-center gap-1">
   160→                  <Globe2 className="w-3 h-3" /> Universal Map — {TOTAL_UNIVERSAL_GALAXIES} galaxies ({TOTAL_NEW_GALAXIES_L6} new)
   161→                </div>
   162→                <Badge variant="outline" className="text-[0.6rem] text-violet-300 border-violet-400/40">
   163→                  🌀 Next rift in {Math.max(0, Math.ceil(riftSpawnTimer))}s
   164→                </Badge>
   165→              </div>
   166→
   167→              <div className="relative w-full aspect-square rounded-md border border-cyan-400/20 bg-gradient-to-br from-slate-950 via-indigo-950/40 to-slate-950 overflow-hidden">
   168→                <svg viewBox={`0 0 ${MAP_SIZE} ${MAP_SIZE}`} className="w-full h-full">
   169→                  {/* Anchor coverage rings — drawn first so galaxies sit on top */}
   170→                  {realityAnchors.map((a) => {
   171→                    const g = UNIVERSAL_GALAXY_MAP[a.galaxyId];
   172→                    if (!g) return null;
   173→                    return (
   174→                      <circle
   175→                        key={`anchor-ring-${a.id}`}
   176→                        cx={g.pos.x}
   177→                        cy={g.pos.y}
   178→                        r={ANCHOR_RADIUS}
   179→                        fill="rgba(59, 130, 246, 0.08)"
   180→                        stroke="rgba(96, 165, 250, 0.45)"
   181→                        strokeWidth={0.4}
   182→                        strokeDasharray="1 1.5"
   183→                      />
   184→                    );
   185→                  })}
   186→
   187→                  {/* Galaxies */}
   188→                  {UNIVERSAL_GALAXIES.map((g) => {
   189→                    const st = stateById[g.id];
   190→                    const riftList = riftsByGalaxyId[g.id] || [];
   191→                    const anchored = isGalaxyAnchored(g.id, anchorGalaxyIds);
   192→                    const owned = (st?.ownerId || g.ownerId) === "player";
   193→                    const radius = owned ? 1.8 : 1.3;
   194→                    const fill = galaxyColor(g);
   195→                    const isSelected = selectedGalaxyId === g.id;
   196→                    return (
   197→                      <g
   198→                        key={g.id}
   199→                        onClick={() => setSelectedGalaxyId(g.id)}
   200→                        style={{ cursor: "pointer" }}
   201→                      >
   202→                        {/* Selection highlight */}
   203→                        {isSelected && (
   204→                          <circle
   205→                            cx={g.pos.x}
   206→                            cy={g.pos.y}
   207→                            r={radius + 1.5}
   208→                            fill="none"
   209→                            stroke="#fcd34d"
   210→                            strokeWidth={0.5}
   211→                          />
   212→                        )}
   213→                        {/* Anchor indicator ring around the galaxy itself */}
   214→                        {anchored && (
   215→                          <circle
   216→                            cx={g.pos.x}
   217→                            cy={g.pos.y}
   218→                            r={radius + 1}
   219→                            fill="none"
   220→                            stroke="#60a5fa"
   221→                            strokeWidth={0.6}
   222→                            className="singularity-anchor-glow"
   223→                          />
   224→                        )}
   225→                        {/* Galaxy body */}
   226→                        <circle
   227→                          cx={g.pos.x}
   228→                          cy={g.pos.y}
   229→                          r={radius}
   230→                          fill={fill}
   231→                          stroke={owned ? "#fde68a" : "rgba(148, 163, 184, 0.4)"}
   232→                          strokeWidth={owned ? 0.4 : 0.2}
   233→                          opacity={owned ? 1 : 0.7}
   234→                        />
   235→                        {/* Rift pulse — red overlay drawn on top of the galaxy */}
   236→                        {riftList.length > 0 && (
   237→                          <>
   238→                            <circle
   239→                              cx={g.pos.x}
   240→                              cy={g.pos.y}
   241→                              r={radius + 0.8}
   242→                              fill="none"
   243→                              stroke="#ef4444"
   244→                              strokeWidth={0.5}
   245→                              className="singularity-rift-pulse"
   246→                            />
   247→                            <circle
   248→                              cx={g.pos.x}
   249→                              cy={g.pos.y}
   250→                              r={radius + 1.6}
   251→                              fill="none"
   252→                              stroke="rgba(239, 68, 68, 0.4)"
   253→                              strokeWidth={0.3}
   254→                              className="singularity-rift-pulse-outer"
   255→                            />
   256→                          </>
   257→                        )}
   258→                        {/* Heresy tint — opacity scales with heresy level */}
   259→                        {((st?.heresy || 0) > 0) && (
   260→                          <circle
   261→                            cx={g.pos.x}
   262→                            cy={g.pos.y}
   263→                            r={radius}
   264→                            fill="#7c1d1d"
   265→                            opacity={Math.min(0.7, (st?.heresy || 0) / 100 * 0.7)}
   266→                          />
   267→                        )}
   268→                      </g>
   269→                    );
   270→                  })}
   271→                </svg>
   272→
   273→                {/* Map legend */}
   274→                <div className="absolute bottom-1 left-1 right-1 flex flex-wrap gap-1.5 text-[0.55rem]">
   275→                  <span className="px-1.5 py-0.5 rounded bg-slate-900/70 border border-slate-700/50 text-amber-300 flex items-center gap-0.5">
   276→                    <span className="inline-block w-1.5 h-1.5 rounded-full bg-amber-400" /> Owned
   277→                  </span>
   278→                  <span className="px-1.5 py-0.5 rounded bg-slate-900/70 border border-slate-700/50 text-slate-300 flex items-center gap-0.5">
   279→                    <span className="inline-block w-1.5 h-1.5 rounded-full bg-slate-500" /> Neutral
   280→                  </span>
   281→                  <span className="px-1.5 py-0.5 rounded bg-slate-900/70 border border-slate-700/50 text-pink-300 flex items-center gap-0.5">
   282→                    <span className="inline-block w-1.5 h-1.5 rounded-full bg-pink-400" /> Rival
   283→                  </span>
   284→                  <span className="px-1.5 py-0.5 rounded bg-slate-900/70 border border-slate-700/50 text-blue-300 flex items-center gap-0.5">
   285→                    <Anchor className="w-2.5 h-2.5" /> Anchored
   286→                  </span>
   287→                  <span className="px-1.5 py-0.5 rounded bg-slate-900/70 border border-slate-700/50 text-red-300 flex items-center gap-0.5">
   288→                    <span className="inline-block w-1.5 h-1.5 rounded-full bg-red-500 animate-pulse" /> Rift
   289→                  </span>
   290→                </div>
   291→              </div>
   292→
   293→              {/* Selected-galaxy detail / actions */}
   294→              {selectedGalaxy && selectedState ? (
   295→                <div className="mt-2 rounded-md border border-cyan-400/30 bg-cyan-950/20 p-2 space-y-1.5">
   296→                  <div className="flex items-center justify-between">
   297→                    <div className="text-sm font-semibold text-cyan-200">
   298→                      {selectedGalaxy.name}
   299→                      {selectedGalaxy.isNewL6 && (
   300→                        <Badge variant="outline" className="ml-1 text-[0.55rem] text-violet-300 border-violet-400/40">
   301→                          NEW L6
   302→                        </Badge>
   303→                      )}
   304→                    </div>
   305→                    <div className="text-[0.6rem] text-muted-foreground">
   306→                      {selectedGalaxy.tagline}
   307→                    </div>
   308→                  </div>
   309→                  <div className="grid grid-cols-2 gap-x-3 gap-y-0.5 text-[0.65rem] font-mono">
   310→                    <div className="flex justify-between">
   311→                      <span className="text-muted-foreground">Owner</span>
   312→                      <span className="text-amber-300">{selectedState.ownerId}</span>
   313→                    </div>
   314→                    <div className="flex justify-between">
   315→                      <span className="text-muted-foreground">Systems</span>
   316→                      <span className="text-cyan-300">
   317→                        {selectedState.systemsClaimed}/{selectedGalaxy.systemCount}
   318→                      </span>
   319→                    </div>
   320→                    <div className="flex justify-between">
   321→                      <span className="text-muted-foreground">Heresy</span>
   322→                      <span className={selectedState.heresy > 50 ? "text-red-300" : "text-slate-300"}>
   323→                        {Math.floor(selectedState.heresy)}%
   324→                      </span>
   325→                    </div>
   326→                    <div className="flex justify-between">
   327→                      <span className="text-muted-foreground">Anchored</span>
   328→                      <span className={selectedAnchored ? "text-blue-300" : "text-slate-500"}>
   329→                        {selectedAnchored ? "YES" : "no"}
   330→                      </span>
   331→                    </div>
   332→                  </div>
   333→
   334→                  {/* Active rifts on this galaxy — close buttons */}
   335→                  {selectedRifts.length > 0 && (
   336→                    <div className="space-y-1 pt-1 border-t border-cyan-400/20">
   337→                      <div className="text-[0.6rem] uppercase text-red-300/80 flex items-center gap-1">
   338→                        <Crosshair className="w-2.5 h-2.5" /> {selectedRifts.length} active rift{selectedRifts.length === 1 ? "" : "s"}
   339→                      </div>
   340→                      {selectedRifts.map((r) => {
   341→                        const cost = riftCloseCost(r.severity);
   342→                        const afford = singularityCores >= cost;
   343→                        return (
   344→                          <div key={r.id} className="flex items-center justify-between gap-2 text-[0.65rem]">
   345→                            <span className="text-red-300">
   346→                              Severity {r.severity} — close cost {cost} Core{cost === 1 ? "" : "s"}
   347→                            </span>
   348→                            <Button
   349→                              size="sm"
   350→                              variant="destructive"
   351→                              className="h-6 text-[0.6rem] px-2"
   352→                              disabled={!afford}
   353→                              onClick={() => {
   354→                                closeRift(r.id);
   355→                              }}
   356→                            >
   357→                              <Shield className="w-2.5 h-2.5 mr-0.5" /> Close
   358→                            </Button>
   359→                          </div>
   360→                        );
   361→                      })}
   362→                    </div>
   363→                  )}
   364→
   365→                  {/* Anchor placement */}
   366→                  {!selectedAnchored && (
   367→                    <div className="pt-1 border-t border-cyan-400/20 flex items-center justify-between gap-2 text-[0.65rem]">
   368→                      <span className="text-blue-300">
   369→                        Anchor this galaxy (+ covers {galaxiesInAnchorRadius(selectedGalaxy.id).length} galaxies in {ANCHOR_RADIUS}u radius)
   370→                      </span>
   371→                      <Button
   372→                        size="sm"
   373→                        variant="outline"
   374→                        className="h-6 text-[0.6rem] px-2 border-blue-400/40 text-blue-300"
   375→                        disabled={singularityCores < nextAnchorCost}
   376→                        onClick={() => placeRealityAnchor(selectedGalaxy.id)}
   377→                        title={`Place Reality Anchor on ${selectedGalaxy.name} — cost ${nextAnchorCost} Singularity Cores`}
   378→                      >
   379→                        <Anchor className="w-2.5 h-2.5 mr-0.5" /> {nextAnchorCost}
   380→                      </Button>
   381→                    </div>
   382→                  )}
   383→                  {selectedAnchored && (
   384→                    <div className="pt-1 border-t border-cyan-400/20 text-[0.6rem] text-blue-300/80 flex items-center gap-1">
   385→                      <CheckCircle2 className="w-2.5 h-2.5" /> Reality Anchor in place — immune to rift corruption.
   386→                    </div>
   387→                  )}
   388→
   389→                  {/* Attack — only for new neutral L6 galaxies */}
   390→                  {selectedIsNewL6 && selectedState.systemsClaimed < selectedGalaxy.systemCount && (
   391→                    <div className="pt-1 border-t border-cyan-400/20 flex items-center justify-between gap-2 text-[0.65rem]">
   392→                      <span className="text-amber-300">
   393→                        Claim system {selectedState.systemsClaimed + 1}/{selectedGalaxy.systemCount} (+198 worlds)
   394→                      </span>
   395→                      <Button
   396→                        size="sm"
   397→                        variant="default"
   398→                        className="h-6 text-[0.6rem] px-2 bg-amber-600/70 hover:bg-amber-600"
   399→                        disabled={faith < selectedAttackCost}
   400→                        onClick={() => attackUniversalGalaxy(selectedGalaxy.id)}
   401→                        title={`Launch attack on ${selectedGalaxy.name} — cost ${selectedAttackCost} Faith`}
   402→                      >
   403→                        <Swords className="w-2.5 h-2.5 mr-0.5" /> {selectedAttackCost} Faith
   404→                      </Button>
   405→                    </div>
   406→                  )}
   407→                  {selectedIsNewL6 && selectedState.systemsClaimed >= selectedGalaxy.systemCount && (
   408→                    <div className="pt-1 border-t border-cyan-400/20 text-[0.6rem] text-emerald-300/80 flex items-center gap-1">
   409→                      <CheckCircle2 className="w-2.5 h-2.5" /> Galaxy fully claimed (+{(selectedGalaxy.systemCount * WORLDS_PER_NEW_SYSTEM_L6).toLocaleString()} worlds total).
   410→                    </div>
   411→                  )}
   412→                </div>
   413→              ) : (
   414→                <div className="mt-2 rounded-md border border-cyan-400/10 bg-cyan-950/10 p-2 text-[0.65rem] text-muted-foreground text-center">
   415→                  Click a galaxy on the map to inspect it, close rifts, place anchors, or launch attacks.
   416→                </div>
   417→              )}
   418→            </section>
   419→
   420→            {/* ===== SIDEBAR ===== */}
   421→            <aside className="space-y-3">
   422→              {/* Universal Map progress */}
   423→              <div className="stat-card">
   424→                <div className="text-[0.7rem] uppercase tracking-wide text-muted-foreground mb-1">
   425→                  L6 → L7 progress
   426→                </div>
   427→                <div className="space-y-1.5">
   428→                  <div>
   429→                    <div className="flex justify-between text-[0.65rem] mb-0.5">
   430→                      <span className="text-red-300/80">Rifts closed</span>
   431→                      <span className="text-red-200 font-mono">{riftsClosed}/{RIFTS_REQUIRED_FOR_L7}</span>
   432→                    </div>
   433→                    <div className="h-1 rounded-full bg-muted/30 overflow-hidden">
   434→                      <div
   435→                        className="h-full bg-gradient-to-r from-red-500 to-rose-400 transition-all"
   436→                        style={{ width: `${Math.min(100, (riftsClosed / RIFTS_REQUIRED_FOR_L7) * 100)}%` }}
   437→                      />
   438→                    </div>
   439→                  </div>
   440→                  <div>
   441→                    <div className="flex justify-between text-[0.65rem] mb-0.5">
   442→                      <span className="text-blue-300/80">Anchors placed</span>
   443→                      <span className="text-blue-200 font-mono">{anchorsPlaced}/{ANCHORS_REQUIRED_FOR_L7}</span>
   444→                    </div>
   445→                    <div className="h-1 rounded-full bg-muted/30 overflow-hidden">
   446→                      <div
   447→                        className="h-full bg-gradient-to-r from-blue-500 to-cyan-400 transition-all"
   448→                        style={{ width: `${Math.min(100, (anchorsPlaced / ANCHORS_REQUIRED_FOR_L7) * 100)}%` }}
   449→                      />
   450→                    </div>
