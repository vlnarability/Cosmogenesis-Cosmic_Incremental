     1→// Apotheosis — Layer 5 prestige content
     2→//
     3→// When a god reaches Apotheosis, civilization recognizes it. The creator
     4→// is no longer silent: Divine Laws bind the run, Worship Modes shape how
     5→// mortals pray, Miracles intervene directly, and Heresy rises whenever
     6→// belief fractures. Manage all four and the god ascends to Singularity.
     7→//
     8→// Four mechanic families live in this file:
     9→//
    10→//   1. Divine Laws (8)        — fundamental rule changes, permanent for
    11→//                               the run, cost Divinity. Up to 8 codified
    12→//                               per run (one of each).
    13→//   2. Worship Modes (6)      — pick one at run start; decides how mortals
    14→//                               relate to the god and which bonuses apply.
    15→//   3. Miracles (5)           — dramatic interventions that cost Divinity,
    16→//                               each with a per-run usage limit.
    17→//   4. Heresy Responses (4)   — manage rising heresy (0..100). The run
    18→//                               ends at HERESY_MAX (100); these tools
    19→//                               reduce heresy at various costs.
    20→//
    21→// State fields (planned for GameState — to be wired in a follow-up task):
    22→//   - lawsCodified:        Record<string, boolean>  // laws codified this run (one of each, max 8)
    23→//   - worshipMode:         string | null             // active worship mode id, or null before pick
    24→//   - miracleUses:         Record<string, number>   // times each miracle used this run
    25→//   - miracleTimers:       Record<string, number>   // remaining seconds for active temporary miracles
    26→//   - heresyResponseUses:  Record<string, number>   // times each heresy response used this run
    27→//   - heresy:              number                    // current heresy level (0..100)
    28→//   - heresySchismResolved: boolean                  // whether a schism was resolved this run
    29→//
    30→// Reference: /home/z/my-project/upload/data.js DATA.DIVINE_LAWS (lines 204-210),
    31→// DATA.WORSHIP_MODES (lines 196-203), DATA.MIRACLES (lines 211-218), and
    32→// DATA.HERESY_RESPONSES (lines 219-224) for the original vanilla-JS shapes.
    33→// The TS port reshapes each mechanic into the Layer 5 spec: 8 / 6 / 5 / 4
    34→// entries with explicit Divinity costs and per-run limits, distinct from
    35→// the vanilla JS (which had 5 / 6 / 6 / 4 entries with sub-currency costs).

    37→// Advancement to Layer 6 (Singularity) requires codifying 3 Divine Laws

    48→/**
    49→ * Base heresy accumulation rate, in points per second, before any laws
    50→ * or worship modes modify it. Effective rate each tick is
    51→ * `BASE_HERESY_RATE * (1 + heresyRateMult(lawsCodified, worshipMode))`.
    52→ *
    53→ * At 0.05/s with no modifiers, heresy fills the 0..100 meter in
    54→ * 100 / 0.05 = 2000 seconds (~33 minutes). Laws and worship modes can
    55→ * speed this up (Zeal +50%, Fear has no effect), slow it down (Order -50%,
    56→ * Mystery worship mode -25%), or even reverse it (Mercy: heresy decays).
    57→ */
    58→export const BASE_HERESY_RATE: number = 0.05;
    59→
    60→/**
    61→ * Maximum heresy value. When `state.heresy` reaches this number the run
    62→ * ends — the god is rejected by creation (a "schism" too large to
    63→ * resolve). The player should manage heresy via Heresy Responses before
    64→ * it hits this cap.
    65→ */
    66→export const HERESY_MAX: number = 100;

    76→// =====================================================================
    77→// 1. Divine Laws (8) — fundamental rule changes, permanent for the run
    78→// =====================================================================
    79→
    80→/**
    81→ * A single Divine Law. Effects are spread across optional fields; the
    82→ * IIFE validator enforces that each law sets at least one. Fields map
    83→ * to consumer wiring as follows:
    84→ *
    85→ *   - `productionMult`        → additive all-production mult (Abundance, Tithe)
    86→ *   - `populationGrowthMult`  → additive pop-growth mult (Fertility)
    87→ *   - `militaryMult`          → additive military mult (Zeal)
    88→ *   - `scienceMult`           → additive science mult (Revelation)
    89→ *   - `goldMult`              → additive gold mult (Tithe)
    90→ *   - `worshipMult`           → multiplicative worship-bonus mult (Faith 2.0)
    91→ *   - `happinessFlat`         → flat per-tick happiness bonus (Mercy +10)
    92→ *   - `happinessTarget`       → clamp happiness to this value while codified (Order 80)
    93→ *   - `heresyDecays`          → UI flag: heresy meter shows "Decaying" (Mercy)
    94→ *   - `heresyFlatOnCodify`    → one-time flat heresy change applied at codification (Faith -20)
    95→ *   - `heresyRateMult`        → additive mult on BASE_HERESY_RATE (Zeal +0.50, Order -0.50, Mercy -2.00)
    96→ *
    97→ * `heresyRateMult` semantics: the effective heresy rate per second is
    98→ * `BASE_HERESY_RATE * (1 + sum_of_law_heresyRateMult + mode.heresyRateMult)`.
    99→ * Mercy's -2.00 makes the factor (1 + -2.00) = -1.0, so heresy decays at
   100→ * the base rate (–0.05/s) when Mercy is codified alone.
   101→ */

   118→  militaryMult?: number;

   120→  scienceMult?: number;
   121→  /** Additive gold multiplier (0.25 = +25% gold). */
   122→  goldMult?: number;
   123→  /** Multiplicative worship-bonus multiplier (2.0 = double the worship bonus). */
   124→  worshipMult?: number;

   506→// =====================================================================
   507→// Module-load sanity checks
   508→// =====================================================================
   509→
   510→/**
   511→ * Validate the apotheosis data at module load:
   512→ *   - All ids within each family are unique.
   513→ *   - Each divine law sets at least one effect field, has cost >= 0,
   514→ *     and (if `worshipMult` set) is >= 1.
   515→ *   - Each worship mode sets at least one effect field.
   516→ *   - Each miracle sets exactly one effect (XOR-style check), has
   517→ *     cost > 0 and limitPerRun >= 1.
   518→ *   - Each heresy response has cost >= 0 and limitPerRun >= 1
   519→ *     (Infinity allowed for Tolerate).
   520→ *   - BASE_HERESY_RATE > 0, HERESY_MAX > 0.
   521→ *
   522→ * Throws on any violation (programmer error, not a runtime / save data
   523→ * error). The check is O(N) and runs once, so it's a no-op on the hot
   524→ * path. Matches the pattern established by `transcendence.ts` and
   525→ * `genesis.ts`.
   526→ */
   527→(function validateApotheosisData(): void {
   528→  if (BASE_HERESY_RATE <= 0) {
   529→    throw new Error(`apotheosis.ts: BASE_HERESY_RATE must be > 0 (got ${BASE_HERESY_RATE})`);
   530→  }
   531→  if (HERESY_MAX <= 0) {

   533→  }

   535→    throw new Error(
   536→      `apotheosis.ts: APOTHEOSIS_COMPLETION_LAWS_REQUIRED must be >= 1 (got ${APOTHEOSIS_COMPLETION_LAWS_REQUIRED})`
   537→    );
   538→  }
   539→
   540→  const seenLaw = new Set<string>();
   541→  for (const l of DIVINE_LAWS) {
   542→    if (seenLaw.has(l.id)) {
   543→      throw new Error(`apotheosis.ts: duplicate divine law id "${l.id}"`);
   544→    }
   545→    seenLaw.add(l.id);
   546→    if (l.cost < 0) {
   547→      throw new Error(`apotheosis.ts: divine law "${l.id}" has negative cost`);
   548→    }
   549→    if (l.worshipMult !== undefined && l.worshipMult < 1) {
   550→      throw new Error(
   551→        `apotheosis.ts: divine law "${l.id}" worshipMult must be >= 1 (got ${l.worshipMult})`
   552→      );
   553→    }
   554→    if (l.happinessTarget !== undefined && (l.happinessTarget < 0 || l.happinessTarget > 100)) {
   555→      throw new Error(
   556→        `apotheosis.ts: divine law "${l.id}" happinessTarget must be in [0, 100] (got ${l.happinessTarget})`
   557→      );
   558→    }
   559→    const effects = [
   560→      l.productionMult !== undefined,
   561→      l.populationGrowthMult !== undefined,
   562→      l.militaryMult !== undefined,
   563→      l.scienceMult !== undefined,
   564→      l.goldMult !== undefined,
   565→      l.worshipMult !== undefined,
   566→      l.happinessFlat !== undefined,
   567→      l.happinessTarget !== undefined,
   568→      l.heresyDecays !== undefined,
   569→      l.heresyFlatOnCodify !== undefined,
   570→      l.heresyRateMult !== undefined,
   571→    ].filter(Boolean).length;
   572→    if (effects < 1) {
   573→      throw new Error(`apotheosis.ts: divine law "${l.id}" must set at least one effect (set ${effects})`);
   574→    }
   575→  }
   576→
   577→  const seenMode = new Set<string>();
   578→  for (const m of WORSHIP_MODES) {
   579→    if (seenMode.has(m.id)) {
   580→      throw new Error(`apotheosis.ts: duplicate worship mode id "${m.id}"`);
   581→    }
   582→    seenMode.add(m.id);
   583→    if (m.happinessDrainMult !== undefined && m.happinessDrainMult < 0) {
   584→      throw new Error(
   585→        `apotheosis.ts: worship mode "${m.id}" happinessDrainMult must be >= 0 (got ${m.happinessDrainMult})`
   586→      );
   587→    }
   588→    const effects = [
   589→      m.productionMult !== undefined,
   590→      m.populationGrowthMult !== undefined,
   591→      m.militaryMult !== undefined,
   592→      m.scienceMult !== undefined,
   593→      m.goldMult !== undefined,
   594→      m.happinessFlat !== undefined,
   595→      m.happinessDrainMult !== undefined,
   596→      m.heresyRateMult !== undefined,
   597→    ].filter(Boolean).length;
   598→    if (effects < 1) {
   599→      throw new Error(`apotheosis.ts: worship mode "${m.id}" must set at least one effect (set ${effects})`);
   600→    }
   601→  }
   602→
   603→  const seenMiracle = new Set<string>();
   604→  for (const m of MIRACLES) {
   605→    if (seenMiracle.has(m.id)) {
   606→      throw new Error(`apotheosis.ts: duplicate miracle id "${m.id}"`);
   607→    }
   608→    seenMiracle.add(m.id);
   609→    if (m.cost <= 0) {
   610→      throw new Error(`apotheosis.ts: miracle "${m.id}" has non-positive cost`);
   611→    }
   612→    if (m.limitPerRun < 1) {
   613→      throw new Error(`apotheosis.ts: miracle "${m.id}" limitPerRun must be >= 1`);
   614→    }
   615→    if (m.grantAllResources !== undefined && m.grantAllResources <= 0) {
   616→      throw new Error(`apotheosis.ts: miracle "${m.id}" grantAllResources must be > 0`);
   617→    }
   618→    if (m.grantPopulation !== undefined && m.grantPopulation <= 0) {
   619→      throw new Error(`apotheosis.ts: miracle "${m.id}" grantPopulation must be > 0`);
   620→    }
   621→    if (m.setHappiness !== undefined && (m.setHappiness < 0 || m.setHappiness > 100)) {
   622→      throw new Error(
   623→        `apotheosis.ts: miracle "${m.id}" setHappiness must be in [0, 100] (got ${m.setHappiness})`
   624→      );
   625→    }
   626→    if (m.heresyFlat !== undefined && m.heresyFlat >= 0) {
   627→      throw new Error(
   628→        `apotheosis.ts: miracle "${m.id}" heresyFlat must be negative (got ${m.heresyFlat})`
   629→      );
   630→    }
   631→    if (m.productionBoost !== undefined) {
   632→      if (m.productionBoost.mult <= 1) {
   633→        throw new Error(
   634→          `apotheosis.ts: miracle "${m.id}" productionBoost.mult must be > 1 (got ${m.productionBoost.mult})`
   635→        );
   636→      }
   637→      if (m.productionBoost.durationSec <= 0) {
   638→        throw new Error(
   639→          `apotheosis.ts: miracle "${m.id}" productionBoost.durationSec must be > 0`
   640→        );
   641→      }
   642→    }
   643→    const effects = [
   644→      m.grantAllResources !== undefined,
   645→      m.grantPopulation !== undefined,
   646→      m.setHappiness !== undefined,
   647→      m.heresyFlat !== undefined,

   649→    ].filter(Boolean).length;
   650→    if (effects !== 1) {
   651→      throw new Error(
   652→        `apotheosis.ts: miracle "${m.id}" must set exactly one effect (set ${effects})`
   653→      );
   654→    }
   655→  }
   656→
   657→  const seenResponse = new Set<string>();
   658→  for (const r of HERESY_RESPONSES) {
   659→    if (seenResponse.has(r.id)) {
   660→      throw new Error(`apotheosis.ts: duplicate heresy response id "${r.id}"`);
   661→    }
   662→    seenResponse.add(r.id);
   663→    if (r.cost < 0) {
   664→      throw new Error(`apotheosis.ts: heresy response "${r.id}" has negative cost`);
   665→    }
   666→    if (r.limitPerRun < 1) {
   667→      throw new Error(`apotheosis.ts: heresy response "${r.id}" limitPerRun must be >= 1`);
   668→    }
   669→    if (r.heresyFlat !== undefined && r.heresyFlat >= 0 && r.id !== "tolerate") {
   670→      // Tolerate has no effects; non-Tolerate responses should reduce heresy.
   671→      throw new Error(

   675→  }

   682→/**
   683→ * Aggregated bonus object returned by `divineLawBonus`.
   684→ *
   685→ * All mult fields are additive on top of the baseline (0 = no change);
   686→ * `worshipMult` is multiplicative (1 = no change, 2 = doubled).
   687→ *
   688→ * The consumer applies these in the tick loop:
   689→ *   - `productionMult`:        multiply production by `(1 + productionMult)`
   690→ *   - `populationGrowthMult`:  multiply pop growth by `(1 + populationGrowthMult)`
   691→ *   - `militaryMult`:           multiply military by `(1 + militaryMult)`
   692→ *   - `scienceMult`:             multiply science by `(1 + scienceMult)`
   693→ *   - `goldMult`:                multiply gold by `(1 + goldMult)`
   694→ *   - `worshipMult`:             multiply the worship-bonus by `worshipMult`
   695→ *   - `happinessFlat`:          add to happiness per tick (or per second)
   696→ *   - `happinessTarget`:        if non-null, clamp happiness to this value
   697→ *   - `heresyDecays`:            UI flag; if true the heresy meter shows "Decaying"
   698→ *   - `heresyRateMult`:          passed through to `heresyRateMult` helper
   699→ */
   700→export interface DivineLawBonus {
   701→  /** Additive all-production multiplier (Abundance +0.15, Tithe +0.25). */
   702→  productionMult: number;
   703→  /** Additive population-growth multiplier (Fertility +0.30). */
   704→  populationGrowthMult: number;
   705→  /** Additive military-power multiplier (Zeal +0.40). */
   706→  militaryMult: number;
   707→  /** Additive science multiplier (Revelation +0.35). */
   708→  scienceMult: number;
   709→  /** Additive gold multiplier (Tithe +0.25). */
   710→  goldMult: number;
   711→  /** Multiplicative worship-bonus multiplier (Faith = 2.0; default 1.0). */
   712→  worshipMult: number;
   713→  /** Flat per-tick happiness bonus (Mercy +10). */
   714→  happinessFlat: number;
   715→  /** Happiness clamp target (Order: 80; null = no clamp). */
   716→  happinessTarget: number | null;
   717→  /** Whether heresy decays naturally while codified (Mercy). */
   718→  heresyDecays: boolean;
   719→  /** Additive multiplier on BASE_HERESY_RATE (Zeal +0.50; Order -0.50; Mercy -2.00). */
   720→  heresyRateMult: number;
   721→}
   722→
   723→/**
   724→ * Sum the bonuses from all codified Divine Laws into a single aggregate.
   725→ *
   726→ * All mult fields are additive (0 = no change) EXCEPT `worshipMult`,
   727→ * which is multiplicative (1 = no change, 2 = doubled by Faith).
   728→ * Compounding multiple laws that set `worshipMult` (currently only
   729→ * Faith) would multiply them; with only Faith in the dataset, the
   730→ * value is 1.0 (no Faith) or 2.0 (Faith codified).
   731→ *
   732→ * Used by the tick loop to compute per-resource multipliers, the
   733→ * pop-growth rate, happiness clamps, and the worship-bonus multiplier.
   734→ * Also feeds the standalone `heresyRateMult` helper for the heresy
   735→ * meter's rate calculation.
   736→ *
   737→ * One-time effects (like Faith's `heresyFlatOnCodify: -20`) are NOT
   738→ * included here — they're applied once by the store at codification
   739→ * time, not per-tick. Read them directly from `DIVINE_LAW_MAP[id]`.
   740→ *
   741→ * @param lawsCodified  Map of law id → codified-this-run flag.
   742→ * @returns Aggregated `DivineLawBonus` (all fields defaulted for empty input).
   743→ *
   744→ * @example
   745→ *   divineLawBonus({})
   746→ *     // → { productionMult:0, ..., worshipMult:1, happinessTarget:null,
   747→ *     //     heresyDecays:false, heresyRateMult:0 }
   748→ *   divineLawBonus({ law_abundance: true })
   749→ *     // → { productionMult:0.15, ..., worshipMult:1, ... }
   750→ *   divineLawBonus({ law_abundance: true, law_tithe: true })
   751→ *     // → { productionMult:0.40, goldMult:0.25, ..., worshipMult:1, ... }
   752→ *   divineLawBonus({ law_faith: true })
   753→ *     // → { ..., worshipMult:2.0, ... }  (multiplicative, not additive)
   754→ *   divineLawBonus({ law_zeal: true, law_order: true })
   755→ *     // → { ..., heresyRateMult:0, ... }  (+0.50 + -0.50 cancel out)
   756→ *   divineLawBonus({ law_mercy: true })
   757→ *     // → { ..., heresyDecays:true, heresyRateMult:-2.00, happinessFlat:10 }
   758→ */
   759→export function divineLawBonus(
   760→  lawsCodified: Record<string, boolean>
   761→): DivineLawBonus {
   762→  const result: DivineLawBonus = {
   763→    productionMult: 0,
   764→    populationGrowthMult: 0,
   765→    militaryMult: 0,
   766→    scienceMult: 0,
   767→    goldMult: 0,
   768→    worshipMult: 1, // multiplicative identity
   769→    happinessFlat: 0,
   770→    happinessTarget: null,
   771→    heresyDecays: false,
   772→    heresyRateMult: 0,
   773→  };
   774→  for (const law of DIVINE_LAWS) {
   775→    if (!lawsCodified[law.id]) continue;
   776→    if (law.productionMult !== undefined) result.productionMult += law.productionMult;
   777→    if (law.populationGrowthMult !== undefined) result.populationGrowthMult += law.populationGrowthMult;
   778→    if (law.militaryMult !== undefined) result.militaryMult += law.militaryMult;
   779→    if (law.scienceMult !== undefined) result.scienceMult += law.scienceMult;
   780→    if (law.goldMult !== undefined) result.goldMult += law.goldMult;
   781→    if (law.worshipMult !== undefined) result.worshipMult *= law.worshipMult;
   782→    if (law.happinessFlat !== undefined) result.happinessFlat += law.happinessFlat;
   783→    if (law.happinessTarget !== undefined) result.happinessTarget = law.happinessTarget;
   784→    if (law.heresyDecays) result.heresyDecays = true;
   785→    if (law.heresyRateMult !== undefined) result.heresyRateMult += law.heresyRateMult;
   786→  }
   787→  return result;
   788→}

   790→/**
   791→ * Return the active Worship Mode's def, or `null` if no mode is set.
   792→ *
   793→ * Only one worship mode can be active per run (`state.worshipMode` is a
   794→ * single id, not a set). The consumer reads the effect fields directly
   795→ * from the returned def:
   796→ *
   797→ *   const mode = worshipModeBonus(state.worshipMode);
   798→ *   if (mode?.productionMult) production *= (1 + mode.productionMult);
   799→ *   if (mode?.happinessDrainMult) drainRate *= mode.happinessDrainMult;
   800→ *
   801→ * @param modeId  The active worship mode id, or `null` if no mode picked yet.
   802→ * @returns The `WorshipModeDef` for the id, or `null`.
   803→ *
   804→ * @example
   805→ *   worshipModeBonus(null)
   806→ *     // → null
   807→ *   worshipModeBonus("reverence")
   808→ *     // → { id:"reverence", name:"Reverence", ..., productionMult:0.10 }
   809→ *   worshipModeBonus("nonexistent_id")
   810→ *     // → null  (unknown id treated as "no mode")
   811→ */
   812→export function worshipModeBonus(
   813→  modeId: string | null
   814→): WorshipModeDef | null {
   815→  if (!modeId) return null;
   816→  return WORSHIP_MODE_MAP[modeId] ?? null;
   817→}

   819→/**
   820→ * Compute the additive multiplier on `BASE_HERESY_RATE` from codified
   821→ * Divine Laws and the active Worship Mode.
   822→ *
   823→ * Sources (all additive):
   824→ *   - Law of Order:    -0.50  (slower heresy: rate × 0.5)
   825→ *   - Law of Zeal:     +0.50  (faster heresy: rate × 1.5)
   826→ *   - Law of Mercy:    -2.00  (heresy decays: rate × -1.0)
   827→ *   - Mystery mode:    -0.25  (slower heresy: rate × 0.75)
   828→ *
   829→ * The effective per-second heresy rate is:
   830→ *   `BASE_HERESY_RATE * (1 + heresyRateMult(lawsCodified, worshipMode))`
   831→ *
   832→ * With no laws / mode: 0 → rate = +0.05/s.
   833→ * With Zeal + Fear:    +0.50 → rate = +0.075/s.
   834→ * With Mercy alone:    -2.00 → rate = -0.05/s (heresy decays).
   835→ * With Order + Mystery: -0.75 → rate = +0.0125/s.
   836→ *
   837→ * @param lawsCodified  Map of law id → codified-this-run flag.
   838→ * @param modeId        Active worship mode id, or `null`.
   839→ * @returns Additive multiplier on `BASE_HERESY_RATE` (0 = no change).
   840→ *
   841→ * @example
   842→ *   heresyRateMult({}, null)
   843→ *     // → 0
   844→ *   heresyRateMult({ law_zeal: true }, null)
   845→ *     // → 0.50
   846→ *   heresyRateMult({ law_mercy: true }, null)
   847→ *     // → -2.00  (heresy decays at base rate)
   848→ *   heresyRateMult({ law_order: true }, "mystery")
   849→ *     // → -0.75  (Order -0.50 + Mystery -0.25)
   850→ *   heresyRateMult({ law_zeal: true, law_order: true }, null)
   851→ *     // → 0  (Zeal +0.50 + Order -0.50 cancel out)
   852→ */
   853→export function heresyRateMult(
   854→  lawsCodified: Record<string, boolean>,
   855→  modeId: string | null
   856→): number {
   857→  let mult = 0;
   858→  // From codified laws.
   859→  mult += divineLawBonus(lawsCodified).heresyRateMult;
   860→  // From active worship mode.
   861→  const mode = worshipModeBonus(modeId);
   862→  if (mode?.heresyRateMult !== undefined) {
   863→    mult += mode.heresyRateMult;
   864→  }
   865→  return mult;
   866→}

   868→/**
   869→ * Whether the Layer 5 (Apotheosis) completion conditions are met.
   870→ *
   871→ * Conditions (both must be true):
   872→ *   1. At least `APOTHEOSIS_COMPLETION_LAWS_REQUIRED` (3) Divine Laws codified.
   873→ *   2. A Heresy schism has been resolved this run.
   874→ *
   875→ * Matches `prestigeLayers.ts apotheosis.completeState = "Codify 3 Divine
   876→ * Laws and resolve a Heresy schism"`. Used by `triggerPrestige()` to gate
   877→ * advancement to Layer 6 (Singularity).
   878→ *
   879→ * @param lawsCodified         Map of law id → codified-this-run flag.
   880→ * @param heresySchismResolved Whether a schism was resolved this run.
   881→ * @returns `true` if both conditions are met.
   882→ *
   883→ * @example
   884→ *   isApotheosisComplete({}, false)
   885→ *     // → false  (no laws codified, no schism resolved)
   886→ *   isApotheosisComplete({ law_abundance: true, law_faith: true }, false)
   887→ *     // → false  (only 2 laws; schism not resolved)
   888→ *   isApotheosisComplete({
   889→ *     law_abundance: true, law_faith: true, law_mercy: true
   890→ *   }, false)
   891→ *     // → false  (3 laws but schism not resolved)
   892→ *   isApotheosisComplete({}, true)
   893→ *     // → false  (schism resolved but no laws codified)
   894→ *   isApotheosisComplete({
   895→ *     law_abundance: true, law_faith: true, law_mercy: true
   896→ *   }, true)
   897→ *     // → true  (3 laws + schism resolved)
   898→ */
   899→export function isApotheosisComplete(
   900→  lawsCodified: Record<string, boolean>,
   901→  heresySchismResolved: boolean
   902→): boolean {
   903→  const codifiedCount = DIVINE_LAWS.filter((l) => lawsCodified[l.id]).length;
   904→  return (
   905→    codifiedCount >= APOTHEOSIS_COMPLETION_LAWS_REQUIRED && heresySchismResolved
   906→  );
   907→}
