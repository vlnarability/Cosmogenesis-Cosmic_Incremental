   545→function ProgressRow({ label, value, max }: { label: string; value: number; max: number }) {
   546→  const pct = max > 0 ? Math.min(100, (value / max) * 100) : 0;
   547→  return (
   548→    <div className="mt-1">
   549→      <div className="flex items-center justify-between text-[0.6rem] text-muted-foreground">
   550→        <span>{label}</span>
   551→        <span>{value}/{max}</span>
   552→      </div>
   553→      <div className="h-1 bg-muted/40 rounded-full overflow-hidden mt-0.5">
   554→        <div className="h-full bg-amber-400/70 transition-all" style={{ width: `${pct}%` }} />
   555→      </div>
   556→    </div>
   557→  );
   558→}

   560→function LayerRow({
   561→  layer,
   562→  unlocked,
   563→  isLast,
   564→}: {
   565→  layer: PrestigeLayer;
   566→  unlocked: boolean;
   567→  isLast: boolean;
   568→}) {
   569→  return (
   570→    <div className={`relative flex items-start gap-3 p-3 rounded-lg border ${unlocked ? "border-border bg-muted/20" : "border-border/40 opacity-60"}`}>
   571→      <div
   572→        className={`w-10 h-10 rounded-lg flex items-center justify-center text-xl shrink-0 ${unlocked ? "" : "grayscale"}`}
   573→        style={{ background: unlocked ? "rgba(252, 211, 77, 0.1)" : "rgba(115, 115, 115, 0.05)" }}
   574→      >
   575→        {unlocked ? layer.icon : <Lock className="w-4 h-4 text-muted-foreground" />}
   576→      </div>
   577→      <div className="flex-1 min-w-0">
   578→        <div className="flex items-center gap-2 flex-wrap">
   579→          <span className="text-sm font-semibold">
   580→            <span className="text-muted-foreground mr-1">#{layer.order}</span>
   581→            {layer.name}
   582→          </span>
   583→          <Badge variant="outline" className="text-[0.6rem]">{layer.currencyName}</Badge>
   584→          {unlocked ? (
   585→            <Badge className="text-[0.6rem] bg-emerald-500/20 text-emerald-300 border-emerald-400/40">Unlocked</Badge>
   586→          ) : (
   587→            <Badge variant="outline" className="text-[0.6rem] text-muted-foreground">{formatUnlock(layer)}</Badge>
   588→          )}
   589→        </div>
   590→        <p className="text-xs text-muted-foreground italic mt-0.5">{layer.tagline}</p>
   591→        <p className="text-xs text-foreground/80 leading-snug mt-1">{layer.story}</p>
   592→      </div>
   593→      {!isLast && (
   594→        <ChevronRight className="w-3 h-3 text-muted-foreground/40 absolute -bottom-2 left-7 rotate-90" />
   595→      )}
   596→    </div>
   597→  );
   598→}
   599→
   600→function formatUnlock(layer: PrestigeLayer): string {
   601→  switch (layer.unlockCondition.type) {
   602→    case "default":
   603→      return "Default";
   604→    case "galacticWins":
   605→      return `${layer.unlockCondition.value} wins`;
   606→    case "challengesCompleted":
   607→      return `${layer.unlockCondition.value} trials`;
   608→    case "totalRuns":
   609→      return `${layer.unlockCondition.value} runs`;
   610→    default:
   611→      return "Locked";
   612→  }
   613→}
