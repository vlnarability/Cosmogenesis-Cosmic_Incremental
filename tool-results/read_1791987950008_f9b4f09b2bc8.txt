    27→    27→// ===== Part 6 — Layer-progress helpers =====
    28→    28→// Per-layer (L_n → L_n+1) unlock progress. Returns the current count and the max (target) count.
    29→    29→// nextLayerKey uses the runtime unlockedLayers keys (e.g. "enlightenment" for L2).
    30→    30→interface NextLayerProgressDef {
    31→    31→  currentLayerName: string; // L_n name (the layer the player needs to have unlocked to progress)
    32→    32→  nextLayerName: string;    // L_n+1 name (the layer the progress bar points toward)
    33→    33→  nextLayerKey: string;     // key in unlockedLayers (or "evolution" for L1)
    34→    34→  current: number;
    35→    35→  max: number;
    36→    36→  metricLabel: string;      // description of the metric (e.g. "Trials completed")
    37→    37→}
    38→    38→
    39→    39→// ===== Part 6 — Layer-next-progress component =====
    40→    40→// Renders a progress bar showing how close the player is to unlocking the NEXT layer (L_(n+1)).
    41→    41→// - If next layer is already unlocked: "✓ {nextLayerName} Unlocked"
    42→    42→// - If current layer (prereq for next) is NOT unlocked: "Requires {currentLayerName}"
    43→    43→// - Otherwise: progress bar with the layer-unlock metric (e.g. "5/10 Foresight nodes").
    44→    44→function LayerNextProgress({
    45→    45→  progress,
    46→    46→  currentLayerUnlocked,
    47→    47→  nextLayerUnlocked,
    48→    48→}: {
    49→    49→  progress: NextLayerProgressDef;
    50→    50→  currentLayerUnlocked: boolean;
    51→    51→  nextLayerUnlocked: boolean;
    52→    52→}) {
    53→    53→  if (nextLayerUnlocked) {
    54→    54→    return (
    55→    55→      <div className="mt-2 flex items-center gap-1.5 text-[0.65rem] text-emerald-300">
    56→    56→        <CheckCircle2 className="w-3 h-3" />
    57→    57→        <span className="font-semibold">{progress.nextLayerName} Unlocked</span>
    58→    58→      </div>
    59→    59→    );
    60→    60→  }
    61→    61→  if (!currentLayerUnlocked) {
    62→    62→    return (
    63→    63→      <div className="mt-2 flex items-center gap-1.5 text-[0.65rem] text-muted-foreground">
    64→    64→        <Lock className="w-3 h-3" />
    65→    65→        <span>Requires <span className="font-semibold text-foreground/80">{progress.currentLayerName}</span> to be unlocked first</span>
    66→    66→      </div>
    67→    67→    );
    68→    68→  }
    69→    69→  const pct = progress.max > 0 ? Math.min(100, (progress.current / progress.max) * 100) : 0;
    70→    70→  return (
    71→    71→    <div className="mt-2">
    72→    72→      <div className="flex items-center justify-between text-[0.6rem] text-muted-foreground">
    73→    73→        <span className="flex items-center gap-1">
    74→    74→          <ArrowRight className="w-2.5 h-2.5" />
    75→    75→          Toward {progress.nextLayerName}
    76→    76→          <span className="text-muted-foreground/70">· {progress.metricLabel}</span>
    77→    77→        </span>
    78→    78→        <span className="tabular-nums">{progress.current}/{progress.max}</span>
    79→    79→      </div>
    80→    80→      <div className="h-1.5 bg-muted/40 rounded-full overflow-hidden mt-0.5">
    81→    81→        <div className="h-full bg-emerald-400/70 transition-all" style={{ width: `${pct}%` }} />
    82→    82→      </div>
    83→    83→    </div>
    84→    84→  );
    85→    85→}
