     1→// Infinity — Layer 9 prestige content
     2→//
     3→// When a god reaches Infinity it stops being a single civilization and
     4→// becomes a curator of timelines. Completed runs leave behind Echoes that
     5→// provide passive bonuses; Forks split decisions into two branches that
     6→// each define a divergent run-long bonus; Future Debt lets the god borrow
     7→// strength now and pay it back later — at compounding interest.
     8→//
     9→// Three mechanic families live in this file:
    10→//
    11→//   1. Echo Types (6)             — completed runs leave behind echoes that
    12→//                                   provide passive bonuses. Each echo type
    13→//                                   can be owned multiple times; the per-echo
    14→//                                   bonus stacks linearly with the count
    15→//                                   owned. Purchased with the Echoes currency
    16→//                                   earned at Layer 9 prestige.
    17→//   2. Fork Scenarios (4)         — split decisions that create two branches.
    18→//                                   Each fork is resolved once per run by
    19→//                                   picking branch A or B; the chosen branch
    20→//                                   grants a run-long bonus. The Final
    21→//                                   Divergence is the capstone fork: branch
    22→//                                   B's "unlock a random divine law for free"
    23→//                                   is dispatched by the consumer via the
    24→//                                   `special` discriminated union.
    25→//   3. Future Debt Tiers (4)       — borrow strength now, pay back later.
    26→//                                   Each tier grants a production boost for
    27→//                                   `boostDuration` seconds, then applies a
    28→//                                   production penalty for `penaltyDuration`
    29→//                                   seconds. Per-run limit caps how many
    30→//                                   times a tier can be invoked per run.
    31→//
    32→// State fields (planned for GameState — to be wired in a follow-up task):
    33→//   - echoesOwned:                Record<string, number>   // echo id → count owned (persistent across runs)
    34→//   - echoes:                     number                    // current Echoes currency pool
    35→//   - forksResolved:              Record<string, "A" | "B"> // fork id → chosen branch (per-run)
    36→//   - activeFutureDebt:           string | null             // currently active debt tier id, or null
    37→//   - futureDebtPhase:            "boost" | "penalty" | null  // current phase of the active debt
    38→//   - futureDebtTimer:            number                    // seconds remaining in the current phase
    39→//   - futureDebtUses:             Record<string, number>   // times each debt tier invoked this run
    40→//   - futureDebtRepaid:           boolean                   // whether any debt has been fully repaid this run
    41→//
    42→// Reference: Infinity is new to the TS port — the vanilla JS had no
    43→// Layer 9 content. The 6 / 4 / 4 spec, the echo bonus values, the fork
    44→// branch magnitudes, and the future-debt durations come straight from
    45→// the L9-data task spec. The Echoes currency is the Layer 9 prestige
    46→// currency (the same `currency: "Infinity"` field on
    47→// `prestigeLayers.ts infinity`); the Echo Type cost field is denominated
    48→// in that currency.
    49→//
    50→// Advancement to Layer 10 (Eternity) requires forking a timeline AND
    51→// repaying the resulting Future Debt — see prestigeLayers.ts
    52→// `infinity.completeState: "Fork a timeline and repay the resulting
    53→// Future Debt"`. See `isInfinityComplete` for the gating logic.
    54→
    55→// =====================================================================
    56→// Constants
    57→// =====================================================================
    58→
    59→/**
    60→ * Number of Fork Scenarios the player must resolve for the Layer 9
    61→ * completion check (`isInfinityComplete`). Matches the spec wording
    62→ * "Fork a timeline and repay the resulting Future Debt" from
    63→ * prestigeLayers.ts. The dataset defines 4 forks total, so this is a
    64→ * quarter of the available pool — the minimum engagement to demonstrate
    65→ * the player understands the divergence mechanic. Forks past the first
    66→ * are optional mastery.
    67→ */
    68→export const INFINITY_COMPLETION_FORKS_REQUIRED: number = 1;

    70→// =====================================================================
    71→// 1. Echo Types (6) — completed runs leave behind echoes
    72→// =====================================================================
    73→
    74→/**
    75→ * The per-echo bonus fields. Each echo type's `bonus` object can set any
    76→ * subset of these; the consumer applies them additively (each owned copy
    77→ * of the echo contributes its bonus once, summed via `echoBonus`).
    78→ *
    79→ * Effect field → consumer wiring:
    80→ *   - `productionMult`         → additive all-production mult (Trade Echo +0.15)
    81→ *   - `populationGrowthMult`   → additive pop-growth mult (Growth Echo +0.10)
    82→ *   - `militaryMult`           → additive military mult (Conquest Echo +0.15)
    83→ *   - `scienceMult`            → additive science mult (Science Echo +0.15)
    84→ *   - `goldMult`               → additive gold mult (Trade Echo +0.15)
    85→ *   - `capsMult`               → additive mult on every resource cap (Stability Echo +0.10)
    86→ *   - `happinessFlat`          → flat per-tick happiness bonus (Stability Echo +10)
    87→ *   - `divinityPerSec`         → flat per-second Divinity grant (Faith Echo +0.02)
    88→ *
    89→ * All mult fields are additive on top of the baseline (0 = no change).
    90→ * `happinessFlat` is a signed flat per-tick grant (Stability Echo's +10
    91→ * is positive; defaults to 0). `divinityPerSec` is a flat per-second
    92→ * grant added directly to `state.resources.divinity` (Faith Echo's +0.02
    93→ * produces 0.02 Divinity/s per copy owned; 5 copies → 0.1 Divinity/s).
    94→ *
    95→ * The `epMult` field is intentionally absent here — Echoes grant
    96→ * production / growth / military / science / gold / caps / happiness /
    97→ * Divinity, NOT evolution-point multipliers. EP multipliers are
    98→ * exclusive to The Final Divergence fork (see `ForkBranchDef`).
    99→ */
   100→export interface EchoBonusFields {
   101→  /** Additive all-production multiplier (0.15 = +15% all production). */
   102→  productionMult?: number;
   103→  /** Additive population-growth multiplier (0.10 = +10% pop growth). */
   104→  populationGrowthMult?: number;
   105→  /** Additive military-power multiplier (0.15 = +15% military). */
   106→  militaryMult?: number;
   107→  /** Additive science multiplier (0.15 = +15% science). */
   108→  scienceMult?: number;
   109→  /** Additive gold multiplier (0.15 = +15% gold). */
   110→  goldMult?: number;
   111→  /** Additive multiplier on every resource's cap (0.10 = +10% caps). */
   112→  capsMult?: number;
   113→  /** Flat per-tick happiness bonus (signed; +10 = +10 happiness/s). */
   114→  happinessFlat?: number;
   115→  /** Flat Divinity generated per second (+0.02 = +0.02 Divinity/s). */
   116→  divinityPerSec?: number;
   117→}
   118→
   119→/**
   120→ * A single Echo Type. Each echo can be owned multiple times — the
   121→ * `echoesOwned` map (persistent across runs, see file header) records
   122→ * the count per echo id. The `bonus` object is applied once per owned
   123→ * copy (5 Growth Echoes → +50% pop growth via `echoBonus`).
   124→ *
   125→ * The `cost` is denominated in the Echoes currency (the Layer 9
   126→ * prestige currency). The wiring task will decide the exact purchase
   127→ * curve — typically each subsequent copy of an echo costs more (e.g.
   128→ * `cost × (1 + count_owned)`), but that scaling is in the store, not
   129→ * here. The base `cost` field is the first-copy price.
   130→ *
   131→ * Echo Types are intentionally NOT mutually exclusive — the player can
   132→ * own all six simultaneously, and the bonuses stack additively across
   133→ * types (Growth + Science + Conquest + Trade + Stability + Faith is a
   134→ * valid end-game echo loadout). This mirrors the per-channel leveling
   135→ * pattern in `divinity_layer.ts` (all six Prayer Channels can be
   136→ * opened) rather than the pick-one pattern in `apotheosis.ts
   137→ * WORSHIP_MODES`.
   138→ */

   146→  /** Emoji glyph shown on the echo card. */

   148→  /** Per-echo bonus object (applied once per owned copy). */
   149→  bonus: EchoBonusFields;
   150→  /** Echoes currency cost for the first copy (subsequent copies may scale — see file header). */

   238→// 2. Fork Scenarios (4) — split decisions that create two branches
   239→// =====================================================================
   240→
   241→/**
   242→ * Discriminated union of fork "special interaction" behaviors. Most
   243→ * fork branches grant a flat stat bonus (production / growth / military
   244→ * / science / gold / caps / ep multipliers); a few instead (or also)
   245→ * dispatch a one-time special action handled by the consumer.
   246→ *
   247→ * Known variants:
   248→ *
   249→ *   - `unlock_random_divine_law`   The Final Divergence branch B
   250→ *                                   (unlocks a random divine law for free,
   251→ *                                   bypassing its Divinity cost).
   252→ */
   253→export type ForkSpecial =
   254→  /** Unlock a random Divine Law from apotheosis.ts for free (Divinity cost waived).
   255→   *  The consumer picks one codifiable law at random (or from the
   256→   *  player's preferred pool) and codifies it instantly. */
   257→  | { kind: "unlock_random_divine_law" };
   258→
   259→/**
   260→ * A single branch of a fork scenario. Each fork has exactly two
   261→ * branches — `branchA` and `branchB` — and the player picks one per run
   262→ * (the chosen branch's bonus applies for the rest of the run). The
   263→ * `label` is a short identifier shown on the fork card; `desc` is
   264→ * optional narrative flavour.
   265→ *
   266→ * Effect field → consumer wiring:
   267→ *   - `productionMult`         → additive all-production mult
   268→ *   - `populationGrowthMult`   → additive pop-growth mult
   269→ *   - `militaryMult`           → additive military mult
   270→ *   - `scienceMult`             → additive science mult
   271→ *   - `goldMult`                → additive gold mult
   272→ *   - `capsMult`                → additive mult on every resource cap
   273→ *   - `epMult`                  → additive evolution-points multiplier
   274→ *                                 (exclusive to The Final Divergence branch A)
   275→ *   - `special`                 → discriminated union dispatched by the
   276→ *                                 consumer (see `ForkSpecial`)
   277→ *
   278→ * `epMult` is the only field not shared with `EchoBonusFields` — Echoes
   279→ * never boost EP gain; only The Final Divergence fork can. This keeps
   280→ * the EP economy gated behind a single, deliberate late-run decision
   281→ * (matching the spec wording "choose between +100% EP OR unlock a
   282→ * random divine law for free").
   283→ *
   284→ * At least one of the bonus fields or `special` must be set on each
   285→ * branch (enforced by the IIFE validator). Most branches set exactly
   286→ * one bonus field for a clean read; Crossroads of Evolution's branchA
   287→ * and branchB each set one; The Final Divergence's branchB sets ONLY
   288→ * `special` (the unlock action is its entire effect).
   289→ */

   298→  populationGrowthMult?: number;
   299→  /** Additive military-power multiplier (0.40 = +40% military). */
   300→  militaryMult?: number;
   301→  /** Additive science multiplier (0.40 = +40% science). */
   302→  scienceMult?: number;
   303→  /** Additive gold multiplier. */
   304→  goldMult?: number;
   305→  /** Additive multiplier on every resource's cap (0.30 = +30% caps). */
   306→  capsMult?: number;
   307→  /** Additive Evolution Points multiplier (1.0 = +100% EP gain). Exclusive to The Final Divergence branch A. */
   308→  epMult?: number;
   309→  /** Optional special interaction (see `ForkSpecial`). */
   310→  special?: ForkSpecial;

   428→// =====================================================================
   429→// 3. Future Debt Tiers (4) — borrow strength now, pay back later
   430→// =====================================================================
   431→
   432→/**
   433→ * A single Future Debt tier. Invoking a debt tier grants a temporary
   434→ * production boost (`boostMult`, additive on production) for
   435→ * `boostDuration` seconds, then immediately applies a production
   436→ * penalty (`penaltyMult`, additive — typically negative) for
   437→ * `penaltyDuration` seconds. The player may invoke a tier up to
   438→ * `perRunLimit` times per run.
   439→ *
   440→ * Effect field → consumer wiring:
   441→ *   - `boostMult`         → during the boost phase, multiply production
   442→ *                            by `(1 + boostMult)` (e.g. +0.50 = ×1.5)
   443→ *   - `boostDuration`     → seconds the boost phase lasts
   444→ *   - `penaltyMult`       → during the penalty phase, multiply
   445→ *                            production by `(1 + penaltyMult)` (e.g.
   446→ *                            -0.25 = ×0.75). Always negative.
   447→ *   - `penaltyDuration`   → seconds the penalty phase lasts
   448→ *   - `perRunLimit`       → max times this tier can be invoked per run
   449→ *
   450→ * The interest curve is intentionally punishing: each tier's penalty
   451→ * duration grows faster than its boost duration, and the penalty
   452→ * multiplier approaches -1.0 (production halted). Critical Debt's
   453→ * -90% for 600s is a near-run-ending penalty — the god pays dearly for
   454→ * the +500% burst. This makes Future Debt a "panic button" rather than
   455→ * a routine accelerator.
   456→ *
   457→ * The completion check (`isInfinityComplete`) requires that at least
   458→ * one debt has been fully repaid — meaning the player has survived a
   459→ * complete boost + penalty cycle without resetting. This proves the
   460→ * god can carry the weight of borrowed power.
   461→ *
   462→ * The tier progression:
   463→ *   - Minor:     +50% for 120s, then -25% for 120s  (1:1 ratio, mild)
   464→ *   - Moderate:  +100% for 120s, then -50% for 180s (1:1.5 ratio, sharp)
   465→ *   - Severe:    +200% for 120s, then -75% for 300s (1:2.5 ratio, harsh)
   466→ *   - Critical:  +500% for 60s, then -90% for 600s  (1:10 ratio, brutal)
   467→ *
   468→ * Per-run limits are inversely scaled to power: Minor can be invoked
   469→ * 5× per run; Critical only once. This prevents stacking Critical
   470→ * Debt boosts back-to-back to skip the penalty phase entirely.
   471→ */
   472→export interface FutureDebtTierDef {
   473→  /** Stable identifier (used as a key in `futureDebtUses` in GameState). */
   474→  id: string;
   475→  /** Human-readable debt tier name. */
   476→  name: string;
   477→  /** Narrative / mechanical flavour describing the bargain and its cost. */
   478→  desc: string;
   479→  /** Emoji glyph shown on the debt card. */
   480→  icon: string;
   481→  /** Additive production multiplier during the boost phase (0.50 = +50% production). */
   482→  boostMult: number;
   483→  /** Duration of the boost phase in seconds. */
   484→  boostDuration: number;
   485→  /** Additive production multiplier during the penalty phase (-0.25 = -25% production). Always negative. */
   486→  penaltyMult: number;
   487→  /** Duration of the penalty phase in seconds. */
   488→  penaltyDuration: number;
   489→  /** Maximum times this tier can be invoked per run. */
   490→  perRunLimit: number;
   491→}

   806→ *

   809→ * `populationGrowthMult` is `5 × 0.10 = 0.50` (+50% pop growth).

   812→ * leave behind echoes that provide passive bonuses" implies they
   813→ * carry forward — the `echoesOwned` map is part of the persistent
   814→ * GameState, not the per-run state). The aggregate is recomputed each
   815→ * tick (it's cheap — 6 entries) so the consumer doesn't need to cache

   837→/**
   838→ * Sum the bonuses from all owned Echo Types into a single aggregate.
   839→ * Each echo type's `bonus` object is multiplied by the count owned
   840→ * (from the `echoesOwned` map) and added to the corresponding result
   841→ * field.
   842→ *
   843→ * Used by the tick loop to compute per-resource multipliers, the
   844→ * pop-growth rate, the cap multipliers, the per-tick happiness bonus,
   845→ * and the per-second Divinity grant. The aggregate is recomputed each
   846→ * tick (it's cheap — 6 entries) so the consumer doesn't need to cache
   847→ * it.
   848→ *
   849→ * @param echoesOwned  Map of echo id → count owned (0 / absent = none owned).
   850→ * @returns Aggregated `EchoBonus` (all fields defaulted for empty input).
   851→ *
   852→ * @example
   853→ *   echoBonus({})
   854→ *     // → { productionMult:0, ..., happinessFlat:0, divinityPerSec:0 }
   855→ *   echoBonus({ growth_echo: 1 })
   856→ *     // → { populationGrowthMult:0.10, ... }  (1 Growth Echo = +10% pop growth)
   857→ *   echoBonus({ growth_echo: 3 })
   858→ *     // → { populationGrowthMult:0.30, ... }  (3 Growth Echoes = +30% pop growth)
   859→ *   echoBonus({ trade_echo: 2 })
   860→ *     // → { goldMult:0.30, productionMult:0.30, ... }  (2 Trade Echoes = +30% gold AND +30% production)
   861→ *   echoBonus({ stability_echo: 2 })
   862→ *     // → { happinessFlat:20, capsMult:0.20, ... }  (2 Stability Echoes = +20 happiness/s, +20% caps)
   863→ *   echoBonus({ faith_echo: 5 })
   864→ *     // → { divinityPerSec:0.10, ... }  (5 Faith Echoes = +0.10 Divinity/s)
   865→ *   echoBonus({ growth_echo: 1, science_echo: 1, conquest_echo: 1 })
   866→ *     // → { populationGrowthMult:0.10, scienceMult:0.15, militaryMult:0.15, ... }
   867→ *   echoBonus({ nonexistent: 5 })
   868→ *     // → { all zeros }  (unknown echo id ignored)
   869→ *   echoBonus({ growth_echo: 0 })
   870→ *     // → { all zeros }  (count <= 0 treated as unowned)
   871→ */
   872→export function echoBonus(
   873→  echoesOwned: Record<string, number>
   874→): EchoBonus {
   875→  const result: EchoBonus = {
   876→    productionMult: 0,
   877→    populationGrowthMult: 0,
   878→    militaryMult: 0,
   879→    scienceMult: 0,
   880→    goldMult: 0,
   881→    capsMult: 0,
   882→    happinessFlat: 0,
   883→    divinityPerSec: 0,
   884→  };
   885→  for (const echo of ECHO_TYPES) {
   886→    const count = echoesOwned[echo.id] ?? 0;
   887→    if (count <= 0) continue;
   888→    const b = echo.bonus;
   889→    if (b.productionMult !== undefined) result.productionMult += b.productionMult * count;
   890→    if (b.populationGrowthMult !== undefined) result.populationGrowthMult += b.populationGrowthMult * count;
   891→    if (b.militaryMult !== undefined) result.militaryMult += b.militaryMult * count;
   892→    if (b.scienceMult !== undefined) result.scienceMult += b.scienceMult * count;
   893→    if (b.goldMult !== undefined) result.goldMult += b.goldMult * count;
   894→    if (b.capsMult !== undefined) result.capsMult += b.capsMult * count;
   895→    if (b.happinessFlat !== undefined) result.happinessFlat += b.happinessFlat * count;
   896→    if (b.divinityPerSec !== undefined) result.divinityPerSec += b.divinityPerSec * count;
   897→  }
   898→  return result;
   899→}

  1004→    capsMult: 0,
  1005→    epMult: 0,
  1006→    special: null,
  1007→  };
  1008→  if (!forkId || !branch) return result;
  1009→  const fork = FORK_MAP[forkId];
  1010→  if (!fork) return result;
  1011→  const branchDef = branch === "A" ? fork.branchA : fork.branchB;
  1012→  if (branchDef.productionMult !== undefined) result.productionMult = branchDef.productionMult;
  1013→  if (branchDef.populationGrowthMult !== undefined) result.populationGrowthMult = branchDef.populationGrowthMult;
  1014→  if (branchDef.militaryMult !== undefined) result.militaryMult = branchDef.militaryMult;
  1015→  if (branchDef.scienceMult !== undefined) result.scienceMult = branchDef.scienceMult;
  1016→  if (branchDef.goldMult !== undefined) result.goldMult = branchDef.goldMult;
  1017→  if (branchDef.capsMult !== undefined) result.capsMult = branchDef.capsMult;
  1018→  if (branchDef.epMult !== undefined) result.epMult = branchDef.epMult;
  1019→  if (branchDef.special !== undefined) result.special = branchDef.special;
  1020→  return result;
  1021→}

  1023→/**
  1024→ * Whether the Layer 9 (Infinity) completion conditions are met.
  1025→ *
  1026→ * Conditions (both must be true):
  1027→ *   1. At least `INFINITY_COMPLETION_FORKS_REQUIRED` (1) Fork Scenarios
  1028→ *      have been resolved — i.e. have a `"A"` or `"B"` entry in the
  1029→ *      `forksResolved` map.
  1030→ *   2. A Future Debt has been fully repaid this run (`debtRepaid === true`).
  1031→ *
  1032→ * Matches `prestigeLayers.ts infinity.completeState = "Fork a timeline
  1033→ * and repay the resulting Future Debt"`. Used by `triggerPrestige()`
  1034→ * to gate advancement to Layer 10 (Eternity).
  1035→ *
  1036→ * Note: "repaid" means the player has survived a complete
  1037→ * boost + penalty cycle on any Future Debt tier — the penalty phase
  1038→ * ended without the player resetting. This proves the god can carry
  1039→ * the weight of borrowed power, which is the thematic crux of the
  1040→ * Infinity layer.
  1041→ *
  1042→ * @param forksResolved  Map of fork id → chosen branch ("A" | "B").
  1043→ * @param debtRepaid     Whether any Future Debt has been fully repaid this run.
  1044→ * @returns `true` if both conditions are met.
  1045→ *
  1046→ * @example
  1047→ *   isInfinityComplete({}, false)
  1048→ *     // → false  (no forks resolved, no debt repaid)
  1049→ *   isInfinityComplete({ crossroads_of_evolution: "A" }, false)
  1050→ *     // → false  (1 fork but no debt repaid)
  1051→ *   isInfinityComplete({}, true)
  1052→ *     // → false  (debt repaid but no fork resolved)
  1053→ *   isInfinityComplete({ crossroads_of_evolution: "A" }, true)
  1054→ *     // → true  (1 fork + debt repaid)
  1055→ *   isInfinityComplete({
  1056→ *     crossroads_of_evolution: "A",
  1057→ *     the_great_schism: "B",
  1058→ *     realm_of_mirrors: "A",
  1059→ *     the_final_divergence: "B"
  1060→ *   }, true)
  1061→ *     // → true  (4 forks + debt repaid — over-satisfied)
  1062→ *   isInfinityComplete({ nonexistent_fork: "A" }, true)
  1063→ *     // → false  (unknown fork id ignored; 0 recognized forks + debt repaid)
  1064→ *   isInfinityComplete({ crossroads_of_evolution: "A" }, false)
  1065→ *     // → false  (fork resolved but debt not repaid)
  1066→ */
  1067→export function isInfinityComplete(
  1068→  forksResolved: Record<string, "A" | "B">,
  1069→  debtRepaid: boolean
  1070→): boolean {
  1071→  const resolvedCount = FORK_SCENARIOS.filter((f) => {
  1072→    const branch = forksResolved[f.id];
  1073→    return branch === "A" || branch === "B";
  1074→  }).length;
  1075→  return (
  1076→    resolvedCount >= INFINITY_COMPLETION_FORKS_REQUIRED && debtRepaid
  1077→  );
  1078→}
