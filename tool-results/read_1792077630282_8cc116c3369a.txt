    40→/**
    41→ * Left-border accent classes for each heresy severity tier.
    42→ * Replaces the old pulsing-border + box-shadow cliché with a single
    43→ * functional left-accent bar — calm cards have no accent; severity
    44→ * grows through amber → orange → red (with a subtle pulse only at the
    45→ * run-ending red tier).
    46→ */
    47→function severityBorderClass(sev: HeresySeverity): string {
    48→  switch (sev) {
    49→    case "red":    return "border-l-2 border-l-red-500 heresy-pulse-red";
    50→    case "orange": return "border-l-2 border-l-orange-500";
    51→    case "yellow": return "border-l-2 border-l-amber-400";
    52→    default:       return "";
    53→  }
    54→}
    55→
    56→function severityTextClass(sev: HeresySeverity): string {
    57→  switch (sev) {
    58→    case "red": return "text-rose-400";
    59→    case "orange": return "text-orange-300";
    60→    case "yellow": return "text-yellow-300";
    61→    default: return "text-muted-foreground";
    62→  }
    63→}
