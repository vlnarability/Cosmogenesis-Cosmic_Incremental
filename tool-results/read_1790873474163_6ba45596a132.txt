    18→const STORAGE_KEY = "evolution_idle_v2";
    19→
    20→function initialRunState(): Partial<GameState> {
    21→  const resources = emptyResources();
    22→  const capacities = emptyCapacities();
    23→  resources.atp = 5;
    24→  resources.glucose = 8;
    25→  resources.happiness = 75;
    26→  return {
    27→    stageIndex: 0,
    28→    time: 0,
    29→    population: 8,
    30→    populationProgress: 0,
    31→    maxPopulation: 8,
    32→    resources,
    33→    capacities,
    34→    ownedSystems: {},
    35→    systemEnabled: {},
    36→    technologies: {},
    37→    lockedTechs: {},
    38→    archetypeAffinity: {},
    39→    lockedArchetype: null,
    40→    log: ["New run started. The first spark stirs."],
    41→    activeChallenge: null,
    42→    challengeRunStart: null,
    43→  };
    44→}
    45→
    46→function initialMetaState(): Partial<GameState> {
    47→  return {
    48→    evolutionPoints: 0,
    49→    totalRuns: 0,
    50→    galacticWins: 0,
    51→    stageClearCounts: Object.fromEntries(STAGE_IDS.map((s) => [s, 0])) as Record<StageId, number>,
    52→    fastestCellClear: 0,
    53→    totalEventsResolved: 0,
    54→    totalPlayTime: 0,
    55→    totalActions: 0,
    56→    totalSystemsBuilt: 0,
    57→    totalTechResearched: 0,
    58→    upgrades: Object.fromEntries(UPGRADES.map((u) => [u.id, 0])),
    59→    storyUnlocked: { first_spark: true },
    60→    storyAcknowledged: {},
    61→    archive: [],
    62→    archivedArchetypes: [],
    63→    unlockedLayers: { evolution: true },
    64→    achievements: {},
    65→    newAchievements: [],
    66→    unlockedChallenges: false,
    67→    activeChallenge: null,
    68→    completedChallenges: {},
    69→    challengeRunStart: null,
    70→    currentTab: "actions",
    71→    speed: 1,
    72→    paused: false,
    73→    showShop: false,
    74→    showEvolve: false,
    75→    showSettings: false,
    76→    showChallenges: false,
    77→    activeStoryPopup: "first_spark",
    78→    activeEvent: null,
    79→    eventCooldown: 60, // first event can fire after 60s
    80→    eventHistory: [],
    81→    hasSeenIntro: false,
    82→    lastSaved: Date.now(),
    83→    tutorialStep: 0,
    84→    tutorialActive: true,
    85→    tutorialDismissed: false,
    86→    eventFrequency: "normal",
    87→  };
    88→}
    89→
    90→export const useGameStore = create<GameStore>()(
    91→  persist(
    92→    (set, get) => ({
    93→      ...initialRunState(),
    94→      ...initialMetaState(),
    95→
    96→      // ============ TICK ============
    97→      tick: (dt) => {
    98→        const s = get();
    99→        if (s.paused) return;
   100→        const realDt = dt * s.speed;
   101→
   102→        const autoLvl = s.upgrades["inherited_efficiency"] || 0;
   103→        const autoMult = 1 + autoLvl * 0.15;
   104→        const techMult = techMultiplier(s, "production");
   105→        const capBoostLvl = s.upgrades["expansive_vaults"] || 0;
   106→        const capMult = 1 + capBoostLvl * 0.20;
   107→        const hasAutoBalancer = (s.upgrades["auto_balancer"] || 0) > 0;
   108→
   109→        // Archetype bonus
   110→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   111→        const archAllMult = archetype?.bonus.allProductionMult || 0;
   112→        const archPopMult = archetype?.bonus.populationGrowthMult || 0;
   113→        const archCapMult = archetype?.bonus.capMult || 0;
   114→
   115→        // ===== Challenge debuff (applied for the duration of an active challenge run)
   116→        const activeChallenge = s.activeChallenge ? CHALLENGE_MAP[s.activeChallenge] : null;
   117→        const challengeEffects = activeChallenge?.debuff.effects || {};
   118→        const chFoodMult = challengeEffects.foodMult ?? 1;
   119→        const chScienceMult = challengeEffects.scienceMult ?? 1;
   120→        const chProductionMult = challengeEffects.productionMult ?? 1;
   121→        const chMilitaryMult = challengeEffects.militaryMult ?? 1;
   122→        const chWaterUpkeepMult = challengeEffects.waterUpkeepMult ?? 1;
   123→        const chHappinessDrain = challengeEffects.happinessDrain ?? 0;
   124→        const chPopGrowthMult = challengeEffects.popGrowthMult ?? 1;
   125→
   126→        const resources = { ...s.resources };
   127→        let capacities = recomputeCapacitiesFresh(s, capMult + archCapMult);
   128→
   129→        // Apply system production — with cap-aware pausing
   130→        for (const [sysId, count] of Object.entries(s.ownedSystems)) {
   131→          if (!count) continue;
   132→          const sys = SYSTEM_MAP[sysId];
   133→          if (!sys) continue;
   134→          if (sys.requiredTech && !s.technologies[sys.requiredTech]) continue;
   135→
   136→          // Check if system is manually disabled
   137→          const manuallyEnabled = s.systemEnabled[sysId] !== false; // default true
   138→          if (!manuallyEnabled && !hasAutoBalancer) continue;
   139→
   140→          // Cap-aware logic: if all output resources are at cap, idle the system
   141→          // (don't produce, don't consume upkeep — no waste)
   142→          const outputsCapped = Object.keys(sys.produces).length > 0 &&
   143→            Object.keys(sys.produces).every((r) => {
   144→              const cap = capacities[r] || 0;
   145→              return (resources[r] || 0) >= cap - 0.01;
   146→            });
   147→
   148→          if (outputsCapped) {
   149→            // System idles — no production, no upkeep
   150→            continue;
   151→          }
   152→
   153→          const stageMult = sys.stage === STAGES[s.stageIndex].id ? 1 : 0.5;
   154→          const mult = autoMult * techMult * stageMult * (1 + archAllMult) * chProductionMult;
   155→
   156→          // Per-resource archetype bonus + challenge resource-specific debuff
   157→          const produces = sys.produces as Record<string, number>;
   158→          for (const [r, v] of Object.entries(produces)) {
   159→            const archResBonus = archetype?.bonus.productionMult?.[r] || 0;
   160→            // Apply per-resource challenge multipliers
   161→            let chResMult = 1;
   162→            if (r === "food") chResMult *= chFoodMult;
   163→            if (r === "science") chResMult *= chScienceMult;
   164→            if (r === "military_power") chResMult *= chMilitaryMult;
   165→            const rate = v * count * mult * (1 + archResBonus) * chResMult;
   166→            resources[r] = (resources[r] || 0) + rate * realDt;
   167→          }
   168→          // Upkeep
   169→          if (sys.upkeep) {
   170→            const upkeep = sys.upkeep as Record<string, number>;
   171→            for (const [r, v] of Object.entries(upkeep)) {
   172→              const upkMult = r === "water" ? chWaterUpkeepMult : 1;
   173→              resources[r] = (resources[r] || 0) - v * count * realDt * upkMult;
   174→            }
   175→          }
   176→        }
   177→
   178→        // Apply challenge happiness drain (flat per second)
   179→        if (chHappinessDrain > 0) {
   180→          const happCap = capacities.happiness || 100;
   181→          resources.happiness = Math.max(0, (resources.happiness || 0) - chHappinessDrain * realDt);
   182→          // also clamp to cap (already done below, but keep happ bounded)
   183→          if (resources.happiness > happCap) resources.happiness = happCap;
   184→        }
   185→
   186→        // Clamp resources to capacities (no overflow)
   187→        for (const r of Object.keys(resources)) {
   188→          const cap = capacities[r];
   189→          if (cap != null && resources[r] > cap) resources[r] = cap;
   190→          if (resources[r] < 0) resources[r] = 0;
   191→        }
   192→
   193→        // Slow passive ATP for cell (keeps early game from stalling)
   194→        if (STAGES[s.stageIndex].id === "cell") {
   195→          resources.atp = Math.min(capacities.atp || 30, resources.atp + 0.1 * realDt);
   196→        }
   197→
   198→        // Happiness drift toward 50 baseline if no source
   199→        // (prevents permanent high happiness from start)
   200→        // — disabled: let systems/tech maintain it
   201→
   202→        // Population growth with progress bar
   203→        // Growth rate scales with happiness and food/water availability
   204→        let popGrowthRate = 0.015; // base per second
   205→        const happiness = resources.happiness || 50;
   206→        popGrowthRate *= (1 + (happiness - 40) / 200); // happiness 100 → +30%, happiness 20 → -10%
   207→        popGrowthRate *= (1 + archPopMult);
   208→        // Apply challenge population-growth debuff
   209→        popGrowthRate *= chPopGrowthMult;
   210→        // Penalty if food/water are critically low (creature+)
   211→        const curStage = STAGES[s.stageIndex];
   212→        if (["creature", "tribal", "civilization", "empire"].includes(curStage.id)) {
   213→          const foodCap = capacities.food || 1;
   214→          const waterCap = capacities.water || 1;
   215→          const foodPct = (resources.food || 0) / foodCap;
   216→          const waterPct = (resources.water || 0) / waterCap;
   217→          if (foodPct < 0.1) popGrowthRate *= 0.3;
   218→          if (waterPct < 0.1) popGrowthRate *= 0.3;
   219→        }
   220→
   221→        let populationProgress = (s.populationProgress || 0) + popGrowthRate * realDt;
   222→        let population = s.population;
   223→        while (populationProgress >= 1) {
   224→          populationProgress -= 1;
   225→          population += 1;
   226→        }
   227→
   228→        const maxPopulation = Math.max(s.maxPopulation || 0, population);
   229→
   230→        const newTime = s.time + realDt;
   231→
   232→        // Check story triggers
   233→        const newStory = { ...s.storyUnlocked };
   234→        let activePopup = s.activeStoryPopup;
   235→        const ctx = { ...s, resources, ownedSystems: s.ownedSystems };
   236→        for (const trig of STORY_TRIGGERS) {
   237→          if (!newStory[trig.id] && trig.condition(ctx)) {
   238→            newStory[trig.id] = true;
   239→            if (!activePopup) activePopup = trig.id;
   240→          }
   241→        }
   242→
   243→        // Check achievements periodically (every ~2 seconds of game time)
   244→        let achievements = s.achievements;
   245→        let newAchievements = s.newAchievements;
   246→        const checkInterval = 2.0;
   247→        const lastCheck = (s as any)._lastAchCheck || 0;
   248→        if (newTime - lastCheck >= checkInterval) {
   249→          const curState = { ...s, resources, time: newTime } as GameState;
   250→          const result = checkAchievements(curState);
   251→          if (result.newOnes.length > 0) {
   252→            achievements = result.earned;
   253→            newAchievements = [...(newAchievements || []), ...result.newOnes];
   254→          } else if (Object.keys(result.earned).length !== Object.keys(achievements || {}).length) {
   255→            achievements = result.earned;
   256→          }
   257→          (s as any)._lastAchCheck = newTime; // not persisted, just a runtime marker
   258→        }
   259→
   260→        // Random event firing — only if no popup/modal is active
   261→        let activeEvent = s.activeEvent;
   262→        let eventCooldown = (s.eventCooldown || 0) - realDt;
   263→        if (eventCooldown < 0) eventCooldown = 0;
   264→        const eventFreq = s.eventFrequency || "normal";
   265→        if (eventFreq !== "off" && !activeEvent && !activePopup && eventCooldown === 0) {
   266→          const stageId = STAGES[s.stageIndex].id;
   267→          const possible = eventsForStage(stageId, newTime);
   268→          if (possible.length > 0) {
   269→            // Weighted random selection based on rarity
   270→            const totalWeight = possible.reduce((sum, e) => sum + eventWeight(e), 0);
   271→            const roll = Math.random() * totalWeight;
   272→            let cum = 0;
   273→            for (const ev of possible) {
   274→              cum += eventWeight(ev);
   275→              if (roll < cum) {
   276→                activeEvent = ev.id;
   277→                // Cooldown depends on frequency setting
   278→                const base = eventFreq === "frequent" ? 30 : 90;
   279→                const range = eventFreq === "frequent" ? 30 : 90;
   280→                eventCooldown = base + Math.random() * range;
   281→                break;
   282→              }
   283→            }
   284→          }
   285→        }
   286→
   287→        set({
   288→          resources,
   289→          capacities,
   290→          time: newTime,
   291→          population,
   292→          maxPopulation,
   293→          populationProgress,
   294→          storyUnlocked: newStory,
   295→          activeStoryPopup: activePopup,
   296→          achievements,
   297→          newAchievements,
   298→          activeEvent,
   299→          eventCooldown,
   300→          totalPlayTime: (s.totalPlayTime || 0) + realDt,
   301→        });
   302→
   303→        // Check if the active challenge's completion condition has been met.
   304→        // If so, mark it complete and clear the active challenge slot.
   305→        // (Done after the set so completion sees updated state.)
   306→        if (s.activeChallenge) {
   307→          get().checkChallengeCompletion();
   308→        }
   309→      },
   310→
   311→      // ============ ACTIONS ============
   312→      performAction: (actionId) => {
   313→        const s = get();
   314→        const action = ACTION_MAP[actionId];
   315→        if (!action) return;
   316→        if (action.stage !== STAGES[s.stageIndex].id) return;
   317→
   318→        const res = { ...s.resources };
   319→
   320→        // Check if all output resources are at cap — if so, block (no waste)
   321→        const outputs = Object.entries(action.produces);
   322→        if (outputs.length > 0) {
   323→          const allCapped = outputs.every(([r]) => {
   324→            const cap = s.capacities[r] || 0;
   325→            return (res[r] || 0) >= cap - 0.001;
   326→          });
   327→          if (allCapped) return; // action wasted — block it
   328→        }
   329→
   330→        // Check costs
   331→        if (action.cost) {
   332→          for (const [r, v] of Object.entries(action.cost)) {
   333→            if ((res[r] || 0) < (v as number)) return;
   334→          }
   335→          for (const [r, v] of Object.entries(action.cost)) {
   336→            res[r] -= v as number;
   337→          }
   338→        }
   339→
   340→        // Manual action multiplier
   341→        const manualLvl = s.upgrades["quickened_hands"] || 0;
   342→        const mult = 1 + manualLvl * 0.25;
   343→        const techMult = techMultiplier(s, "manual");
   344→        const finalMult = mult * techMult;
   345→
   346→        // Archetype per-resource bonus
   347→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   348→
   349→        for (const [r, v] of Object.entries(action.produces)) {
   350→          const cap = s.capacities[r] || 0;
   351→          const archResBonus = archetype?.bonus.productionMult?.[r] || 0;
   352→          const totalArchMult = 1 + archResBonus + (archetype?.bonus.allProductionMult || 0);
   353→          const amount = (v as number) * finalMult * totalArchMult;
   354→          res[r] = Math.min(cap, (res[r] || 0) + amount);
   355→        }
   356→
   357→        set({
   358→          resources: res,
   359→          totalActions: (s.totalActions || 0) + 1,
   360→        });
   361→      },
   362→
   363→      buySystem: (systemId, qty) => {
   364→        const s = get();
   365→        const sys = SYSTEM_MAP[systemId];
   366→        if (!sys) return;
   367→        if (sys.stage !== STAGES[s.stageIndex].id) return;
   368→        // Check tech prerequisite
   369→        if (sys.requiredTech && !s.technologies[sys.requiredTech]) return;
   370→
   371→        const costReduction = (s.upgrades["frugality"] || 0) * 0.04;
   372→        let owned = s.ownedSystems[systemId] || 0;
   373→        if (sys.maxOwned && owned >= sys.maxOwned) return;
   374→
   375→        const res = { ...s.resources };
   376→        const ownedSystems = { ...s.ownedSystems };
   377→        const systemEnabled = { ...s.systemEnabled };
   378→        const archetypeAffinity = { ...s.archetypeAffinity };
   379→
   380→        let bought = 0;
   381→        for (let i = 0; i < qty; i++) {
   382→          if (sys.maxOwned && owned + bought >= sys.maxOwned) break;
   383→          const cost = systemCost(sys, owned + bought, costReduction);
   384→          let canAfford = true;
   385→          for (const [r, v] of Object.entries(cost)) {
   386→            if ((res[r] || 0) < (v as number)) {
   387→              canAfford = false;
   388→              break;
   389→            }
   390→          }
   391→          if (!canAfford) break;
   392→          for (const [r, v] of Object.entries(cost)) {
   393→            res[r] -= v as number;
   394→          }
   395→          bought++;
   396→        }
   397→
   398→        if (bought === 0) return;
   399→
   400→        ownedSystems[systemId] = owned + bought;
   401→        // New systems are enabled by default
   402→        if (systemEnabled[systemId] === undefined) systemEnabled[systemId] = true;
   403→
   404→        // Apply archetype affinity grant
   405→        if (sys.grantsAffinity) {
   406→          const arch = sys.grantsAffinity.archetype;
   407→          archetypeAffinity[arch] = (archetypeAffinity[arch] || 0) + sys.grantsAffinity.amount * bought;
   408→        }
   409→
   410→        // Trigger story
   411→        const newStory = { ...s.storyUnlocked };
   412→        let activePopup = s.activeStoryPopup;
   413→        for (const trig of STORY_TRIGGERS) {
   414→          if (!newStory[trig.id] && trig.condition({ ...s, resources: res, ownedSystems })) {
   415→            newStory[trig.id] = true;
   416→            if (!activePopup) activePopup = trig.id;
   417→          }
   418→        }
   419→
   420→        set({
   421→          resources: res,
   422→          ownedSystems,
   423→          systemEnabled,
   424→          archetypeAffinity,
   425→          storyUnlocked: newStory,
   426→          activeStoryPopup: activePopup,
   427→          totalSystemsBuilt: (s.totalSystemsBuilt || 0) + bought,
   428→        });
   429→        get().addToLog(`Built ${bought}× ${sys.name}.`);
   430→      },
   431→
   432→      toggleSystem: (systemId) => {
   433→        const s = get();
   434→        if (!s.ownedSystems[systemId]) return;
   435→        const systemEnabled = { ...s.systemEnabled };
   436→        systemEnabled[systemId] = systemEnabled[systemId] === false ? true : false;
   437→        set({ systemEnabled });
   438→      },
   439→
   440→      enableAllSystems: (enabled) => {
   441→        const s = get();
   442→        const systemEnabled: Record<string, boolean> = {};
   443→        for (const id of Object.keys(s.ownedSystems)) {
   444→          systemEnabled[id] = enabled;
   445→        }
   446→        set({ systemEnabled });
   447→      },
   448→
   449→      buyTech: (techId) => {
   450→        const s = get();
   451→        const tech = TECH_MAP[techId];
   452→        if (!tech) return;
   453→        if (s.technologies[techId]) return;
   454→        // Locked techs cannot be researched (a mutually-exclusive sibling was already chosen)
   455→        if (s.lockedTechs && s.lockedTechs[techId]) return;
   456→        if (tech.stage !== STAGES[s.stageIndex].id) return;
   457→        if (tech.requires) {
   458→          for (const req of tech.requires) {
   459→            if (!s.technologies[req]) return;
   460→          }
   461→        }
   462→        const res = { ...s.resources };
   463→        for (const [r, v] of Object.entries(tech.cost)) {
   464→          if ((res[r] || 0) < (v as number)) return;
   465→        }
   466→        for (const [r, v] of Object.entries(tech.cost)) {
   467→          res[r] -= v as number;
   468→        }
   469→        const technologies = { ...s.technologies, [techId]: true };
   470→        const archetypeAffinity = { ...s.archetypeAffinity };
   471→        if (tech.grantsAffinity) {
   472→          archetypeAffinity[tech.grantsAffinity.archetype] =
   473→            (archetypeAffinity[tech.grantsAffinity.archetype] || 0) + tech.grantsAffinity.amount;
   474→        }
   475→        // Enforce mutually-exclusive tech path choices: lock all siblings
   476→        // listed in `exclusiveWith` so they can never be researched this run.
   477→        const lockedTechs = { ...(s.lockedTechs || {}) };
   478→        if (tech.exclusiveWith && tech.exclusiveWith.length > 0) {
   479→          for (const siblingId of tech.exclusiveWith) {
   480→            // Don't lock a sibling that is somehow already researched (defensive)
   481→            if (!technologies[siblingId]) {
   482→              lockedTechs[siblingId] = true;
   483→            }
   484→          }
   485→        }
   486→        set({
   487→          resources: res,
   488→          technologies,
   489→          lockedTechs,
   490→          archetypeAffinity,
   491→          totalTechResearched: (s.totalTechResearched || 0) + 1,
   492→        });
   493→        get().addToLog(`Researched ${tech.name}.`);
   494→      },
   495→
   496→      buyUpgrade: (upgradeId) => {
   497→        const s = get();
   498→        const up = UPGRADE_MAP[upgradeId];
   499→        if (!up) return;
   500→        if (up.requiresWins && s.galacticWins < up.requiresWins) return;
   501→        const owned = s.upgrades[upgradeId] || 0;
   502→        if (owned >= up.maxLevel) return;
   503→        const cost = upgradeCost(up, owned);
   504→        if (s.evolutionPoints < cost) return;
   505→        const upgrades = { ...s.upgrades, [upgradeId]: owned + 1 };
   506→        set({
   507→          evolutionPoints: s.evolutionPoints - cost,
   508→          upgrades,
   509→        });
   510→        get().addToLog(`Purchased ${up.name} (Lv ${owned + 1}).`);
   511→      },
   512→
   513→      evolveStage: () => {
   514→        const s = get();
   515→        const stage = STAGES[s.stageIndex];
   516→        const next = nextStage(stage.id);
   517→        if (!next) {
   518→          get().triggerPrestige();
   519→          return;
   520→        }
   521→        const reqs = stage.evolveRequires;
   522→        const evolveBoost = 1 - Math.min(0.30, (s.upgrades["evolutionary_momentum"] || 0) * 0.05);
   523→        const totalSystems = Object.values(s.ownedSystems).reduce((a, b) => a + b, 0);
   524→        const totalTech = Object.keys(s.technologies).length;
   525→
   526→        if (s.population < reqs.minPopulation * evolveBoost) return;
   527→        if (totalSystems < reqs.minSystems * evolveBoost) return;
   528→        if (totalTech < reqs.minTech * evolveBoost) return;
   529→
   530→        const nextStageIdx = s.stageIndex + 1;

  1065→function techMultiplier(s: GameState, target: "production" | "manual" | "capacity"): number {
  1066→  let mult = 1;
  1067→  for (const [techId, owned] of Object.entries(s.technologies)) {
  1068→    if (!owned) continue;
  1069→    const tech = TECH_MAP[techId];
  1070→    if (!tech || !tech.multiplier) continue;
  1071→    if (tech.multiplier.target === target || tech.multiplier.target === "production") {
  1072→      mult += tech.multiplier.value;
  1073→    }
  1074→  }
  1075→  // Achievement production bonus
  1076→  const ctx = buildAchievementCtx(s);
  1077→  mult += achievementBonus(s.achievements || {}, ctx, "production");
  1078→  // Challenge mastery bonus (only for completed challenges, persists across runs)
  1079→  // — applied as a production multiplier (all 10 challenges grant "production" type).
  1080→  if (target === "production") {
  1081→    mult += challengeMasteryBonus(s.completedChallenges || {}, "production");
  1082→  }
  1083→  return mult;
  1084→}

  1120→function computeScore(s: GameState): number {
  1121→  const totalSystems = Object.values(s.ownedSystems).reduce((a, b) => a + b, 0);
  1122→  const totalTech = Object.keys(s.technologies).length;
  1123→  const resourceSum = Object.values(s.resources).reduce((a, b) => a + b, 0);
  1124→  return Math.floor(
  1125→    s.population * 2 +
  1126→    totalSystems * 5 +
  1127→    totalTech * 10 +
  1128→    resourceSum * 0.5 +
  1129→    s.time * 0.1
  1130→  );
  1131→}

  1133→// Recompute all capacities from scratch based on owned systems
  1134→function recomputeCapacitiesFresh(s: GameState, capMult: number): Record<string, number> {
  1135→  const base = emptyCapacities();
  1136→  const caps: Record<string, number> = {};
  1137→  for (const r of Object.keys(base)) {
  1138→    let cap = (base[r] || 50) * capMult;
  1139→    for (const sys of SYSTEMS) {
  1140→      const count = s.ownedSystems[sys.id] || 0;
  1141→      if (!count || !sys.capacityBoost) continue;
  1142→      const boost = sys.capacityBoost[r];
  1143→      if (boost) cap += (boost as number) * count * capMult;
  1144→    }
  1145→    caps[r] = cap;
  1146→  }
  1147→  return caps;
  1148→}

  1150→// Export helper for components to compute whether an action is capped
  1151→export function isActionCapped(
  1152→  produces: Record<string, number>,
  1153→  resources: Record<string, number>,
  1154→  capacities: Record<string, number>
  1155→): boolean {
  1156→  const outputs = Object.entries(produces);
  1157→  if (outputs.length === 0) return false;
  1158→  return outputs.every(([r]) => {
  1159→    const cap = capacities[r] || 0;
  1160→    return (resources[r] || 0) >= cap - 0.001;
  1161→  });
  1162→}
