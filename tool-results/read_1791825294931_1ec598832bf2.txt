    22→} from "@/game/data/constellations";
    23→import {

    27→} from "@/game/data/foresight";

    29→import { Button } from "@/components/ui/button";
    30→import { Badge } from "@/components/ui/badge";
    31→import { Separator } from "@/components/ui/separator";
    32→import { ScrollArea } from "@/components/ui/scroll-area";
    33→import { Tooltip, TooltipContent, TooltipProvider, TooltipTrigger } from "@/components/ui/tooltip";

    37→
    38→// SVG viewBox dimensions for the star map.
    39→const MAP_W = 1000;
    40→const MAP_H = 600;
    41→
    42→type AbilityMode = null | "supernova" | "blackhole";
    43→
    44→/**
    45→ * Layer 2 — Enlightenment (Constellation Map) — full redesign.
    46→ *
    47→ * Renders a star-map SVG (1000×600 viewBox) of all curated constellations.
    48→ * Each star is a dim dot by default; lit stars glow. Constellation connection
    49→ * lines appear dim until completed (or revealed by Stargaze), then bright.
    50→ * Click a star to illuminate it (calls `illuminateStar`). Sidebar shows
    51→ * progress stats, an active-constellation selector that pans/zooms the SVG,
    52→ * and the three active abilities (Stargaze, Supernova, Black Hole).
    53→ */
    54→export function EnlightenmentModal() {
    55→  const show = useGameStore((s) => s.showEnlightenment);
    56→  const setShow = useGameStore((s) => s.setShowEnlightenment);
    57→  const unlockedLayers = useGameStore((s) => s.unlockedLayers);
    58→  const divinity = useGameStore((s) => s.divinity);
    59→  // L2 state
    60→  const constellationStars = useGameStore((s) => s.constellationStars || {});
    61→  const constellationProgress = useGameStore((s) => s.constellationProgress || {});
    62→  const constellationsStarted = useGameStore((s) => s.constellationsStarted || {});
    63→  const constellationsCompleted = useGameStore((s) => s.constellationsCompleted || {});
    64→  const constellationRevealed = useGameStore((s) => s.constellationRevealed || {});
    65→  const foresightNodes = useGameStore((s) => s.foresightNodes || {});
    66→  const activeRoute = useGameStore((s) => s.activeForesightRoute);
    67→  // Layer shop upgrades
    68→  const upgrades = useGameStore((s) => s.upgrades || {});
    69→
    70→  // Actions
    71→  const illuminateStar = useGameStore((s) => s.illuminateStar);
    72→  const stargazeReveal = useGameStore((s) => s.stargazeReveal);
    73→  const supernovaIlluminate = useGameStore((s) => s.supernovaIlluminate);
    74→  const blackHoleReset = useGameStore((s) => s.blackHoleReset);
    75→  const setForesightRoute = useGameStore((s) => s.setForesightRoute);
    76→
    77→  // Local UI state
    78→  const [focusedConstellationId, setFocusedConstellationId] = useState<string | null>(null);
    79→  const [abilityMode, setAbilityMode] = useState<AbilityMode>(null);
    80→  const [flashedStars, setFlashedStars] = useState<Set<string>>(new Set());
    81→  const [collapsingStars, setCollapsingStars] = useState<Set<string>>(new Set());
    82→  const [hoveredStar, setHoveredStar] = useState<string | null>(null);
    83→  const flashTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    84→  const collapseTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    85→
    86→  // Clean up timers on unmount.
    87→  useEffect(() => {
    88→    return () => {
    89→      if (flashTimer.current) clearTimeout(flashTimer.current);
    90→      if (collapseTimer.current) clearTimeout(collapseTimer.current);
    91→    };
    92→  }, []);
    93→
    94→  const isUnlocked = !!unlockedLayers.enlightenment;
    95→
    96→  const constellationState: ConstellationState = useMemo(
    97→    () => ({
    98→      constellationStars,
    99→      constellationProgress,
   100→      constellationsStarted,
   101→      constellationsCompleted,
   102→    }),
   103→    [constellationStars, constellationProgress, constellationsStarted, constellationsCompleted]
   104→  );
   105→
   106→  const fs = foresightBonus(foresightNodes); // legacy — returns zeros for new saves
   107→  const rb = routeBonus(activeRoute);
   108→  const cb = constellationBonus(constellationState);
   109→  const completedCount = countCompletedConstellations(constellationState);
   110→  const litStarsCount = countLitStars(constellationState);
   111→  const startedCount = Object.values(constellationsStarted).filter(Boolean).length;
   112→  const totalStars = useMemo(
   113→    () => CONSTELLATIONS.reduce((acc, c) => acc + c.stars.length, 0),
   114→    []
   115→  );
   116→
   117→  // Foresight discount level (shop upgrade)
   118→  const fdLvl = (upgrades?.foresight_discount as number) || 0;
   119→  const foresightDiscountMult = Math.pow(0.80, fdLvl);
   120→
   121→  // Compute the cost for the focused constellation's next star (or first star).
   122→  const focusedNextCost = useMemo(() => {
   123→    const cid = focusedConstellationId;
   124→    if (!cid) return null;
   125→    const cdef = CONSTELLATION_MAP[cid];
   126→    if (!cdef) return null;
   127→    // First unlit star in the focused constellation
   128→    const nextStar = cdef.stars.find((st) => !constellationStars[st.id]);
   129→    if (!nextStar) return { starId: null, cost: 0, isNew: false, constellationId: cid, name: cdef.name };
   130→    const cost = starCost(cid, constellationState, nextStar.id, { foresightDiscountMult });
   131→    const isNew = (constellationProgress[cid] || 0) === 0;
   132→    return { starId: nextStar.id, cost, isNew, constellationId: cid, name: cdef.name };
   133→  }, [focusedConstellationId, constellationStars, constellationState, foresightDiscountMult, constellationProgress]);
   134→
   135→  // Compute a "first star of any unstarted constellation" cost preview (for the
   136→  // "new constellation penalty" tooltip).
   137→  const newConstellationPreview = useMemo(() => {
   138→    const cid = CONSTELLATIONS.find((c) => !constellationsStarted[c.id])?.id;
   139→    if (!cid) return null;
   140→    const cdef = CONSTELLATION_MAP[cid];
   141→    if (!cdef) return null;
   142→    const firstStar = cdef.stars[0];
   143→    const cost = starCost(cid, constellationState, firstStar.id, { foresightDiscountMult });
   144→    return { constellationId: cid, name: cdef.name, cost };
   145→  }, [constellationsStarted, constellationState, foresightDiscountMult]);
   146→
   147→  // SVG pan/zoom target based on focused constellation.
   148→  const viewBox = useMemo(() => {
   149→    if (!focusedConstellationId) return `0 0 ${MAP_W} ${MAP_H}`;
   150→    const cdef = CONSTELLATION_MAP[focusedConstellationId];
   151→    if (!cdef) return `0 0 ${MAP_W} ${MAP_H}`;
   152→    const xs = cdef.stars.map((s) => s.x);
   153→    const ys = cdef.stars.map((s) => s.y);
   154→    const minX = Math.min(...xs);
   155→    const maxX = Math.max(...xs);
   156→    const minY = Math.min(...ys);
   157→    const maxY = Math.max(...ys);
   158→    const pad = 6;
   159→    const w = (maxX - minX + pad * 2) * (MAP_W / 100);
   160→    const h = (maxY - minY + pad * 2) * (MAP_H / 100);
   161→    const cx = (minX + maxX) / 2 * (MAP_W / 100);
   162→    const cy = (minY + maxY) / 2 * (MAP_H / 100);
   163→    // Clamp to map bounds.
   164→    const halfW = w / 2;
   165→    const halfH = h / 2;
   166→    const x0 = Math.max(0, Math.min(MAP_W - w, cx - halfW));
   167→    const y0 = Math.max(0, Math.min(MAP_H - h, cy - halfH));
   168→    return `${x0} ${y0} ${w} ${h}`;
   169→  }, [focusedConstellationId]);
   170→
   171→  // Click a star: illuminate (normal mode) or apply ability (ability mode).
   172→  const onStarClick = useCallback(
   173→    (constellationId: string, starId: string) => {
   174→      if (!isUnlocked) return;
   175→      if (abilityMode === "supernova") {
   176→        supernovaIlluminate(constellationId, starId);
   177→        // Flash the supernova'd cluster (target + connected stars).
   178→        const cdef = CONSTELLATION_MAP[constellationId];
   179→        if (cdef) {
   180→          const affected = new Set<string>([starId]);
   181→          for (const [a, b] of cdef.connections) {
   182→            if (a === starId) affected.add(b);
   183→            else if (b === starId) affected.add(a);
   184→          }
   185→          setFlashedStars(affected);
   186→          if (flashTimer.current) clearTimeout(flashTimer.current);
   187→          flashTimer.current = setTimeout(() => setFlashedStars(new Set()), 600);
   188→        }
   189→        setAbilityMode(null);
   190→        return;
   191→      }
   192→      if (abilityMode === "blackhole") {
   193→        const cdef = CONSTELLATION_MAP[constellationId];
   194→        if (!cdef) return;
   195→        // Capture the lit stars in this constellation for the collapse animation.
   196→        const collapsing = new Set<string>(
   197→          cdef.stars.filter((st) => constellationStars[st.id]).map((st) => st.id)
   198→        );
   199→        blackHoleReset(constellationId);
   200→        if (collapsing.size > 0) {
   201→          setCollapsingStars(collapsing);
   202→          if (collapseTimer.current) clearTimeout(collapseTimer.current);
   203→          collapseTimer.current = setTimeout(() => setCollapsingStars(new Set()), 600);
   204→        }
   205→        setAbilityMode(null);
   206→        return;
   207→      }
   208→      illuminateStar(constellationId, starId);
   209→    },
   210→    [abilityMode, constellationStars, illuminateStar, supernovaIlluminate, blackHoleReset, isUnlocked]
   211→  );
   212→
   213→  return (
   214→    <Dialog open={show} onOpenChange={setShow}>
   215→      <DialogContent className="max-w-6xl max-h-[92vh] overflow-hidden glass-strong flex flex-col">
   216→        <DialogHeader>
   217→          <DialogTitle className="flex items-center gap-2 text-lg flex-wrap">
   218→            <Eye className="w-5 h-5" />
   219→            Layer 2 — Enlightenment (Constellation Map)
   220→            <BackgroundLayerBadge layerId="enlightenment" />
   221→          </DialogTitle>
   222→          <DialogDescription className="flex items-center justify-between flex-wrap gap-2">
   223→            <span>
   224→              Illuminate stars within {CONSTELLATIONS.length} IAU constellations. Completing a constellation grants a permanent bonus.
   225→            </span>
   226→            <Badge variant="outline" className="text-violet-300 border-violet-400/40">
   227→              {formatNumber(divinity, 0)} Divinity
   228→            </Badge>
   229→          </DialogDescription>
   230→        </DialogHeader>
   231→
   232→        {!isUnlocked ? (
   233→          <div className="text-center text-muted-foreground py-10">
   234→            <Lock className="w-8 h-8 mx-auto mb-2 opacity-50" />
   235→            <p className="text-sm">Enlightenment is sealed.</p>
   236→            <p className="text-xs mt-1">Master 3 distinct Layer 1 trials to unlock the constellation map.</p>
   237→          </div>
   238→        ) : (
   239→          <div className="flex-1 overflow-hidden grid grid-cols-1 lg:grid-cols-[1fr_320px] gap-3 min-h-0">
   240→            {/* ===== STAR MAP ===== */}
   241→            <div className="relative rounded-lg overflow-hidden border border-violet-500/20 bg-gradient-to-b from-[#020817] via-[#0a0a23] to-[#050518]">
   242→              {/* Subtle starfield background */}
   243→              <div
   244→                className="absolute inset-0 opacity-60 pointer-events-none"
   245→                style={{
   246→                  backgroundImage: `
   247→                    radial-gradient(1px 1px at 12% 18%, #fff, transparent),
   248→                    radial-gradient(1px 1px at 28% 42%, #fff, transparent),
   249→                    radial-gradient(1.5px 1.5px at 47% 12%, #fff, transparent),
   250→                    radial-gradient(1px 1px at 63% 56%, #fff, transparent),
   251→                    radial-gradient(1px 1px at 78% 28%, #fff, transparent),
   252→                    radial-gradient(1.5px 1.5px at 88% 70%, #fff, transparent),
   253→                    radial-gradient(1px 1px at 8% 72%, #fff, transparent),
   254→                    radial-gradient(1px 1px at 33% 88%, #fff, transparent),
   255→                    radial-gradient(1px 1px at 70% 85%, #fff, transparent),
   256→                    radial-gradient(1px 1px at 95% 8%, #fff, transparent),
   257→                    radial-gradient(1px 1px at 18% 50%, #fff, transparent),
   258→                    radial-gradient(1px 1px at 53% 75%, #fff, transparent)
   259→                  `,
   260→                  backgroundRepeat: "no-repeat",
   261→                  backgroundSize: "100% 100%",
   262→                }}
   263→              />
   264→              <svg
   265→                viewBox={viewBox}
   266→                className="relative w-full h-full"
   267→                style={{ aspectRatio: `${MAP_W} / ${MAP_H}`, transition: "viewBox 0.6s ease" as any }}
   268→                preserveAspectRatio="xMidYMid meet"
   269→              >
   270→                {/* Connection lines */}
   271→                {CONSTELLATIONS.flatMap((c) =>
   272→                  c.connections.map(([aId, bId], i) => {
   273→                    const aStar = c.stars.find((s) => s.id === aId);
   274→                    const bStar = c.stars.find((s) => s.id === bId);
   275→                    if (!aStar || !bStar) return null;
   276→                    const x1 = (aStar.x / 100) * MAP_W;
   277→                    const y1 = (aStar.y / 100) * MAP_H;
   278→                    const x2 = (bStar.x / 100) * MAP_W;
   279→                    const y2 = (bStar.y / 100) * MAP_H;
   280→                    const isCompleted = !!constellationsCompleted[c.id];
   281→                    const isStarted = !!constellationsStarted[c.id];
   282→                    const isRevealed = !!constellationRevealed[c.id];
   283→                    const dim = !isStarted && !isRevealed;
   284→                    const stroke = isCompleted
   285→                      ? CATEGORY_ACCENT[c.category]
   286→                      : isStarted
   287→                        ? `${CATEGORY_ACCENT[c.category]}80`
   288→                        : "#1e293b";
   289→                    return (
   290→                      <line
   291→                        key={`${c.id}-${i}`}
   292→                        x1={x1}
   293→                        y1={y1}
   294→                        x2={x2}
   295→                        y2={y2}
   296→                        stroke={stroke}
   297→                        strokeWidth={isCompleted ? 2.2 : 1}
   298→                        strokeLinecap="round"
   299→                        opacity={dim ? 0.35 : 0.9}
   300→                      />
   301→                    );
   302→                  })
   303→                )}
   304→                {/* Stars */}
   305→                {CONSTELLATIONS.map((c) =>
   306→                  c.stars.map((star) => {
   307→                    const cx = (star.x / 100) * MAP_W;
   308→                    const cy = (star.y / 100) * MAP_H;
   309→                    const lit = !!constellationStars[star.id];
   310→                    const isCompleted = !!constellationsCompleted[c.id];
   311→                    const isStarted = !!constellationsStarted[c.id];
   312→                    const isRevealed = !!constellationRevealed[c.id];
   313→                    const accent = CATEGORY_ACCENT[c.category];
   314→                    const isFocused = focusedConstellationId === c.id;
   315→                    const isHovered = hoveredStar === star.id;
   316→                    const isFlashing = flashedStars.has(star.id);
   317→                    const isCollapsing = collapsingStars.has(star.id);
   318→                    const isNewConstellation = (constellationProgress[c.id] || 0) === 0;
   319→                    const cost = starCost(c.id, constellationState, star.id, { foresightDiscountMult });
   320→                    const canAfford = divinity >= cost;
   321→                    // In ability mode, all stars are clickable regardless of cost.
   322→                    const inAbilityMode = abilityMode !== null;
   323→                    const clickable = !lit && (canAfford || inAbilityMode);
   324→                    const r = lit ? 4.5 : 2.2;
   325→                    return (
   326→                      <g
   327→                        key={star.id}
   328→                        transform={`translate(${cx},${cy})`}
   329→                        style={{ cursor: clickable ? "pointer" : "default" }}
   330→                        onClick={() => clickable && onStarClick(c.id, star.id)}
   331→                        onMouseEnter={() => setHoveredStar(star.id)}
   332→                        onMouseLeave={() => setHoveredStar(null)}
   333→                      >
   334→                        {/* Glow halo for lit/hovered */}
   335→                        {lit && (
   336→                          <circle r={r * 3} fill={accent} opacity={0.18} />
   337→                        )}
   338→                        {isHovered && !lit && (
   339→                          <circle r={r * 2.5} fill={accent} opacity={0.12} />
   340→                        )}
   341→                        {/* Flash overlay (supernova) */}
   342→                        {isFlashing && (
   343→                          <circle r={r * 5} fill="#fbbf24" opacity={0.55} />
   344→                        )}
   345→                        {/* Collapse overlay (black hole) */}
   346→                        {isCollapsing && (
   347→                          <circle r={r * 1.4} fill="#1f2937" opacity={0.8} stroke="#ef4444" strokeWidth={0.8} />
   348→                        )}
   349→                        {/* The star itself */}
   350→                        <circle
   351→                          r={r}
   352→                          fill={lit ? accent : isStarted ? "#475569" : "#1e293b"}
   353→                          stroke={lit ? "#fff" : isFocused ? accent : "transparent"}
   354→                          strokeWidth={lit || isFocused ? 0.8 : 0}
   355→                          opacity={lit ? 1 : isStarted ? 0.8 : 0.55}
   356→                        />
   357→                        {/* Cost label on hover for unlit affordable stars (normal mode) */}
   358→                        {isHovered && !lit && !inAbilityMode && (
   359→                          <text
   360→                            y={-8}
   361→                            textAnchor="middle"
   362→                            fill={canAfford ? "#a3e635" : "#f87171"}
   363→                            fontSize="9"
   364→                            fontFamily="monospace"
   365→                            pointerEvents="none"
   366→                          >
   367→                            {isNewConstellation && startedCount > 0 ? `${cost}✦` : cost}
   368→                          </text>
   369→                        )}
   370→                        {/* Ability-mode hint */}
   371→                        {isHovered && abilityMode === "supernova" && (
   372→                          <text y={-8} textAnchor="middle" fill="#fbbf24" fontSize="9" fontFamily="monospace" pointerEvents="none">
   373→                            ✦ nova
   374→                          </text>
   375→                        )}
   376→                        {isHovered && abilityMode === "blackhole" && (
   377→                          <text y={-8} textAnchor="middle" fill="#f87171" fontSize="9" fontFamily="monospace" pointerEvents="none">
   378→                            ✦ collapse
   379→                          </text>
   380→                        )}
   381→                      </g>
   382→                    );
   383→                  })
   384→                )}
   385→                {/* Constellation labels */}
   386→                {CONSTELLATIONS.map((c) => {
   387→                  const xs = c.stars.map((s) => s.x);
   388→                  const ys = c.stars.map((s) => s.y);
   389→                  const maxX = Math.max(...xs);
   390→                  const minY = Math.min(...ys);
   391→                  const lx = (maxX / 100) * MAP_W + 8;
   392→                  const ly = (minY / 100) * MAP_H + 4;
   393→                  const isCompleted = !!constellationsCompleted[c.id];
   394→                  const isStarted = !!constellationsStarted[c.id];
   395→                  const isRevealed = !!constellationRevealed[c.id];
   396→                  const dim = !isStarted && !isRevealed;
   397→                  return (
   398→                    <text
   399→                      key={`label-${c.id}`}
   400→                      x={lx}
   401→                      y={ly}
   402→                      fill={isCompleted ? CATEGORY_ACCENT[c.category] : isStarted ? "#cbd5e1" : "#475569"}
   403→                      fontSize="10"
   404→                      fontFamily="serif"
   405→                      fontStyle="italic"
   406→                      opacity={dim ? 0.35 : 0.95}
   407→                      pointerEvents="none"
   408→                    >
   409→                      {c.i18nName}
   410→                    </text>
   411→                  );
   412→                })}
   413→              </svg>
   414→              {/* Ability-mode banner */}
   415→              {abilityMode !== null && (
   416→                <div className="absolute top-2 left-2 right-2 flex items-center justify-between gap-2 rounded-md border border-amber-400/60 bg-amber-500/15 px-3 py-1.5 text-amber-200 text-xs">
   417→                  <span className="flex items-center gap-1.5">
   418→                    <Target className="w-3.5 h-3.5" />
   419→                    {abilityMode === "supernova"
   420→                      ? "Supernova armed — click a star to ignite it + its connected stars (50 Div)."
   421→                      : "Black Hole armed — click a star to collapse that constellation & refund 50% of its cost."}
   422→                  </span>
   423→                  <Button size="sm" variant="ghost" onClick={() => setAbilityMode(null)} className="h-6 px-2 text-[0.65rem]">
   424→                    Cancel
   425→                  </Button>
   426→                </div>
   427→              )}
   428→              {/* Map legend */}
   429→              <div className="absolute bottom-2 left-2 flex flex-wrap items-center gap-x-3 gap-y-1 rounded-md border border-slate-700/50 bg-slate-950/70 px-2.5 py-1 text-[0.6rem] text-slate-300 backdrop-blur">
   430→                {(["zodiac", "northern", "equatorial", "southern"] as ConstellationCategory[]).map((cat) => (
   431→                  <span key={cat} className="flex items-center gap-1">
   432→                    <span className="inline-block w-2 h-2 rounded-full" style={{ background: CATEGORY_ACCENT[cat] }} />
   433→                    {CATEGORY_LABEL[cat]}
   434→                  </span>
   435→                ))}
   436→              </div>
   437→            </div>
   438→
   439→            {/* ===== SIDEBAR ===== */}
   440→            <div className="flex flex-col gap-2 overflow-hidden min-h-0">
   441→              <ScrollArea className="flex-1 pr-2">
   442→                <div className="space-y-3">
   443→                  {/* Progress stats */}
   444→                  <div className="stat-card">
   445→                    <div className="text-[0.7rem] uppercase tracking-wide text-muted-foreground mb-1.5">
   446→                      Constellation progress
   447→                    </div>
   448→                    <div className="grid grid-cols-2 gap-2 text-xs">
   449→                      <div className="flex flex-col">
   450→                        <span className="text-[0.6rem] text-muted-foreground">Completed</span>
   451→                        <span className="text-amber-300 font-semibold">
   452→                          {completedCount} / {CONSTELLATIONS_REQUIRED_FOR_LAYER_3}
   453→                          <span className="text-muted-foreground text-[0.6rem] ml-1">→ L3</span>
   454→                        </span>
   455→                      </div>
   456→                      <div className="flex flex-col">
