    52→// Display-friendly key label. Space → "Space", Escape → "Esc", letters
    53→// uppercased for readability.
    54→function keyLabel(key: string): string {
    55→  if (key === " ") return "Space";
    56→  if (key === "Escape") return "Esc";
    57→  if (key.length === 1) return key.toUpperCase();
    58→  return key;
    59→}

   718→// ============================================================
   719→// ShortcutsTab — rebindable keyboard shortcut list.
   720→// Each row shows: action name · current key (click to rebind) ·
   721→// for locked-layer actions: a "???" badge + the key button is disabled.
   722→// ============================================================
   723→function ShortcutsTab() {
   724→  const shortcuts = useGameStore((s) => s.keyboardShortcuts) || {};
   725→  const setKeyboardShortcut = useGameStore((s) => s.setKeyboardShortcut);
   726→  const resetKeyboardShortcuts = useGameStore((s) => s.resetKeyboardShortcuts);
   727→  const unlockedChallenges = useGameStore((s) => s.unlockedChallenges);
   728→  const unlockedLayers = useGameStore((s) => s.unlockedLayers);
   729→
   730→  // Which action is currently in "listening for key" mode.
   731→  const [listeningAction, setListeningAction] = useState<string | null>(null);
   732→
   733→  const isLocked = useCallback((actionId: string): boolean => {
   734→    const layerId = SHORTCUT_LAYER_GATE[actionId];
   735→    if (!layerId) return false;
   736→    if (layerId === "challenges") return !unlockedChallenges;
   737→    return !unlockedLayers?.[layerId as keyof typeof unlockedLayers];
   738→  }, [unlockedChallenges, unlockedLayers]);
   739→
   740→  // Global key listener while an action is in "listening" mode. Captures any
   741→  // key (including Escape, which would otherwise close the modal via the main
   742→  // keyboard-shortcut handler) and binds it to the active action.
   743→  useEffect(() => {
   744→    if (!listeningAction) return;
   745→    const handler = (e: KeyboardEvent) => {
   746→      e.preventDefault();
   747→      // stopImmediatePropagation is required (not just stopPropagation) —
   748→      // both this listener AND the main useKeyboardShortcuts handler are
   749→      // registered on `window`. stopImmediatePropagation prevents other
   750→      // listeners on the SAME target from firing.
   751→      e.stopImmediatePropagation();
   752→      // Ignore pure modifier keys — they shouldn't be a bind on their own.
   753→      if (e.key === "Shift" || e.key === "Control" || e.key === "Alt" || e.key === "Meta") return;
   754→      setKeyboardShortcut(listeningAction, e.key);
   755→      setListeningAction(null);
   756→    };
   757→    // Capture-phase so we beat the main shortcut handler to the event.
   758→    window.addEventListener("keydown", handler, true);
   759→    return () => window.removeEventListener("keydown", handler, true);
   760→  }, [listeningAction, setKeyboardShortcut]);
   761→
   762→  // Stable action-id ordering (defaults order, then any custom extras).
   763→  const actionIds = Object.keys(DEFAULT_KEYBOARD_SHORTCUTS);
   764→
   765→  return (
   766→    <div className="space-y-3">
   767→      <div className="flex items-start justify-between gap-3">
   768→        <div>
   769→          <h3 className="text-sm font-semibold flex items-center gap-1.5">
   770→            <Keyboard className="w-4 h-4 text-primary" />
   771→            Keyboard Shortcuts
   772→          </h3>
   773→          <p className="text-xs text-muted-foreground mt-0.5">
   774→            Click any key to rebind. Press the new key to set it. Locked layers show <span className="text-amber-300 font-mono">???</span> and can&apos;t be triggered until unlocked.
   775→          </p>
   776→        </div>
   777→        <Button
   778→          size="sm"
   779→          variant="outline"
   780→          onClick={resetKeyboardShortcuts}
   781→          className="h-8 shrink-0"
   782→          title="Reset all shortcuts to defaults"
   783→        >
   784→          <RotateCcw className="w-3 h-3 mr-1" />
   785→          Reset
   786→        </Button>
   787→      </div>
   788→
   789→      <Separator />
   790→
   791→      <div className="space-y-1.5">
   792→        {actionIds.map((actionId) => {
   793→          const key = shortcuts[actionId] ?? DEFAULT_KEYBOARD_SHORTCUTS[actionId];
   794→          const locked = isLocked(actionId);
   795→          const listening = listeningAction === actionId;
   796→          return (
   797→            <ShortcutRow
   798→              key={actionId}
   799→              label={SHORTCUT_LABELS[actionId] ?? actionId}
   800→              keyStr={key}
   801→              locked={locked}
   802→              listening={listening}
   803→              onRebind={() => setListeningAction(listening ? null : actionId)}
   804→              onCancel={() => setListeningAction(null)}
   805→            />
   806→          );
   807→        })}
   808→      </div>
   809→    </div>
   810→  );
   811→}

   813→function ShortcutRow({
   814→  label,
   815→  keyStr,
   816→  locked,
   817→  listening,
   818→  onRebind,
   819→  onCancel,
   820→}: {
   821→  label: string;
   822→  keyStr: string;
   823→  locked: boolean;
   824→  listening: boolean;
   825→  onRebind: () => void;
   826→  onCancel: () => void;
   827→}) {
   828→  // Auto-cancel "listening" mode after 3s of no input.
   829→  const cancelTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   830→  useEffect(() => {
   831→    if (!listening) return;
   832→    cancelTimer.current = setTimeout(() => onCancel(), 3000);
   833→    return () => {
   834→      if (cancelTimer.current) clearTimeout(cancelTimer.current);
   835→    };
   836→  }, [listening, onCancel]);
   837→
   838→  return (
   839→    <div className="flex items-center justify-between gap-3 py-1.5 px-2 rounded border border-border/50 bg-muted/20">
   840→      <div className="flex items-center gap-2 min-w-0">
   841→        <span className="text-xs font-medium truncate">{label}</span>
   842→        {locked && (
   843→          <Badge variant="outline" className="text-[0.55rem] text-amber-300 border-amber-500/40 px-1 py-0 shrink-0">
   844→            ???
   845→          </Badge>
   846→        )}
   847→      </div>
   848→      <button
   849→        onClick={onRebind}
   850→        disabled={locked && !listening}
   851→        className={cn(
   852→          "px-2.5 py-1 rounded border text-xs font-mono font-semibold min-w-[3.5rem] text-center shrink-0 transition-colors",
   853→          listening
   854→            ? "border-primary bg-primary/15 text-primary animate-pulse"
   855→            : locked
   856→              ? "border-border/40 bg-muted/10 text-muted-foreground/50 cursor-not-allowed"
   857→              : "border-border bg-muted/40 hover:bg-muted/60 text-foreground"
   858→        )}
   859→        title={
   860→          listening
   861→            ? "Press any key to bind…"
   862→            : locked
   863→              ? "Locked layer"
   864→              : "Click to rebind"
   865→        }
   866→      >
   867→        {listening ? "…" : locked ? "???" : keyLabel(keyStr)}
   868→      </button>
   869→    </div>
   870→  );
   871→}
