     1→// Random events — sporadic occurrences that give the player choices
     2→// Each event has a condition (when it can fire), a description, and choices with effects
     3→
     4→import type { ResourceId } from "../state/types";
     5→
     6→export interface EventChoice {
     7→  id: string;
     8→  label: string;
     9→  desc: string;
    10→  // Effects: positive numbers add, negative subtract
    11→  resourceChanges?: Partial<Record<ResourceId, number>>;
    12→  populationChange?: number;
    13→  happinessChange?: number;
    14→  affinityChange?: { archetype: string; amount: number }[];
    15→  // L10-ENDING-L5-UI-UNLOCKTECH — when set, resolving this choice also
    16→  // unlocks the named tech (sets `technologies[unlockTech] = true`). Useful
    17→  // for narrative "discovery" events that should grant a tech without
    18→  // the player having to spend science on the Tech tab.
    19→  unlockTech?: string;
    20→  // Log message to add after choosing
    21→  logMsg?: string;
    22→}
    23→
    24→export interface GameEvent {
    25→  id: string;
    26→  name: string;
    27→  desc: string;
    28→  icon: string;
    29→  // Which stages this event can fire in
    30→  stages: string[];
    31→  // Minimum game time before this event can fire (seconds)
    32→  minTime: number;
    33→  // Weight for random selection (higher = more common)
    34→  weight: number;
    35→  choices: EventChoice[];
    36→  // ARCHETYPES-REWORK — if set, the event only fires when the player's locked
    37→  // archetype matches this id (used by the 90 archetype-specific lineage events).
    38→  requiredArchetype?: string;
    39→  // IMMUNITY-TECHS — if set, the event is excluded from the possible pool when
    40→  // the player has researched the tech with this id (e.g. dna_repair blocks
    41→  // mutation events).
    42→  immuneTech?: string;
    43→}
