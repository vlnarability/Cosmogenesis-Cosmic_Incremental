    28→// ===== Part 6 — Layer-progress helpers =====
    29→// Per-layer (L_n → L_n+1) unlock progress. Returns the current count and the max (target) count.
    30→// nextLayerKey uses the runtime unlockedLayers keys (e.g. "enlightenment" for L2).
    31→interface NextLayerProgressDef {
    32→  currentLayerName: string; // L_n name (the layer the player needs to have unlocked to progress)
    33→  nextLayerName: string;    // L_n+1 name (the layer the progress bar points toward)
    34→  nextLayerKey: string;     // key in unlockedLayers (or "evolution" for L1)
    35→  current: number;
    36→  max: number;
    37→  metricLabel: string;      // description of the metric (e.g. "Trials completed")
    38→}
    39→
    40→// ===== Part 6 — Layer-next-progress component =====
    41→// Renders a progress bar showing how close the player is to unlocking the NEXT layer (L_(n+1)).
    42→// - If next layer is already unlocked: "✓ {nextLayerName} Unlocked"
    43→// - If current layer (prereq for next) is NOT unlocked: "Requires {currentLayerName}"
    44→// - Otherwise: progress bar with the layer-unlock metric (e.g. "5/10 Foresight nodes").
    45→function LayerNextProgress({
    46→  progress,
    47→  currentLayerUnlocked,
    48→  nextLayerUnlocked,
    49→}: {
    50→  progress: NextLayerProgressDef;
    51→  currentLayerUnlocked: boolean;
    52→  nextLayerUnlocked: boolean;
    53→}) {
    54→  if (nextLayerUnlocked) {
    55→    return (
    56→      <div className="mt-2 flex items-center gap-1.5 text-[0.65rem] text-emerald-300">
    57→        <CheckCircle2 className="w-3 h-3" />
    58→        <span className="font-semibold">{progress.nextLayerName} Unlocked</span>
    59→      </div>
    60→    );
    61→  }
    62→  if (!currentLayerUnlocked) {
    63→    return (
    64→      <div className="mt-2 flex items-center gap-1.5 text-[0.65rem] text-muted-foreground">
    65→        <Lock className="w-3 h-3" />
    66→        <span>Requires <span className="font-semibold text-foreground/80">{progress.currentLayerName}</span> to be unlocked first</span>
    67→      </div>
    68→    );
    69→  }
    70→  const pct = progress.max > 0 ? Math.min(100, (progress.current / progress.max) * 100) : 0;
    71→  return (
    72→    <div className="mt-2">
    73→      <div className="flex items-center justify-between text-[0.6rem] text-muted-foreground">
    74→        <span className="flex items-center gap-1">
    75→          <ArrowRight className="w-2.5 h-2.5" />
    76→          Toward {progress.nextLayerName}
    77→          <span className="text-muted-foreground/70">· {progress.metricLabel}</span>
    78→        </span>
    79→        <span className="tabular-nums">{progress.current}/{progress.max}</span>
    80→      </div>
    81→      <div className="h-1.5 bg-muted/40 rounded-full overflow-hidden mt-0.5">
    82→        <div className="h-full bg-emerald-400/70 transition-all" style={{ width: `${pct}%` }} />
    83→      </div>
    84→    </div>
    85→  );
    86→}

   754→function ProgressRow({ label, value, max }: { label: string; value: number; max: number }) {
   755→  const pct = max > 0 ? Math.min(100, (value / max) * 100) : 0;
   756→  return (
   757→    <div className="mt-1">
   758→      <div className="flex items-center justify-between text-[0.6rem] text-muted-foreground">
   759→        <span>{label}</span>
   760→        <span>{value}/{max}</span>
   761→      </div>
   762→      <div className="h-1 bg-muted/40 rounded-full overflow-hidden mt-0.5">
   763→        <div className="h-full bg-amber-400/70 transition-all" style={{ width: `${pct}%` }} />
   764→      </div>
   765→    </div>
   766→  );
   767→}

   809→function formatUnlock(layer: PrestigeLayer): string {
   810→  switch (layer.unlockCondition.type) {
   811→    case "default":
   812→      return "Default";
   813→    case "galacticWins":
   814→      return `${layer.unlockCondition.value} wins`;
   815→    case "challengesCompleted":
   816→      return `${layer.unlockCondition.value} trials`;
   817→    case "totalRuns":
   818→      return `${layer.unlockCondition.value} runs`;
   819→    default:
   820→      return "Locked";
   821→  }
   822→}
