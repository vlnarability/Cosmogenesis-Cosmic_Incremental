    44→// ----- helpers -----
    45→
    46→function ruleShortName(ruleId: string): string {
    47→  const r = UNIVERSE_RULES.find((x) => x.id === ruleId);
    48→  return r ? r.shortName : ruleId;
    49→}
    50→
    51→function ruleIcon(ruleId: string): string {
    52→  const r = UNIVERSE_RULES.find((x) => x.id === ruleId);
    53→  return r ? r.icon : "?";
    54→}
    55→
    56→function amountLabel(amt: number): string {
    57→  return `${amt >= 0 ? "+" : ""}${amt}`;
    58→}
    59→
    60→// ----- main component -----
    61→
    62→export function EternityContent() {
    63→  const closeLayerModal = useGameStore((s) => s.closeLayerModal);
    64→  const unlockedLayers = useGameStore((s) => s.unlockedLayers);
    65→  const divinity = useGameStore((s) => s.divinity || 0);
    66→  const cosmicBoonStacks = useGameStore((s) => s.cosmicBoonStacks || 0);
    67→  const chosenEnding = useGameStore((s) => s.chosenEnding);
    68→
    69→  // Genesis Engine state
    70→  const divineRelics = useGameStore((s) => s.divineRelics || []);
    71→  const crucibleRelicIds = useGameStore((s) => s.crucibleRelicIds || []);
    72→  const discoveredArchetypes = useGameStore((s) => s.discoveredArchetypes || {});
    73→  const universeSimulating = useGameStore((s) => s.universeSimulating);
    74→  const simulationStartTime = useGameStore((s) => s.simulationStartTime || 0);
    75→  const lastSimResult = useGameStore((s) => s.lastSimulationResult);
    76→  // Subscribe to time so the progress bar updates each tick.
    77→  const time = useGameStore((s) => s.time || 0);
    78→  const totalPlayTime = useGameStore((s) => s.totalPlayTime || 0);
    79→  const totalRuns = useGameStore((s) => s.totalRuns || 0);
    80→
    81→  // Actions
    82→  const addRelicToCrucible = useGameStore((s) => s.addRelicToCrucible);
    83→  const clearCrucible = useGameStore((s) => s.clearCrucible);
    84→  const revealRelicRule = useGameStore((s) => s.revealRelicRule);
    85→  const revealRelicAmount = useGameStore((s) => s.revealRelicAmount);
    86→  const triggerBigBang = useGameStore((s) => s.triggerBigBang);
    87→  const chooseEnding = useGameStore((s) => s.chooseEnding);
    88→
    89→  const isUnlocked = !!unlockedLayers.eternity;
    90→  const discoveredCount = Object.values(discoveredArchetypes).filter(Boolean).length;
    91→  const allDiscovered = discoveredCount >= ARCHETYPE_COUNT;
    92→  const boon = cosmicBoonBonus(cosmicBoonStacks);
    93→  // L10-ENDING-L5-UI-UNLOCKTECH — visited dimensions always starts with
    94→  // ["mundane"]; we read it here so the EndingScreen's dimension picker
    95→  // can determine which dimensions are selectable.
    96→  const visitedDimensions = useGameStore((s) => s.visitedDimensions || ["mundane"]);
    97→
    98→  // Compute crucible rule view (recomputed each render — cheap).
    99→  const crucibleRelics: DivineRelic[] = useMemo(() => {
   100→    return crucibleRelicIds
   101→      .map((id) => divineRelics.find((r) => r.id === id))
   102→      .filter((r): r is DivineRelic => !!r);
   103→  }, [crucibleRelicIds, divineRelics]);
   104→
   105→  const crucibleComp = useMemo(
   106→    () => computeCrucibleRules(crucibleRelics),
   107→    [crucibleRelics]
   108→  );
   109→
   110→  const simElapsed = universeSimulating ? Math.min(BIG_BANG_DURATION_SEC, time - simulationStartTime) : 0;
   111→  // P7 — simProgressPct removed; Big Bang now shows a skeleton screen instead
   112→  // of a progress bar. simElapsed is still used for the seconds-remaining count.
   113→  const simRemainingSec = universeSimulating ? Math.max(0, BIG_BANG_DURATION_SEC - simElapsed) : 0;
   114→
   115→  // Ending confirmation dialog
   116→  const [pendingEnding, setPendingEnding] = useState<"preserve" | "reset" | null>(null);
   117→  const confirmEnding = () => {
   118→    if (pendingEnding) {
   119→      chooseEnding(pendingEnding);
   120→      setPendingEnding(null);
   121→    }
   122→  };
   123→
   124→  // Total score for the ending screen
   125→  const finalScore = useMemo(() => {
   126→    // Resources: sum of all primary resources.
   127→    const resourceIds: Array<string> = ["atp", "glucose", "proteins", "lipids", "elements",
   128→      "food", "water", "materials", "organic_matter", "knowledge", "culture",
   129→      "wood", "lumber", "stone", "clay", "science", "happiness", "military_power",
   130→      "production", "gold", "influence", "energy", "alloys", "data",
   131→      "divinity", "evolution_points", "enlightenment_points", "transcendence_points"];
   132→    const resources = useGameStore.getState().resources || {};
   133→    let sum = 0;
   134→    for (const id of resourceIds) {
   135→      sum += (resources as any)[id] || 0;
   136→    }
   137→    // Layers completed count
   138→    const unlocked = useGameStore.getState().unlockedLayers || {};
   139→    const layersCompleted = Object.values(unlocked).filter(Boolean).length;
   140→    // Worlds: authored worlds + claimed galaxy systems
   141→    const authoredWorlds = (useGameStore.getState() as any).authoredWorlds || [];
   142→    const galaxySystems = (useGameStore.getState() as any).galaxySystems || [];
   143→    const claimedGalaxies = galaxySystems.filter((g: any) => g?.claimed).length;
   144→    const worldCount = authoredWorlds.length + claimedGalaxies;
   145→    // Final score: weighted sum
   146→    return Math.floor(sum * 0.01 + worldCount * 100 + layersCompleted * 500 + discoveredCount * 1000);
   147→  }, [discoveredCount, allDiscovered]);
   148→
   149→  return (
   150→    <>
   151→      <DialogHeader>
   152→        <DialogTitle className="flex items-center gap-2 text-lg flex-wrap">
   153→          <Star className="w-5 h-5" />
   154→          Layer 10 — Eternity (Genesis Engine)
   155→          <BackgroundLayerBadge layerId="eternity" />
   156→        </DialogTitle>
   157→        <DialogDescription className="flex items-center justify-between flex-wrap gap-2">
   158→          <span>
   159→            Load relics into the Crucible, trigger Big Bangs, and discover all 100 archetypes.
   160→          </span>
   161→          <div className="flex items-center gap-1.5">
   162→            <Badge variant="outline" className="text-violet-300 border-violet-400/40">
   163→              ✨ {formatNumber(divinity, 0)} Divinity
   164→            </Badge>
   165→            <Badge variant="outline" className="text-amber-300 border-amber-400/40">
   166→              🌟 {discoveredCount}/{ARCHETYPE_COUNT}
   167→            </Badge>
   168→            {cosmicBoonStacks > 0 && (
   169→              <Badge variant="outline" className="text-indigo-300 border-indigo-400/40">
   170→                🌠 {cosmicBoonStacks} Boon{cosmicBoonStacks !== 1 ? "s" : ""}
   171→              </Badge>
   172→            )}
   173→          </div>
   174→        </DialogDescription>
   175→      </DialogHeader>
   176→
   177→        {!isUnlocked ? (
   178→          <div className="text-center text-muted-foreground py-10">
   179→            <Lock className="w-8 h-8 mx-auto mb-2 opacity-50" />
   180→            <p className="text-sm">The Genesis Engine lies dormant.</p>
   181→            <p className="text-xs mt-1">Defeat all 3 Old Gods (Layer 9 — Divine War) to ignite the Crucible.</p>
   182→          </div>
   183→        ) : allDiscovered && chosenEnding === null ? (
   184→          <EndingScreen
   185→            discoveredCount={discoveredCount}
   186→            totalPlayTime={totalPlayTime}
   187→            thisRunTime={time}
   188→            totalRuns={totalRuns}
   189→            finalScore={finalScore}
   190→            boon={boon}
   191→            cosmicBoonStacks={cosmicBoonStacks}
   192→            visitedDimensions={visitedDimensions}
   193→            onChooseEnding={(id) => setPendingEnding(id)}
   194→          />
   195→        ) : chosenEnding !== null ? (
   196→          <div className="text-center py-10">
   197→            <Sparkles className="w-8 h-8 mx-auto mb-2 text-amber-300" />
   198→            <p className="text-sm">Ending chosen: {ENDING_CHOICES.find((e) => e.id === chosenEnding)?.name}</p>
   199→            <p className="text-xs mt-1 text-muted-foreground">
   200→              {ENDING_CHOICES.find((e) => e.id === chosenEnding)?.desc}
   201→            </p>
   202→            <Button variant="outline" className="mt-4" onClick={() => closeLayerModal()}>Close</Button>
   203→          </div>
   204→        ) : (
   205→          <div className="space-y-4">
   206→            {/* Relic Inventory */}
   207→            <section>
   208→              <div className="text-[0.7rem] uppercase tracking-wide text-muted-foreground mb-1 flex items-center gap-1 justify-between">
   209→                <span className="flex items-center gap-1">
   210→                  <Sparkles className="w-3 h-3" /> Divine Relic Inventory ({divineRelics.length})
   211→                </span>
   212→                <span className="text-[0.6rem] normal-case">
   213→                  Reveal rule name: {REVEAL_COST_RULE} Div · amount: {REVEAL_COST_AMOUNT} Div
   214→                </span>
   215→              </div>
   216→              {divineRelics.length === 0 ? (
   217→                // P7 — Empty state: no relics yet. Use the shared EmptyState
   218→                // component for consistent styling. The StarterRelicHint below
   219→                // points the player at the Big Bang button (which grants the
   220→                // first relic without needing a prior simulation).
   221→                <>
   222→                  <EmptyState
   223→                    compact
   224→                    icon={<Atom className="w-6 h-6" />}
   225→                    title="No relics yet."
   226→                    description="Complete a universe simulation (Big Bang) to earn relics. Use the starter relic below to trigger your first Big Bang."
   227→                  />
   228→                  <div className="mt-2">
   229→                    <StarterRelicHint />
   230→                  </div>
   231→                </>
   232→              ) : (
   233→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 max-h-72 overflow-y-auto pr-1">
   234→                  {divineRelics.map((r) => {
   235→                    const inCrucible = crucibleRelicIds.includes(r.id);
   236→                    return (
   237→                      <div
   238→                        key={r.id}
   239→                        className={`stat-card p-2 ${inCrucible ? "border-amber-400/70 bg-amber-500/10" : "border-slate-600/40"}`}
   240→                      >
   241→                        <div className="flex items-center gap-1 mb-1">
   242→                          <Atom className="w-3 h-3 text-amber-300" />
   243→                          <span className="text-xs font-medium truncate flex-1">{r.name}</span>
   244→                          {inCrucible && (
   245→                            <Badge variant="outline" className="text-[0.55rem] text-amber-300 border-amber-400/60">in crucible</Badge>
   246→                          )}
   247→                        </div>
   248→                        <div className="space-y-1">
   249→                          {r.rules.map((rr, idx) => {
   250→                            const showRule = rr.revealed.rule;
   251→                            const showAmt = rr.revealed.amount;
   252→                            return (
   253→                              <div key={idx} className="flex items-center gap-1 text-[0.65rem]">
   254→                                <span className="w-4 text-center">{showRule ? ruleIcon(rr.ruleId) : "?"}</span>
   255→                                <span className={`flex-1 truncate ${showRule ? "text-foreground" : "text-muted-foreground/70"}`}>
   256→                                  {showRule ? ruleShortName(rr.ruleId) : "???"}
   257→                                </span>
   258→                                <span className={`font-mono ${showAmt ? (rr.amount >= 0 ? "text-emerald-300" : "text-rose-300") : "text-muted-foreground/70"}`}>
   259→                                  {showAmt ? amountLabel(rr.amount) : "???"}
   260→                                </span>
   261→                                <div className="flex items-center gap-0.5">
   262→                                  {!showRule && (
   263→                                    <button
   264→                                      onClick={() => revealRelicRule(r.id, idx)}
   265→                                      disabled={divinity < REVEAL_COST_RULE || universeSimulating}
   266→                                      className="text-[0.55rem] px-1 py-0.5 rounded border border-cyan-400/40 text-cyan-300 hover:bg-cyan-500/10 disabled:opacity-40 disabled:cursor-not-allowed"
   267→                                      title={`Reveal rule name (${REVEAL_COST_RULE} Divinity)`}
   268→                                    >
   269→                                      <Eye className="w-2.5 h-2.5" />
   270→                                    </button>
   271→                                  )}
   272→                                  {showRule && !showAmt && (
   273→                                    <button
   274→                                      onClick={() => revealRelicAmount(r.id, idx)}
   275→                                      disabled={divinity < REVEAL_COST_AMOUNT || universeSimulating}
   276→                                      className="text-[0.55rem] px-1 py-0.5 rounded border border-amber-400/40 text-amber-300 hover:bg-amber-500/10 disabled:opacity-40 disabled:cursor-not-allowed"
   277→                                      title={`Reveal amount (${REVEAL_COST_AMOUNT} Divinity)`}
   278→                                    >
   279→                                      <Eye className="w-2.5 h-2.5" />
   280→                                    </button>
   281→                                  )}
   282→                                  {showRule && showAmt && (
   283→                                    <CheckCircle2 className="w-2.5 h-2.5 text-emerald-400" />
   284→                                  )}
   285→                                </div>
   286→                              </div>
   287→                            );
   288→                          })}
   289→                        </div>
   290→                        {!inCrucible && (
   291→                          <Button
   292→                            size="sm"
   293→                            variant="outline"
   294→                            className="w-full h-6 mt-1.5 text-[0.6rem] text-amber-300 border-amber-400/40"
   295→                            disabled={crucibleRelicIds.length >= 5 || universeSimulating}
   296→                            onClick={() => addRelicToCrucible(r.id)}
   297→                          >
   298→                            <Plus className="w-3 h-3 mr-1" />
   299→                            Add to Crucible ({crucibleRelicIds.length}/5)
   300→                          </Button>
   301→                        )}
   302→                      </div>
   303→                    );
   304→                  })}
   305→                </div>
   306→              )}
   307→            </section>
   308→
   309→            {/* Crucible */}
   310→            <section>
   311→              <div className="text-[0.7rem] uppercase tracking-wide text-muted-foreground mb-1 flex items-center gap-1 justify-between">
   312→                <span className="flex items-center gap-1">
   313→                  <Atom className="w-3 h-3" /> The Crucible ({crucibleRelicIds.length}/5 relics)
   314→                </span>
   315→                <div className="flex items-center gap-1">
   316→                  <Button
   317→                    size="sm"
   318→                    variant="outline"
   319→                    className="h-5 text-[0.55rem] text-rose-300 border-rose-400/40"
   320→                    disabled={crucibleRelicIds.length === 0 || universeSimulating}
   321→                    onClick={() => clearCrucible()}
   322→                    title="Clear all relics from the crucible"
   323→                  >
   324→                    <RefreshCw className="w-2.5 h-2.5 mr-0.5" /> Clear
   325→                  </Button>
   326→                </div>
   327→              </div>
   328→              <div className="stat-card p-2">
   329→                {/* Crucible slots */}
   330→                <div className="grid grid-cols-5 gap-1 mb-2">
   331→                  {[0, 1, 2, 3, 4].map((slotIdx) => {
   332→                    const relic = crucibleRelics[slotIdx];
   333→                    return (
   334→                      <div
   335→                        key={slotIdx}
   336→                        className={`rounded border p-1 text-center text-[0.55rem] ${relic ? "border-amber-400/60 bg-amber-500/10 text-amber-200" : "border-slate-700/40 text-muted-foreground/60"}`}
   337→                      >
   338→                        {relic ? (
   339→                          <div className="truncate" title={relic.name}>
   340→                            <Atom className="w-2.5 h-2.5 inline mr-0.5" />
   341→                            {relic.name.replace("Relic of the ", "")}
   342→                          </div>
   343→                        ) : (
   344→                          <span>empty</span>
   345→                        )}
   346→                      </div>
   347→                    );
   348→                  })}
   349→                </div>
   350→                {/* Computed rule levels */}
   351→                <div className="space-y-1">
   352→                  {crucibleComp.rules.map((rv) => {
   353→                    const stable = rv.finalValue >= RULE_STABLE_MIN && rv.finalValue <= RULE_STABLE_MAX;
   354→                    const low = rv.finalValue < RULE_STABLE_MIN;
   355→                    const high = rv.finalValue > RULE_STABLE_MAX;
   356→                    return (
   357→                      <div key={rv.ruleId} className="flex items-center gap-2 text-[0.7rem]">
   358→                        <span className="w-5 text-center">{ruleIcon(rv.ruleId)}</span>
   359→                        <span className="w-28 truncate">{ruleShortName(rv.ruleId)}</span>
   360→                        {/* Displayed value: revealed portion + mystery marker */}
   361→                        <span className={`font-mono w-14 text-right ${stable ? "text-emerald-300" : low ? "text-rose-300" : high ? "text-amber-300" : "text-foreground"}`}>
   362→                          {rv.revealedValue}
   363→                          {rv.hasMystery ? "±?" : ""}
   364→                        </span>
   365→                        <span className="text-muted-foreground/60 text-[0.6rem]">→</span>
   366→                        {/* Final value (after overflow) */}
   367→                        <span className={`font-mono w-8 text-right font-bold ${stable ? "text-emerald-300" : low ? "text-rose-300" : high ? "text-amber-300" : "text-foreground"}`}>
   368→                          {rv.finalValue}
   369→                        </span>
   370→                        {(low || high) && (
   371→                          <AlertTriangle className={`w-3 h-3 ${low ? "text-rose-400" : "text-amber-400"}`} />
   372→                        )}
   373→                      </div>
   374→                    );
   375→                  })}
   376→                </div>
   377→                <div className="flex items-center justify-between mt-2 pt-2 border-t border-slate-700/40 text-[0.65rem]">
   378→                  <span className="text-muted-foreground">
   379→                    Total: <span className={`font-mono font-bold ${crucibleComp.total === RULE_TOTAL_TARGET ? "text-emerald-300" : "text-rose-300"}`}>{crucibleComp.total}</span> / {RULE_TOTAL_TARGET}
   380→                  </span>
   381→                  <span className="text-muted-foreground">
   382→                    Base per rule: {RULE_BASE} · Stability: <span className={crucibleComp.unstable ? "text-rose-300" : "text-emerald-300"}>
   383→                      {crucibleComp.unstable ? `UNSTABLE (instability ${crucibleComp.instability})` : "stable"}
   384→                    </span>
   385→                  </span>
   386→                </div>
   387→              </div>
   388→              {/* Big Bang button */}
   389→              <div className="mt-2">
   390→                {universeSimulating ? (
   391→                  // P7 — Loading state: skeleton screen during the 30-second
   392→                  // Big Bang simulation. Show pulsing placeholder relic slots
   393→                  // (instead of just a progress bar) + a "Simulating universe..."
   394→                  // status line. Keep the seconds-remaining indicator so the
   395→                  // player knows roughly how long is left.
   396→                  <div className="stat-card p-3 space-y-2">
   397→                    <div className="flex items-center justify-between text-[0.7rem]">
   398→                      <span className="text-amber-200 flex items-center gap-1 animate-pulse">
   399→                        <Sparkles className="w-3 h-3" /> Simulating universe…
   400→                      </span>
   401→                      <span className="font-mono text-amber-300">{simRemainingSec.toFixed(0)}s</span>
   402→                    </div>
   403→                    {/* Pulsing placeholder cards — match the relic grid layout
   404→                        so the layout doesn't shift when the result returns. */}
   405→                    <div className="grid grid-cols-2 sm:grid-cols-4 gap-1.5">
   406→                      {Array.from({ length: 4 }).map((_, i) => (
   407→                        <div
   408→                          key={i}
   409→                          className="h-12 rounded-md border border-amber-400/20 bg-amber-500/5 animate-pulse"
   410→                          style={{ animationDelay: `${i * 150}ms` }}
   411→                          aria-hidden
   412→                        />
   413→                      ))}
   414→                    </div>
   415→                    <div className="text-[0.6rem] text-muted-foreground text-center">
   416→                      The crucible blazes — reality crystallizes…
   417→                    </div>
   418→                  </div>
   419→                ) : (
   420→                  <Button
   421→                    className="w-full h-9 bg-gradient-to-r from-amber-600 via-rose-600 to-violet-600 hover:from-amber-500 hover:via-rose-500 hover:to-violet-500 text-white font-bold"
   422→                    disabled={crucibleRelicIds.length < 1}
   423→                    onClick={() => triggerBigBang()}
   424→                  >
   425→                    <Sparkles className="w-4 h-4 mr-2" />
   426→                    Trigger Big Bang {crucibleRelicIds.length < 1 ? "(add a relic first)" : `(${crucibleRelicIds.length} relic${crucibleRelicIds.length !== 1 ? "s" : ""})`}
   427→                  </Button>
   428→                )}
   429→              </div>
   430→            </section>
   431→
   432→            {/* Last simulation result */}
   433→            {lastSimResult && (
   434→              <section>
   435→                <div className="text-[0.7rem] uppercase tracking-wide text-muted-foreground mb-1">
   436→                  Last Big Bang Result
   437→                </div>
   438→                <div className={`stat-card p-2 ${lastSimResult.stable ? "border-emerald-400/60 bg-emerald-500/5" : "border-rose-400/60 bg-rose-500/5"}`}>
   439→                  {lastSimResult.stable ? (
   440→                    <div className="text-[0.7rem]">
   441→                      <div className="flex items-center gap-1 text-emerald-300 font-semibold">
   442→                        <CheckCircle2 className="w-3 h-3" /> STABLE
   443→                      </div>
   444→                      <div className="text-muted-foreground mt-1">
   445→                        {lastSimResult.archetypeId ? (
   446→                          <>Discovered archetype: <span className="text-amber-300 font-semibold">{ARCHETYPE_MAP[lastSimResult.archetypeId]?.name || "Unknown"}</span></>
   447→                        ) : (
   448→                          <>No archetype matched this combination.</>
   449→                        )}
