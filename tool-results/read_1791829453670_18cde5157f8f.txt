    66→/**
    67→ * Layer 2 — Enlightenment (Constellation Map) — full redesign.
    68→ *
    69→ * Renders a star-map SVG (1000×600 viewBox) of all curated constellations.
    70→ * Each star is a dim dot by default; lit stars glow. Constellation connection
    71→ * lines appear dim until completed (or revealed by Stargaze), then bright.
    72→ * Click a star to illuminate it (calls `illuminateStar`). Sidebar shows
    73→ * progress stats, an active-constellation selector that pans/zooms the SVG,
    74→ * and the three active abilities (Stargaze, Supernova, Black Hole).
    75→ */
    76→export function EnlightenmentModal() {
    77→  const show = useGameStore((s) => s.showEnlightenment);
    78→  const setShow = useGameStore((s) => s.setShowEnlightenment);
    79→  const unlockedLayers = useGameStore((s) => s.unlockedLayers);
    80→  const divinity = useGameStore((s) => s.divinity);
    81→  // L2 state
    82→  const constellationStars = useGameStore((s) => s.constellationStars || {});
    83→  const constellationProgress = useGameStore((s) => s.constellationProgress || {});
    84→  const constellationsStarted = useGameStore((s) => s.constellationsStarted || {});
    85→  const constellationsCompleted = useGameStore((s) => s.constellationsCompleted || {});
    86→  const constellationRevealed = useGameStore((s) => s.constellationRevealed || {});
    87→  const cosmicBridgeLink = useGameStore((s) => s.cosmicBridgeLink || null);
    88→  const stargazeRecommendedActive = useGameStore((s) => s.stargazeRecommendedActive || false);
    89→  const activeAbilityCooldowns = useGameStore((s) => s.activeAbilityCooldowns || {});
    90→  const foresightNodes = useGameStore((s) => s.foresightNodes || {});
    91→  const activeRoute = useGameStore((s) => s.activeForesightRoute);
    92→  // Layer shop upgrades
    93→  const upgrades = useGameStore((s) => s.upgrades || {});
    94→
    95→  // Actions
    96→  const illuminateStar = useGameStore((s) => s.illuminateStar);
    97→  const stargazeReveal = useGameStore((s) => s.stargazeReveal);
    98→  const supernovaIlluminate = useGameStore((s) => s.supernovaIlluminate);
    99→  const blackHoleReset = useGameStore((s) => s.blackHoleReset);
   100→  const activateCosmicBridge = useGameStore((s) => s.activateCosmicBridge);
   101→  const setForesightRoute = useGameStore((s) => s.setForesightRoute);
   102→
   103→  // Local UI state
   104→  const [focusedConstellationId, setFocusedConstellationId] = useState<string | null>(null);
   105→  const [abilityMode, setAbilityMode] = useState<AbilityMode>(null);
   106→  const [bridgeFirst, setBridgeFirst] = useState<string | null>(null); // first constellation picked for bridge
   107→  const [flashedStars, setFlashedStars] = useState<Set<string>>(new Set());
   108→  const [collapsingStars, setCollapsingStars] = useState<Set<string>>(new Set());
   109→  const [hoveredStar, setHoveredStar] = useState<string | null>(null);
   110→  const flashTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   111→  const collapseTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   112→
   113→  // Clean up timers on unmount.
   114→  useEffect(() => {
   115→    return () => {
   116→      if (flashTimer.current) clearTimeout(flashTimer.current);
   117→      if (collapseTimer.current) clearTimeout(collapseTimer.current);
   118→    };
   119→  }, []);
   120→
   121→  const isUnlocked = !!unlockedLayers.enlightenment;
   122→
   123→  const constellationState: ConstellationState = useMemo(
   124→    () => ({
   125→      constellationStars,
   126→      constellationProgress,
   127→      constellationsStarted,
   128→      constellationsCompleted,
   129→    }),
   130→    [constellationStars, constellationProgress, constellationsStarted, constellationsCompleted]
   131→  );
   132→
   133→  const fs = foresightBonus(foresightNodes); // legacy — returns zeros for new saves
   134→  const rb = routeBonus(activeRoute);
   135→  const cb = constellationBonus(constellationState);
   136→  const completedCount = countCompletedConstellations(constellationState);
   137→  const litStarsCount = countLitStars(constellationState);
   138→  const startedCount = Object.values(constellationsStarted).filter(Boolean).length;
   139→  const totalStars = useMemo(
   140→    () => CONSTELLATIONS.reduce((acc, c) => acc + c.stars.length, 0),
   141→    []
   142→  );
   143→
   144→  // Foresight discount level (shop upgrade)
   145→  const fdLvl = (upgrades?.foresight_discount as number) || 0;
   146→  const foresightDiscountMult = Math.pow(0.80, fdLvl);
   147→
   148→  // Compute the cost for the focused constellation's next star (or first star).
   149→  const focusedNextCost = useMemo(() => {
   150→    const cid = focusedConstellationId;
   151→    if (!cid) return null;
   152→    const cdef = CONSTELLATION_MAP[cid];
   153→    if (!cdef) return null;
   154→    // First unlit star in the focused constellation
   155→    const nextStar = cdef.stars.find((st) => !constellationStars[st.id]);
   156→    if (!nextStar) return { starId: null, cost: 0, isNew: false, constellationId: cid, name: cdef.name };
   157→    const cost = starCost(cid, constellationState, nextStar.id, { foresightDiscountMult });
   158→    const isNew = (constellationProgress[cid] || 0) === 0;
   159→    return { starId: nextStar.id, cost, isNew, constellationId: cid, name: cdef.name };
   160→  }, [focusedConstellationId, constellationStars, constellationState, foresightDiscountMult, constellationProgress]);
   161→
   162→  // Compute a "first star of any unstarted constellation" cost preview (for the
   163→  // "new constellation penalty" tooltip).
   164→  const newConstellationPreview = useMemo(() => {
   165→    const cid = CONSTELLATIONS.find((c) => !constellationsStarted[c.id])?.id;
   166→    if (!cid) return null;
   167→    const cdef = CONSTELLATION_MAP[cid];
   168→    if (!cdef) return null;
   169→    const firstStar = cdef.stars[0];
   170→    const cost = starCost(cid, constellationState, firstStar.id, { foresightDiscountMult });
   171→    return { constellationId: cid, name: cdef.name, cost };
   172→  }, [constellationsStarted, constellationState, foresightDiscountMult]);
   173→
   174→  // SVG pan/zoom target based on focused constellation.
   175→  const viewBox = useMemo(() => {
   176→    if (!focusedConstellationId) return `0 0 ${MAP_W} ${MAP_H}`;
   177→    const cdef = CONSTELLATION_MAP[focusedConstellationId];
   178→    if (!cdef) return `0 0 ${MAP_W} ${MAP_H}`;
   179→    const xs = cdef.stars.map((s) => s.x);
   180→    const ys = cdef.stars.map((s) => s.y);
   181→    const minX = Math.min(...xs);
   182→    const maxX = Math.max(...xs);
   183→    const minY = Math.min(...ys);
   184→    const maxY = Math.max(...ys);
   185→    const pad = 6;
   186→    const w = (maxX - minX + pad * 2) * (MAP_W / 100);
   187→    const h = (maxY - minY + pad * 2) * (MAP_H / 100);
   188→    const cx = (minX + maxX) / 2 * (MAP_W / 100);
   189→    const cy = (minY + maxY) / 2 * (MAP_H / 100);
   190→    // Clamp to map bounds.
   191→    const halfW = w / 2;
   192→    const halfH = h / 2;
   193→    const x0 = Math.max(0, Math.min(MAP_W - w, cx - halfW));
   194→    const y0 = Math.max(0, Math.min(MAP_H - h, cy - halfH));
   195→    return `${x0} ${y0} ${w} ${h}`;
   196→  }, [focusedConstellationId]);
   197→
   198→  // Click a star: illuminate (normal mode) or apply ability (ability mode).
   199→  const onStarClick = useCallback(
   200→    (constellationId: string, starId: string) => {
   201→      if (!isUnlocked) return;
   202→      if (abilityMode === "supernova") {
   203→        supernovaIlluminate(constellationId, starId);
   204→        // Flash the supernova'd cluster (target + connected stars).
   205→        const cdef = CONSTELLATION_MAP[constellationId];
   206→        if (cdef) {
   207→          const affected = new Set<string>([starId]);
   208→          for (const [a, b] of cdef.connections) {
   209→            if (a === starId) affected.add(b);
   210→            else if (b === starId) affected.add(a);
   211→          }
   212→          setFlashedStars(affected);
   213→          if (flashTimer.current) clearTimeout(flashTimer.current);
   214→          flashTimer.current = setTimeout(() => setFlashedStars(new Set()), 600);
   215→        }
   216→        setAbilityMode(null);
   217→        return;
   218→      }
   219→      if (abilityMode === "blackhole") {
   220→        const cdef = CONSTELLATION_MAP[constellationId];
   221→        if (!cdef) return;
   222→        // Capture the lit stars in this constellation for the collapse animation.
   223→        const collapsing = new Set<string>(
   224→          cdef.stars.filter((st) => constellationStars[st.id]).map((st) => st.id)
   225→        );
   226→        blackHoleReset(constellationId);
   227→        if (collapsing.size > 0) {
   228→          setCollapsingStars(collapsing);
   229→          if (collapseTimer.current) clearTimeout(collapseTimer.current);
   230→          collapseTimer.current = setTimeout(() => setCollapsingStars(new Set()), 600);
   231→        }
   232→        setAbilityMode(null);
   233→        return;
   234→      }
   235→      if (abilityMode === "cosmicbridge_first") {
   236→        // Pick first endpoint.
   237→        setBridgeFirst(constellationId);
   238→        setAbilityMode("cosmicbridge_second");
   239→        return;
   240→      }
   241→      if (abilityMode === "cosmicbridge_second") {
   242→        if (bridgeFirst && bridgeFirst !== constellationId) {
   243→          activateCosmicBridge(bridgeFirst, constellationId);
   244→        }
   245→        setBridgeFirst(null);
   246→        setAbilityMode(null);
   247→        return;
   248→      }
   249→      illuminateStar(constellationId, starId);
   250→    },
   251→    [abilityMode, bridgeFirst, constellationStars, illuminateStar, supernovaIlluminate, blackHoleReset, activateCosmicBridge, isUnlocked]
   252→  );
   253→
   254→  // Compute the cheapest affordable unlit next star across ALL constellations
   255→  // (for the Stargaze "recommended next" glow). Returns starId or null.
   256→  const recommendedNextStarId = useMemo(() => {
   257→    if (!stargazeRecommendedActive) return null;
   258→    let best: { id: string; cost: number } | null = null;
   259→    for (const c of CONSTELLATIONS) {
   260→      for (const st of c.stars) {
   261→        if (constellationStars[st.id]) continue;
   262→        const cost = starCost(c.id, constellationState, st.id, { foresightDiscountMult });
   263→        if (cost > divinity) continue; // must be affordable
   264→        if (!best || cost < best.cost) best = { id: st.id, cost };
   265→      }
   266→    }
   267→    return best?.id ?? null;
   268→  }, [stargazeRecommendedActive, constellationStars, constellationState, foresightDiscountMult, divinity]);
   269→
   270→  // Cooldown helpers (ceil to seconds for display).
   271→  const stargazeCd = Math.ceil(activeAbilityCooldowns[STARGAZE_CD_KEY] || 0);
   272→  const supernovaCd = Math.ceil(activeAbilityCooldowns[SUPERNOVA_CD_KEY] || 0);
   273→  const blackHoleCd = Math.ceil(activeAbilityCooldowns[BLACK_HOLE_CD_KEY] || 0);
   274→  const cosmicBridgeCd = Math.ceil(activeAbilityCooldowns[COSMIC_BRIDGE_CD_KEY] || 0);
   275→
   276→  return (
   277→    <Dialog open={show} onOpenChange={setShow}>
   278→      <DialogContent className="max-w-6xl max-h-[92vh] overflow-hidden glass-strong flex flex-col">
   279→        <DialogHeader>
   280→          <DialogTitle className="flex items-center gap-2 text-lg flex-wrap">
   281→            <Eye className="w-5 h-5" />
   282→            Layer 2 — Enlightenment (Constellation Map)
   283→            <BackgroundLayerBadge layerId="enlightenment" />
   284→          </DialogTitle>
   285→          <DialogDescription className="flex items-center justify-between flex-wrap gap-2">
   286→            <span>
   287→              Illuminate stars within {CONSTELLATIONS.length} IAU constellations. Completing a constellation grants a permanent bonus.
   288→            </span>
   289→            <Badge variant="outline" className="text-violet-300 border-violet-400/40">
   290→              {formatNumber(divinity, 0)} Divinity
   291→            </Badge>
   292→          </DialogDescription>
   293→        </DialogHeader>
   294→
   295→        {!isUnlocked ? (
   296→          <div className="text-center text-muted-foreground py-10">
   297→            <Lock className="w-8 h-8 mx-auto mb-2 opacity-50" />
   298→            <p className="text-sm">Enlightenment is sealed.</p>
   299→            <p className="text-xs mt-1">Master 3 distinct Layer 1 trials to unlock the constellation map.</p>
   300→          </div>
   301→        ) : (
   302→          <div className="flex-1 overflow-hidden grid grid-cols-1 lg:grid-cols-[1fr_320px] gap-3 min-h-0">
   303→            {/* ===== STAR MAP ===== */}
   304→            <div className="relative rounded-lg overflow-hidden border border-violet-500/20 bg-gradient-to-b from-[#020817] via-[#0a0a23] to-[#050518]">
   305→              {/* Subtle starfield background */}
   306→              <div
   307→                className="absolute inset-0 opacity-60 pointer-events-none"
   308→                style={{
   309→                  backgroundImage: `
   310→                    radial-gradient(1px 1px at 12% 18%, #fff, transparent),
   311→                    radial-gradient(1px 1px at 28% 42%, #fff, transparent),
   312→                    radial-gradient(1.5px 1.5px at 47% 12%, #fff, transparent),
   313→                    radial-gradient(1px 1px at 63% 56%, #fff, transparent),
   314→                    radial-gradient(1px 1px at 78% 28%, #fff, transparent),
   315→                    radial-gradient(1.5px 1.5px at 88% 70%, #fff, transparent),
   316→                    radial-gradient(1px 1px at 8% 72%, #fff, transparent),
   317→                    radial-gradient(1px 1px at 33% 88%, #fff, transparent),
   318→                    radial-gradient(1px 1px at 70% 85%, #fff, transparent),
   319→                    radial-gradient(1px 1px at 95% 8%, #fff, transparent),
   320→                    radial-gradient(1px 1px at 18% 50%, #fff, transparent),
   321→                    radial-gradient(1px 1px at 53% 75%, #fff, transparent)
   322→                  `,
   323→                  backgroundRepeat: "no-repeat",
   324→                  backgroundSize: "100% 100%",
   325→                }}
   326→              />
   327→              <svg
   328→                viewBox={viewBox}
   329→                className="relative w-full h-full"
   330→                style={{ aspectRatio: `${MAP_W} / ${MAP_H}`, transition: "viewBox 0.6s ease" as any }}
   331→                preserveAspectRatio="xMidYMid meet"
   332→              >
   333→                {/* Connection lines */}
   334→                {CONSTELLATIONS.flatMap((c) =>
   335→                  c.connections.map(([aId, bId], i) => {
   336→                    const aStar = c.stars.find((s) => s.id === aId);
   337→                    const bStar = c.stars.find((s) => s.id === bId);
   338→                    if (!aStar || !bStar) return null;
   339→                    const x1 = (aStar.x / 100) * MAP_W;
   340→                    const y1 = (aStar.y / 100) * MAP_H;
   341→                    const x2 = (bStar.x / 100) * MAP_W;
   342→                    const y2 = (bStar.y / 100) * MAP_H;
   343→                    const isCompleted = !!constellationsCompleted[c.id];
   344→                    const isStarted = !!constellationsStarted[c.id];
   345→                    const isRevealed = !!constellationRevealed[c.id];
   346→                    const dim = !isStarted && !isRevealed;
   347→                    const stroke = isCompleted
   348→                      ? CATEGORY_ACCENT[c.category]
   349→                      : isStarted
   350→                        ? `${CATEGORY_ACCENT[c.category]}80`
   351→                        : "#1e293b";
   352→                    return (
   353→                      <line
   354→                        key={`${c.id}-${i}`}
   355→                        x1={x1}
   356→                        y1={y1}
   357→                        x2={x2}
   358→                        y2={y2}
   359→                        stroke={stroke}
   360→                        strokeWidth={isCompleted ? 2.2 : 1}
   361→                        strokeLinecap="round"
   362→                        opacity={dim ? 0.35 : 0.9}
   363→                      />
   364→                    );
   365→                  })
   366→                )}
   367→                {/* Stars */}
   368→                {CONSTELLATIONS.map((c) =>
   369→                  c.stars.map((star) => {
   370→                    const cx = (star.x / 100) * MAP_W;
   371→                    const cy = (star.y / 100) * MAP_H;
   372→                    const lit = !!constellationStars[star.id];
   373→                    const isCompleted = !!constellationsCompleted[c.id];
   374→                    const isStarted = !!constellationsStarted[c.id];
   375→                    const isRevealed = !!constellationRevealed[c.id];
   376→                    const accent = CATEGORY_ACCENT[c.category];
   377→                    const isFocused = focusedConstellationId === c.id;
   378→                    const isHovered = hoveredStar === star.id;
   379→                    const isFlashing = flashedStars.has(star.id);
   380→                    const isCollapsing = collapsingStars.has(star.id);
   381→                    const isNewConstellation = (constellationProgress[c.id] || 0) === 0;
   382→                    const cost = starCost(c.id, constellationState, star.id, { foresightDiscountMult });
   383→                    const canAfford = divinity >= cost;
   384→                    // In ability mode, all stars are clickable regardless of cost.
   385→                    const inAbilityMode = abilityMode !== null;
   386→                    const clickable = !lit && (canAfford || inAbilityMode);
   387→                    const r = lit ? 4.5 : 2.2;
   388→                    // Stargaze "recommended next" glow — pulsing amber ring around
   389→                    // the single cheapest affordable unlit star in the sky.
   390→                    const isRecommended = stargazeRecommendedActive && recommendedNextStarId === star.id;
   391→                    // Cosmic Bridge: highlight stars in the linked pair.
   392→                    const bridgeActive = !!cosmicBridgeLink && cosmicBridgeLink.remaining > 0 &&
   393→                      (cosmicBridgeLink.from === c.id || cosmicBridgeLink.to === c.id);
   394→                    const isBridgePick1 = (abilityMode === "cosmicbridge_second") && bridgeFirst === c.id;
   395→                    return (
   396→                      <g
   397→                        key={star.id}
   398→                        transform={`translate(${cx},${cy})`}
   399→                        style={{ cursor: clickable ? "pointer" : "default" }}
   400→                        onClick={() => clickable && onStarClick(c.id, star.id)}
   401→                        onMouseEnter={() => setHoveredStar(star.id)}
   402→                        onMouseLeave={() => setHoveredStar(null)}
   403→                      >
   404→                        {/* Glow halo for lit/hovered */}
   405→                        {lit && (
   406→                          <circle r={r * 3} fill={accent} opacity={0.18} />
   407→                        )}
   408→                        {isHovered && !lit && (
   409→                          <circle r={r * 2.5} fill={accent} opacity={0.12} />
   410→                        )}
   411→                        {/* Stargaze recommended-next glow */}
   412→                        {isRecommended && (
   413→                          <circle r={r * 3.5} fill="none" stroke="#fbbf24" strokeWidth={0.9} opacity={0.85}>
   414→                            <animate attributeName="r" values={`${r * 3.5};${r * 5};${r * 3.5}`} dur="1.6s" repeatCount="indefinite" />
   415→                            <animate attributeName="opacity" values="0.85;0.3;0.85" dur="1.6s" repeatCount="indefinite" />
   416→                          </circle>
   417→                        )}
   418→                        {/* Cosmic Bridge highlight ring around stars of a bridged constellation */}
   419→                        {bridgeActive && !lit && (
   420→                          <circle r={r * 2.2} fill="none" stroke="#a78bfa" strokeWidth={0.7} opacity={0.6} />
   421→                        )}
   422→                        {/* Highlight first-picked constellation in cosmic bridge mode */}
   423→                        {isBridgePick1 && (
   424→                          <circle r={r * 2.8} fill="none" stroke="#a78bfa" strokeWidth={1.1} opacity={0.95} />
   425→                        )}
   426→                        {/* Flash overlay (supernova) */}
   427→                        {isFlashing && (
   428→                          <circle r={r * 5} fill="#fbbf24" opacity={0.55} />
   429→                        )}
   430→                        {/* Collapse overlay (black hole) */}
   431→                        {isCollapsing && (
   432→                          <circle r={r * 1.4} fill="#1f2937" opacity={0.8} stroke="#ef4444" strokeWidth={0.8} />
   433→                        )}
   434→                        {/* The star itself */}
   435→                        <circle
   436→                          r={r}
   437→                          fill={lit ? accent : isStarted ? "#475569" : "#1e293b"}
   438→                          stroke={lit ? "#fff" : isFocused ? accent : "transparent"}
   439→                          strokeWidth={lit || isFocused ? 0.8 : 0}
   440→                          opacity={lit ? 1 : isStarted ? 0.8 : 0.55}
   441→                        />
   442→                        {/* Cost label on hover for unlit affordable stars (normal mode) */}
   443→                        {isHovered && !lit && !inAbilityMode && (
   444→                          <text
   445→                            y={-8}
   446→                            textAnchor="middle"
   447→                            fill={canAfford ? "#a3e635" : "#f87171"}
