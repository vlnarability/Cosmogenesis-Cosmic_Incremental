    20→function initialRunState(): Partial<GameState> {
    21→  const resources = emptyResources();
    22→  const capacities = emptyCapacities();
    23→  resources.atp = 5;
    24→  resources.glucose = 8;
    25→  resources.happiness = 75;
    26→  return {
    27→    stageIndex: 0,
    28→    time: 0,
    29→    population: 8,
    30→    populationProgress: 0,
    31→    maxPopulation: 8,
    32→    resources,
    33→    capacities,
    34→    ownedSystems: {},
    35→    systemEnabled: {},
    36→    technologies: {},
    37→    lockedTechs: {},
    38→    lockedSystems: {},
    39→    archetypeAffinity: {},
    40→    lockedArchetype: null,
    41→    log: ["New run started. The first spark stirs."],
    42→    activeChallenge: null,
    43→    challengeRunStart: null,
    44→  };
    45→}
    46→
    47→function initialMetaState(): Partial<GameState> {
    48→  return {
    49→    evolutionPoints: 0,
    50→    totalRuns: 0,
    51→    galacticWins: 0,
    52→    stageClearCounts: Object.fromEntries(STAGE_IDS.map((s) => [s, 0])) as Record<StageId, number>,
    53→    fastestCellClear: 0,
    54→    totalEventsResolved: 0,
    55→    totalPlayTime: 0,
    56→    totalActions: 0,
    57→    totalSystemsBuilt: 0,
    58→    totalTechResearched: 0,
    59→    upgrades: Object.fromEntries(UPGRADES.map((u) => [u.id, 0])),
    60→    storyUnlocked: { first_spark: true },
    61→    storyAcknowledged: {},
    62→    archive: [],
    63→    archivedArchetypes: [],
    64→    unlockedLayers: { evolution: true },
    65→    achievements: {},
    66→    newAchievements: [],
    67→    unlockedChallenges: false,
    68→    activeChallenge: null,
    69→    completedChallenges: {},
    70→    challengeRunStart: null,
    71→    currentTab: "actions",
    72→    speed: 1,
    73→    paused: false,
    74→    showShop: false,
    75→    showEvolve: false,
    76→    showSettings: false,
    77→    showChallenges: false,
    78→    activeStoryPopup: "first_spark",
    79→    activeEvent: null,
    80→    eventCooldown: 60, // first event can fire after 60s
    81→    eventHistory: [],
    82→    hasSeenIntro: false,
    83→    lastSaved: Date.now(),
    84→    tutorialStep: 0,
    85→    tutorialActive: true,

    87→    eventFrequency: "normal",
    88→  };

    90→
    91→export const useGameStore = create<GameStore>()(
    92→  persist(
    93→    (set, get) => ({
    94→      ...initialRunState(),
    95→      ...initialMetaState(),
    96→
    97→      // ============ TICK ============
    98→      tick: (dt) => {
    99→        const s = get();
   100→        if (s.paused) return;
   101→        const realDt = dt * s.speed;
   102→
   103→        const autoLvl = s.upgrades["inherited_efficiency"] || 0;
   104→        const autoMult = 1 + autoLvl * 0.15;
   105→        const techMult = techMultiplier(s, "production");
   106→        const capBoostLvl = s.upgrades["expansive_vaults"] || 0;
   107→        const capMult = 1 + capBoostLvl * 0.20;
   108→        const hasAutoBalancer = (s.upgrades["auto_balancer"] || 0) > 0;
   109→
   110→        // Archetype bonus
   111→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   112→        const archAllMult = archetype?.bonus.allProductionMult || 0;
   113→        const archPopMult = archetype?.bonus.populationGrowthMult || 0;
   114→        const archCapMult = archetype?.bonus.capMult || 0;
   115→
   116→        // ===== Challenge debuff (applied for the duration of an active challenge run)
   117→        const activeChallenge = s.activeChallenge ? CHALLENGE_MAP[s.activeChallenge] : null;
   118→        const challengeEffects = activeChallenge?.debuff.effects || {};
   119→        const chFoodMult = challengeEffects.foodMult ?? 1;
   120→        const chScienceMult = challengeEffects.scienceMult ?? 1;
   121→        const chProductionMult = challengeEffects.productionMult ?? 1;
   122→        const chMilitaryMult = challengeEffects.militaryMult ?? 1;
   123→        const chWaterUpkeepMult = challengeEffects.waterUpkeepMult ?? 1;
   124→        const chHappinessDrain = challengeEffects.happinessDrain ?? 0;
   125→        const chPopGrowthMult = challengeEffects.popGrowthMult ?? 1;
   126→
   127→        const resources = { ...s.resources };
   128→        let capacities = recomputeCapacitiesFresh(s, capMult + archCapMult);
   129→
   130→        // Apply system production — with cap-aware pausing
   131→        for (const [sysId, count] of Object.entries(s.ownedSystems)) {
   132→          if (!count) continue;
   133→          const sys = SYSTEM_MAP[sysId];
   134→          if (!sys) continue;
   135→          if (sys.requiredTech && !s.technologies[sys.requiredTech]) continue;
   136→
   137→          // Check if system is manually disabled
   138→          const manuallyEnabled = s.systemEnabled[sysId] !== false; // default true
   139→          if (!manuallyEnabled && !hasAutoBalancer) continue;
   140→
   141→          // Cap-aware logic: if all output resources are at cap, idle the system
   142→          // (don't produce, don't consume upkeep — no waste)
   143→          const outputsCapped = Object.keys(sys.produces).length > 0 &&
   144→            Object.keys(sys.produces).every((r) => {
   145→              const cap = capacities[r] || 0;
   146→              return (resources[r] || 0) >= cap - 0.01;
   147→            });
   148→
   149→          if (outputsCapped) {
   150→            // System idles — no production, no upkeep
   151→            continue;
   152→          }
   153→
   154→          const stageMult = sys.stage === STAGES[s.stageIndex].id ? 1 : 0.5;
   155→          const mult = autoMult * techMult * stageMult * (1 + archAllMult) * chProductionMult;
   156→
   157→          // Per-resource archetype bonus + challenge resource-specific debuff
   158→          const produces = sys.produces as Record<string, number>;
   159→          for (const [r, v] of Object.entries(produces)) {
   160→            const archResBonus = archetype?.bonus.productionMult?.[r] || 0;
   161→            // Apply per-resource challenge multipliers
   162→            let chResMult = 1;
   163→            if (r === "food") chResMult *= chFoodMult;
   164→            if (r === "science") chResMult *= chScienceMult;
   165→            if (r === "military_power") chResMult *= chMilitaryMult;
   166→            const rate = v * count * mult * (1 + archResBonus) * chResMult;
   167→            resources[r] = (resources[r] || 0) + rate * realDt;
   168→          }
   169→          // Upkeep
   170→          if (sys.upkeep) {
   171→            const upkeep = sys.upkeep as Record<string, number>;
   172→            for (const [r, v] of Object.entries(upkeep)) {
   173→              const upkMult = r === "water" ? chWaterUpkeepMult : 1;
   174→              resources[r] = (resources[r] || 0) - v * count * realDt * upkMult;
   175→            }
   176→          }
   177→        }
   178→
   179→        // Apply challenge happiness drain (flat per second)
   180→        if (chHappinessDrain > 0) {
   181→          const happCap = capacities.happiness || 100;
   182→          resources.happiness = Math.max(0, (resources.happiness || 0) - chHappinessDrain * realDt);
   183→          // also clamp to cap (already done below, but keep happ bounded)
   184→          if (resources.happiness > happCap) resources.happiness = happCap;
   185→        }
   186→
   187→        // Clamp resources to capacities (no overflow)
   188→        for (const r of Object.keys(resources)) {
   189→          const cap = capacities[r];
   190→          if (cap != null && resources[r] > cap) resources[r] = cap;
   191→          if (resources[r] < 0) resources[r] = 0;
   192→        }
   193→
   194→        // Slow passive ATP for cell (keeps early game from stalling)
   195→        if (STAGES[s.stageIndex].id === "cell") {
   196→          resources.atp = Math.min(capacities.atp || 30, resources.atp + 0.1 * realDt);
   197→        }
   198→
   199→        // Happiness drift toward 50 baseline if no source
   200→        // (prevents permanent high happiness from start)
   201→        // — disabled: let systems/tech maintain it
   202→
   203→        // Population growth with progress bar
   204→        // Growth rate scales with happiness and food/water availability
   205→        let popGrowthRate = 0.015; // base per second
   206→        const happiness = resources.happiness || 50;
   207→        popGrowthRate *= (1 + (happiness - 40) / 200); // happiness 100 → +30%, happiness 20 → -10%
   208→        popGrowthRate *= (1 + archPopMult);
   209→        // Apply challenge population-growth debuff
   210→        popGrowthRate *= chPopGrowthMult;
   211→        // Penalty if food/water are critically low (creature+)
   212→        const curStage = STAGES[s.stageIndex];
   213→        if (["creature", "tribal", "civilization", "empire"].includes(curStage.id)) {
   214→          const foodCap = capacities.food || 1;
   215→          const waterCap = capacities.water || 1;
   216→          const foodPct = (resources.food || 0) / foodCap;
   217→          const waterPct = (resources.water || 0) / waterCap;
   218→          if (foodPct < 0.1) popGrowthRate *= 0.3;
   219→          if (waterPct < 0.1) popGrowthRate *= 0.3;
   220→        }
   221→
   222→        let populationProgress = (s.populationProgress || 0) + popGrowthRate * realDt;
   223→        let population = s.population;
   224→        while (populationProgress >= 1) {
   225→          populationProgress -= 1;
   226→          population += 1;
   227→        }
   228→
   229→        const maxPopulation = Math.max(s.maxPopulation || 0, population);
   230→
   231→        const newTime = s.time + realDt;
   232→
   233→        // Check story triggers
   234→        const newStory = { ...s.storyUnlocked };
   235→        let activePopup = s.activeStoryPopup;
   236→        const ctx = { ...s, resources, ownedSystems: s.ownedSystems };
   237→        for (const trig of STORY_TRIGGERS) {
   238→          if (!newStory[trig.id] && trig.condition(ctx)) {
   239→            newStory[trig.id] = true;
   240→            if (!activePopup) activePopup = trig.id;
   241→          }
   242→        }
   243→
   244→        // Check achievements periodically (every ~2 seconds of game time)
   245→        let achievements = s.achievements;
   246→        let newAchievements = s.newAchievements;
   247→        const checkInterval = 2.0;
   248→        const lastCheck = (s as any)._lastAchCheck || 0;
   249→        if (newTime - lastCheck >= checkInterval) {
   250→          const curState = { ...s, resources, time: newTime } as GameState;
   251→          const result = checkAchievements(curState);
   252→          if (result.newOnes.length > 0) {
   253→            achievements = result.earned;
   254→            newAchievements = [...(newAchievements || []), ...result.newOnes];
   255→          } else if (Object.keys(result.earned).length !== Object.keys(achievements || {}).length) {
   256→            achievements = result.earned;
   257→          }
   258→          (s as any)._lastAchCheck = newTime; // not persisted, just a runtime marker
   259→        }
   260→
   261→        // Random event firing — only if no popup/modal is active
   262→        let activeEvent = s.activeEvent;
   263→        let eventCooldown = (s.eventCooldown || 0) - realDt;
   264→        if (eventCooldown < 0) eventCooldown = 0;
   265→        const eventFreq = s.eventFrequency || "normal";
   266→        if (eventFreq !== "off" && !activeEvent && !activePopup && eventCooldown === 0) {
   267→          const stageId = STAGES[s.stageIndex].id;
   268→          const possible = eventsForStage(stageId, newTime);
   269→          if (possible.length > 0) {
   270→            // Weighted random selection based on rarity
   271→            const totalWeight = possible.reduce((sum, e) => sum + eventWeight(e), 0);
   272→            const roll = Math.random() * totalWeight;
   273→            let cum = 0;
   274→            for (const ev of possible) {
   275→              cum += eventWeight(ev);
   276→              if (roll < cum) {
   277→                activeEvent = ev.id;
   278→                // Cooldown depends on frequency setting
   279→                const base = eventFreq === "frequent" ? 30 : 90;
   280→                const range = eventFreq === "frequent" ? 30 : 90;
   281→                eventCooldown = base + Math.random() * range;
   282→                break;
   283→              }
   284→            }
   285→          }
   286→        }
   287→
   288→        set({
   289→          resources,
   290→          capacities,
   291→          time: newTime,
   292→          population,
   293→          maxPopulation,
   294→          populationProgress,
   295→          storyUnlocked: newStory,
   296→          activeStoryPopup: activePopup,
   297→          achievements,
   298→          newAchievements,
   299→          activeEvent,
   300→          eventCooldown,
   301→          totalPlayTime: (s.totalPlayTime || 0) + realDt,
   302→        });
   303→
   304→        // Check if the active challenge's completion condition has been met.
   305→        // If so, mark it complete and clear the active challenge slot.
   306→        // (Done after the set so completion sees updated state.)
   307→        if (s.activeChallenge) {
   308→          get().checkChallengeCompletion();
   309→        }
   310→      },
   311→
   312→      // ============ ACTIONS ============
   313→      performAction: (actionId) => {
   314→        const s = get();
   315→        const action = ACTION_MAP[actionId];
   316→        if (!action) return;
   317→        if (action.stage !== STAGES[s.stageIndex].id) return;
   318→
   319→        const res = { ...s.resources };
   320→
   321→        // Check if all output resources are at cap — if so, block (no waste)
   322→        const outputs = Object.entries(action.produces);
   323→        if (outputs.length > 0) {
   324→          const allCapped = outputs.every(([r]) => {
   325→            const cap = s.capacities[r] || 0;
   326→            return (res[r] || 0) >= cap - 0.001;
   327→          });
   328→          if (allCapped) return; // action wasted — block it
   329→        }
   330→
   331→        // Check costs
   332→        if (action.cost) {
   333→          for (const [r, v] of Object.entries(action.cost)) {
   334→            if ((res[r] || 0) < (v as number)) return;
   335→          }
   336→          for (const [r, v] of Object.entries(action.cost)) {
   337→            res[r] -= v as number;
   338→          }
   339→        }
   340→
   341→        // Manual action multiplier
   342→        const manualLvl = s.upgrades["quickened_hands"] || 0;
   343→        const mult = 1 + manualLvl * 0.25;
   344→        const techMult = techMultiplier(s, "manual");
   345→        const finalMult = mult * techMult;
   346→
   347→        // Archetype per-resource bonus
   348→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   349→
   350→        for (const [r, v] of Object.entries(action.produces)) {
   351→          const cap = s.capacities[r] || 0;
   352→          const archResBonus = archetype?.bonus.productionMult?.[r] || 0;
   353→          const totalArchMult = 1 + archResBonus + (archetype?.bonus.allProductionMult || 0);
   354→          const amount = (v as number) * finalMult * totalArchMult;
   355→          res[r] = Math.min(cap, (res[r] || 0) + amount);
   356→        }
   357→
   358→        set({
   359→          resources: res,
   360→          totalActions: (s.totalActions || 0) + 1,
   361→        });
   362→      },
   363→
   364→      buySystem: (systemId, qty) => {
   365→        const s = get();
   366→        const sys = SYSTEM_MAP[systemId];
   367→        if (!sys) return;
   368→        if (sys.stage !== STAGES[s.stageIndex].id) return;
   369→        // Check tech prerequisite
   370→        if (sys.requiredTech && !s.technologies[sys.requiredTech]) return;
   371→        // Locked systems (via exclusiveWith variant choices) cannot be purchased.
   372→        if (s.lockedSystems && s.lockedSystems[systemId]) return;
   373→
   374→        const costReduction = (s.upgrades["frugality"] || 0) * 0.04;
   375→        let owned = s.ownedSystems[systemId] || 0;
   376→        if (sys.maxOwned && owned >= sys.maxOwned) return;
   377→
   378→        const res = { ...s.resources };
   379→        const ownedSystems = { ...s.ownedSystems };
   380→        const systemEnabled = { ...s.systemEnabled };
   381→        const archetypeAffinity = { ...s.archetypeAffinity };
   382→
   383→        let bought = 0;
   384→        for (let i = 0; i < qty; i++) {
   385→          if (sys.maxOwned && owned + bought >= sys.maxOwned) break;
   386→          const cost = systemCost(sys, owned + bought, costReduction);
   387→          let canAfford = true;
   388→          for (const [r, v] of Object.entries(cost)) {
   389→            if ((res[r] || 0) < (v as number)) {
   390→              canAfford = false;
   391→              break;
   392→            }
   393→          }
   394→          if (!canAfford) break;
   395→          for (const [r, v] of Object.entries(cost)) {
   396→            res[r] -= v as number;
   397→          }
   398→          bought++;
   399→        }
   400→
   401→        if (bought === 0) return;
   402→
   403→        ownedSystems[systemId] = owned + bought;
   404→        // New systems are enabled by default
   405→        if (systemEnabled[systemId] === undefined) systemEnabled[systemId] = true;
   406→
   407→        // Apply archetype affinity grant
   408→        if (sys.grantsAffinity) {
   409→          const arch = sys.grantsAffinity.archetype;
   410→          archetypeAffinity[arch] = (archetypeAffinity[arch] || 0) + sys.grantsAffinity.amount * bought;
   411→        }
   412→
   413→        // Trigger story
   414→        const newStory = { ...s.storyUnlocked };
   415→        let activePopup = s.activeStoryPopup;
   416→        for (const trig of STORY_TRIGGERS) {
   417→          if (!newStory[trig.id] && trig.condition({ ...s, resources: res, ownedSystems })) {
   418→            newStory[trig.id] = true;
   419→            if (!activePopup) activePopup = trig.id;
   420→          }
   421→        }
   422→
   423→        // Enforce mutually-exclusive variant choices: when buying the first
   424→        // instance of a system that is part of an exclusive group, permanently
   425→        // lock all sibling variants so they can never be purchased this run.
   426→        // (Only the FIRST purchase locks siblings — once chosen, the choice is
   427→        // final for this run, but the chosen system can still be bought up to
   428→        // its maxOwned.)
   429→        const lockedSystems = { ...(s.lockedSystems || {}) };
   430→        if (sys.exclusiveWith && sys.exclusiveWith.length > 0 && owned === 0 && bought > 0) {
   431→          for (const siblingId of sys.exclusiveWith) {
   432→            // Don't lock a sibling that is somehow already owned (defensive)
   433→            if (!(ownedSystems[siblingId] > 0)) {
   434→              lockedSystems[siblingId] = true;
   435→            }
   436→          }
   437→        }
   438→
   439→        set({
   440→          resources: res,
   441→          ownedSystems,
   442→          systemEnabled,
   443→          archetypeAffinity,
   444→          storyUnlocked: newStory,
   445→          activeStoryPopup: activePopup,
   446→          lockedSystems,
   447→          totalSystemsBuilt: (s.totalSystemsBuilt || 0) + bought,
   448→        });
   449→        get().addToLog(`Built ${bought}× ${sys.name}.`);
   450→      },
   451→
   452→      toggleSystem: (systemId) => {
   453→        const s = get();
   454→        if (!s.ownedSystems[systemId]) return;
   455→        const systemEnabled = { ...s.systemEnabled };
   456→        systemEnabled[systemId] = systemEnabled[systemId] === false ? true : false;
   457→        set({ systemEnabled });
   458→      },
   459→
   460→      enableAllSystems: (enabled) => {
   461→        const s = get();
   462→        const systemEnabled: Record<string, boolean> = {};
   463→        for (const id of Object.keys(s.ownedSystems)) {
   464→          systemEnabled[id] = enabled;
   465→        }
   466→        set({ systemEnabled });
   467→      },
   468→
   469→      buyTech: (techId) => {
   470→        const s = get();
   471→        const tech = TECH_MAP[techId];
   472→        if (!tech) return;
   473→        if (s.technologies[techId]) return;
   474→        // Locked techs cannot be researched (a mutually-exclusive sibling was already chosen)
   475→        if (s.lockedTechs && s.lockedTechs[techId]) return;
   476→        if (tech.stage !== STAGES[s.stageIndex].id) return;
   477→        if (tech.requires) {
   478→          for (const req of tech.requires) {
   479→            if (!s.technologies[req]) return;
   480→          }
   481→        }
   482→        const res = { ...s.resources };
   483→        for (const [r, v] of Object.entries(tech.cost)) {
   484→          if ((res[r] || 0) < (v as number)) return;
   485→        }
   486→        for (const [r, v] of Object.entries(tech.cost)) {
   487→          res[r] -= v as number;
   488→        }
   489→        const technologies = { ...s.technologies, [techId]: true };
   490→        const archetypeAffinity = { ...s.archetypeAffinity };
   491→        if (tech.grantsAffinity) {
   492→          archetypeAffinity[tech.grantsAffinity.archetype] =
   493→            (archetypeAffinity[tech.grantsAffinity.archetype] || 0) + tech.grantsAffinity.amount;
   494→        }
   495→        // Enforce mutually-exclusive tech path choices: lock all siblings
   496→        // listed in `exclusiveWith` so they can never be researched this run.
   497→        const lockedTechs = { ...(s.lockedTechs || {}) };
   498→        if (tech.exclusiveWith && tech.exclusiveWith.length > 0) {
   499→          for (const siblingId of tech.exclusiveWith) {
   500→            // Don't lock a sibling that is somehow already researched (defensive)
   501→            if (!technologies[siblingId]) {
   502→              lockedTechs[siblingId] = true;
   503→            }
   504→          }
   505→        }
   506→        set({
   507→          resources: res,
   508→          technologies,
   509→          lockedTechs,
   510→          archetypeAffinity,
   511→          totalTechResearched: (s.totalTechResearched || 0) + 1,
   512→        });
   513→        get().addToLog(`Researched ${tech.name}.`);
   514→      },
   515→
   516→      buyUpgrade: (upgradeId) => {
   517→        const s = get();
   518→        const up = UPGRADE_MAP[upgradeId];
   519→        if (!up) return;
   520→        if (up.requiresWins && s.galacticWins < up.requiresWins) return;
   521→        const owned = s.upgrades[upgradeId] || 0;
   522→        if (owned >= up.maxLevel) return;
   523→        const cost = upgradeCost(up, owned);
   524→        if (s.evolutionPoints < cost) return;
   525→        const upgrades = { ...s.upgrades, [upgradeId]: owned + 1 };
   526→        set({
   527→          evolutionPoints: s.evolutionPoints - cost,
   528→          upgrades,
   529→        });

  1086→function techMultiplier(s: GameState, target: "production" | "manual" | "capacity"): number {
  1087→  let mult = 1;
  1088→  for (const [techId, owned] of Object.entries(s.technologies)) {
  1089→    if (!owned) continue;
  1090→    const tech = TECH_MAP[techId];
  1091→    if (!tech || !tech.multiplier) continue;
  1092→    if (tech.multiplier.target === target || tech.multiplier.target === "production") {
  1093→      mult += tech.multiplier.value;
  1094→    }
  1095→  }
  1096→  // Achievement production bonus
  1097→  const ctx = buildAchievementCtx(s);
  1098→  mult += achievementBonus(s.achievements || {}, ctx, "production");
  1099→  // Challenge mastery bonus (only for completed challenges, persists across runs)
  1100→  // — applied as a production multiplier (all 10 challenges grant "production" type).
  1101→  if (target === "production") {
  1102→    mult += challengeMasteryBonus(s.completedChallenges || {}, "production");
  1103→  }
  1104→  return mult;
  1105→}

  1141→function computeScore(s: GameState): number {
  1142→  const totalSystems = Object.values(s.ownedSystems).reduce((a, b) => a + b, 0);
  1143→  const totalTech = Object.keys(s.technologies).length;
  1144→  const resourceSum = Object.values(s.resources).reduce((a, b) => a + b, 0);
  1145→  return Math.floor(
  1146→    s.population * 2 +
  1147→    totalSystems * 5 +
  1148→    totalTech * 10 +
  1149→    resourceSum * 0.5 +
  1150→    s.time * 0.1
  1151→  );
  1152→}

  1154→// Recompute all capacities from scratch based on owned systems
  1155→function recomputeCapacitiesFresh(s: GameState, capMult: number): Record<string, number> {
  1156→  const base = emptyCapacities();
  1157→  const caps: Record<string, number> = {};
  1158→  for (const r of Object.keys(base)) {
  1159→    let cap = (base[r] || 50) * capMult;
  1160→    for (const sys of SYSTEMS) {
  1161→      const count = s.ownedSystems[sys.id] || 0;
  1162→      if (!count || !sys.capacityBoost) continue;
  1163→      const boost = sys.capacityBoost[r];
  1164→      if (boost) cap += (boost as number) * count * capMult;
  1165→    }
  1166→    caps[r] = cap;
  1167→  }
  1168→  return caps;
  1169→}

  1171→// Export helper for components to compute whether an action is capped
  1172→export function isActionCapped(
  1173→  produces: Record<string, number>,
  1174→  resources: Record<string, number>,
  1175→  capacities: Record<string, number>
  1176→): boolean {
  1177→  const outputs = Object.entries(produces);

  1179→  return outputs.every(([r]) => {
  1180→    const cap = capacities[r] || 0;
  1181→    return (resources[r] || 0) >= cap - 0.001;
  1182→  });
  1183→}
