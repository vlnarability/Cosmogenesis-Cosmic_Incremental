    63→// ----- Hex layout constants -----
    64→const HEX_SIZE = 24;
    65→const HEX_POINTS = hexPolygonPoints(HEX_SIZE);

    67→// Compute pixel positions for all 50 systems (once per module load).
    68→const PIXEL_LAYOUT = (() => {
    69→  let minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity;
    70→  const positions = GALAXY_SYSTEMS.map((s) => {
    71→    const p = hexToPixel(s.hex.q, s.hex.r, HEX_SIZE);
    72→    if (p.x < minX) minX = p.x;
    73→    if (p.x > maxX) maxX = p.x;
    74→    if (p.y < minY) minY = p.y;
    75→    if (p.y > maxY) maxY = p.y;
    76→    return { id: s.id, x: p.x, y: p.y };
    77→  });
    78→  const pad = HEX_SIZE + 14;
    79→  return {
    80→    positions,
    81→    minX: minX - pad,
    82→    minY: minY - pad,
    83→    width: (maxX - minX) + pad * 2,
    84→    height: (maxY - minY) + pad * 2,
    85→  };
    86→})();
