   342→// Returns the layer overlay theme id for the highest unlocked layer (used by auto theme mode).
   343→// Order matches the layer progression order in LAYER_ID_TO_THEME keys.
   344→const LAYER_ORDER_FOR_THEME = [
   345→  "eternity",
   346→  "infinity",
   347→  "divinity",
   348→  "omnipotence",
   349→  "singularity",
   350→  "apotheosis",

   354→  "evolution",
   355→];
   356→function highestUnlockedLayerThemeId(unlockedLayers: Record<string, boolean>): string | null {
   357→  for (const layerId of LAYER_ORDER_FOR_THEME) {
   358→    if (unlockedLayers[layerId]) {
   359→      return LAYER_ID_TO_THEME[layerId] || null;
   360→    }
   361→  }
   362→  return null;
   363→}

   837→        // ARCHETYPES-REWORK — extended archetype bonus fields.
   838→        // archAllMult is composed of:
   839→        //   - base allProductionMult (e.g. Humanoid +10%, Necroid Ascendant +20%)
   840→        //   - productionPerSystemAllMult × totalOwnedSystems (Fungoid network)
   841→        //   - productionInWarmStagesMult (Reptilian — Tribal+)
   842→        //   - productionWhenLowResourcesMult (Extremophile — when avg fill < 20%)

   844→        const totalOwnedSystems = Object.values(s.ownedSystems).reduce((a, b) => a + b, 0);

   888→        // pipelines pick them up without duplicating logic.
   889→        const gov = s.activeGovernment ? GOVERNMENT_MAP[s.activeGovernment] : null;
   890→        const govAllMult = gov?.productionMult || 0;
   891→        const govPopMult = gov?.popGrowthMult || 0;
   892→        const govCapMult = gov?.capacityMult || 0;
   893→        const govMilitaryMult = gov?.militaryMult || 0;

   895→        // Republic) is scaled to a per-second delta and folded into the

   897→        const govHappinessPerTick = (gov?.happinessMult || 0) * HAPPINESS_DRIFT_SCALE;

   905→
   906→        const allAllMult = archAllMult + govAllMult;
   907→        const allPopMult = archPopMult + govPopMult;
   908→        const allCapMult = archCapMult + govCapMult;
   909→        const allMilitaryMult = archMilitaryMult + govMilitaryMult;
   910→        const allHappinessPerTick = archHappinessPerTick + govHappinessPerTick;
   911→
   912→        const resources = { ...s.resources };
   913→        let capacities = recomputeCapacitiesFresh(s, capMult + allCapMult);
   914→
   915→        // Cost reduction (frugality) — also used by auto-buyer
   916→        // Layers 2-6: foresight cost reduction (additive, negative = cheaper).
   917→        // New L2: completed constellations can also grant costMult reduction.
   918→        const layerCostReduction = foresight.costMult + constB.costMult;
   919→        const costReduction = (s.upgrades["frugality"] || 0) * 0.04 + Math.max(0, -layerCostReduction);
   920→        // Challenge cost multiplier (debuff)
   921→        const challengeCostMult = challengeDebuff.costMult ?? 1;
   922→
   923→        // Apply system production — with cap-aware pausing
   924→        // BACKGROUND-SIMULATION: accumulate the per-second total resource production
   925→        // (before cap-clamping) so we can track `peakProductionRate` — the best
   926→        // per-second production ever achieved. Used to scale passive Divinity
   927→        // generation from background simulations after the first Galactic prestige.
   928→        let totalProductionRate = 0;
   929→        for (const [sysId, count] of Object.entries(s.ownedSystems)) {
   930→          if (!count) continue;
   931→          const sys = SYSTEM_MAP[sysId];
   932→          if (!sys) continue;
   933→          if (sys.requiredTech && !s.technologies[sys.requiredTech]) continue;
   934→
   935→          // Check if system is manually disabled
   936→          const manuallyEnabled = s.systemEnabled[sysId] !== false; // default true
   937→          // ARCHETYPES-REWORK — Necroid grants partial production from disabled
   938→          // systems (productionFromDisabledSystemsMult). When active, disabled
   939→          // systems still produce at the fractional rate but consume no upkeep.
   940→          const isDisabledProducer = !manuallyEnabled && archDisabledFrac > 0;
   941→          if (!manuallyEnabled && !hasAutoBalancer && !isDisabledProducer) continue;
   942→
   943→          // Cap-aware logic: if all output resources are at cap, idle the system
   944→          // (don't produce, don't consume upkeep — no waste). Disabled producers
   945→          // (necroid) skip this check — they still produce even when capped, but
   946→          // only at the fractional rate.
   947→          if (!isDisabledProducer) {
   948→            const outputsCapped = Object.keys(sys.produces).length > 0 &&
   949→              Object.keys(sys.produces).every((r) => {
   950→                const cap = capacities[r] || 0;
   951→                return (resources[r] || 0) >= cap - 0.01;
   952→              });
   953→
   954→            if (outputsCapped) {
   955→              // System idles — no production, no upkeep
   956→              continue;
   957→            }
   958→          }
   959→
   960→          const stageMult = sys.stage === STAGES[s.stageIndex].id ? 1 : 0.5;
   961→          // ARCHETYPES-REWORK — disabled producers (necroid) get only the
   962→          // fractional production rate. They still benefit from archAllMult +
   963→          // stage scaling, but their effective count is multiplied by the
