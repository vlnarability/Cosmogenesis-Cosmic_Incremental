     2→// LAYER 2 — ENLIGHTENMENT (CONSTELLATION MAP)
     3→// ===========================================================================
     4→// After 3 trials are mastered, the god opens its eyes. The Enlightenment layer
     5→// is a star map: the player illuminates individual stars within real IAU
     6→// constellations. Completing a constellation grants a permanent bonus.
     7→// Currency = Divinity (earned at prestige while this layer is unlocked).
     8→// Complete CONSTELLATIONS_REQUIRED_FOR_LAYER_3 constellations to advance to
     9→// Layer 3 (Transcendence).
    10→//
    11→// ===========================================================================
    12→// ARCHITECTURE NOTE — The full IAU list has 88 constellations. This file ships
    13→// ALL 88 (12 zodiac + 23 northern + 23 equatorial + 30 southern, grouped by
    14→// approximate declination). Star positions are normalized 0-100 and packed for
    15→// visual variety per category — they're NOT a literal equatorial projection.
    16→// The data shape (`ConstellationDef`) is constellation-count-agnostic, so
    17→// adding/removing entries requires no other code changes.
    18→// ===========================================================================
    19→
    20→export type ConstellationCategory =
    21→  | "zodiac"
    22→  | "northern"
    23→  | "southern"
    24→  | "equatorial";

    27→  /** Globally-unique star id, e.g. "orion_alpha". */
    28→  id: string;
    29→  /** Normalized X coordinate on the star map, 0-100 (left → right). */
    30→  x: number;
    31→  /** Normalized Y coordinate on the star map, 0-100 (top → bottom). */
    32→  y: number;
    33→}

    35→export interface ConstellationBonus {
    36→  /** Additive production multiplier (0.05 = +5%). */
    37→  productionMult?: number;
    38→  /** Additive capacity multiplier. */
    39→  capMult?: number;
    40→  /** Additive EP multiplier (applied at prestige). */
    41→  epMult?: number;
    42→  /** Additive population growth multiplier. */
    43→  popGrowthMult?: number;
    44→  /** Cost reduction (negative = cheaper systems/techs). */
    45→  costMult?: number;
    46→  /** Bonus Divinity earned on prestige. */
    47→  divinityMult?: number;

    50→export interface ConstellationDef {
    51→  /** Slug id, e.g. "orion". */
    52→  id: string;
    53→  /** Display name, e.g. "Orion". */
    54→  name: string;
    55→  /** IAU designation (Latin), e.g. "Orion". */
    56→  i18nName: string;
    57→  /** Stars with normalized 0-100 coordinates on the star map. */
    58→  stars: ConstellationStar[];
    59→  /** Pairs of star ids that form the constellation's connecting lines. */
    60→  connections: [string, string][];
    61→  /** Permanent reward for completing this constellation. */
    62→  bonus: ConstellationBonus;
    63→  /** Flavor grouping. */
    64→  category: ConstellationCategory;
    65→}

    67→// ---------------------------------------------------------------------------
    68→// COST MODEL CONSTANTS
    69→// ---------------------------------------------------------------------------
    70→// **Critical**: matches the user-supplied spec exactly.
    71→//   baseCost = 10 Divinity.
    72→//   withinConstellationGrowth = 1.5 — each subsequent star in the SAME
    73→//     constellation multiplies the cost by 1.5 (cumulative).
    74→//   costGrowth = 2.0 — multiplier per OTHER started constellation (i.e. any
    75→//     constellation the player has already begun illuminating that isn't the
    76→//     one being illuminated right now).
    77→//
    78→// Concretely (baseCost = 10):
    79→//   A·star1:  10 × 1.5^0 × 2.0^0 = 10       (A is the first started → N=0)
    80→//   A·star2:  10 × 1.5^1 × 2.0^0 = 15        (A is the only started, N=0 still)
    81→//   A·star3:  10 × 1.5^2 × 2.0^0 = 22.5
    82→//   B·star1:  10 × 1.5^0 × 2.0^1 = 20        (A is started, B is new → N=1)
    83→//   B·star2:  10 × 1.5^1 × 2.0^1 = 30        (A+B started, B excluded → N=1)
    84→//   Cass·star1: 10 × 1.5^0 × 2.0^2 = 40      (A+B started, Cass is new → N=2)
    85→//
    86→// Where N = (# of started constellations) - (1 if THIS constellation is
    87→// already started, else 0). I.e. N counts OTHER started constellations.
    88→//
    89→// The "new constellation penalty" the user describes is the costGrowth^N
    90→// term: it only kicks in when this is a brand-new constellation (because at
    91→// that point every other started constellation counts toward N). For
    92→// already-started constellations, N stays fixed at the count when you began
    93→// that constellation, which means subsequent stars in a "penalized"
    94→// constellation inherit the same N multiplier — encouraging the player to
    95→// finish a constellation before starting a new one.
    96→// ---------------------------------------------------------------------------
    97→export const CONSTELLATION_BASE_COST = 10;
    98→export const WITHIN_CONSTELLATION_GROWTH = 1.5;
    99→export const COST_GROWTH_PER_STARTED = 2.0;
   100→export const NEW_CONSTELLATION_PENALTY = 2.0;
