     1→// Achievement system — persistent goals across all runs.
     2→// Each achievement grants a small permanent bonus (0.5-3% per achievement).
     3→// Total at full completion: ~15-20% production, ~7% capacity, ~10% EP, ~5% pop.
     4→//
     5→// 4 categories: stages / layers / milestones / fun
     6→//   - stages    (35) : 7 stages × 5 (reach / first system / first tech /
     7→//                      completionist / builder)
     8→//   - layers    (50) : 10 layers × 5 (enter / first action / progress /
     9→//                      complete / master)
    10→//   - milestones(~95): 36 user-specified + tiered progression counters
    11→//   - fun       (~30): UI/easter-egg + retained legacy curiosities
    12→//
    13→// REMOVED per user request: OOM Survivor, Cosmic Visionary (light all 497
    14→// stars). ADDED per user request: True Masochistic Perfection.
    15→//
    16→// Bonus types:
    17→//   - production : additive bonus to all resource production
    18→//   - capacity   : additive bonus to resource capacities
    19→//   - ep         : additive bonus to Evolution Points gained on prestige
    20→//   - pop        : additive bonus to population growth rate
    21→

    51→export type AchievementCategory =
    52→  | "stages"
    53→  | "layers"
    54→  | "milestones"
    55→  | "fun";

    57→export type AchievementBonusType =

    59→  | "capacity"

    63→export interface AchievementDef {
    64→  id: string;
    65→  name: string;
    66→  desc: string;
    67→  category: AchievementCategory;
    68→  icon: string;
    69→  // Check if achievement is earned given the meta state
    70→  check: (meta: AchievementCheckCtx) => boolean;
    71→  bonus: { type: AchievementBonusType; value: number };
    72→  // For display: tiers/progress (optional, not implemented yet)
    73→  tier?: number;
    74→}

    91→  totalEventsResolved?: number; // across all runs
    92→
    93→  // ---- New fields for 200+ achievements (all optional — callers may
    94→  //      build a partial ctx; check functions use `|| 0` / `|| false`
    95→  //      defaults so the achievement simply stays locked if a field is
    96→  //      missing on a given save). ----
    97→
    98→  // Stage + meta progression
    99→  stageIndex?: number;
   100→  totalPlayTime?: number; // seconds across all runs

   117→  // L2 — Constellations
