    44→function mulberry32(seed: number): () => number {
    45→  let t = seed >>> 0;
    46→  return () => {
    47→    t = (t + 0x6d2b79f5) >>> 0;
    48→    let r = Math.imul(t ^ (t >>> 15), 1 | t);
    49→    r = (r + Math.imul(r ^ (r >>> 7), 61 | r)) ^ r;
    50→    return ((r ^ (r >>> 14)) >>> 0) / 4294967296;
    51→  };
    52→}

    63→// Generate stable 3D positions for worlds. Each world gets a deterministic
    64→// random position in a sphere. We use a sqrt-radius distribution so worlds
    65→// are spread more on the outer shell (less clumping at the center) and a
    66→// wide radius range (60..380) so dots are visibly separate in 3D space.
    67→function computeWorldDots(worldTiers: Record<string, number>): WorldDot[] {
    68→  const rng = mulberry32(0xfeedface);
    69→  const dots: WorldDot[] = [];
    70→  for (const tier of FUSION_TIERS) {
    71→    const count = Math.max(0, Math.floor(worldTiers[tier.id] || 0));
    72→    if (count === 0) continue;
    73→    // Cap the visualized count per tier so the 3D scene stays performant.
    74→    const maxDots = Math.min(count, tier.id === "1" ? 220 : 40);
    75→    for (let i = 0; i < maxDots; i++) {
    76→      // Uniform direction (theta + phi via acos trick).
    77→      const theta = rng() * Math.PI * 2;
    78→      const phi = Math.acos(2 * rng() - 1);
    79→      // sqrt distribution → more dots near the outer shell (galaxy look).
    80→      // Radius range 60..380 gives ~6× spread between nearest + farthest,
    81→      // so dots are clearly separated at any rotation angle.
    82→      const r = 60 + Math.sqrt(rng()) * 320;
    83→      dots.push({
    84→        x: r * Math.sin(phi) * Math.cos(theta),
    85→        y: r * Math.sin(phi) * Math.sin(theta),
    86→        z: r * Math.cos(phi),
    87→        tier: tier.id,
    88→        color: TIER_COLORS[tier.id] || "#fff",
    89→        size: TIER_SIZES[tier.id] || 5,
    90→      });
    91→    }
    92→  }
    93→  return dots;
    94→}

    96→// Background starfield — distant points positioned in 3D, rendered with
    97→// reduced opacity for parallax depth. Pushed far back so they don't
    98→// overlap with the world sphere.
    99→function computeStarfield(): WorldDot[] {
   100→  const rng = mulberry32(0x12345678);
   101→  const stars: WorldDot[] = [];
   102→  for (let i = 0; i < 100; i++) {
   103→    const theta = rng() * Math.PI * 2;
   104→    const phi = Math.acos(2 * rng() - 1);
   105→    const r = 500 + rng() * 200; // far outside the world sphere
   106→    stars.push({
   107→      x: r * Math.sin(phi) * Math.cos(theta),
   108→      y: r * Math.sin(phi) * Math.sin(theta),
   109→      z: r * Math.cos(phi),
   110→      tier: "1",
   111→      color: "#ffffff",
   112→      size: 1.5 + rng() * 1.5,
   113→    });
   114→  }
   115→  return stars;
   116→}

   118→// ===========================================================================
   119→// WorldVisualization3D — pure CSS 3D transforms. No external deps.
   120→// Features:
   121→//   • Drag to rotate 360° (rotateX clamped ±89°, rotateY unbounded).
   122→//   • Auto-rotate 6°/s on Y axis when not dragging.
   123→//   • Zoom in/out via buttons (+/-), mouse wheel, or pinch (touch).
   124→//   • Sharper dot rendering: solid dot + thin border + small glow (no
   125→//     soft blur). `will-change: transform` to hint the compositor.
   126→//   • Reset button restores default angle + zoom.

   129→const ZOOM_MIN = 0.4;
   130→const ZOOM_MAX = 3.0;
   131→const ZOOM_STEP = 0.15;
   132→
   133→export function WorldVisualization3D({ dots }: { dots: WorldDot[] }) {
   134→  // Rotation state — degrees. Drag to update; auto-rotate when idle.
   135→  const [rotX, setRotX] = useState(-18);
   136→  const [rotY, setRotY] = useState(25);
   137→  const [zoom, setZoom] = useState(1.0);
   138→  const [isDragging, setIsDragging] = useState(false);
   139→  const dragStart = useRef<{ x: number; y: number; rotX: number; rotY: number } | null>(null);
   140→  // Pinch-tracking refs (touch). Stores the initial distance + zoom.
   141→  const pinchStart = useRef<{ dist: number; zoom: number } | null>(null);
   142→  const containerRef = useRef<HTMLDivElement>(null);
   143→  const stars = useMemo(() => computeStarfield(), []);
   144→
   145→  // Auto-rotate when not dragging. Subtle continuous Y rotation.
   146→  useEffect(() => {
   147→    if (isDragging) return;
   148→    let raf: number;
   149→    let last = performance.now();
   150→    const tick = (now: number) => {
   151→      const dt = (now - last) / 1000;
   152→      last = now;
   153→      // 6°/s slow rotation around Y axis.
   154→      setRotY((prev) => (prev + 6 * dt) % 360);
   155→      raf = requestAnimationFrame(tick);
   156→    };
   157→    raf = requestAnimationFrame(tick);
   158→    return () => cancelAnimationFrame(raf);
   159→  }, [isDragging]);
   160→
   161→  const onPointerDown = useCallback((e: React.PointerEvent) => {
   162→    e.preventDefault();
   163→    try { (e.target as Element).setPointerCapture(e.pointerId); } catch {}
   164→    dragStart.current = { x: e.clientX, y: e.clientY, rotX, rotY };
   165→    setIsDragging(true);
   166→  }, [rotX, rotY]);
   167→
   168→  const onPointerMove = useCallback((e: React.PointerEvent) => {
   169→    if (!dragStart.current) return;
   170→    const dx = e.clientX - dragStart.current.x;
   171→    const dy = e.clientY - dragStart.current.y;
   172→    // 0.4° per pixel of drag — feels responsive without being twitchy.
   173→    const newRotY = (dragStart.current.rotY + dx * 0.4) % 360;
   174→    let newRotX = dragStart.current.rotX - dy * 0.4;
   175→    // Clamp X rotation to [-89°, 89°] so the scene never fully flips.
   176→    newRotX = Math.max(-89, Math.min(89, newRotX));
   177→    setRotY(newRotY);
   178→    setRotX(newRotX);
   179→  }, []);
   180→
   181→  const onPointerUp = useCallback((e: React.PointerEvent) => {
   182→    if (dragStart.current) {
   183→      try { (e.target as Element).releasePointerCapture(e.pointerId); } catch {}
   184→    }
   185→    dragStart.current = null;
   186→    pinchStart.current = null;
   187→    setIsDragging(false);
   188→  }, []);
   189→
   190→  // Zoom helpers — clamp to [ZOOM_MIN, ZOOM_MAX].
   191→  const clampZoom = useCallback((v: number) => Math.max(ZOOM_MIN, Math.min(ZOOM_MAX, v)), []);
   192→  const zoomIn = useCallback(() => setZoom((z) => clampZoom(z + ZOOM_STEP * 2)), [clampZoom]);
   193→  const zoomOut = useCallback(() => setZoom((z) => clampZoom(z - ZOOM_STEP * 2)), [clampZoom]);
   194→
   195→  // Mouse-wheel zoom — intercepts wheel so the page doesn't scroll.
   196→  const onWheel = useCallback((e: React.WheelEvent) => {
   197→    e.preventDefault();
   198→    // deltaY > 0 = scroll down = zoom out. Step scaled by scroll amount.
   199→    const step = ZOOM_STEP * Math.min(3, Math.abs(e.deltaY) / 50);
   200→    const delta = e.deltaY > 0 ? -step : step;
   201→    setZoom((z) => clampZoom(z + delta));
   202→  }, [clampZoom]);
   203→
   204→  // Touch pinch — when 2 fingers are down, the distance between them
   205→  // drives zoom. We track the initial distance + zoom and apply the ratio.
   206→  const onTouchMove = useCallback((e: React.TouchEvent) => {
   207→    if (e.touches.length !== 2) return;
   208→    e.preventDefault();
   209→    const t1 = e.touches[0];
   210→    const t2 = e.touches[1];
   211→    const dist = Math.hypot(t2.clientX - t1.clientX, t2.clientY - t1.clientY);
   212→    if (!pinchStart.current) {
   213→      pinchStart.current = { dist, zoom };
   214→      return;
   215→    }
   216→    const ratio = dist / pinchStart.current.dist;
   217→    setZoom(clampZoom(pinchStart.current.zoom * ratio));
   218→  }, [zoom, clampZoom]);
   219→
   220→  const resetView = useCallback(() => {
   221→    setRotX(-18);
   222→    setRotY(25);
   223→    setZoom(1.0);
   224→  }, []);
   225→
   226→  return (
   227→    <div className="relative w-full" style={{ height: "clamp(360px, 60vh, 600px)" }}>
   228→      <div
   229→        ref={containerRef}
   230→        className="relative w-full h-full overflow-hidden rounded-lg border border-violet-500/20 bg-gradient-to-b from-[#020817] via-[#0a0a23] to-[#050518] touch-none select-none"
   231→        // Perspective scales inversely with zoom: higher zoom = smaller
   232→        // perspective = closer view. Base 1100px gives comfortable depth.
   233→        style={{ perspective: `${1100 / zoom}px`, perspectiveOrigin: "50% 50%" }}
   234→        onPointerDown={onPointerDown}
   235→        onPointerMove={onPointerMove}
   236→        onPointerUp={onPointerUp}
   237→        onPointerCancel={onPointerUp}
   238→        onWheel={onWheel}
   239→        onTouchMove={onTouchMove}
   240→        role="application"
   241→        aria-label="Drag to rotate · scroll to zoom · 3D world visualization"
   242→      >
   243→        {/* Inner 3D scene — preserve-3d so children are positioned in 3D space.
   244→            Scale by zoom so the whole sphere grows/shrinks together. */}
   245→        <div
   246→          className="absolute left-1/2 top-1/2"
   247→          style={{
   248→            transformStyle: "preserve-3d",
   249→            transform: `translate(-50%, -50%) scale3d(${zoom}, ${zoom}, ${zoom}) rotateX(${rotX}deg) rotateY(${rotY}deg)`,
   250→            willChange: "transform",
   251→          }}
   252→        >
   253→          {/* Starfield — far back, low opacity. */}
   254→          {stars.map((s, i) => (
   255→            <div
   256→              key={`star-${i}`}
   257→              className="absolute rounded-full"
   258→              style={{
   259→                width: `${s.size}px`,
   260→                height: `${s.size}px`,
   261→                background: s.color,
   262→                opacity: 0.4,
   263→                transform: `translate3d(${s.x}px, ${s.y}px, ${s.z}px)`,
   264→                transformStyle: "preserve-3d",
   265→                willChange: "transform",
   266→              }}
   267→            />
   268→          ))}
   269→
   270→          {/* World dots — solid dot + thin border + small glow. */}
   271→          {dots.map((dot, i) => (
   272→            <div
   273→              key={`world-${i}`}
   274→              className="absolute rounded-full"
   275→              style={{
   276→                width: `${dot.size}px`,
   277→                height: `${dot.size}px`,
   278→                marginLeft: `-${dot.size / 2}px`,
   279→                marginTop: `-${dot.size / 2}px`,
   280→                background: dot.color,
   281→                // Sharp 1px border for definition + small glow for visibility.
   282→                // Old boxShadow was 2-layer with 1.6× radius → too soft.
   283→                border: `1px solid ${dot.color}`,
   284→                boxShadow: `0 0 ${Math.max(4, dot.size * 0.6)}px ${dot.size * 0.25}px ${dot.color}66`,
   285→                opacity: 0.95,
   286→                transform: `translate3d(${dot.x}px, ${dot.y}px, ${dot.z}px)`,
   287→                transformStyle: "preserve-3d",
   288→                willChange: "transform",
   289→              }}
   290→              title={`${FUSION_TIERS.find(t => t.id === dot.tier)?.name || dot.tier} world`}
   291→            >
   292→              {dot.tier === "void" && (
   293→                <div
   294→                  className="absolute inset-0 rounded-full border-2"
   295→                  style={{
   296→                    borderColor: "#a855f7",
   297→                    transform: "scale(1.8)",
   298→                    opacity: 0.6,
   299→                  }}
   300→                />
   301→              )}
   302→            </div>
   303→          ))}
   304→        </div>
   305→
   306→        {/* Top-left: hint badge + zoom controls + Reset */}
   307→        <div className="absolute top-2 left-2 flex items-center gap-1 pointer-events-none flex-wrap">
   308→          <Badge variant="outline" className="text-[0.55rem] text-violet-200 border-violet-400/40 bg-slate-950/70 backdrop-blur pointer-events-auto">
   309→            <Orbit className="w-2.5 h-2.5 mr-0.5" /> Drag · scroll to zoom
   310→          </Badge>
   311→          <div className="flex items-center gap-0.5 pointer-events-auto bg-slate-950/70 backdrop-blur rounded-md border border-violet-400/40">
   312→            <Button
   313→              size="sm"
   314→              variant="ghost"
   315→              className="h-6 w-6 p-0 text-violet-200 hover:bg-violet-500/20"
   316→              onClick={zoomOut}
   317→              disabled={zoom <= ZOOM_MIN + 0.001}
   318→              title="Zoom out"
   319→              aria-label="Zoom out"
   320→            >
   321→              <ZoomOut className="w-3 h-3" />
   322→            </Button>
   323→            <span className="text-[0.55rem] text-violet-200 font-mono px-1 min-w-[2.4rem] text-center">
   324→              {zoom.toFixed(2)}×
   325→            </span>
   326→            <Button

   352→          style={{ perspective: "200px" }}

   412→  const forgeWorldBulk = useGameStore((s) => s.forgeWorldBulk);
   413→  const fuseWorlds = useGameStore((s) => s.fuseWorlds);
   414→  const fuseWorldsBulk = useGameStore((s) => s.fuseWorldsBulk);
   415→
   416→  const rawCount = rawWorldCount(worldTiers);
   417→  const effectiveCount = effectiveWorldCount(worldTiers);
   418→  const flameMult = eternalFlameMultiplier(eternalFlameInvested);
   419→  const dots = useMemo(() => computeWorldDots(worldTiers), [worldTiers]);
   420→
   421→  // Speed slider → Divinity/s multiplier: 1 + speed × 0.10 (0 → ×1.0, 10 → ×2.0).
   422→  const sliderSpeedMult = 1 + (evolutionSliders.speed || 0) * 0.10;
   423→  const sliderSum =
   424→    (evolutionSliders.speed || 0) +
   425→    (evolutionSliders.aggression || 0) +
   426→    (evolutionSliders.curiosity || 0) +
   427→    (evolutionSliders.faith || 0);
   428→
   429→  const divPerSec = effectiveCount > 0 && peakProductionRate > 0
   430→    ? effectiveCount * peakProductionRate * 0.01 * flameMult * (activeWorldArchetype ? 1.30 : 1) * sliderSpeedMult
   431→    : 0;
   432→
   433→  // Bulk upgrade ownership — bypassed in debug mode (debugFreeBuy).
   434→  const hasBulkForge10 = debugFreeBuy || (upgrades["bulk_forge_10x"] || 0) > 0;
   435→  const hasBulkForge100 = debugFreeBuy || (upgrades["bulk_forge_100x"] || 0) > 0;
   436→  const hasBulkForgeMax = debugFreeBuy || (upgrades["bulk_forge_max"] || 0) > 0;
   437→  const hasBulkFuse10 = debugFreeBuy || (upgrades["bulk_fuse_10x"] || 0) > 0;
   438→  const hasBulkFuse100 = debugFreeBuy || (upgrades["bulk_fuse_100x"] || 0) > 0;
   439→  const hasBulkFuseMax = debugFreeBuy || (upgrades["bulk_fuse_max"] || 0) > 0;
   440→
   441→  // Cost of the next single forge = ceil(50 × 1.5^worldsForged).
   442→  const nextForgeCost = Math.ceil(50 * Math.pow(1.5, worldsForged));
   443→  const canAffordNextForge = debugFreeBuy || divinity >= nextForgeCost;
   444→
   445→  // Compute how many worlds the player can afford to forge right now,
   446→  // starting from `worldsForged`. Capped at 10000 (safety cap). The cost
   447→  // grows exponentially so the loop terminates quickly even with very
   448→  // large divinity pools.
   449→  const maxForgeable = useMemo(() => {
   450→    if (divinity <= 0 && !debugFreeBuy) return 0;
   451→    let remaining = divinity;
   452→    let count = 0;
   453→    for (let i = 0; i < 10000; i++) {
   454→      const cost = Math.ceil(50 * Math.pow(1.5, worldsForged + i));
   455→      if (remaining < cost) break;
