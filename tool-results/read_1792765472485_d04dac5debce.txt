    43→function mulberry32(seed: number): () => number {
    44→  let t = seed >>> 0;
    45→  return () => {
    46→    t = (t + 0x6d2b79f5) >>> 0;
    47→    let r = Math.imul(t ^ (t >>> 15), 1 | t);
    48→    r = (r + Math.imul(r ^ (r >>> 7), 61 | r)) ^ r;
    49→    return ((r ^ (r >>> 14)) >>> 0) / 4294967296;
    50→  };
    51→}

    53→interface WorldDot {
    54→  x: number;
    55→  y: number;
    56→  z: number;

    62→// Generate stable 3D positions for worlds. Each world gets a deterministic
    63→// random position in a sphere. We use a sqrt-radius distribution so worlds
    64→// are spread more on the outer shell (less clumping at the center) and a
    65→// wide radius range (60..380) so dots are visibly separate in 3D space.
    66→function computeWorldDots(worldTiers: Record<string, number>): WorldDot[] {
    67→  const rng = mulberry32(0xfeedface);
    68→  const dots: WorldDot[] = [];
    69→  for (const tier of FUSION_TIERS) {
    70→    const count = Math.max(0, Math.floor(worldTiers[tier.id] || 0));
    71→    if (count === 0) continue;
    72→    // Cap the visualized count per tier so the 3D scene stays performant.
    73→    const maxDots = Math.min(count, tier.id === "1" ? 220 : 40);
    74→    for (let i = 0; i < maxDots; i++) {
    75→      // Uniform direction (theta + phi via acos trick).
    76→      const theta = rng() * Math.PI * 2;
    77→      const phi = Math.acos(2 * rng() - 1);
    78→      // sqrt distribution → more dots near the outer shell (galaxy look).
    79→      // Radius range 60..380 gives ~6× spread between nearest + farthest,
    80→      // so dots are clearly separated at any rotation angle.
    81→      const r = 60 + Math.sqrt(rng()) * 320;
    82→      dots.push({
    83→        x: r * Math.sin(phi) * Math.cos(theta),
    84→        y: r * Math.sin(phi) * Math.sin(theta),
    85→        z: r * Math.cos(phi),
    86→        tier: tier.id,
    87→        color: TIER_COLORS[tier.id] || "#fff",
    88→        size: TIER_SIZES[tier.id] || 5,
    89→      });
    90→    }
    91→  }
    92→  return dots;
    93→}

    95→// Background starfield — distant points positioned in 3D, rendered with
    96→// reduced opacity for parallax depth. Pushed far back so they don't
    97→// overlap with the world sphere.
    98→function computeStarfield(): WorldDot[] {
    99→  const rng = mulberry32(0x12345678);
   100→  const stars: WorldDot[] = [];
   101→  for (let i = 0; i < 100; i++) {
   102→    const theta = rng() * Math.PI * 2;
   103→    const phi = Math.acos(2 * rng() - 1);
   104→    const r = 500 + rng() * 200; // far outside the world sphere
   105→    stars.push({
   106→      x: r * Math.sin(phi) * Math.cos(theta),
   107→      y: r * Math.sin(phi) * Math.sin(theta),
   108→      z: r * Math.cos(phi),
   109→      tier: "1",
   110→      color: "#ffffff",
   111→      size: 1.5 + rng() * 1.5,
   112→    });
   113→  }
   114→  return stars;
   115→}

   117→// ===========================================================================
   118→// WorldVisualization3D — pure CSS 3D transforms. No external deps.
   119→// Features:
   120→//   • Drag to rotate 360° (rotateX clamped ±89°, rotateY unbounded).
   121→//   • Auto-rotate 6°/s on Y axis when not dragging.
   122→//   • Zoom in/out via buttons (+/-), mouse wheel, or pinch (touch).
   123→//   • Sharper dot rendering: solid dot + thin border + small glow (no
   124→//     soft blur). `will-change: transform` to hint the compositor.
   125→//   • Reset button restores default angle + zoom.
   126→// ===========================================================================
   127→
   128→const ZOOM_MIN = 0.4;
   129→const ZOOM_MAX = 3.0;
   130→const ZOOM_STEP = 0.15;
   131→
   132→export function WorldVisualization3D({ dots }: { dots: WorldDot[] }) {
   133→  // Rotation state — degrees. Drag to update; auto-rotate when idle.
   134→  const [rotX, setRotX] = useState(-18);
   135→  const [rotY, setRotY] = useState(25);
   136→  const [zoom, setZoom] = useState(1.0);
   137→  const [isDragging, setIsDragging] = useState(false);
   138→  const dragStart = useRef<{ x: number; y: number; rotX: number; rotY: number } | null>(null);
   139→  // Pinch-tracking refs (touch). Stores the initial distance + zoom.
   140→  const pinchStart = useRef<{ dist: number; zoom: number } | null>(null);
   141→  const containerRef = useRef<HTMLDivElement>(null);
   142→  const stars = useMemo(() => computeStarfield(), []);
   143→
   144→  // Auto-rotate when not dragging. Subtle continuous Y rotation.
   145→  useEffect(() => {
   146→    if (isDragging) return;
   147→    let raf: number;
   148→    let last = performance.now();
   149→    const tick = (now: number) => {
   150→      const dt = (now - last) / 1000;
   151→      last = now;
   152→      // 6°/s slow rotation around Y axis.
   153→      setRotY((prev) => (prev + 6 * dt) % 360);
   154→      raf = requestAnimationFrame(tick);
   155→    };
   156→    raf = requestAnimationFrame(tick);
   157→    return () => cancelAnimationFrame(raf);
   158→  }, [isDragging]);
   159→
   160→  const onPointerDown = useCallback((e: React.PointerEvent) => {
   161→    e.preventDefault();
   162→    try { (e.target as Element).setPointerCapture(e.pointerId); } catch {}
   163→    dragStart.current = { x: e.clientX, y: e.clientY, rotX, rotY };
   164→    setIsDragging(true);
   165→  }, [rotX, rotY]);
   166→
   167→  const onPointerMove = useCallback((e: React.PointerEvent) => {
   168→    if (!dragStart.current) return;
   169→    const dx = e.clientX - dragStart.current.x;
   170→    const dy = e.clientY - dragStart.current.y;
   171→    // 0.4° per pixel of drag — feels responsive without being twitchy.
   172→    const newRotY = (dragStart.current.rotY + dx * 0.4) % 360;
   173→    let newRotX = dragStart.current.rotX - dy * 0.4;
   174→    // Clamp X rotation to [-89°, 89°] so the scene never fully flips.
   175→    newRotX = Math.max(-89, Math.min(89, newRotX));
   176→    setRotY(newRotY);
   177→    setRotX(newRotX);
   178→  }, []);
   179→
   180→  const onPointerUp = useCallback((e: React.PointerEvent) => {
   181→    if (dragStart.current) {
   182→      try { (e.target as Element).releasePointerCapture(e.pointerId); } catch {}
   183→    }
   184→    dragStart.current = null;
   185→    pinchStart.current = null;
   186→    setIsDragging(false);
   187→  }, []);
   188→
   189→  // Zoom helpers — clamp to [ZOOM_MIN, ZOOM_MAX].
   190→  const clampZoom = useCallback((v: number) => Math.max(ZOOM_MIN, Math.min(ZOOM_MAX, v)), []);
   191→  const zoomIn = useCallback(() => setZoom((z) => clampZoom(z + ZOOM_STEP * 2)), [clampZoom]);
   192→  const zoomOut = useCallback(() => setZoom((z) => clampZoom(z - ZOOM_STEP * 2)), [clampZoom]);
   193→
   194→  // Mouse-wheel zoom — intercepts wheel so the page doesn't scroll.
   195→  const onWheel = useCallback((e: React.WheelEvent) => {
   196→    e.preventDefault();
   197→    // deltaY > 0 = scroll down = zoom out. Step scaled by scroll amount.
   198→    const step = ZOOM_STEP * Math.min(3, Math.abs(e.deltaY) / 50);
   199→    const delta = e.deltaY > 0 ? -step : step;
   200→    setZoom((z) => clampZoom(z + delta));
   201→  }, [clampZoom]);
   202→
   203→  // Touch pinch — when 2 fingers are down, the distance between them
   204→  // drives zoom. We track the initial distance + zoom and apply the ratio.
   205→  const onTouchMove = useCallback((e: React.TouchEvent) => {
   206→    if (e.touches.length !== 2) return;
   207→    e.preventDefault();
   208→    const t1 = e.touches[0];
   209→    const t2 = e.touches[1];
   210→    const dist = Math.hypot(t2.clientX - t1.clientX, t2.clientY - t1.clientY);
   211→    if (!pinchStart.current) {
   212→      pinchStart.current = { dist, zoom };
   213→      return;
   214→    }
   215→    const ratio = dist / pinchStart.current.dist;
   216→    setZoom(clampZoom(pinchStart.current.zoom * ratio));
   217→  }, [zoom, clampZoom]);
   218→
   219→  const resetView = useCallback(() => {
   220→    setRotX(-18);
   221→    setRotY(25);
   222→    setZoom(1.0);
   223→  }, []);
   224→
   225→  return (
   226→    <div className="relative w-full" style={{ height: "clamp(360px, 60vh, 600px)" }}>
   227→      <div
   228→        ref={containerRef}
   229→        className="relative w-full h-full overflow-hidden rounded-lg border border-violet-500/20 bg-gradient-to-b from-[#020817] via-[#0a0a23] to-[#050518] touch-none select-none"
   230→        // Perspective scales inversely with zoom: higher zoom = smaller
   231→        // perspective = closer view. Base 1100px gives comfortable depth.
   232→        style={{ perspective: `${1100 / zoom}px`, perspectiveOrigin: "50% 50%" }}
   233→        onPointerDown={onPointerDown}
   234→        onPointerMove={onPointerMove}
   235→        onPointerUp={onPointerUp}
   236→        onPointerCancel={onPointerUp}
   237→        onWheel={onWheel}
   238→        onTouchMove={onTouchMove}
   239→        role="application"
   240→        aria-label="Drag to rotate · scroll to zoom · 3D world visualization"
   241→      >
   242→        {/* Inner 3D scene — preserve-3d so children are positioned in 3D space.
   243→            Scale by zoom so the whole sphere grows/shrinks together. */}
   244→        <div
   245→          className="absolute left-1/2 top-1/2"
   246→          style={{
   247→            transformStyle: "preserve-3d",
   248→            transform: `translate(-50%, -50%) scale3d(${zoom}, ${zoom}, ${zoom}) rotateX(${rotX}deg) rotateY(${rotY}deg)`,
   249→            willChange: "transform",
   250→          }}
   251→        >
   252→          {/* Starfield — far back, low opacity. */}
   253→          {stars.map((s, i) => (
   254→            <div
   255→              key={`star-${i}`}
   256→              className="absolute rounded-full"
   257→              style={{
   258→                width: `${s.size}px`,
   259→                height: `${s.size}px`,
   260→                background: s.color,
   261→                opacity: 0.4,
   262→                transform: `translate3d(${s.x}px, ${s.y}px, ${s.z}px)`,
   263→                transformStyle: "preserve-3d",
   264→                willChange: "transform",
   265→              }}
   266→            />
   267→          ))}
   268→
   269→          {/* World dots — solid dot + thin border + small glow. */}
   270→          {dots.map((dot, i) => (
   271→            <div
   272→              key={`world-${i}`}
   273→              className="absolute rounded-full"
   274→              style={{
   275→                width: `${dot.size}px`,
   276→                height: `${dot.size}px`,
   277→                marginLeft: `-${dot.size / 2}px`,
   278→                marginTop: `-${dot.size / 2}px`,
   279→                background: dot.color,
   280→                // Sharp 1px border for definition + small glow for visibility.
   281→                // Old boxShadow was 2-layer with 1.6× radius → too soft.
   282→                border: `1px solid ${dot.color}`,
   283→                boxShadow: `0 0 ${Math.max(4, dot.size * 0.6)}px ${dot.size * 0.25}px ${dot.color}66`,
   284→                opacity: 0.95,
   285→                transform: `translate3d(${dot.x}px, ${dot.y}px, ${dot.z}px)`,
   286→                transformStyle: "preserve-3d",
   287→                willChange: "transform",
   288→              }}
   289→              title={`${FUSION_TIERS.find(t => t.id === dot.tier)?.name || dot.tier} world`}
   290→            >
   291→              {dot.tier === "void" && (
   292→                <div
   293→                  className="absolute inset-0 rounded-full border-2"
   294→                  style={{
   295→                    borderColor: "#a855f7",
   296→                    transform: "scale(1.8)",
   297→                    opacity: 0.6,
   298→                  }}
   299→                />
   300→              )}
   301→            </div>
   302→          ))}
   303→        </div>
   304→
   305→        {/* Top-left: hint badge + zoom controls + Reset */}
   306→        <div className="absolute top-2 left-2 flex items-center gap-1 pointer-events-none flex-wrap">
   307→          <Badge variant="outline" className="text-[0.55rem] text-violet-200 border-violet-400/40 bg-slate-950/70 backdrop-blur pointer-events-auto">
   308→            <Orbit className="w-2.5 h-2.5 mr-0.5" /> Drag · scroll to zoom
   309→          </Badge>
   310→          <div className="flex items-center gap-0.5 pointer-events-auto bg-slate-950/70 backdrop-blur rounded-md border border-violet-400/40">
   311→            <Button
   312→              size="sm"
   313→              variant="ghost"
   314→              className="h-6 w-6 p-0 text-violet-200 hover:bg-violet-500/20"
   315→              onClick={zoomOut}
   316→              disabled={zoom <= ZOOM_MIN + 0.001}
   317→              title="Zoom out"
   318→              aria-label="Zoom out"
   319→            >
   320→              <ZoomOut className="w-3 h-3" />
   321→            </Button>
   322→            <span className="text-[0.55rem] text-violet-200 font-mono px-1 min-w-[2.4rem] text-center">
   323→              {zoom.toFixed(2)}×
   324→            </span>
   325→            <Button
   326→              size="sm"
   327→              variant="ghost"
   328→              className="h-6 w-6 p-0 text-violet-200 hover:bg-violet-500/20"
   329→              onClick={zoomIn}
   330→              disabled={zoom >= ZOOM_MAX - 0.001}
   331→              title="Zoom in"
   332→              aria-label="Zoom in"
   333→            >
   334→              <ZoomIn className="w-3 h-3" />
   335→            </Button>
   336→          </div>
   337→          <Button
   338→            size="sm"
   339→            variant="outline"
   340→            className="h-6 text-[0.55rem] text-violet-200 border-violet-400/40 bg-slate-950/70 backdrop-blur pointer-events-auto"
   341→            onClick={resetView}
   342→            title="Reset view to default angle + zoom"
   343→          >
   344→            <RotateCcw className="w-2.5 h-2.5 mr-0.5" /> Reset
   345→          </Button>
   346→        </div>
   347→
   348→        {/* Coordinate axis hint — small 3D triad in the corner */}
   349→        <div
   350→          className="absolute bottom-2 right-2 pointer-events-none"
   351→          style={{ perspective: "200px" }}
   352→          aria-hidden
   353→        >
   354→          <div
   355→            style={{
   356→              transformStyle: "preserve-3d",
   357→              transform: `rotateX(${rotX}deg) rotateY(${rotY}deg)`,
   358→              width: "44px",
   359→              height: "44px",
   360→              position: "relative",
   361→            }}
   362→          >
   363→            {/* X axis - red */}
   364→            <div style={{ position: "absolute", left: "22px", top: "22px", width: "18px", height: "2px", background: "#ef4444", transform: "rotateY(0deg)", transformOrigin: "left center", borderRadius: "1px" }} />
   365→            {/* Y axis - green */}
   366→            <div style={{ position: "absolute", left: "22px", top: "22px", width: "2px", height: "18px", background: "#10b981", transform: "rotateX(0deg)", transformOrigin: "center top", borderRadius: "1px" }} />
   367→            {/* Z axis - blue (toward viewer) */}
   368→            <div style={{ position: "absolute", left: "22px", top: "22px", width: "18px", height: "2px", background: "#3b82f6", transform: "rotateY(90deg)", transformOrigin: "left center", borderRadius: "1px" }} />

   380→        </div>
   381→      </div>
   382→    </div>
   383→  );
   384→}
