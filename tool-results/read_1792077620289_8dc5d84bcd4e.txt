    44→// ----- helpers -----
    45→
    46→function ruleShortName(ruleId: string): string {
    47→  const r = UNIVERSE_RULES.find((x) => x.id === ruleId);
    48→  return r ? r.shortName : ruleId;
    49→}
    50→
    51→function ruleIcon(ruleId: string): string {
    52→  const r = UNIVERSE_RULES.find((x) => x.id === ruleId);
    53→  return r ? r.icon : "?";
    54→}
    55→
    56→function amountLabel(amt: number): string {
    57→  return `${amt >= 0 ? "+" : ""}${amt}`;
    58→}
    59→
    60→// ----- main component -----
    61→
    62→export function EternityContent() {
    63→  const closeLayerModal = useGameStore((s) => s.closeLayerModal);
    64→  const unlockedLayers = useGameStore((s) => s.unlockedLayers);
    65→  const divinity = useGameStore((s) => s.divinity || 0);
    66→  const cosmicBoonStacks = useGameStore((s) => s.cosmicBoonStacks || 0);
    67→  const chosenEnding = useGameStore((s) => s.chosenEnding);
    68→
    69→  // Genesis Engine state
    70→  const divineRelics = useGameStore((s) => s.divineRelics || []);
    71→  const crucibleRelicIds = useGameStore((s) => s.crucibleRelicIds || []);
    72→  const discoveredArchetypes = useGameStore((s) => s.discoveredArchetypes || {});
    73→  const universeSimulating = useGameStore((s) => s.universeSimulating);
    74→  const simulationStartTime = useGameStore((s) => s.simulationStartTime || 0);
    75→  const lastSimResult = useGameStore((s) => s.lastSimulationResult);
    76→  // Subscribe to time so the progress bar updates each tick.
    77→  const time = useGameStore((s) => s.time || 0);
    78→  const totalPlayTime = useGameStore((s) => s.totalPlayTime || 0);
    79→  const totalRuns = useGameStore((s) => s.totalRuns || 0);
    80→
    81→  // Actions
    82→  const addRelicToCrucible = useGameStore((s) => s.addRelicToCrucible);
    83→  const clearCrucible = useGameStore((s) => s.clearCrucible);
    84→  const revealRelicRule = useGameStore((s) => s.revealRelicRule);
    85→  const revealRelicAmount = useGameStore((s) => s.revealRelicAmount);
    86→  const triggerBigBang = useGameStore((s) => s.triggerBigBang);
    87→  const chooseEnding = useGameStore((s) => s.chooseEnding);
    88→
    89→  const isUnlocked = !!unlockedLayers.eternity;
    90→  const discoveredCount = Object.values(discoveredArchetypes).filter(Boolean).length;
    91→  const allDiscovered = discoveredCount >= ARCHETYPE_COUNT;
    92→  const boon = cosmicBoonBonus(cosmicBoonStacks);
    93→
    94→  // Compute crucible rule view (recomputed each render — cheap).
    95→  const crucibleRelics: DivineRelic[] = useMemo(() => {
    96→    return crucibleRelicIds
    97→      .map((id) => divineRelics.find((r) => r.id === id))
    98→      .filter((r): r is DivineRelic => !!r);
    99→  }, [crucibleRelicIds, divineRelics]);
   100→
   101→  const crucibleComp = useMemo(
   102→    () => computeCrucibleRules(crucibleRelics),
   103→    [crucibleRelics]
   104→  );
   105→
   106→  const simElapsed = universeSimulating ? Math.min(BIG_BANG_DURATION_SEC, time - simulationStartTime) : 0;
   107→  // P7 — simProgressPct removed; Big Bang now shows a skeleton screen instead
   108→  // of a progress bar. simElapsed is still used for the seconds-remaining count.
   109→  const simRemainingSec = universeSimulating ? Math.max(0, BIG_BANG_DURATION_SEC - simElapsed) : 0;
   110→
   111→  // Ending confirmation dialog
   112→  const [pendingEnding, setPendingEnding] = useState<"preserve" | "reset" | null>(null);
   113→  const confirmEnding = () => {
   114→    if (pendingEnding) {
   115→      chooseEnding(pendingEnding);
   116→      setPendingEnding(null);
   117→    }
   118→  };
   119→
   120→  // Total score for the ending screen
   121→  const finalScore = useMemo(() => {
   122→    // Resources: sum of all primary resources.
   123→    const resourceIds: Array<string> = ["atp", "glucose", "proteins", "lipids", "elements",
   124→      "food", "water", "materials", "organic_matter", "knowledge", "culture",
   125→      "wood", "lumber", "stone", "clay", "science", "happiness", "military_power",
   126→      "production", "gold", "influence", "energy", "alloys", "data",
   127→      "divinity", "evolution_points", "enlightenment_points", "transcendence_points"];
   128→    const resources = useGameStore.getState().resources || {};
   129→    let sum = 0;
   130→    for (const id of resourceIds) {
   131→      sum += (resources as any)[id] || 0;
   132→    }
   133→    // Layers completed count
   134→    const unlocked = useGameStore.getState().unlockedLayers || {};
   135→    const layersCompleted = Object.values(unlocked).filter(Boolean).length;
   136→    // Worlds: authored worlds + claimed galaxy systems
   137→    const authoredWorlds = (useGameStore.getState() as any).authoredWorlds || [];
   138→    const galaxySystems = (useGameStore.getState() as any).galaxySystems || [];
   139→    const claimedGalaxies = galaxySystems.filter((g: any) => g?.claimed).length;
   140→    const worldCount = authoredWorlds.length + claimedGalaxies;
   141→    // Final score: weighted sum
   142→    return Math.floor(sum * 0.01 + worldCount * 100 + layersCompleted * 500 + discoveredCount * 1000);
   143→  }, [discoveredCount, allDiscovered]);
   144→
   145→  return (
   146→    <>
   147→      <DialogHeader>
   148→        <DialogTitle className="flex items-center gap-2 text-lg flex-wrap">
   149→          <Star className="w-5 h-5" />
   150→          Layer 10 — Eternity (Genesis Engine)
   151→          <BackgroundLayerBadge layerId="eternity" />
   152→        </DialogTitle>
   153→        <DialogDescription className="flex items-center justify-between flex-wrap gap-2">
   154→          <span>
   155→            Load relics into the Crucible, trigger Big Bangs, and discover all 100 archetypes.
   156→          </span>
   157→          <div className="flex items-center gap-1.5">
   158→            <Badge variant="outline" className="text-violet-300 border-violet-400/40">
   159→              ✨ {formatNumber(divinity, 0)} Divinity
   160→            </Badge>
   161→            <Badge variant="outline" className="text-amber-300 border-amber-400/40">
   162→              🌟 {discoveredCount}/{ARCHETYPE_COUNT}
   163→            </Badge>
   164→            {cosmicBoonStacks > 0 && (
   165→              <Badge variant="outline" className="text-indigo-300 border-indigo-400/40">
   166→                🌠 {cosmicBoonStacks} Boon{cosmicBoonStacks !== 1 ? "s" : ""}
   167→              </Badge>
   168→            )}
   169→          </div>
   170→        </DialogDescription>
   171→      </DialogHeader>
   172→
   173→        {!isUnlocked ? (
   174→          <div className="text-center text-muted-foreground py-10">
   175→            <Lock className="w-8 h-8 mx-auto mb-2 opacity-50" />
   176→            <p className="text-sm">The Genesis Engine lies dormant.</p>
   177→            <p className="text-xs mt-1">Defeat all 3 Old Gods (Layer 9 — Divine War) to ignite the Crucible.</p>
   178→          </div>
   179→        ) : allDiscovered && chosenEnding === null ? (
   180→          <EndingScreen
   181→            discoveredCount={discoveredCount}
   182→            totalPlayTime={totalPlayTime}
   183→            thisRunTime={time}
   184→            totalRuns={totalRuns}
   185→            finalScore={finalScore}
   186→            boon={boon}
   187→            cosmicBoonStacks={cosmicBoonStacks}
   188→            onChooseEnding={(id) => setPendingEnding(id)}
   189→          />
   190→        ) : chosenEnding !== null ? (
   191→          <div className="text-center py-10">
   192→            <Sparkles className="w-8 h-8 mx-auto mb-2 text-amber-300" />
   193→            <p className="text-sm">Ending chosen: {ENDING_CHOICES.find((e) => e.id === chosenEnding)?.name}</p>
   194→            <p className="text-xs mt-1 text-muted-foreground">
   195→              {ENDING_CHOICES.find((e) => e.id === chosenEnding)?.desc}
   196→            </p>
   197→            <Button variant="outline" className="mt-4" onClick={() => closeLayerModal()}>Close</Button>
   198→          </div>
   199→        ) : (
   200→          <div className="space-y-4">
   201→            {/* Relic Inventory */}
   202→            <section>
   203→              <div className="text-[0.7rem] uppercase tracking-wide text-muted-foreground mb-1 flex items-center gap-1 justify-between">
   204→                <span className="flex items-center gap-1">
   205→                  <Sparkles className="w-3 h-3" /> Divine Relic Inventory ({divineRelics.length})
   206→                </span>
   207→                <span className="text-[0.6rem] normal-case">
   208→                  Reveal rule name: {REVEAL_COST_RULE} Div · amount: {REVEAL_COST_AMOUNT} Div
   209→                </span>
   210→              </div>
   211→              {divineRelics.length === 0 ? (
   212→                // P7 — Empty state: no relics yet. Use the shared EmptyState
   213→                // component for consistent styling. The StarterRelicHint below
   214→                // points the player at the Big Bang button (which grants the
   215→                // first relic without needing a prior simulation).
   216→                <>
   217→                  <EmptyState
   218→                    compact
   219→                    icon={<Atom className="w-6 h-6" />}
   220→                    title="No relics yet."
   221→                    description="Complete a universe simulation (Big Bang) to earn relics. Use the starter relic below to trigger your first Big Bang."
   222→                  />
   223→                  <div className="mt-2">
   224→                    <StarterRelicHint />
   225→                  </div>
   226→                </>
   227→              ) : (
   228→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 max-h-72 overflow-y-auto pr-1">
   229→                  {divineRelics.map((r) => {
   230→                    const inCrucible = crucibleRelicIds.includes(r.id);
   231→                    return (
   232→                      <div
   233→                        key={r.id}
   234→                        className={`stat-card p-2 ${inCrucible ? "border-amber-400/70 bg-amber-500/10" : "border-slate-600/40"}`}
   235→                      >
   236→                        <div className="flex items-center gap-1 mb-1">
   237→                          <Atom className="w-3 h-3 text-amber-300" />
   238→                          <span className="text-xs font-medium truncate flex-1">{r.name}</span>
   239→                          {inCrucible && (
   240→                            <Badge variant="outline" className="text-[0.55rem] text-amber-300 border-amber-400/60">in crucible</Badge>
   241→                          )}
   242→                        </div>
   243→                        <div className="space-y-1">
   244→                          {r.rules.map((rr, idx) => {
   245→                            const showRule = rr.revealed.rule;
   246→                            const showAmt = rr.revealed.amount;
   247→                            return (
   248→                              <div key={idx} className="flex items-center gap-1 text-[0.65rem]">
   249→                                <span className="w-4 text-center">{showRule ? ruleIcon(rr.ruleId) : "?"}</span>
   250→                                <span className={`flex-1 truncate ${showRule ? "text-foreground" : "text-muted-foreground/70"}`}>
   251→                                  {showRule ? ruleShortName(rr.ruleId) : "???"}
   252→                                </span>
   253→                                <span className={`font-mono ${showAmt ? (rr.amount >= 0 ? "text-emerald-300" : "text-rose-300") : "text-muted-foreground/70"}`}>
   254→                                  {showAmt ? amountLabel(rr.amount) : "???"}
   255→                                </span>
   256→                                <div className="flex items-center gap-0.5">
   257→                                  {!showRule && (
   258→                                    <button
   259→                                      onClick={() => revealRelicRule(r.id, idx)}
   260→                                      disabled={divinity < REVEAL_COST_RULE || universeSimulating}
   261→                                      className="text-[0.55rem] px-1 py-0.5 rounded border border-cyan-400/40 text-cyan-300 hover:bg-cyan-500/10 disabled:opacity-40 disabled:cursor-not-allowed"
   262→                                      title={`Reveal rule name (${REVEAL_COST_RULE} Divinity)`}
   263→                                    >
   264→                                      <Eye className="w-2.5 h-2.5" />
   265→                                    </button>
   266→                                  )}
   267→                                  {showRule && !showAmt && (
   268→                                    <button
   269→                                      onClick={() => revealRelicAmount(r.id, idx)}
   270→                                      disabled={divinity < REVEAL_COST_AMOUNT || universeSimulating}
   271→                                      className="text-[0.55rem] px-1 py-0.5 rounded border border-amber-400/40 text-amber-300 hover:bg-amber-500/10 disabled:opacity-40 disabled:cursor-not-allowed"
   272→                                      title={`Reveal amount (${REVEAL_COST_AMOUNT} Divinity)`}
   273→                                    >
   274→                                      <Eye className="w-2.5 h-2.5" />
   275→                                    </button>
   276→                                  )}
   277→                                  {showRule && showAmt && (
   278→                                    <CheckCircle2 className="w-2.5 h-2.5 text-emerald-400" />
   279→                                  )}
   280→                                </div>
   281→                              </div>
   282→                            );
   283→                          })}
   284→                        </div>
   285→                        {!inCrucible && (
   286→                          <Button
   287→                            size="sm"
   288→                            variant="outline"
   289→                            className="w-full h-6 mt-1.5 text-[0.6rem] text-amber-300 border-amber-400/40"
   290→                            disabled={crucibleRelicIds.length >= 5 || universeSimulating}
   291→                            onClick={() => addRelicToCrucible(r.id)}
   292→                          >
   293→                            <Plus className="w-3 h-3 mr-1" />
   294→                            Add to Crucible ({crucibleRelicIds.length}/5)
   295→                          </Button>
   296→                        )}
   297→                      </div>
   298→                    );
   299→                  })}
   300→                </div>
   301→              )}
   302→            </section>
   303→
   304→            {/* Crucible */}
   305→            <section>
   306→              <div className="text-[0.7rem] uppercase tracking-wide text-muted-foreground mb-1 flex items-center gap-1 justify-between">
   307→                <span className="flex items-center gap-1">
   308→                  <Atom className="w-3 h-3" /> The Crucible ({crucibleRelicIds.length}/5 relics)
   309→                </span>
   310→                <div className="flex items-center gap-1">
   311→                  <Button
   312→                    size="sm"
   313→                    variant="outline"
   314→                    className="h-5 text-[0.55rem] text-rose-300 border-rose-400/40"
   315→                    disabled={crucibleRelicIds.length === 0 || universeSimulating}
   316→                    onClick={() => clearCrucible()}
   317→                    title="Clear all relics from the crucible"
   318→                  >
   319→                    <RefreshCw className="w-2.5 h-2.5 mr-0.5" /> Clear
   320→                  </Button>
   321→                </div>
   322→              </div>
   323→              <div className="stat-card p-2">
   324→                {/* Crucible slots */}
   325→                <div className="grid grid-cols-5 gap-1 mb-2">
   326→                  {[0, 1, 2, 3, 4].map((slotIdx) => {
   327→                    const relic = crucibleRelics[slotIdx];
   328→                    return (
   329→                      <div
   330→                        key={slotIdx}
   331→                        className={`rounded border p-1 text-center text-[0.55rem] ${relic ? "border-amber-400/60 bg-amber-500/10 text-amber-200" : "border-slate-700/40 text-muted-foreground/60"}`}
   332→                      >
   333→                        {relic ? (
   334→                          <div className="truncate" title={relic.name}>
   335→                            <Atom className="w-2.5 h-2.5 inline mr-0.5" />
   336→                            {relic.name.replace("Relic of the ", "")}
   337→                          </div>
   338→                        ) : (
   339→                          <span>empty</span>
   340→                        )}
   341→                      </div>
   342→                    );
   343→                  })}
   344→                </div>
   345→                {/* Computed rule levels */}
   346→                <div className="space-y-1">
   347→                  {crucibleComp.rules.map((rv) => {
   348→                    const stable = rv.finalValue >= RULE_STABLE_MIN && rv.finalValue <= RULE_STABLE_MAX;
   349→                    const low = rv.finalValue < RULE_STABLE_MIN;
   350→                    const high = rv.finalValue > RULE_STABLE_MAX;
   351→                    return (
   352→                      <div key={rv.ruleId} className="flex items-center gap-2 text-[0.7rem]">
   353→                        <span className="w-5 text-center">{ruleIcon(rv.ruleId)}</span>
   354→                        <span className="w-28 truncate">{ruleShortName(rv.ruleId)}</span>
   355→                        {/* Displayed value: revealed portion + mystery marker */}
   356→                        <span className={`font-mono w-14 text-right ${stable ? "text-emerald-300" : low ? "text-rose-300" : high ? "text-amber-300" : "text-foreground"}`}>
   357→                          {rv.revealedValue}
   358→                          {rv.hasMystery ? "±?" : ""}
   359→                        </span>
   360→                        <span className="text-muted-foreground/60 text-[0.6rem]">→</span>
   361→                        {/* Final value (after overflow) */}
   362→                        <span className={`font-mono w-8 text-right font-bold ${stable ? "text-emerald-300" : low ? "text-rose-300" : high ? "text-amber-300" : "text-foreground"}`}>
   363→                          {rv.finalValue}
   364→                        </span>
   365→                        {(low || high) && (
   366→                          <AlertTriangle className={`w-3 h-3 ${low ? "text-rose-400" : "text-amber-400"}`} />
   367→                        )}
   368→                      </div>
   369→                    );
   370→                  })}
   371→                </div>
   372→                <div className="flex items-center justify-between mt-2 pt-2 border-t border-slate-700/40 text-[0.65rem]">
   373→                  <span className="text-muted-foreground">
   374→                    Total: <span className={`font-mono font-bold ${crucibleComp.total === RULE_TOTAL_TARGET ? "text-emerald-300" : "text-rose-300"}`}>{crucibleComp.total}</span> / {RULE_TOTAL_TARGET}
   375→                  </span>
   376→                  <span className="text-muted-foreground">
   377→                    Base per rule: {RULE_BASE} · Stability: <span className={crucibleComp.unstable ? "text-rose-300" : "text-emerald-300"}>
   378→                      {crucibleComp.unstable ? `UNSTABLE (instability ${crucibleComp.instability})` : "stable"}
   379→                    </span>
   380→                  </span>
   381→                </div>
   382→              </div>
   383→              {/* Big Bang button */}
   384→              <div className="mt-2">
   385→                {universeSimulating ? (
   386→                  // P7 — Loading state: skeleton screen during the 30-second
   387→                  // Big Bang simulation. Show pulsing placeholder relic slots
   388→                  // (instead of just a progress bar) + a "Simulating universe..."
   389→                  // status line. Keep the seconds-remaining indicator so the
   390→                  // player knows roughly how long is left.
   391→                  <div className="stat-card p-3 space-y-2">
   392→                    <div className="flex items-center justify-between text-[0.7rem]">
   393→                      <span className="text-amber-200 flex items-center gap-1 animate-pulse">
   394→                        <Sparkles className="w-3 h-3" /> Simulating universe…
   395→                      </span>
   396→                      <span className="font-mono text-amber-300">{simRemainingSec.toFixed(0)}s</span>
   397→                    </div>
   398→                    {/* Pulsing placeholder cards — match the relic grid layout
   399→                        so the layout doesn't shift when the result returns. */}
   400→                    <div className="grid grid-cols-2 sm:grid-cols-4 gap-1.5">
   401→                      {Array.from({ length: 4 }).map((_, i) => (
   402→                        <div
   403→                          key={i}
   404→                          className="h-12 rounded-md border border-amber-400/20 bg-amber-500/5 animate-pulse"
   405→                          style={{ animationDelay: `${i * 150}ms` }}
   406→                          aria-hidden
   407→                        />
   408→                      ))}
   409→                    </div>
   410→                    <div className="text-[0.6rem] text-muted-foreground text-center">
   411→                      The crucible blazes — reality crystallizes…
   412→                    </div>
   413→                  </div>
   414→                ) : (
   415→                  <Button
   416→                    className="w-full h-9 bg-gradient-to-r from-amber-600 via-rose-600 to-violet-600 hover:from-amber-500 hover:via-rose-500 hover:to-violet-500 text-white font-bold"
   417→                    disabled={crucibleRelicIds.length < 1}
   418→                    onClick={() => triggerBigBang()}
   419→                  >
   420→                    <Sparkles className="w-4 h-4 mr-2" />
   421→                    Trigger Big Bang {crucibleRelicIds.length < 1 ? "(add a relic first)" : `(${crucibleRelicIds.length} relic${crucibleRelicIds.length !== 1 ? "s" : ""})`}
   422→                  </Button>
   423→                )}
   424→              </div>
   425→            </section>
   426→
   427→            {/* Last simulation result */}
   428→            {lastSimResult && (
   429→              <section>
   430→                <div className="text-[0.7rem] uppercase tracking-wide text-muted-foreground mb-1">
   431→                  Last Big Bang Result
   432→                </div>
   433→                <div className={`stat-card p-2 ${lastSimResult.stable ? "border-emerald-400/60 bg-emerald-500/5" : "border-rose-400/60 bg-rose-500/5"}`}>
   434→                  {lastSimResult.stable ? (
   435→                    <div className="text-[0.7rem]">
   436→                      <div className="flex items-center gap-1 text-emerald-300 font-semibold">
   437→                        <CheckCircle2 className="w-3 h-3" /> STABLE
   438→                      </div>
   439→                      <div className="text-muted-foreground mt-1">
   440→                        {lastSimResult.archetypeId ? (
   441→                          <>Discovered archetype: <span className="text-amber-300 font-semibold">{ARCHETYPE_MAP[lastSimResult.archetypeId]?.name || "Unknown"}</span></>
   442→                        ) : (
   443→                          <>No archetype matched this combination.</>
   444→                        )}
   445→                      </div>
   446→                      <div className="text-muted-foreground mt-0.5">
   447→                        +{lastSimResult.relicsGained} new relic(s).
   448→                      </div>
   449→                    </div>
   450→                  ) : (
   451→                    <div className="text-[0.7rem]">
   452→                      <div className="flex items-center gap-1 text-rose-300 font-semibold">
   453→                        <AlertTriangle className="w-3 h-3" /> UNSTABLE
   454→                      </div>
   455→                      <div className="text-muted-foreground mt-1">
   456→                        Instability level: {lastSimResult.instability}. Lost {lastSimResult.relicsLost} relic(s) from inventory.
   457→                      </div>
   458→                    </div>
   459→                  )}
   460→                </div>
   461→              </section>
   462→            )}
   463→
   464→            {/* Archetype Collection */}
   465→            <section>
   466→              <div className="text-[0.7rem] uppercase tracking-wide text-muted-foreground mb-1 flex items-center justify-between">
   467→                <span className="flex items-center gap-1">
   468→                  <Star className="w-3 h-3" /> Archetype Collection ({discoveredCount}/{ARCHETYPE_COUNT})
   469→                </span>
   470→                <span className="text-[0.6rem] normal-case">
   471→                  {allDiscovered ? "All discovered — ending unlocked!" : `${ARCHETYPE_COUNT - discoveredCount} remaining`}
   472→                </span>
   473→              </div>
   474→              <Progress value={(discoveredCount / ARCHETYPE_COUNT) * 100} className="h-2 mb-2 bg-slate-800/60" />
   475→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-1 max-h-48 overflow-y-auto pr-1">
   476→                {ARCHETYPES.map((a) => {
   477→                  const found = !!discoveredArchetypes[a.id];
   478→                  return (
   479→                    <div
   480→                      key={a.id}
   481→                      className={`stat-card p-1.5 text-[0.65rem] ${found ? "border-amber-400/60 bg-amber-500/10" : "border-slate-700/40 opacity-80"}`}
   482→                    >
   483→                      <div className="flex items-center gap-1">
   484→                        {found ? (
   485→                          <CheckCircle2 className="w-2.5 h-2.5 text-amber-300" />

   487→                          <Lock className="w-2.5 h-2.5 text-muted-foreground/60" />
   488→                        )}
   489→                        <span className={`font-medium truncate flex-1 ${found ? "text-amber-200" : "text-muted-foreground"}`}>
   490→                          {found ? a.name : "???"}
   491→                        </span>
   492→                        {found && (
   493→                          <Badge variant="outline" className="text-[0.5rem] text-emerald-300 border-emerald-400/40">+1%</Badge>
   494→                        )}
   495→                      </div>
   496→                      <div className="text-[0.55rem] text-muted-foreground/80 mt-0.5 leading-tight">
   497→                        {found ? `Rules: ${a.rules.join("/")}` : a.hint}
   498→                      </div>
   499→                    </div>
   500→                  );
   501→                })}
   502→              </div>
   503→            </section>
   504→
   505→            {/* Bonus summary */}
   506→            <section>
   507→              <div className="stat-card p-2 text-[0.7rem]">
   508→                <div className="text-[0.6rem] uppercase tracking-wide text-muted-foreground mb-1">Current bonuses</div>
   509→                <div className="flex flex-wrap gap-1">
   510→                  <Badge variant="outline" className="text-emerald-300 border-emerald-400/40">
   511→                    +{(discoveredCount * 1).toFixed(0)}% production (archetypes)
   512→                  </Badge>
   513→                  {cosmicBoonStacks > 0 && (
   514→                    <Badge variant="outline" className="text-indigo-300 border-indigo-400/40">
   515→                      +{(boon * 100).toFixed(0)}% production (cosmic boons)
   516→                    </Badge>
   517→                  )}
   518→                </div>
   519→              </div>
   520→            </section>
   521→          </div>
   522→        )}
   523→
   524→        <Separator className="my-2" />
   525→        <div className="text-[0.65rem] text-muted-foreground px-1 flex items-center gap-1">
   526→          <RefreshCw className="w-3 h-3" />
   527→          Discover all {ARCHETYPE_COUNT} archetypes to unlock the ending screen. Each archetype grants +1% production.
   528→        </div>
   529→        <div className="flex justify-end pt-2">
   530→          <Button variant="outline" onClick={() => closeLayerModal()}>Close</Button>
   531→        </div>
   532→
   533→        {/* Ending-screen confirmation dialog */}
   534→        <AlertDialog open={pendingEnding !== null} onOpenChange={(open) => { if (!open) setPendingEnding(null); }}>
   535→          <AlertDialogContent>
   536→            <AlertDialogHeader>
   537→              <AlertDialogTitle>
   538→                {pendingEnding === "preserve" ? "Continue in this Realm?" : "Explore another Dimension?"}
   539→              </AlertDialogTitle>
   540→              <AlertDialogDescription>
   541→                {pendingEnding === "preserve"
   542→                  ? "Continue playing with all bonuses intact. The Genesis Engine remains open for further experiments."
   543→                  : "Reset the universe and begin a new dimension. You keep all discovered archetypes and gain a permanent Cosmic Boon stack (+10% all production). This cannot be undone."}
   544→              </AlertDialogDescription>
   545→            </AlertDialogHeader>
   546→            <AlertDialogFooter>
   547→              <AlertDialogCancel>Cancel</AlertDialogCancel>
   548→              <AlertDialogAction
   549→                onClick={confirmEnding}
   550→                className={pendingEnding === "reset" ? "bg-rose-600 hover:bg-rose-700 text-white" : ""}
   551→              >
   552→                {pendingEnding === "preserve" ? "Continue" : "Reset"}
   553→              </AlertDialogAction>
   554→            </AlertDialogFooter>
   555→          </AlertDialogContent>
   556→        </AlertDialog>
   557→    </>
   558→  );
   559→}
