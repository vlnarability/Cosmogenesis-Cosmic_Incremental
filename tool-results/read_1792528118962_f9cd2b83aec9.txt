    85→const SKY_W = 5000;
    86→const SKY_H = 500;

    88→// Full-screen overlay uses an even wider sky so panning feels roomier.
    89→const EXPANDED_SKY_W = 8000;
    90→// EXPANDED_SKY_H is purely documentary — the expanded viewport renders
    91→// height: 100% of its flex-1 parent. The constant documents the conceptual
    92→// height that vertical percentage coordinates are calibrated against.
    93→const EXPANDED_SKY_H = 600;

    95→// Per-instance offsets for the infinite-wrap 3-copy trick.
    96→// We render the sky at three x-offsets: -W (left), 0 (center), +W (right).
    97→// The wrapper translates them all by `scrollX`. As the user scrolls, when
    98→// |scrollX| exceeds W, we wrap by adding/subtracting W so the visible area
    99→// always shows at least one fully-populated instance. The wrap is invisible
   100→// because all 3 instances render identical content.
   101→const SKY_OFFSETS: number[] = [-SKY_W, 0, SKY_W];
   102→const EXPANDED_SKY_OFFSETS: number[] = [-EXPANDED_SKY_W, 0, EXPANDED_SKY_W];
   103→
   104→// FIX-UI-TABS / react-hooks/set-state-in-effect — module-level helper for the
   105→// infinite-wrap. Moved out of useEffect (which linted as a cascading-render
   106→// trigger) and applied at each setScrollX() call site.
   107→// CONSTELLATION-REDESIGN-NOIRLAB: parameterized over `w` so it works for both
   108→// the standard (SKY_W) and expanded (EXPANDED_SKY_W) skies.
   109→function wrapScrollX(x: number, w: number = SKY_W): number {
   110→  if (x < -w) return x + w;
   111→  if (x > w) return x - w;
   112→  return x;
   113→}

   177→// Per-constellation label anchor (top-right of the constellation's bounding
   178→// box). Precomputed so we don't run Math.min/max on every render.
   179→interface LabelAnchor { cId: string; left: number; top: number; }
   180→const LABEL_ANCHORS: LabelAnchor[] = CONSTELLATIONS.map((c) => {
   181→  const xs = c.stars.map((s) => s.x);
   182→  const ys = c.stars.map((s) => s.y);
   183→  const maxX = Math.max(...xs);
   184→  const minY = Math.min(...ys);
   185→  return { cId: c.id, left: maxX, top: minY };
   186→});

   188→// Greek-letter → readable glyph, for star designations in the popup.
   189→const GREEK: Record<string, string> = {
   190→  alpha: "α", beta: "β", gamma: "γ", delta: "δ", epsilon: "ε", eps: "ε",
   191→  zeta: "ζ", eta: "η", theta: "θ", iota: "ι", kappa: "κ", lambda: "λ",
   192→  mu: "μ", nu: "ν", xi: "ξ", omicron: "ο", pi: "π", rho: "ρ", sigma: "σ",
   193→  tau: "τ", upsilon: "υ", phi: "φ", chi: "χ", psi: "ψ", omega: "ω",
   194→};
   195→
   196→// Convert a star id like "aries_alpha" or "taurus_eps" or "orion_pi3" into a
   197→// readable designation like "α Arietis". Falls back to the raw id if parsing
   198→// fails (no constellation ever uses an unknown suffix pattern, but be safe).
   199→function starDesignation(starId: string, constellationName: string): string {
   200→  const parts = starId.split("_");
   201→  if (parts.length < 2) return starId;
   202→  const suffix = parts.slice(1).join("_");
   203→  const m = suffix.match(/^([a-z]+)(\d*)$/);
   204→  if (m) {
   205→    const greek = GREEK[m[1]];
   206→    if (greek) return `${greek}${m[2]} ${constellationName}`;
   207→  }
   208→  return `${suffix} ${constellationName}`;
   209→}

   211→// Format the constellation's completion bonus as a list of pill badges.
   212→function bonusSummary(bonus: ConstellationDef["bonus"]): { label: string; value: string; cls: string }[] {
   213→  const out: { label: string; value: string; cls: string }[] = [];
   214→  if (bonus.productionMult) out.push({ label: "Production", value: `+${(bonus.productionMult * 100).toFixed(1)}%`, cls: "text-cyan-300 border-cyan-400/40" });
   215→  if (bonus.capMult) out.push({ label: "Capacity", value: `+${(bonus.capMult * 100).toFixed(1)}%`, cls: "text-amber-300 border-amber-400/40" });
   216→  if (bonus.epMult) out.push({ label: "EP", value: `+${(bonus.epMult * 100).toFixed(1)}%`, cls: "text-emerald-300 border-emerald-400/40" });
   217→  if (bonus.popGrowthMult) out.push({ label: "Pop growth", value: `+${(bonus.popGrowthMult * 100).toFixed(1)}%`, cls: "text-violet-300 border-violet-400/40" });
   218→  if (bonus.costMult) out.push({ label: "Cost", value: `${(bonus.costMult * 100).toFixed(1)}%`, cls: "text-pink-300 border-pink-400/40" });
   219→  if (bonus.divinityMult) out.push({ label: "Divinity", value: `+${(bonus.divinityMult * 100).toFixed(0)}%`, cls: "text-yellow-300 border-yellow-400/40" });
   220→  return out;
   221→}

   398→  // ===== Render a single instance of the wide sky at the given x offset =====

   401→  // CONSTELLATION-REDESIGN-NOIRLAB: parameterized over `skyW` so the same

   426→            each copy independently — avoids edge seams between instances). */}
   427→        <div
   428→          className="absolute inset-0 opacity-60 pointer-events-none"
   429→          style={{
   430→            backgroundImage: `
   431→              radial-gradient(1px 1px at 12% 18%, #fff, transparent),
   432→              radial-gradient(1px 1px at 28% 42%, #fff, transparent),
   433→              radial-gradient(1.5px 1.5px at 47% 12%, #fff, transparent),

   437→              radial-gradient(1px 1px at 8% 72%, #fff, transparent),
   438→              radial-gradient(1px 1px at 33% 88%, #fff, transparent),
   439→              radial-gradient(1px 1px at 70% 85%, #fff, transparent),
   440→              radial-gradient(1px 1px at 95% 8%, #fff, transparent),
   441→              radial-gradient(1px 1px at 18% 50%, #fff, transparent),
   442→              radial-gradient(1px 1px at 53% 75%, #fff, transparent)
   443→            `,
   444→            backgroundRepeat: "no-repeat",
   445→            backgroundSize: "100% 100%",
   446→          }}
   447→        />
   448→        {/* CONSTELLATION-REDESIGN-NOIRLAB — NOIRLab outline overlay.
   449→            For each constellation that is COMPLETED (all stars lit), render
   450→            the NOIRLab outline SVG as a subtle (opacity 0.25) overlay
   451→            positioned over the constellation's bounding box. The overlay is
   452→            at zIndex 1 so it sits BEHIND the stars (which are zIndex 10+) but
   453→            ABOVE the SVG connection lines (auto z-index = 0). object-fit:
   454→            contain stretches the 360×360 viewBox SVG into the constellation's
   455→            bounding box while preserving aspect ratio. */}
   456→        {LABEL_ANCHORS.filter((la) => constellationsCompleted[la.cId]).map((la) => {
   457→          const cdef = CONSTELLATION_MAP[la.cId];
   458→          if (!cdef) return null;
   459→          // Compute the constellation's bounding box from its star coords.
