   306→export const useGameStore = create<GameStore>()(
   307→  persist(
   308→    (set, get) => ({
   309→      ...initialRunState(),
   310→      ...initialMetaState(),
   311→
   312→      // ============ TICK ============
   313→      tick: (dt) => {
   314→        const s = get();
   315→        if (s.paused) return;
   316→
   317→        // ---- Temporal Acceleration: +0.5× effective speed per level ----
   318→        const temporalLvl = s.upgrades["temporal_acceleration"] || 0;
   319→        const temporalMult = 1 + temporalLvl * 0.5;
   320→        // ---- Prestige Speed Bonus: +50% per total run (Part 2c) ----
   321→        const prestigeSpeedMult = 1 + (s.totalRuns || 0) * 0.5;
   322→        // ---- QUICK WIN 5 — Universal Speed upgrade: +10% per level ----
   323→        const universalSpeedLvlTick = s.upgrades["universal_speed"] || 0;
   324→        const universalSpeedMultTick = 1 + universalSpeedLvlTick * 0.10;
   325→        const realDt = dt * s.speed * temporalMult * prestigeSpeedMult * universalSpeedMultTick;
   326→
   327→        // ---- Deep Memory: +5% production per level per total run ----
   328→        const deepMemLvl = s.upgrades["deep_memory"] || 0;
   329→        const deepMemMult = 1 + deepMemLvl * 0.05 * (s.totalRuns || 0);
   330→
   331→        const autoLvl = s.upgrades["inherited_efficiency"] || 0;
   332→        let autoMult = (1 + autoLvl * 0.15) * deepMemMult;
   333→
   334→        // ---- Active challenge debuff (scaled by repeat level) ----
   335→        let challengeDebuff: ChallengeEffects = {};
   336→        if (s.activeChallenge && s.unlockedChallenges) {
   337→          const ch = CHALLENGE_MAP[s.activeChallenge];
   338→          if (ch) {
   339→            const repeat = s.challengeRepeatCounts?.[s.activeChallenge] || 0;
   340→            challengeDebuff = getChallengeDebuffAtRepeat(ch, repeat);
   341→            if (challengeDebuff.productionMult !== undefined) {
   342→              autoMult *= challengeDebuff.productionMult;
   343→            }
   344→          }
   345→        }
   346→
   347→        // ---- Layers 2-6: aggregate additive bonuses (production, cap, pop, ep, cost) ----
   348→        const foresight = foresightBonus(s.foresightNodes || {});
   349→        const route = routeBonus(s.activeForesightRoute);
   350→        const ritualProd = activeRitualProductionBonus(s.purchasedRituals || {}, s.activeTemporaryRituals || {});
   351→        const ritualCap = activeRitualCapBonus(s.purchasedRituals || {}, s.activeTemporaryRituals || {});
   352→        const ritualPop = activeRitualPopBonus(s.purchasedRituals || {}, s.activeTemporaryRituals || {});
   353→        const wProd = worldProductionMult(s.authoredWorlds || []);
   354→        const wCap = worldCapMult(s.authoredWorlds || []);
   355→        const wPop = worldPopGrowthMult(s.authoredWorlds || []);
   356→        const dLaw = divineLawBonus(s.enactedDivineLaws || {});
   357→        const wMode = worshipModeBonus(s.activeWorshipMode);
   358→        const hResp = heresyResponseProductionMult(s.activeHeresyResponse);
   359→        const relics = relicBonus(s.equippedRelics || {});
   360→        const cores = logicCoreBonus(s.activeLogicCores || {});
   361→        // Layer 7 — Omnipotence (hybrids + stance)
   362→        const hybrids = hybridBonus(s.equippedHybrids || {}, s.activeOmnipotenceStance);
   363→        // Layer 8 — Divinity (prayer channels + mask + polarity)
   364→        const chanBonus = prayerChannelBonus(s.prayerChannelLevels || {});
   365→        const maskBonus = divineMaskBonus(s.activeDivineMask);
   366→        const polarityBonus = worshipPolarityBonus(s.activeWorshipPolarity);
   367→        // Layer 9 — Infinity (echoes + forks + active debt)
   368→        const echoB = echoBonus(s.purchasedEchoes || {});
   369→        const forkB = forkBonus(s.resolvedForks || {});
   370→        const debtB = activeDebtBonus(s.takenFutureDebts || {}, s.repaidFutureDebts || {});
   371→        // Layer 10 — Eternity (testament + canonizations + weaves + cosmic boon)
   372→        const testB = testamentClauseBonus(s.purchasedTestamentClauses || {});
   373→        const canonB = canonizationBonus(s.activeCanonizations || {});
   374→        const weaveB = permanenceWeaveBonus(s.activePermanenceWeaves || {});
   375→        const cosmicBoon = cosmicBoonBonus(s.cosmicBoonStacks || 0);
   376→
   377→        const layerProdBonus =
   378→          foresight.productionMult + route.productionMult + ritualProd +
   379→          wProd + dLaw.productionMult + wMode.productionMult + hResp +
   380→          relics.productionMult + cores.productionMult +
   381→          hybrids.productionMult + chanBonus.productionMult + maskBonus.productionMult +
   382→          polarityBonus.productionMult + echoB.productionMult + forkB.productionMult +
   383→          debtB.productionMult + testB.productionMult + canonB.productionMult +
   384→          weaveB.productionMult + cosmicBoon;
   385→        const layerCapBonus =
   386→          foresight.capMult + route.capMult + ritualCap + wCap +
   387→          dLaw.capMult + relics.capMult + cores.capMult +
   388→          hybrids.capMult + chanBonus.capMult + maskBonus.capMult +
   389→          polarityBonus.capMult + echoB.capMult + forkB.capMult +
   390→          debtB.capMult + testB.capMult + canonB.capMult + weaveB.capMult;
   391→        const layerPopBonus =
   392→          foresight.popGrowthMult + route.popGrowthMult + ritualPop + wPop +
   393→          dLaw.popGrowthMult + relics.popGrowthMult +
   394→          hybrids.popGrowthMult + chanBonus.popGrowthMult + maskBonus.popGrowthMult +
   395→          polarityBonus.popGrowthMult + echoB.popGrowthMult + forkB.popGrowthMult +
   396→          debtB.popGrowthMult + testB.popGrowthMult + canonB.popGrowthMult +
   397→          weaveB.popGrowthMult;
   398→
   399→        // QUICK WIN 2 — Active abilities: Surge (×3 prod) + Overclock (×2 layer bonuses)
   400→        const surgeActive = (s.activeAbilityEffects?.surge || 0) > 0;
   401→        const overclockActive = (s.activeAbilityEffects?.overclock || 0) > 0;
   402→        // QUICK WIN 3 — Divine Combo: +50% all production while ritualComboTimer > 0
   403→        const divineComboActive = (s.ritualComboTimer || 0) > 0;
   404→        // QUICK WIN 5 — Universal Boost upgrade: +5% all production per level
   405→        const universalBoostLvl = s.upgrades["universal_boost"] || 0;
   406→        const universalBoostMult = 1 + universalBoostLvl * 0.05;
   407→
   408→        // Miracle: double production (one-shot temporary effect)
   409→        const miracleDouble = (s.miracleTimers?.mir_genesis || 0) > 0;
   410→
   411→        // Apply: overclock multiplies layer bonuses (foresight, ritual, relics, etc.) by 2
   412→        const layerProdBonusApplied = layerProdBonus * (overclockActive ? 2 : 1);
   413→        // Surge multiplies total production output (after layer bonus & miracle)
   414→        const surgeMult = surgeActive ? 3 : 1;
   415→        const divineComboMult = divineComboActive ? 1.5 : 1;
   416→        autoMult *= (1 + layerProdBonusApplied) * (miracleDouble ? 2 : 1) * surgeMult * divineComboMult * universalBoostMult;
   417→
   418→        const techMult = techMultiplier(s, "production");
   419→        const capBoostLvl = s.upgrades["expansive_vaults"] || 0;
   420→        let capMult = 1 + capBoostLvl * 0.20;
   421→        capMult *= (1 + layerCapBonus);
   422→        if (challengeDebuff.capMult !== undefined) capMult *= challengeDebuff.capMult;
   423→        const hasAutoBalancer = (s.upgrades["auto_balancer"] || 0) > 0;
   424→
   425→        // Archetype bonus
   426→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   427→        const archAllMult = archetype?.bonus.allProductionMult || 0;
   428→        const archPopMult = archetype?.bonus.populationGrowthMult || 0;
   429→        const archCapMult = archetype?.bonus.capMult || 0;
   430→
   431→        const resources = { ...s.resources };
   432→        let capacities = recomputeCapacitiesFresh(s, capMult + archCapMult);
   433→
   434→        // Cost reduction (frugality) — also used by auto-buyer
   435→        // Layers 2-6: foresight cost reduction (additive, negative = cheaper)
   436→        const layerCostReduction = foresight.costMult;
   437→        const costReduction = (s.upgrades["frugality"] || 0) * 0.04 + Math.max(0, -layerCostReduction);
   438→        // Challenge cost multiplier (debuff)
   439→        const challengeCostMult = challengeDebuff.costMult ?? 1;
   440→
   441→        // Apply system production — with cap-aware pausing
   442→        for (const [sysId, count] of Object.entries(s.ownedSystems)) {
   443→          if (!count) continue;
   444→          const sys = SYSTEM_MAP[sysId];
   445→          if (!sys) continue;
   446→          if (sys.requiredTech && !s.technologies[sys.requiredTech]) continue;
   447→
   448→          // Check if system is manually disabled
   449→          const manuallyEnabled = s.systemEnabled[sysId] !== false; // default true
   450→          if (!manuallyEnabled && !hasAutoBalancer) continue;
   451→
   452→          // Cap-aware logic: if all output resources are at cap, idle the system
   453→          // (don't produce, don't consume upkeep — no waste)
   454→          const outputsCapped = Object.keys(sys.produces).length > 0 &&
   455→            Object.keys(sys.produces).every((r) => {
   456→              const cap = capacities[r] || 0;
   457→              return (resources[r] || 0) >= cap - 0.01;
   458→            });
   459→
   460→          if (outputsCapped) {
   461→            // System idles — no production, no upkeep
   462→            continue;
   463→          }
   464→
   465→          const stageMult = sys.stage === STAGES[s.stageIndex].id ? 1 : 0.5;
   466→          const mult = autoMult * techMult * stageMult * (1 + archAllMult);
   467→
   468→          // Per-resource archetype bonus
   469→          const produces = sys.produces as Record<string, number>;
   470→          for (const [r, v] of Object.entries(produces)) {
   471→            const archResBonus = archetype?.bonus.productionMult?.[r] || 0;
   472→            const rate = v * count * mult * (1 + archResBonus);
   473→            resources[r] = (resources[r] || 0) + rate * realDt;
   474→          }
   475→          // Upkeep
   476→          if (sys.upkeep) {
   477→            const upkeep = sys.upkeep as Record<string, number>;
   478→            for (const [r, v] of Object.entries(upkeep)) {
   479→              resources[r] = (resources[r] || 0) - v * count * realDt;
   480→            }
   481→          }
   482→        }
   483→
   484→        // Clamp resources to capacities (no overflow)
   485→        for (const r of Object.keys(resources)) {
   486→          const cap = capacities[r];
   487→          if (cap != null && resources[r] > cap) resources[r] = cap;
   488→          if (resources[r] < 0) resources[r] = 0;
   489→        }
   490→
   491→        // Slow passive ATP for cell (keeps early game from stalling)
   492→        if (STAGES[s.stageIndex].id === "cell") {
   493→          resources.atp = Math.min(capacities.atp || 30, resources.atp + 0.5 * realDt);
   494→        }
   495→
   496→        // Population growth with progress bar
   497→        // QUICK WIN 1: 3× faster base rate (0.015 → 0.045) — waiting for pop is the #1 fun killer
   498→        let popGrowthRate = 0.045; // base per second
   499→        const happiness = resources.happiness || 50;
   500→        popGrowthRate *= (1 + (happiness - 40) / 200);
   501→        popGrowthRate *= (1 + archPopMult);
   502→        // Layers 2-6: aggregate pop-growth bonus
   503→        popGrowthRate *= (1 + layerPopBonus);
   504→        const curStage = STAGES[s.stageIndex];
   505→        if (["creature", "tribal", "civilization", "empire"].includes(curStage.id)) {
   506→          const foodCap = capacities.food || 1;
   507→          const waterCap = capacities.water || 1;
   508→          const foodPct = (resources.food || 0) / foodCap;
   509→          const waterPct = (resources.water || 0) / waterCap;
   510→          if (foodPct < 0.1) popGrowthRate *= 0.3;
   511→          if (waterPct < 0.1) popGrowthRate *= 0.3;
   512→        }
   513→        // Challenge pop-growth debuff
