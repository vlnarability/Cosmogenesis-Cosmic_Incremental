    26→function initialRunState(): Partial<GameState> {
    27→  const resources = emptyResources();
    28→  const capacities = emptyCapacities();
    29→  resources.atp = 5;
    30→  resources.glucose = 8;
    31→  resources.happiness = 75;
    32→  return {
    33→    stageIndex: 0,
    34→    time: 0,
    35→    population: 8,
    36→    populationProgress: 0,
    37→    maxPopulation: 8,
    38→    resources,
    39→    capacities,
    40→    ownedSystems: {},
    41→    systemEnabled: {},
    42→    technologies: {},
    43→    lockedTechs: {},
    44→    lockedSystems: {},
    45→    archetypeAffinity: {},
    46→    lockedArchetype: null,
    47→    log: ["New run started. The first spark stirs."],
    48→    activeChallenge: null,
    49→    challengeRunStart: null,
    50→  };
    51→}

    91→    activeEvent: null,
    92→    eventCooldown: 60, // first event can fire after 60s
    93→    eventHistory: [],
    94→    hasSeenIntro: false,
    95→    lastSaved: Date.now(),
    96→    tutorialStep: 0,

   102→
   103→export const useGameStore = create<GameStore>()(
   104→  persist(
   105→    (set, get) => ({
   106→      ...initialRunState(),
   107→      ...initialMetaState(),
   108→
   109→      // ============ TICK ============
   110→      tick: (dt) => {
   111→        const s = get();
   112→        if (s.paused) return;
   113→        const realDt = dt * s.speed;
   114→
   115→        const autoLvl = s.upgrades["inherited_efficiency"] || 0;
   116→        const autoMult = 1 + autoLvl * 0.15;
   117→        const techMult = techMultiplier(s, "production");
   118→        const capBoostLvl = s.upgrades["expansive_vaults"] || 0;
   119→        const capMult = 1 + capBoostLvl * 0.20;
   120→        const hasAutoBalancer = (s.upgrades["auto_balancer"] || 0) > 0;
   121→
   122→        // Archetype bonus
   123→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   124→        const archAllMult = archetype?.bonus.allProductionMult || 0;
   125→        const archPopMult = archetype?.bonus.populationGrowthMult || 0;
   126→        const archCapMult = archetype?.bonus.capMult || 0;
   127→
   128→        // ===== Enlightenment route bonus (Layer 2)
   129→        // The chosen route provides persistent bonuses that stack with archetype
   130→        // and challenge-mastery bonuses. Foresight passives apply on top.
   131→        const route = s.selectedRoute ? ROUTE_MAP[s.selectedRoute] : null;
   132→        const routeAllMult = route?.bonus.allProductionMult || 0;
   133→        const routePopMult = route?.bonus.populationGrowthMult || 0;
   134→        const foresightAllMult = foresightBonus(s.foresightNodes || {}, "production");
   135→        const foresightPopMult = foresightBonus(s.foresightNodes || {}, "popGrowth");
   136→
   137→        // ===== Challenge debuff (applied for the duration of an active challenge run)
   138→        const activeChallenge = s.activeChallenge ? CHALLENGE_MAP[s.activeChallenge] : null;
   139→        const challengeEffects = activeChallenge?.debuff.effects || {};
   140→        const chFoodMult = challengeEffects.foodMult ?? 1;
   141→        const chScienceMult = challengeEffects.scienceMult ?? 1;
   142→        const chProductionMult = challengeEffects.productionMult ?? 1;
   143→        const chMilitaryMult = challengeEffects.militaryMult ?? 1;
   144→        const chWaterUpkeepMult = challengeEffects.waterUpkeepMult ?? 1;
   145→        const chHappinessDrain = challengeEffects.happinessDrain ?? 0;
   146→        const chPopGrowthMult = challengeEffects.popGrowthMult ?? 1;
   147→
   148→        const resources = { ...s.resources };
   149→        let capacities = recomputeCapacitiesFresh(s, capMult + archCapMult);
   150→
   151→        // Apply system production — with cap-aware pausing
   152→        for (const [sysId, count] of Object.entries(s.ownedSystems)) {
   153→          if (!count) continue;
   154→          const sys = SYSTEM_MAP[sysId];
   155→          if (!sys) continue;
   156→          if (sys.requiredTech && !s.technologies[sys.requiredTech]) continue;
   157→
   158→          // Check if system is manually disabled
   159→          const manuallyEnabled = s.systemEnabled[sysId] !== false; // default true
   160→          if (!manuallyEnabled && !hasAutoBalancer) continue;
   161→
   162→          // Cap-aware logic: if all output resources are at cap, idle the system
   163→          // (don't produce, don't consume upkeep — no waste)
   164→          const outputsCapped = Object.keys(sys.produces).length > 0 &&
   165→            Object.keys(sys.produces).every((r) => {
   166→              const cap = capacities[r] || 0;
   167→              return (resources[r] || 0) >= cap - 0.01;
   168→            });
   169→
   170→          if (outputsCapped) {
   171→            // System idles — no production, no upkeep
   172→            continue;
   173→          }
   174→
   175→          const stageMult = sys.stage === STAGES[s.stageIndex].id ? 1 : 0.5;
   176→          const mult = autoMult * techMult * stageMult *
   177→            (1 + archAllMult + routeAllMult + foresightAllMult) * chProductionMult;
   178→
   179→          // Per-resource archetype + route + foresight bonus + challenge resource-specific debuff
   180→          const produces = sys.produces as Record<string, number>;
   181→          for (const [r, v] of Object.entries(produces)) {
   182→            const archResBonus = archetype?.bonus.productionMult?.[r] || 0;
   183→            const routeResBonus = route?.bonus.productionMult?.[r] || 0;
   184→            const foresightResBonus = foresightBonus(s.foresightNodes || {}, "resource", r);
   185→            // Apply per-resource challenge multipliers
   186→            let chResMult = 1;
   187→            if (r === "food") chResMult *= chFoodMult;
   188→            if (r === "science") chResMult *= chScienceMult;
   189→            if (r === "military_power") chResMult *= chMilitaryMult;
   190→            const rate = v * count * mult *
   191→              (1 + archResBonus + routeResBonus + foresightResBonus) * chResMult;
   192→            resources[r] = (resources[r] || 0) + rate * realDt;
   193→          }
   194→          // Upkeep
   195→          if (sys.upkeep) {
   196→            const upkeep = sys.upkeep as Record<string, number>;
   197→            for (const [r, v] of Object.entries(upkeep)) {
   198→              const upkMult = r === "water" ? chWaterUpkeepMult : 1;
   199→              resources[r] = (resources[r] || 0) - v * count * realDt * upkMult;
   200→            }
   201→          }
   202→        }
   203→
   204→        // Apply challenge happiness drain (flat per second)
   205→        if (chHappinessDrain > 0) {
   206→          const happCap = capacities.happiness || 100;
   207→          resources.happiness = Math.max(0, (resources.happiness || 0) - chHappinessDrain * realDt);
   208→          // also clamp to cap (already done below, but keep happ bounded)
   209→          if (resources.happiness > happCap) resources.happiness = happCap;
   210→        }
   211→
   212→        // Clamp resources to capacities (no overflow)
   213→        for (const r of Object.keys(resources)) {
   214→          const cap = capacities[r];
   215→          if (cap != null && resources[r] > cap) resources[r] = cap;
   216→          if (resources[r] < 0) resources[r] = 0;
   217→        }
   218→
   219→        // Slow passive ATP for cell (keeps early game from stalling)
   220→        if (STAGES[s.stageIndex].id === "cell") {
   221→          resources.atp = Math.min(capacities.atp || 30, resources.atp + 0.1 * realDt);
   222→        }
   223→
   224→        // Happiness drift toward 50 baseline if no source
   225→        // (prevents permanent high happiness from start)
   226→        // — disabled: let systems/tech maintain it
   227→
   228→        // Population growth with progress bar
   229→        // Growth rate scales with happiness and food/water availability
   230→        let popGrowthRate = 0.015; // base per second
   231→        const happiness = resources.happiness || 50;
   232→        popGrowthRate *= (1 + (happiness - 40) / 200); // happiness 100 → +30%, happiness 20 → -10%
   233→        popGrowthRate *= (1 + archPopMult + routePopMult + foresightPopMult);
   234→        // Apply challenge population-growth debuff
   235→        popGrowthRate *= chPopGrowthMult;
   236→        // Penalty if food/water are critically low (creature+)
   237→        const curStage = STAGES[s.stageIndex];
   238→        if (["creature", "tribal", "civilization", "empire"].includes(curStage.id)) {
   239→          const foodCap = capacities.food || 1;
   240→          const waterCap = capacities.water || 1;
   241→          const foodPct = (resources.food || 0) / foodCap;
   242→          const waterPct = (resources.water || 0) / waterCap;
   243→          if (foodPct < 0.1) popGrowthRate *= 0.3;
   244→          if (waterPct < 0.1) popGrowthRate *= 0.3;
   245→        }
   246→
   247→        let populationProgress = (s.populationProgress || 0) + popGrowthRate * realDt;
   248→        let population = s.population;
   249→        while (populationProgress >= 1) {
   250→          populationProgress -= 1;
   251→          population += 1;
   252→        }
   253→
   254→        const maxPopulation = Math.max(s.maxPopulation || 0, population);
   255→
   256→        const newTime = s.time + realDt;
   257→
   258→        // Check story triggers
   259→        const newStory = { ...s.storyUnlocked };
   260→        let activePopup = s.activeStoryPopup;
   261→        const ctx = { ...s, resources, ownedSystems: s.ownedSystems };
   262→        for (const trig of STORY_TRIGGERS) {
   263→          if (!newStory[trig.id] && trig.condition(ctx)) {
   264→            newStory[trig.id] = true;
   265→            if (!activePopup) activePopup = trig.id;
   266→          }
   267→        }
   268→
   269→        // Check achievements periodically (every ~2 seconds of game time)
   270→        let achievements = s.achievements;
   271→        let newAchievements = s.newAchievements;
   272→        const checkInterval = 2.0;
   273→        const lastCheck = (s as any)._lastAchCheck || 0;
   274→        if (newTime - lastCheck >= checkInterval) {
   275→          const curState = { ...s, resources, time: newTime } as GameState;
   276→          const result = checkAchievements(curState);
   277→          if (result.newOnes.length > 0) {
   278→            achievements = result.earned;
   279→            newAchievements = [...(newAchievements || []), ...result.newOnes];
   280→          } else if (Object.keys(result.earned).length !== Object.keys(achievements || {}).length) {
   281→            achievements = result.earned;
   282→          }
   283→          (s as any)._lastAchCheck = newTime; // not persisted, just a runtime marker
   284→        }
   285→
   286→        // Random event firing — only if no popup/modal is active
   287→        let activeEvent = s.activeEvent;
   288→        let eventCooldown = (s.eventCooldown || 0) - realDt;
   289→        if (eventCooldown < 0) eventCooldown = 0;
   290→        const eventFreq = s.eventFrequency || "normal";
   291→        if (eventFreq !== "off" && !activeEvent && !activePopup && eventCooldown === 0) {
   292→          const stageId = STAGES[s.stageIndex].id;
   293→          const possible = eventsForStage(stageId, newTime);
   294→          if (possible.length > 0) {
   295→            // Weighted random selection based on rarity
   296→            const totalWeight = possible.reduce((sum, e) => sum + eventWeight(e), 0);
   297→            const roll = Math.random() * totalWeight;
   298→            let cum = 0;
   299→            for (const ev of possible) {
   300→              cum += eventWeight(ev);
   301→              if (roll < cum) {
   302→                activeEvent = ev.id;
   303→                // Cooldown depends on frequency setting
   304→                const base = eventFreq === "frequent" ? 30 : 90;
   305→                const range = eventFreq === "frequent" ? 30 : 90;
   306→                eventCooldown = base + Math.random() * range;
   307→                break;
   308→              }
   309→            }
   310→          }
   311→        }
   312→
   313→        set({
   314→          resources,
   315→          capacities,
   316→          time: newTime,
   317→          population,
   318→          maxPopulation,
   319→          populationProgress,
   320→          storyUnlocked: newStory,
   321→          activeStoryPopup: activePopup,
   322→          achievements,
   323→          newAchievements,
   324→          activeEvent,
   325→          eventCooldown,
   326→          totalPlayTime: (s.totalPlayTime || 0) + realDt,
   327→        });
   328→
   329→        // Check if the active challenge's completion condition has been met.
   330→        // If so, mark it complete and clear the active challenge slot.
   331→        // (Done after the set so completion sees updated state.)
   332→        if (s.activeChallenge) {
   333→          get().checkChallengeCompletion();
   334→        }
   335→      },
   336→
   337→      // ============ ACTIONS ============
   338→      performAction: (actionId) => {
   339→        const s = get();
   340→        const action = ACTION_MAP[actionId];
   341→        if (!action) return;
   342→        if (action.stage !== STAGES[s.stageIndex].id) return;
   343→
   344→        const res = { ...s.resources };
   345→
   346→        // Check if all output resources are at cap — if so, block (no waste)
   347→        const outputs = Object.entries(action.produces);
   348→        if (outputs.length > 0) {
   349→          const allCapped = outputs.every(([r]) => {
   350→            const cap = s.capacities[r] || 0;
   351→            return (res[r] || 0) >= cap - 0.001;
   352→          });
   353→          if (allCapped) return; // action wasted — block it
   354→        }
   355→
   356→        // Check costs
   357→        if (action.cost) {
   358→          for (const [r, v] of Object.entries(action.cost)) {
   359→            if ((res[r] || 0) < (v as number)) return;
   360→          }
   361→          for (const [r, v] of Object.entries(action.cost)) {
   362→            res[r] -= v as number;
   363→          }
   364→        }
   365→
   366→        // Manual action multiplier
   367→        const manualLvl = s.upgrades["quickened_hands"] || 0;
   368→        const mult = 1 + manualLvl * 0.25;
   369→        const techMult = techMultiplier(s, "manual");
   370→        const finalMult = mult * techMult;
   371→
   372→        // Archetype + Enlightenment route per-resource bonus
   373→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   374→        const route = s.selectedRoute ? ROUTE_MAP[s.selectedRoute] : null;
   375→        const routeAllMult = route?.bonus.allProductionMult || 0;
   376→        const foresightAllMult = foresightBonus(s.foresightNodes || {}, "production");
   377→
   378→        for (const [r, v] of Object.entries(action.produces)) {
   379→          const cap = s.capacities[r] || 0;
   380→          const archResBonus = archetype?.bonus.productionMult?.[r] || 0;
   381→          const routeResBonus = route?.bonus.productionMult?.[r] || 0;
   382→          const foresightResBonus = foresightBonus(s.foresightNodes || {}, "resource", r);
   383→          const totalMult = 1 + archResBonus + routeResBonus + foresightResBonus +
   384→            (archetype?.bonus.allProductionMult || 0) + routeAllMult + foresightAllMult;
   385→          const amount = (v as number) * finalMult * totalMult;
   386→          res[r] = Math.min(cap, (res[r] || 0) + amount);
   387→        }
   388→
   389→        set({
   390→          resources: res,
   391→          totalActions: (s.totalActions || 0) + 1,
   392→        });
   393→      },
   394→
   395→      buySystem: (systemId, qty) => {
   396→        const s = get();
   397→        const sys = SYSTEM_MAP[systemId];
   398→        if (!sys) return;
   399→        if (sys.stage !== STAGES[s.stageIndex].id) return;
   400→        // Check tech prerequisite
   401→        if (sys.requiredTech && !s.technologies[sys.requiredTech]) return;
   402→        // Locked systems (via exclusiveWith variant choices) cannot be purchased.
   403→        if (s.lockedSystems && s.lockedSystems[systemId]) return;
   404→
   405→        const costReduction = (s.upgrades["frugality"] || 0) * 0.04;
   406→        let owned = s.ownedSystems[systemId] || 0;
   407→        if (sys.maxOwned && owned >= sys.maxOwned) return;
   408→
   409→        const res = { ...s.resources };
   410→        const ownedSystems = { ...s.ownedSystems };
   411→        const systemEnabled = { ...s.systemEnabled };
   412→        const archetypeAffinity = { ...s.archetypeAffinity };
   413→
   414→        let bought = 0;
   415→        for (let i = 0; i < qty; i++) {
   416→          if (sys.maxOwned && owned + bought >= sys.maxOwned) break;
   417→          const cost = systemCost(sys, owned + bought, costReduction);
   418→          let canAfford = true;
   419→          for (const [r, v] of Object.entries(cost)) {
   420→            if ((res[r] || 0) < (v as number)) {
   421→              canAfford = false;
   422→              break;
   423→            }
   424→          }
   425→          if (!canAfford) break;
   426→          for (const [r, v] of Object.entries(cost)) {
   427→            res[r] -= v as number;
   428→          }
   429→          bought++;
   430→        }
   431→
   432→        if (bought === 0) return;
   433→
   434→        ownedSystems[systemId] = owned + bought;
   435→        // New systems are enabled by default
   436→        if (systemEnabled[systemId] === undefined) systemEnabled[systemId] = true;
   437→
   438→        // Apply archetype affinity grant
   439→        if (sys.grantsAffinity) {
   440→          const arch = sys.grantsAffinity.archetype;
   441→          archetypeAffinity[arch] = (archetypeAffinity[arch] || 0) + sys.grantsAffinity.amount * bought;
   442→        }
   443→
   444→        // Trigger story
   445→        const newStory = { ...s.storyUnlocked };
   446→        let activePopup = s.activeStoryPopup;
   447→        for (const trig of STORY_TRIGGERS) {
   448→          if (!newStory[trig.id] && trig.condition({ ...s, resources: res, ownedSystems })) {
   449→            newStory[trig.id] = true;
   450→            if (!activePopup) activePopup = trig.id;
   451→          }
   452→        }
   453→
   454→        // Enforce mutually-exclusive variant choices: when buying the first
   455→        // instance of a system that is part of an exclusive group, permanently
   456→        // lock all sibling variants so they can never be purchased this run.
   457→        // (Only the FIRST purchase locks siblings — once chosen, the choice is
   458→        // final for this run, but the chosen system can still be bought up to
   459→        // its maxOwned.)
   460→        const lockedSystems = { ...(s.lockedSystems || {}) };
   461→        if (sys.exclusiveWith && sys.exclusiveWith.length > 0 && owned === 0 && bought > 0) {
   462→          for (const siblingId of sys.exclusiveWith) {
   463→            // Don't lock a sibling that is somehow already owned (defensive)
   464→            if (!(ownedSystems[siblingId] > 0)) {
   465→              lockedSystems[siblingId] = true;
   466→            }
   467→          }
   468→        }
   469→
   470→        set({
   471→          resources: res,
   472→          ownedSystems,
   473→          systemEnabled,
   474→          archetypeAffinity,
   475→          storyUnlocked: newStory,
   476→          activeStoryPopup: activePopup,
   477→          lockedSystems,
   478→          totalSystemsBuilt: (s.totalSystemsBuilt || 0) + bought,
   479→        });
   480→        get().addToLog(`Built ${bought}× ${sys.name}.`);
   481→      },
   482→
   483→      toggleSystem: (systemId) => {
   484→        const s = get();
   485→        if (!s.ownedSystems[systemId]) return;
   486→        const systemEnabled = { ...s.systemEnabled };
   487→        systemEnabled[systemId] = systemEnabled[systemId] === false ? true : false;
   488→        set({ systemEnabled });
   489→      },
   490→
   491→      enableAllSystems: (enabled) => {
   492→        const s = get();
   493→        const systemEnabled: Record<string, boolean> = {};
   494→        for (const id of Object.keys(s.ownedSystems)) {
   495→          systemEnabled[id] = enabled;
   496→        }
   497→        set({ systemEnabled });
   498→      },
   499→
   500→      buyTech: (techId) => {
   501→        const s = get();
   502→        const tech = TECH_MAP[techId];
   503→        if (!tech) return;
   504→        if (s.technologies[techId]) return;
   505→        // Locked techs cannot be researched (a mutually-exclusive sibling was already chosen)
   506→        if (s.lockedTechs && s.lockedTechs[techId]) return;
   507→        if (tech.stage !== STAGES[s.stageIndex].id) return;
   508→        if (tech.requires) {
   509→          for (const req of tech.requires) {
   510→            if (!s.technologies[req]) return;
   511→          }
   512→        }
   513→        const res = { ...s.resources };
   514→        for (const [r, v] of Object.entries(tech.cost)) {
   515→          if ((res[r] || 0) < (v as number)) return;
   516→        }
   517→        for (const [r, v] of Object.entries(tech.cost)) {
   518→          res[r] -= v as number;
   519→        }
   520→        const technologies = { ...s.technologies, [techId]: true };
   521→        const archetypeAffinity = { ...s.archetypeAffinity };
   522→        if (tech.grantsAffinity) {
   523→          archetypeAffinity[tech.grantsAffinity.archetype] =
   524→            (archetypeAffinity[tech.grantsAffinity.archetype] || 0) + tech.grantsAffinity.amount;
   525→        }
   526→        // Enforce mutually-exclusive tech path choices: lock all siblings
   527→        // listed in `exclusiveWith` so they can never be researched this run.
   528→        const lockedTechs = { ...(s.lockedTechs || {}) };
   529→        if (tech.exclusiveWith && tech.exclusiveWith.length > 0) {
   530→          for (const siblingId of tech.exclusiveWith) {
   531→            // Don't lock a sibling that is somehow already researched (defensive)
   532→            if (!technologies[siblingId]) {
   533→              lockedTechs[siblingId] = true;
   534→            }
   535→          }
   536→        }
   537→        set({
   538→          resources: res,
   539→          technologies,
   540→          lockedTechs,
   541→          archetypeAffinity,
   542→          totalTechResearched: (s.totalTechResearched || 0) + 1,
   543→        });
   544→        get().addToLog(`Researched ${tech.name}.`);
   545→      },
   546→
   547→      buyUpgrade: (upgradeId) => {
   548→        const s = get();
   549→        const up = UPGRADE_MAP[upgradeId];
   550→        if (!up) return;
   551→        if (up.requiresWins && s.galacticWins < up.requiresWins) return;
   552→        const owned = s.upgrades[upgradeId] || 0;
   553→        if (owned >= up.maxLevel) return;
   554→        const cost = upgradeCost(up, owned);
   555→        if (s.evolutionPoints < cost) return;
   556→        const upgrades = { ...s.upgrades, [upgradeId]: owned + 1 };
   557→        set({
   558→          evolutionPoints: s.evolutionPoints - cost,
   559→          upgrades,
   560→        });
   561→        get().addToLog(`Purchased ${up.name} (Lv ${owned + 1}).`);
   562→      },
   563→
   564→      evolveStage: () => {
   565→        const s = get();
   566→        const stage = STAGES[s.stageIndex];
   567→        const next = nextStage(stage.id);
   568→        if (!next) {
   569→          get().triggerPrestige();
   570→          return;
   571→        }
   572→        const reqs = stage.evolveRequires;
   573→        const evolveBoost = 1 - Math.min(0.30, (s.upgrades["evolutionary_momentum"] || 0) * 0.05);
   574→        const totalSystems = Object.values(s.ownedSystems).reduce((a, b) => a + b, 0);
   575→        const totalTech = Object.keys(s.technologies).length;
   576→
   577→        if (s.population < reqs.minPopulation * evolveBoost) return;
   578→        if (totalSystems < reqs.minSystems * evolveBoost) return;
   579→        if (totalTech < reqs.minTech * evolveBoost) return;
   580→
   581→        const nextStageIdx = s.stageIndex + 1;
   582→        const nextStageDef = STAGES[nextStageIdx];
   583→
   584→        const resources = { ...s.resources };
   585→        // Give starting resources for the new stage so the player can build the first systems
   586→        // This prevents the "can't build anything" progression blocker
   587→        const stageStartBonus: Record<string, number> = {
   588→          creature: { food: 15, water: 10, materials: 5 },
   589→          tribal: { wood: 15, stone: 5, clay: 3 },
   590→          civilization: { production: 15, gold: 5 },
   591→          empire: { influence: 5, gold: 20 },
   592→          solar: { energy: 20, alloys: 5 },
   593→          galactic: { data: 10, energy: 30 },
   594→        };
   595→        const startBonus = stageStartBonus[nextStageDef.id];
   596→        if (startBonus) {
   597→          for (const [r, v] of Object.entries(startBonus)) {
   598→            resources[r] = (resources[r] || 0) + v;
   599→          }
   600→        }
   601→        const capBoostLvl = s.upgrades["expansive_vaults"] || 0;
   602→        const capMult = 1 + capBoostLvl * 0.20;
   603→        const archetype = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype] : null;
   604→        const archCapMult = archetype?.bonus.capMult || 0;
   605→        const capacities = recomputeCapacitiesFresh(s, capMult + archCapMult);
   606→
   607→        const newStory = { ...s.storyUnlocked };
   608→        const storyId = STORY_MAP[
   609→          nextStageDef.id === "creature" ? "first_body_stirs"
   610→          : nextStageDef.id === "tribal" ? "fire_remembered"
   611→          : nextStageDef.id === "civilization" ? "cities_learn_to_dream"
   612→          : nextStageDef.id === "empire" ? "distance_begins_to_obey"
   613→          : nextStageDef.id === "solar" ? "sky_stops_being_a_wall"
   614→          : "galaxy_takes_a_shape"
   615→        ];
   616→        if (storyId) newStory[storyId.id] = true;
   617→
   618→        // Lock archetype when evolving from Creature → Tribal
   619→        let lockedArchetype = s.lockedArchetype;
   620→        if (stage.id === "creature" && !lockedArchetype) {
   621→          const dom = dominantArchetype(s.archetypeAffinity);
   622→          if (dom) {
   623→            lockedArchetype = dom;
   624→            get().addToLog(`Lineage locked: ${ARCHETYPE_MAP[dom].name}.`);
   625→          } else {
   626→            lockedArchetype = "humanoid"; // default fallback
   627→            get().addToLog(`Lineage locked: Humanoid (no drift).`);
   628→          }
   629→        }
   630→
   631→        const stageClearCounts = { ...s.stageClearCounts, [stage.id]: (s.stageClearCounts[stage.id] || 0) + 1 };
   632→
   633→        // Track fastest Cell clear
   634→        let fastestCellClear = s.fastestCellClear || 0;
   635→        if (stage.id === "cell") {
   636→          const clearTime = s.time;
   637→          if (fastestCellClear === 0 || clearTime < fastestCellClear) {
   638→            fastestCellClear = clearTime;
   639→          }
   640→        }
   641→
   642→        // Check achievements after stage clear
   643→        const postEvolveState = { ...s, stageClearCounts, lockedArchetype, fastestCellClear } as GameState;
   644→        const achResult = checkAchievements(postEvolveState);
   645→        const newAchievements = achResult.newOnes.length > 0
   646→          ? [...(s.newAchievements || []), ...achResult.newOnes]
   647→          : s.newAchievements;
   648→
   649→        set({
   650→          stageIndex: nextStageIdx,
   651→          resources,
   652→          capacities,
   653→          storyUnlocked: newStory,
   654→          activeStoryPopup: storyId?.id || s.activeStoryPopup,
   655→          stageClearCounts,
   656→          fastestCellClear,
   657→          lockedArchetype,
   658→          achievements: achResult.earned,
   659→          newAchievements,
   660→        });
   661→        get().addToLog(`Evolved to ${nextStageDef.name}. ${nextStageDef.tagline}`);
   662→      },
   663→
   664→      triggerPrestige: () => {
   665→        const s = get();
   666→        const stage = STAGES[s.stageIndex];
   667→        const score = computeScore(s);
   668→        const stageMult = [1, 2, 4, 8, 16, 32, 64][s.stageIndex] || 1;
   669→        const epEarned = Math.max(1, Math.floor(Math.sqrt(score / 100) * stageMult));
   670→
   671→        const archive = [...s.archive];
   672→        const archName = s.lockedArchetype ? ARCHETYPE_MAP[s.lockedArchetype]?.name : "Humanoid";
   673→        archive.unshift({
   674→          runId: `run_${s.totalRuns + 1}`,
   675→          timestamp: Date.now(),
   676→          stage: stage.id,
   677→          archetype: s.lockedArchetype || "humanoid",
   678→          score,
   679→          epEarned,
   680→          duration: s.time,
   681→          ending: s.stageIndex >= 6 ? "Galactic Ascension" : `${stage.name} Evolution (${archName})`,
   682→        });
   683→
   684→        const archivedArchetypes = Array.from(new Set([
   685→          ...(s.archivedArchetypes || []),
   686→          ...(s.lockedArchetype ? [s.lockedArchetype] : []),
   687→        ]));
   688→
   689→        // ===== Challenges: unlock on first Galactic win + abandon any active challenge
   690→        const wasFirstGalacticWin = s.stageIndex >= 6 && !s.unlockedChallenges;
   691→        const unlockedChallenges = s.unlockedChallenges || (s.stageIndex >= 6);
   692→        const completedChallenges = s.completedChallenges || {};
   693→        // If a challenge was active when prestiging without completing it, that's
   694→        // an abandonment — log it. The fresh run will start with no active challenge.
   695→        if (s.activeChallenge) {
   696→          const chName = CHALLENGE_MAP[s.activeChallenge]?.name || s.activeChallenge;
   697→          // Defer logging until after state has been set so it appears after the
   698→          // prestige log entry.
   699→          setTimeout(() => get().addToLog(`Challenge abandoned: ${chName}. The trial is incomplete.`), 0);
   700→        }
   701→
   702→        // ===== Enlightenment (Layer 2): unlock when all 10 Evolution challenges are mastered.
   703→        // Also re-checked defensively here in case checkChallengeCompletion missed
   704→        // it (e.g., legacy saves where the 10th challenge was completed before
   705→        // this code shipped).
   706→        const completedChallengesCount = Object.values(completedChallenges).filter(Boolean).length;
   707→        const wasEnlightenmentUnlocked = !!s.enlightenmentUnlocked;
   708→        const enlightenmentUnlocked = wasEnlightenmentUnlocked ||
   709→          completedChallengesCount >= CHALLENGES.length;
   710→
   711→        // Divinity is earned only once Enlightenment is unlocked. Strong runs
   712→        // (higher stage + higher score) yield proportionally more.
   713→        const divinityEarned = enlightenmentUnlocked
   714→          ? Math.max(1, Math.floor(score / 500 * stageMult))
   715→          : 0;
   716→
   717→        const fresh = initialRunState();
   718→        const metaState: Partial<GameState> = {
   719→          evolutionPoints: s.evolutionPoints + epEarned,
   720→          totalRuns: s.totalRuns + 1,
   721→          galacticWins: s.galacticWins + (s.stageIndex >= 6 ? 1 : 0),
   722→          archive,
   723→          archivedArchetypes,
   724→          upgrades: s.upgrades,
   725→          storyUnlocked: { ...s.storyUnlocked, god_opens_its_eyes: true },
   726→          storyAcknowledged: s.storyAcknowledged,
   727→          unlockedLayers: s.unlockedLayers,
   728→          stageClearCounts: s.stageClearCounts,
   729→          fastestCellClear: s.fastestCellClear,
   730→          totalEventsResolved: s.totalEventsResolved,
   731→          totalPlayTime: s.totalPlayTime,
   732→          totalActions: s.totalActions,
   733→          totalSystemsBuilt: s.totalSystemsBuilt,
   734→          totalTechResearched: s.totalTechResearched,
   735→          achievements: s.achievements,
   736→          newAchievements: s.newAchievements,
   737→          // Persist challenge state across prestige
   738→          unlockedChallenges,
   739→          completedChallenges,
   740→          // Persist Enlightenment state across prestige (Divinity + route + foresight are permanent)
   741→          divinity: (s.divinity || 0) + divinityEarned,
   742→          enlightenmentUnlocked,
   743→          selectedRoute: s.selectedRoute,
   744→          foresightNodes: s.foresightNodes || {},
   745→        };
   746→
   747→        // Check achievements after prestige
   748→        const postState = { ...s, ...metaState } as GameState;
   749→        const achResult = checkAchievements(postState);
   750→        metaState.achievements = achResult.earned;
   751→        if (achResult.newOnes.length > 0) {
   752→          metaState.newAchievements = [...(s.newAchievements || []), ...achResult.newOnes];
   753→        }
   754→
   755→        // Determine which story popup should fire on prestige.
   756→        // Priority:
   757→        //   1. first Galactic win → Evolution Layer unveiled
   758→        //   2. first prestige where Enlightenment just unlocked → Enlightenment unveiled
   759→        //   3. the "god_opens_its_eyes" beat for the very first prestige
   760→        let prestigePopup: string | null = null;
   761→        if (wasFirstGalacticWin) {
   762→          prestigePopup = "evolution_layer_unveiled";
   763→          metaState.storyUnlocked = { ...metaState.storyUnlocked!, evolution_layer_unveiled: true };
   764→        } else if (!wasEnlightenmentUnlocked && enlightenmentUnlocked) {
   765→          // Enlightenment just unlocked (defensive — the unlock normally fires
   766→          // inside checkChallengeCompletion when the 10th challenge is mastered).
   767→          prestigePopup = "enlightenment_unlocked";
   768→          metaState.storyUnlocked = { ...metaState.storyUnlocked!, enlightenment_unlocked: true };
   769→        } else if (s.totalRuns === 0) {
   770→          prestigePopup = "god_opens_its_eyes";
   771→        }
   772→
   773→        set({
   774→          ...fresh,
   775→          ...metaState,
   776→          currentTab: "actions",
   777→          speed: 1,
   778→          paused: false,
   779→          showShop: false,
   780→          showEvolve: false,
   781→          showChallenges: false,
   782→          showEnlightenment: false,
   783→          activeStoryPopup: prestigePopup,
   784→          hasSeenIntro: true,
   785→          lastSaved: Date.now(),
   786→          tutorialActive: false,
   787→        });
   788→        get().addToLog(
   789→          `Prestige! Earned ${epEarned} Evolution Points` +
   790→          (divinityEarned > 0 ? ` and ${divinityEarned} Divinity.` : `.`),
   791→        );
   792→      },
   793→
   794→      setTab: (tab) => set({ currentTab: tab }),
   795→      setSpeed: (speed) => set({ speed }),
   796→      togglePause: () => set((s) => ({ paused: !s.paused })),
   797→      setShowShop: (v) => set({ showShop: v }),
   798→      setShowEvolve: (v) => set({ showEvolve: v }),
   799→      setShowSettings: (v) => set({ showSettings: v }),
   800→      setShowChallenges: (v) => set({ showChallenges: v }),
   801→      setShowEnlightenment: (v) => set({ showEnlightenment: v }),
   802→
   803→      // ============ CHALLENGES ============
   804→      setActiveChallenge: (id) => {
   805→        const s = get();
   806→        // Only one active challenge at a time. If we are setting a new challenge,
   807→        // the previous one (if any) is implicitly abandoned.
   808→        if (s.activeChallenge && id && s.activeChallenge !== id) {
   809→          const prev = CHALLENGE_MAP[s.activeChallenge];
   810→          if (prev) get().addToLog(`Abandoned "${prev.name}" before completion.`);
   811→        }
   812→        if (id === null) {
   813→          // Explicit abandon / clear
   814→          if (s.activeChallenge) {
   815→            const prev = CHALLENGE_MAP[s.activeChallenge];
   816→            if (prev) get().addToLog(`Abandoned "${prev.name}" before completion.`);
   817→          }
   818→          set({ activeChallenge: null, challengeRunStart: null });
   819→          return;
   820→        }
   821→        const ch = CHALLENGE_MAP[id];
   822→        if (!ch) return;
   823→        // Don't restart an already-completed challenge
   824→        if (s.completedChallenges && s.completedChallenges[id]) {
   825→          get().addToLog(`"${ch.name}" is already mastered.`);
   826→          return;
   827→        }
   828→        // Apply special "zeroStart" — drain all run resources + population progress
   829→        // to 0 so the player truly starts the trial with nothing.
   830→        const updates: Partial<GameState> = {
   831→          activeChallenge: id,
   832→          challengeRunStart: s.time,
   833→        };
   834→        if (ch.debuff.effects.zeroStart) {
   835→          const resources = { ...s.resources };
   836→          for (const k of Object.keys(resources)) {
   837→            // Keep happiness at a minimal baseline so the run isn't dead-on-arrival,
   838→            // but zero out every produced resource.
   839→            if (k === "happiness") {
   840→              resources[k] = 5;
   841→            } else {
   842→              resources[k] = 0;
   843→            }
   844→          }
   845→          updates.resources = resources;
   846→          updates.populationProgress = 0;
   847→        }
   848→        // Apply forced archetype (lock the lineage immediately if not already)
   849→        if (ch.debuff.effects.forcedArchetype && s.lockedArchetype !== ch.debuff.effects.forcedArchetype) {
   850→          updates.lockedArchetype = ch.debuff.effects.forcedArchetype;
   851→        }
   852→        set(updates);
   853→        get().addToLog(`Challenge begun: ${ch.name}. ${ch.debuff.label}.`);
   854→      },
   855→
   856→      checkChallengeCompletion: () => {
   857→        const s = get();
   858→        if (!s.activeChallenge) return;
   859→        const ch = CHALLENGE_MAP[s.activeChallenge];
   860→        if (!ch) return;
   861→        const cw = ch.completeWhen;
   862→        let done = false;
   863→        const totalSystems = Object.values(s.ownedSystems).reduce((a, b) => a + b, 0);
   864→        const totalTech = Object.keys(s.technologies).length;
   865→        switch (cw.type) {
   866→          case "reachPopulation":
   867→            done = s.population >= Number(cw.value);
   868→            break;
   869→          case "reachStage":
   870→            // Stage reached (or beyond)
   871→            {
   872→              const stageIds = ["cell", "creature", "tribal", "civilization", "empire", "solar", "galactic"];
   873→              const targetIdx = stageIds.indexOf(String(cw.value));
   874→              done = targetIdx >= 0 && s.stageIndex >= targetIdx;
   875→            }
   876→            break;
   877→          case "stockpileResource":
   878→            done = (s.resources[cw.resourceId || ""] || 0) >= Number(cw.value);
   879→            break;
   880→          case "clearStage":
   881→            done = (s.stageClearCounts?.[cw.value as keyof typeof s.stageClearCounts] || 0) >= 1;
   882→            break;
   883→          case "reachGalactic":
   884→            // Reached Galactic stage during this run
   885→            done = s.stageIndex >= 6;
   886→            break;
   887→          case "researchTechs":
   888→            done = totalTech >= Number(cw.value);
   889→            break;
   890→          case "buildSystems":
   891→            done = totalSystems >= Number(cw.value);
   892→            break;
   893→          case "clearCellFast":
   894→            // Cell stage must be cleared in under N seconds. We use the run-time
   895→            // when reaching Creature (stageIndex >= 1). If they have already
   896→            // cleared Cell during this challenge run, check the elapsed time.
   897→            if (s.stageIndex >= 1 && s.challengeRunStart != null) {
   898→              const elapsed = s.time - s.challengeRunStart;
   899→              done = elapsed <= Number(cw.value);
   900→            }
   901→            break;
   902→        }
   903→        if (!done) return;
   904→
   905→        // Mark challenge complete, clear the active slot, grant mastery bonus
   906→        const completedChallenges = { ...(s.completedChallenges || {}), [ch.id]: true };
   907→        set({
   908→          completedChallenges,
   909→          activeChallenge: null,
   910→          challengeRunStart: null,
   911→        });
   912→        get().addToLog(
   913→          `Challenge mastered: ${ch.name}! ${ch.mastery.label}. The god is pleased.`
   914→        );
   915→
   916→        // ===== Enlightenment unlock check (Layer 2)
   917→        // When the player masters the 10th and final Evolution challenge, the
   918→        // god opens its eyes fully. Enlightenment unlocks immediately, and the
   919→        // Enlightenment story popup fires INSTEAD of the generic challenge_mastered
   920→        // popup — that single moment of awakening deserves its own beat.
   921→        const completedCount = Object.values(completedChallenges).filter(Boolean).length;
   922→        const unlocksEnlightenmentNow = !s.enlightenmentUnlocked &&
   923→          completedCount >= CHALLENGES.length;
   924→        if (unlocksEnlightenmentNow) {
   925→          const storyUnlocked = { ...s.storyUnlocked, enlightenment_unlocked: true };
   926→          set({
   927→            enlightenmentUnlocked: true,
   928→            storyUnlocked,
   929→            activeStoryPopup: "enlightenment_unlocked",
   930→          });
   931→          get().addToLog(
   932→            `Enlightenment unlocked — the god opens its eyes fully. Divinity awakens.`
   933→          );
   934→        } else {
   935→          // Trigger a story popup so the player feels the reward
   936→          set({ activeStoryPopup: "challenge_mastered" });
   937→        }
   938→      },
   939→
   940→      // ============ ENLIGHTENMENT (Layer 2) ============
   941→      spendDivinity: (amount, _nodeId) => {
   942→        const s = get();
   943→        if ((s.divinity || 0) < amount) return false;
   944→        set({ divinity: s.divinity - amount });
   945→        return true;
   946→      },
   947→
   948→      selectRoute: (routeId) => {
   949→        const s = get();
   950→        // Route selection is free but requires Enlightenment unlocked
   951→        if (!s.enlightenmentUnlocked) return;
   952→        const route = ROUTE_MAP[routeId];
   953→        if (!route) return;
   954→        // Toggle off if already selected (let the player change their mind)
   955→        const newRoute = s.selectedRoute === routeId ? null : routeId;
   956→        set({ selectedRoute: newRoute });
   957→        get().addToLog(
   958→          newRoute
   959→            ? `Route chosen: ${route.name}. ${route.effectLabel}.`
   960→            : `Route unchosen. The god is open to all paths once more.`
   961→        );
   962→      },
   963→
   964→      purchaseForesightNode: (nodeId) => {
   965→        const s = get();
   966→        if (!s.enlightenmentUnlocked) return false;
   967→        const node = FORESIGHT_NODE_MAP[nodeId];
   968→        if (!node) return false;
   969→        // "Coming soon" nodes cannot be purchased
   970→        if (node.implemented === false) return false;
   971→        // Already purchased
   972→        if (s.foresightNodes?.[nodeId]) return false;
   973→        // Check prerequisites
   974→        if (node.requires) {
   975→          for (const req of node.requires) {
   976→            if (!s.foresightNodes?.[req]) return false;
   977→          }
   978→        }
   979→        // Check cost
   980→        if ((s.divinity || 0) < node.cost) return false;
   981→        const foresightNodes = { ...(s.foresightNodes || {}), [nodeId]: true };
   982→        set({
   983→          divinity: s.divinity - node.cost,
   984→          foresightNodes,
   985→        });
   986→        get().addToLog(
   987→          `Foresight acquired: ${node.name}. ${node.passive?.label || node.effect}.`
   988→        );
   989→        return true;
   990→      },
   991→
   992→      dismissTutorial: () => set({ tutorialActive: false, tutorialDismissed: true }),
   993→      setEventFrequency: (freq) => set({ eventFrequency: freq }),
   994→
   995→      resolveEvent: (eventId, choiceId) => {
   996→        const s = get();
   997→        const ev = EVENT_MAP[eventId];
   998→        if (!ev) {
   999→          set({ activeEvent: null });
  1000→          return;
  1001→        }
  1002→        const choice = ev.choices.find((c) => c.id === choiceId);
  1003→        if (!choice) {
  1004→          set({ activeEvent: null });
  1005→          return;
  1006→        }
  1007→
  1008→        // Check for risk — roll the dice
  1009→        const riskTriggered = choice.riskChance && Math.random() < choice.riskChance;
  1010→        const effects = riskTriggered && choice.riskEffect ? choice.riskEffect : choice;
  1011→
  1012→        // Apply effects
  1013→        const res = { ...s.resources };
  1014→        if (effects.resourceChanges) {
  1015→          for (const [r, v] of Object.entries(effects.resourceChanges)) {
  1016→            const cap = s.capacities[r] || Infinity;
  1017→            res[r] = Math.max(0, Math.min(cap, (res[r] || 0) + (v as number)));
  1018→          }
  1019→        }
  1020→        let pop = s.population;
  1021→        if (effects.populationChange) {
  1022→          pop = Math.max(0, pop + effects.populationChange);
  1023→        }
  1024→        if (effects.happinessChange || choice.happinessChange) {

  1237→function techMultiplier(s: GameState, target: "production" | "manual" | "capacity"): number {
  1238→  let mult = 1;
  1239→  for (const [techId, owned] of Object.entries(s.technologies)) {
  1240→    if (!owned) continue;
  1241→    const tech = TECH_MAP[techId];
  1242→    if (!tech || !tech.multiplier) continue;
  1243→    if (tech.multiplier.target === target || tech.multiplier.target === "production") {
  1244→      mult += tech.multiplier.value;
  1245→    }
  1246→  }
  1247→  // Achievement production bonus
  1248→  const ctx = buildAchievementCtx(s);
  1249→  mult += achievementBonus(s.achievements || {}, ctx, "production");
  1250→  // Challenge mastery bonus (only for completed challenges, persists across runs)
  1251→  // — applied as a production multiplier (all 10 challenges grant "production" type).
  1252→  if (target === "production") {
  1253→    mult += challengeMasteryBonus(s.completedChallenges || {}, "production");
  1254→  }
  1255→  // Foresight (Enlightenment Layer 2) — purchased foresight nodes grant a
  1256→  // small production / manual / capacity bonus that stacks with achievements
  1257→  // and challenge mastery. Route bonus is applied separately in tick() and
  1258→  // performAction (per-resource), not here, because routes are per-resource.
  1259→  if (target === "production" || target === "manual") {
  1260→    mult += foresightBonus(s.foresightNodes || {}, target);
  1261→  }
  1262→  return mult;
  1263→}

  1265→function buildAchievementCtx(s: GameState): AchievementCheckCtx {

  1267→  const techResearchedCount = Object.keys(s.technologies).filter((id) => s.technologies[id]).length;
  1268→  const upgradesOwnedCount = Object.values(s.upgrades).filter((c) => c > 0).length;

  1285→// Check all achievements; return new ones earned
  1286→function checkAchievements(s: GameState): { earned: Record<string, boolean>; newOnes: string[] } {
  1287→  const ctx = buildAchievementCtx(s);
  1288→  const earned = { ...(s.achievements || {}) };
  1289→  const newOnes: string[] = [];
  1290→  for (const ach of ACHIEVEMENTS) {
  1291→    if (!earned[ach.id] && ach.check(ctx)) {
  1292→      earned[ach.id] = true;
  1293→      newOnes.push(ach.id);
  1294→    }
  1295→  }
  1296→  return { earned, newOnes };

  1299→function computeScore(s: GameState): number {
  1300→  const totalSystems = Object.values(s.ownedSystems).reduce((a, b) => a + b, 0);
  1301→  const totalTech = Object.keys(s.technologies).length;
  1302→  const resourceSum = Object.values(s.resources).reduce((a, b) => a + b, 0);
  1303→  return Math.floor(
  1304→    s.population * 2 +
  1305→    totalSystems * 5 +
  1306→    totalTech * 10 +
  1307→    resourceSum * 0.5 +
  1308→    s.time * 0.1
  1309→  );
  1310→}

  1312→// Recompute all capacities from scratch based on owned systems
  1313→function recomputeCapacitiesFresh(s: GameState, capMult: number): Record<string, number> {
  1314→  const base = emptyCapacities();
  1315→  const caps: Record<string, number> = {};
  1316→  for (const r of Object.keys(base)) {
  1317→    let cap = (base[r] || 50) * capMult;
  1318→    for (const sys of SYSTEMS) {
  1319→      const count = s.ownedSystems[sys.id] || 0;
  1320→      if (!count || !sys.capacityBoost) continue;
  1321→      const boost = sys.capacityBoost[r];
  1322→      if (boost) cap += (boost as number) * count * capMult;
  1323→    }
  1324→    caps[r] = cap;
  1325→  }
  1326→  return caps;
  1327→}

  1329→// Export helper for components to compute whether an action is capped
  1330→export function isActionCapped(
  1331→  produces: Record<string, number>,
  1332→  resources: Record<string, number>,
  1333→  capacities: Record<string, number>
  1334→): boolean {
  1335→  const outputs = Object.entries(produces);

  1337→  return outputs.every(([r]) => {
  1338→    const cap = capacities[r] || 0;
  1339→    return (resources[r] || 0) >= cap - 0.001;
  1340→  });
