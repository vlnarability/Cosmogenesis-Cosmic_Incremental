   279→// ===== Active realm view — shows the actual mini-game UI =====
   280→function ActiveRealmView({
   281→  realmId,
   282→  state,
   283→  onClose,
   284→  onBreakthrough,
   285→  onDiscontentAction,
   286→  stageIndex,
   287→  liveElapsed = 0,
   288→}: {
   289→  realmId: string;
   290→  state: Record<string, any>;
   291→  onClose: () => void;
   292→  onBreakthrough: () => void;
   293→  onDiscontentAction: (action: "celebrate" | "tax" | "ignore") => void;
   294→  stageIndex: number;
   295→  liveElapsed?: number; // FEATURE 1 — live timer
   296→}) {
   297→  const realm = TRIAL_REALMS.find((r) => r.id === realmId);
   298→  if (!realm) return null;
   299→
   300→  return (
   301→    <div className="space-y-3">
   302→      <div className="flex items-center justify-between">
   303→        <div className="flex items-center gap-2">
   304→          <span className="text-3xl">{realm.icon}</span>
   305→          <div>
   306→            <div className="text-base font-semibold">{realm.name}</div>
   307→            <div className="text-[0.65rem] text-muted-foreground">{realm.tagline}</div>
   308→          </div>
   309→          {/* FEATURE 1 — live timer in the active realm header */}
   310→          <Badge variant="outline" className="ml-2 text-[0.6rem] text-cyan-300 border-cyan-400/40 animate-pulse">
   311→            <Hourglass className="w-2.5 h-2.5 mr-0.5" /> Time: {Math.floor(liveElapsed)}s
   312→          </Badge>
   313→        </div>
   314→        <Button size="sm" variant="outline" onClick={onClose}>Exit Realm</Button>
   315→      </div>
   316→
   317→      {realmId === "realm_growth" && (
   318→        <RealmGrowthGame state={state} onBreakthrough={onBreakthrough} />
   319→      )}
   320→      {realmId === "realm_discontent" && (
   321→        <RealmDiscontentGame
   322→          state={state}
   323→          onAction={onDiscontentAction}
   324→        />
   325→      )}
   326→      {realmId === "realm_swiftness" && (
   327→        <RealmSwiftnessGame state={state} stageIndex={stageIndex} />
   328→      )}
   329→    </div>
   330→  );
   331→}

   333→// ===== Realm of Growth — energy bar that fills passively, click to Break Through =====
   334→function RealmGrowthGame({ state, onBreakthrough }: { state: Record<string, any>; onBreakthrough: () => void }) {
   335→  const energy = state.energy || 0;
   336→  const breaks = state.breaks || 0;
   337→  const completed = !!state.completed;
   338→  const canBreak = energy >= 100 && !completed;
   339→
   340→  return (
   341→    <div className="stat-card space-y-3">
   342→      <div className="text-[0.7rem] text-muted-foreground">
   343→        Energy fills at 0.5/s. When the bar is full, click Break Through to convert energy into +50% population.
   344→      </div>
   345→
   346→      <div className="space-y-1">
   347→        <div className="flex justify-between text-xs">
   348→          <span className="text-emerald-300/80">Energy</span>
   349→          <span className="font-mono">{energy.toFixed(1)} / 100</span>
   350→        </div>
   351→        <div className="h-6 rounded-md bg-emerald-950/40 border border-emerald-500/30 overflow-hidden">
   352→          <div
   353→            className="h-full bg-gradient-to-r from-emerald-600/60 to-emerald-400/80 transition-all"
   354→            style={{ width: `${Math.min(100, energy)}%` }}
   355→          />
   356→        </div>
   357→      </div>
   358→
   359→      <div className="flex items-center justify-between">
   360→        <div className="text-xs">
   361→          <span className="text-muted-foreground">Break-Throughs: </span>
   362→          <span className="font-semibold text-amber-300">{breaks} / {REALM_GROWTH_BREAKS_GOAL}</span>
   363→        </div>
   364→        {completed ? (
   365→          <Badge className="bg-emerald-500/20 text-emerald-300 border-emerald-400/40">
   366→            <Trophy className="w-3 h-3 mr-1" /> Conquered
   367→          </Badge>
   368→        ) : (
   369→          <Button size="sm" disabled={!canBreak} onClick={onBreakthrough}>
   370→            <Sparkles className="w-3 h-3 mr-1" /> Break Through
   371→          </Button>
   372→        )}
   373→      </div>
   374→
   375→      {/* Progress dots */}
   376→      <div className="flex gap-1">
   377→        {Array.from({ length: REALM_GROWTH_BREAKS_GOAL }, (_, i) => (
   378→          <div
   379→            key={i}
   380→            className={`h-1.5 flex-1 rounded-sm ${i < breaks ? "bg-emerald-400" : "bg-muted-foreground/20"}`}
   381→          />
   382→        ))}
   383→      </div>
   384→    </div>
   385→  );
   386→}

Keep it above <span className="text-rose-300">{REALM_DISCONTENT_HAPPINESS_THRESHOLD}</span> while stockpiling <span className="text-amber-300">{REALM_DISCONTENT_GOLD_GOAL} gold</span>. Survive {REALM_DISCONTENT_SURVIVE_SECONDS}s above the threshold.

   520→function formatGoal(ch: typeof CHALLENGES[number]): string {
   521→  switch (ch.completeWhen.type) {
   522→    case "reachPopulation":
   523→      return `Reach a population of ${ch.completeWhen.value}`;
   524→    case "reachStage":
   525→      return `Reach the ${cap(ch.completeWhen.stage)} stage`;
   526→    case "clearStage":
   527→      return `Clear the ${cap(ch.completeWhen.stage)} stage (evolve past it)`;
   528→    case "reachGalactic":
   529→      return `Reach the Galactic stage`;
   530→    case "stockpileResource":
   531→      return `Stockpile ${ch.completeWhen.value} of ${ch.completeWhen.resource.replace("_", " ")}`;
   532→    default:
   533→      return "Unknown goal";
   534→  }
   535→}

   537→function formatEffects(eff: Record<string, unknown>): string {
   538→  const parts: string[] = [];
   539→  if (eff.productionMult !== undefined) {
   540→    const v = eff.productionMult as number;
   541→    parts.push(v >= 0 ? `+${(v * 100).toFixed(0)}% prod` : `${(v * 100).toFixed(0)}% prod`);
   542→  }
   543→  if (eff.costMult !== undefined) {
   544→    const v = eff.costMult as number;
   545→    if (v >= 1) parts.push(`costs ×${v.toFixed(2)}`);
   546→    else parts.push(`costs ${(v * 100).toFixed(0)}%`);
   547→  }
   548→  if (eff.popGrowthMult !== undefined) {
   549→    const v = eff.popGrowthMult as number;
   550→    parts.push(v >= 1 ? `pop ×${v.toFixed(2)}` : `pop ${(v * 100).toFixed(0)}%`);
   551→  }
   552→  if (eff.capMult !== undefined) {
   553→    const v = eff.capMult as number;
   554→    parts.push(v >= 1 ? `cap ×${v.toFixed(2)}` : `cap ${(v * 100).toFixed(0)}%`);
   555→  }
   556→  if (eff.epMult !== undefined) {
   557→    const v = eff.epMult as number;
   558→    parts.push(`+${(v * 100).toFixed(0)}% EP`);
   559→  }
   560→  if (eff.disableTech) parts.push("no tech");
   561→  if (eff.disableMilitary) parts.push("no military");
   562→  if (eff.disableEvents) parts.push("no events");
   563→  if (eff.disablePause) parts.push("no pause");
   564→  if (eff.startPop !== undefined) parts.push(`start pop = ${eff.startPop}`);
   565→  if (eff.maxSystems !== undefined) parts.push(`max ${eff.maxSystems} systems`);
   566→  return parts.join(" · ");
   567→}

   569→function cap(s: string): string {
   570→  return s.charAt(0).toUpperCase() + s.slice(1);
   571→}

   573→// QUICK WIN 4 — Real-time progress for an active challenge, based on its completeWhen condition.
   574→function ChallengeProgress({
   575→  cond,
   576→  population,
   577→  stageIndex,
   578→  resources,
   579→}: {
   580→  cond: ChallengeCompleteWhen;
   581→  population: number;
   582→  stageIndex: number;
   583→  resources: Record<string, number>;
   584→}) {
   585→  const curStageId = STAGES[stageIndex]?.id;
   586→  switch (cond.type) {
   587→    case "reachPopulation": {
   588→      const cur = Math.min(population, cond.value);
   589→      const pct = Math.min(100, (cur / cond.value) * 100);
   590→      return (
   591→        <div className="rounded-md border border-amber-400/30 bg-amber-500/10 p-1.5">
   592→          <div className="flex items-center justify-between text-amber-300/90">
   593→            <span className="font-semibold">Population progress</span>
   594→            <span>{formatNumber(cur, 0)}/{formatNumber(cond.value, 0)}</span>
   595→          </div>
   596→          <div className="h-1.5 rounded-full bg-muted/40 overflow-hidden mt-1">
   597→            <div className="h-full bg-amber-400/80 transition-all" style={{ width: `${pct}%` }} />
   598→          </div>
   599→        </div>
   600→      );
   601→    }
   602→    case "stockpileResource": {
   603→      const cur = Math.min(resources[cond.resource] || 0, cond.value);
   604→      const pct = Math.min(100, (cur / cond.value) * 100);
   605→      return (
   606→        <div className="rounded-md border border-amber-400/30 bg-amber-500/10 p-1.5">
   607→          <div className="flex items-center justify-between text-amber-300/90">
   608→            <span className="font-semibold">Resources ({cond.resource.replace("_", " ")})</span>
   609→            <span>{formatNumber(cur, 0)}/{formatNumber(cond.value, 0)}</span>
   610→          </div>
   611→          <div className="h-1.5 rounded-full bg-muted/40 overflow-hidden mt-1">
   612→            <div className="h-full bg-amber-400/80 transition-all" style={{ width: `${pct}%` }} />
   613→          </div>
   614→        </div>
   615→      );
   616→    }
   617→    case "reachStage": {
   618→      const targetIdx = STAGE_IDS.indexOf(cond.stage);
   619→      const reached = stageIndex >= targetIdx;
   620→      return (
   621→        <div className="rounded-md border border-amber-400/30 bg-amber-500/10 p-1.5">
   622→          <div className="flex items-center justify-between text-amber-300/90">
   623→            <span className="font-semibold">Stage</span>
   624→            <span>Current: {cap(curStageId || "")} · need {cap(cond.stage)}</span>
   625→          </div>
   626→          <div className="h-1.5 rounded-full bg-muted/40 overflow-hidden mt-1">
   627→            <div className="h-full bg-amber-400/80 transition-all" style={{ width: `${reached ? 100 : (stageIndex / Math.max(targetIdx, 1)) * 100}%` }} />
   628→          </div>
   629→        </div>
   630→      );
   631→    }
   632→    case "clearStage": {
   633→      const targetIdx = STAGE_IDS.indexOf(cond.stage);
   634→      const cleared = stageIndex > targetIdx;
   635→      const pct = cleared ? 100 : Math.min(100, (stageIndex / Math.max(targetIdx + 1, 1)) * 100);
   636→      return (
   637→        <div className="rounded-md border border-amber-400/30 bg-amber-500/10 p-1.5">
   638→          <div className="flex items-center justify-between text-amber-300/90">
   639→            <span className="font-semibold">Clear stage</span>
   640→            <span>{cleared ? "Cleared!" : `At ${cap(curStageId || "")} — need to evolve past ${cap(cond.stage)}`}</span>
   641→          </div>
   642→          <div className="h-1.5 rounded-full bg-muted/40 overflow-hidden mt-1">
   643→            <div className="h-full bg-amber-400/80 transition-all" style={{ width: `${pct}%` }} />
   644→          </div>
   645→        </div>
   646→      );
   647→    }
   648→    case "reachGalactic": {
   649→      const totalStages = STAGES.length;
   650→      const pct = Math.min(100, ((stageIndex + 1) / totalStages) * 100);
   651→      const reached = stageIndex >= totalStages - 1;
   652→      return (
   653→        <div className="rounded-md border border-amber-400/30 bg-amber-500/10 p-1.5">
   654→          <div className="flex items-center justify-between text-amber-300/90">
   655→            <span className="font-semibold">Reach Galactic</span>
   656→            <span>{reached ? "Galactic!" : `Stage ${stageIndex + 1}/${totalStages}`}</span>
   657→          </div>
   658→          <div className="h-1.5 rounded-full bg-muted/40 overflow-hidden mt-1">
   659→            <div className="h-full bg-amber-400/80 transition-all" style={{ width: `${pct}%` }} />
   660→          </div>
   661→        </div>
   662→      );
   663→    }
   664→    default:
   665→      return null;
   666→  }
   667→}
