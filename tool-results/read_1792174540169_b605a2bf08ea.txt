    27→// ===== Part 6 — Layer-progress helpers =====
    28→// Per-layer (L_n → L_n+1) unlock progress. Returns the current count and the max (target) count.
    29→// nextLayerKey uses the runtime unlockedLayers keys (e.g. "enlightenment" for L2).
    30→interface NextLayerProgressDef {
    31→  currentLayerName: string; // L_n name (the layer the player needs to have unlocked to progress)
    32→  nextLayerName: string;    // L_n+1 name (the layer the progress bar points toward)
    33→  nextLayerKey: string;     // key in unlockedLayers (or "evolution" for L1)
    34→  current: number;
    35→  max: number;
    36→  metricLabel: string;      // description of the metric (e.g. "Trials completed")
    37→}
    38→
    39→// ===== Part 6 — Layer-next-progress component =====
    40→// Renders a progress bar showing how close the player is to unlocking the NEXT layer (L_(n+1)).
    41→// - If next layer is already unlocked: "✓ {nextLayerName} Unlocked"
    42→// - If current layer (prereq for next) is NOT unlocked: "Requires {currentLayerName}"
    43→// - Otherwise: progress bar with the layer-unlock metric (e.g. "5/10 Foresight nodes").
    44→function LayerNextProgress({
    45→  progress,
    46→  currentLayerUnlocked,
    47→  nextLayerUnlocked,
    48→}: {
    49→  progress: NextLayerProgressDef;
    50→  currentLayerUnlocked: boolean;
    51→  nextLayerUnlocked: boolean;
    52→}) {
    53→  if (nextLayerUnlocked) {
    54→    return (
    55→      <div className="mt-2 flex items-center gap-1.5 text-[0.65rem] text-emerald-300">
    56→        <CheckCircle2 className="w-3 h-3" />
    57→        <span className="font-semibold">{progress.nextLayerName} Unlocked</span>
    58→      </div>
    59→    );
    60→  }
    61→  if (!currentLayerUnlocked) {
    62→    return (
    63→      <div className="mt-2 flex items-center gap-1.5 text-[0.65rem] text-muted-foreground">
    64→        <Lock className="w-3 h-3" />
    65→        <span>Requires <span className="font-semibold text-foreground/80">{progress.currentLayerName}</span> to be unlocked first</span>
    66→      </div>
    67→    );
    68→  }
    69→  const pct = progress.max > 0 ? Math.min(100, (progress.current / progress.max) * 100) : 0;
    70→  return (
    71→    <div className="mt-2">
    72→      <div className="flex items-center justify-between text-[0.6rem] text-muted-foreground">
    73→        <span className="flex items-center gap-1">
    74→          <ArrowRight className="w-2.5 h-2.5" />
    75→          Toward {progress.nextLayerName}
    76→          <span className="text-muted-foreground/70">· {progress.metricLabel}</span>
    77→        </span>
    78→        <span className="tabular-nums">{progress.current}/{progress.max}</span>
    79→      </div>
    80→      <div className="h-1.5 bg-muted/40 rounded-full overflow-hidden mt-0.5">
    81→        <div className="h-full bg-emerald-400/70 transition-all" style={{ width: `${pct}%` }} />
    82→      </div>
    83→    </div>
    84→  );
    85→}

   695→function ProgressRow({ label, value, max }: { label: string; value: number; max: number }) {
   696→  const pct = max > 0 ? Math.min(100, (value / max) * 100) : 0;
   697→  return (
   698→    <div className="mt-1">
   699→      <div className="flex items-center justify-between text-[0.6rem] text-muted-foreground">
   700→        <span>{label}</span>
   701→        <span>{value}/{max}</span>
   702→      </div>
   703→      <div className="h-1 bg-muted/40 rounded-full overflow-hidden mt-0.5">
   704→        <div className="h-full bg-amber-400/70 transition-all" style={{ width: `${pct}%` }} />
   705→      </div>
   706→    </div>
   707→  );
   708→}

   711→  layer,
   712→  unlocked,

   752→function formatUnlock(layer: PrestigeLayer): string {
   753→  switch (layer.unlockCondition.type) {
   754→    case "default":
   755→      return "Default";
   756→    case "galacticWins":
   757→      return `${layer.unlockCondition.value} wins`;
   758→    case "challengesCompleted":
   759→      return `${layer.unlockCondition.value} trials`;
   760→    case "totalRuns":
   761→      return `${layer.unlockCondition.value} runs`;
   762→    default:
   763→      return "Locked";
   764→  }
   765→}
