   612→function totalSystems(s: any): number {
   613→  return Object.values(s.ownedSystems || {}).reduce((a: number, b: any) => a + (b || 0), 0) as number;
   614→}
   615→
   616→function stageSystemsCount(s: any, stage: StageId): number {
   617→  return SYSTEMS
   618→    .filter((x) => x.stage === stage)
   619→    .reduce((sum, def) => sum + (s.ownedSystems?.[def.id] || 0), 0);
   620→}
   621→
   622→function hasResourceAtCap(s: any): boolean {
   623→  if (!s.resources || !s.capacities) return false;
   624→  for (const r of Object.keys(s.resources)) {
   625→    const cap = s.capacities[r];
   626→    if (cap && cap > 0 && (s.resources[r] || 0) >= cap - 0.5) return true;
   627→  }
   628→  return false;
   629→}
